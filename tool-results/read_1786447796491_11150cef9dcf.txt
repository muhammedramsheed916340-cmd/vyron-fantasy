     1→'use client';
     2→
     3→import React, { useState, useCallback, useMemo } from 'react';
     4→import {
     5→  Trophy, X, CheckCircle2, Loader2, ChevronRight, ChevronLeft,
     6→  AlertCircle, Users, DollarSign, Filter, RefreshCw,
     7→} from 'lucide-react';
     8→import {
     9→  JCMatch, JCContest, JCTeam, JCJoinItem, JCProgress,
    10→  parseContest, formatCurrency, getContestTypeColor,
    11→  getExistingPlatformTeams, buildJoinItems, getJoinKey,
    12→} from '@/lib/join-contest-service';
    13→
    14→// ============ Step Enum ============
    15→type Step = 'matches' | 'teams' | 'contests' | 'joining' | 'result';
    16→
    17→// ============ Props ============
    18→interface JoinContestDialogProps {
    19→  open: boolean;
    20→  onClose: () => void;
    21→  matches: JCMatch[];
    22→  fantasyAccounts: Record<string, { authToken: string; mobileNumber: string; my11circleChallenge?: string | null }>;
    23→}
    24→
    25→// ============ Main Dialog ============
    26→export default function JoinContestDialog({
    27→  open, onClose, matches, fantasyAccounts,
    28→}: JoinContestDialogProps) {
    29→  // Step state
    30→  const [step, setStep] = useState<Step>('matches');
    31→  const [platform, setPlatform] = useState<string>('dream11');
    32→
    33→  // Match selection
    34→  const [selectedMatchIds, setSelectedMatchIds] = useState<Set<string | number>>(new Set());
    35→
    36→  // Team selection — EXISTING PLATFORM TEAMS
    37→  const [platformTeams, setPlatformTeams] = useState<JCTeam[]>([]);
    38→  const [selectedTeamIds, setSelectedTeamIds] = useState<Set<string | number>>(new Set());
    39→  const [mixedTeamMode, setMixedTeamMode] = useState(false);
    40→  const [loadingTeams, setLoadingTeams] = useState(false);
    41→  const [teamsError, setTeamsError] = useState<string | null>(null);
    42→  const [teamsTokenExpired, setTeamsTokenExpired] = useState(false);
    43→
    44→  // Contest selection
    45→  const [contestsMap, setContestsMap] = useState<Map<string, JCContest[]>>(new Map());
    46→  const [selectedContestIds, setSelectedContestIds] = useState<Set<string>>(new Set());
    47→  const [loadingContests, setLoadingContests] = useState(false);
    48→
    49→  // Join progress
    50→  const [joinItems, setJoinItems] = useState<JCJoinItem[]>([]);
    51→  const [progress, setProgress] = useState<JCProgress>({ current: 0, total: 0, status: 'idle' });
    52→
    53→  // Result
    54→  const [resultItems, setResultItems] = useState<JCJoinItem[]>([]);
    55→
    56→  // Available platforms from linked accounts
    57→  const availablePlatforms = useMemo(() => {
    58→    const platforms: string[] = [];
    59→    if (fantasyAccounts.dream11?.authToken) platforms.push('dream11');
    60→    if (fantasyAccounts.my11circle?.authToken) platforms.push('my11circle');
    61→    return platforms;
    62→  }, [fantasyAccounts]);
    63→
    64→  const account = fantasyAccounts[platform];
    65→
    66→  // Reset on close
    67→  const handleClose = () => {
    68→    setStep('matches');
    69→    setSelectedMatchIds(new Set());
    70→    setPlatformTeams([]);
    71→    setSelectedTeamIds(new Set());
    72→    setMixedTeamMode(false);
    73→    setLoadingTeams(false);
    74→    setTeamsError(null);
    75→    setTeamsTokenExpired(false);
    76→    setContestsMap(new Map());
    77→    setSelectedContestIds(new Set());
    78→    setJoinItems([]);
    79→    setProgress({ current: 0, total: 0, status: 'idle' });
    80→    setResultItems([]);
    81→    onClose();
    82→  };
    83→
    84→  // ============ Match Selection ============
    85→  const toggleMatch = (id: string | number) => {
    86→    setSelectedMatchIds(prev => {
    87→      const next = new Set(prev);
    88→      next.has(id) ? next.delete(id) : next.add(id);
    89→      return next;
    90→    });
    91→  };
    92→
    93→  const selectAllMatches = () => {
    94→    setSelectedMatchIds(new Set(matches.map(m => m.id)));
    95→  };
    96→
    97→  // ============ Load Existing Platform Teams ============
    98→  const loadPlatformTeams = useCallback(async (selectedPlatform?: string) => {
    99→    const plat = selectedPlatform || platform;
   100→    const acc = fantasyAccounts[plat];
   101→    if (!acc?.authToken) {
   102→      setTeamsError('No connected account found. Connect your platform account to load existing teams.');
   103→      setPlatformTeams([]);
   104→      return;
   105→    }
   106→
   107→    setLoadingTeams(true);
   108→    setTeamsError(null);
   109→    setTeamsTokenExpired(false);
   110→    setSelectedTeamIds(new Set());
   111→
   112→    const allTeams: JCTeam[] = [];
   113→    let firstError: string | null = null;
   114→    let anyTokenExpired = false;
   115→
   116→    for (const matchId of selectedMatchIds) {
   117→      const result = await getExistingPlatformTeams(plat, matchId, acc.authToken);
   118→      if (result.teams.length > 0) {
   119→        allTeams.push(...result.teams);
   120→      }
   121→      if (result.tokenExpired) {
   122→        anyTokenExpired = true;
   123→        firstError = result.error || 'Session expired';
   124→      } else if (result.error && !firstError) {
   125→        firstError = result.error;
   126→      }
   127→    }
   128→
   129→    setPlatformTeams(allTeams);
   130→    setTeamsError(allTeams.length === 0 ? firstError : null);
   131→    setTeamsTokenExpired(anyTokenExpired);
   132→    setLoadingTeams(false);
   133→  }, [selectedMatchIds, platform, fantasyAccounts]);
   134→
   135→  // ============ Team Selection ============
   136→  const toggleTeam = (id: string | number) => {
   137→    setSelectedTeamIds(prev => {
   138→      const next = new Set(prev);
   139→      next.has(id) ? next.delete(id) : next.add(id);
   140→      return next;
   141→    });
   142→  };
   143→
   144→  const selectAllTeams = () => {
   145→    setSelectedTeamIds(new Set(platformTeams.map(t => t.id)));
   146→  };
   147→
   148→  const deselectAllTeams = () => {
   149→    setSelectedTeamIds(new Set());
   150→  };
   151→
   152→  // ============ Contest Loading ============
   153→  const loadContests = useCallback(async () => {
   154→    if (!account?.authToken) return;
   155→    setLoadingContests(true);
   156→    const newMap = new Map<string, JCContest[]>();
   157→
   158→    for (const matchId of selectedMatchIds) {
   159→      const match = matches.find(m => m.id === matchId);
   160→      try {
   161→        const res = await fetch('/api/fantasy/list-contests', {
   162→          method: 'POST',
   163→          headers: { 'Content-Type': 'application/json' },
   164→          body: JSON.stringify({
   165→            fantasyApp: platform,
   166→            matchId,
   167→            authToken: account.authToken,
   168→            sportIndex: match?.sport_index ?? 0,
   169→          }),
   170→        });
   171→        const data = await res.json();
   172→        if (data.status === 'success' && Array.isArray(data.data?.contests)) {
   173→          newMap.set(String(matchId), data.data.contests.map((c: Record<string, unknown>) => parseContest(c)));
   174→        } else {
   175→          newMap.set(String(matchId), []);
   176→        }
   177→      } catch {
   178→        newMap.set(String(matchId), []);
   179→      }
   180→    }
   181→
   182→    setContestsMap(newMap);
   183→    setLoadingContests(false);
   184→  }, [selectedMatchIds, platform, account, matches]);
   185→
   186→  const toggleContest = (id: string) => {
   187→    setSelectedContestIds(prev => {
   188→      const next = new Set(prev);
   189→      next.has(id) ? next.delete(id) : next.add(id);
   190→      return next;
   191→    });
   192→  };
   193→
   194→  // ============ Join Execution ============
   195→  const executeJoin = useCallback(async () => {
   196→    const selectedMatches = matches.filter(m => selectedMatchIds.has(m.id));
   197→    const selectedContestsForJoin = new Map<string, JCContest[]>();
   198→    for (const [matchId, contests] of contestsMap.entries()) {
   199→      selectedContestsForJoin.set(matchId, contests.filter(c => selectedContestIds.has(c.id)));
   200→    }
   201→
   202→    const items = buildJoinItems(selectedMatches, selectedContestsForJoin, selectedTeamIds, platformTeams, platform);
   203→
   204→    if (items.length === 0) {
   205→      setStep('result');
   206→      return;
   207→    }
   208→
   209→    setJoinItems(items);
   210→    setProgress({ current: 0, total: items.length, status: 'joining' });
   211→    setStep('joining');
   212→
   213→    const updatedItems = [...items];
   214→
   215→    for (let i = 0; i < items.length; i++) {
   216→      const item = items[i];
   217→      setProgress(prev => ({ ...prev, current: i + 1 }));
   218→      setJoinItems(prev => prev.map((it, idx) =>
   219→        idx === i ? { ...it, status: 'processing' } : it
   220→      ));
   221→
   222→      try {
   223→        const match = matches.find(m => String(m.id) === String(item.matchId));
   224→        const res = await fetch('/api/fantasy/join-contest', {
   225→          method: 'POST',
   226→          headers: { 'Content-Type': 'application/json' },
   227→          body: JSON.stringify({
   228→            fantasyApp: item.platform || platform,
   229→            matchId: item.matchId,
   230→            authToken: account?.authToken,
   231→            teamId: item.teamId,    // REAL platform team ID
   232→            contestId: item.contestId,
   233→            sportIndex: match?.sport_index ?? 0,
   234→          }),
   235→        });
   236→        const data = await res.json();
   237→
   238→        const status = data.status === 'success' ? 'success'
   239→          : data.status === 'already_joined' ? 'already_joined' : 'fail';
   240→
   241→        updatedItems[i] = { ...updatedItems[i], status, message: data.message };
   242→        setJoinItems([...updatedItems]);
   243→      } catch {
   244→        updatedItems[i] = { ...updatedItems[i], status: 'fail', message: 'Network error' };
   245→        setJoinItems([...updatedItems]);
   246→      }
   247→
   248→      // Rate limit between joins
   249→      await new Promise(r => setTimeout(r, 300));
   250→    }
   251→
   252→    setResultItems([...updatedItems]);
   253→    setProgress(prev => ({ ...prev, status: 'done' }));
   254→    setStep('result');
   255→  }, [matches, selectedMatchIds, contestsMap, selectedContestIds, selectedTeamIds, platformTeams, platform, account]);
   256→
   257→  // ============ Result Stats ============
   258→  const resultStats = useMemo(() => {
   259→    const items = resultItems.length > 0 ? resultItems : joinItems;
   260→    return {
   261→      total: items.length,
   262→      success: items.filter(i => i.status === 'success').length,
   263→      alreadyJoined: items.filter(i => i.status === 'already_joined').length,
   264→      fail: items.filter(i => i.status === 'fail').length,
   265→    };
   266→  }, [resultItems, joinItems]);
   267→
   268→  if (!open) return null;
   269→
   270→  // ============ RENDER ============
   271→  return (
   272→    <div className="fixed inset-0 z-50 bg-black/60 flex items-center justify-center p-4" onClick={handleClose}>
   273→      <div className="bg-white rounded-2xl shadow-2xl w-full max-w-2xl max-h-[90vh] overflow-hidden flex flex-col" onClick={e => e.stopPropagation()}>
   274→        {/* Header */}
   275→        <div className="flex items-center justify-between px-6 py-4 border-b bg-gradient-to-r from-green-600 to-emerald-600 text-white">
   276→          <div className="flex items-center gap-3">
   277→            <Trophy className="w-6 h-6" />
   278→            <h2 className="text-lg font-bold">Join Contest</h2>
   279→          </div>
   280→          <button onClick={handleClose} className="p-1 hover:bg-white/20 rounded-lg transition-all">
   281→            <X className="w-5 h-5" />
   282→          </button>
   283→        </div>
   284→
   285→        {/* Step Indicator */}
   286→        <div className="flex items-center gap-1 px-6 py-3 bg-gray-50 border-b">
   287→          {(['matches', 'teams', 'contests', 'joining', 'result'] as Step[]).map((s, i) => {
   288→            const labels = ['Match', 'Teams', 'Contests', 'Join', 'Result'];
   289→            const stepOrder = ['matches', 'teams', 'contests', 'joining', 'result'];
   290→            const isActive = step === s;
   291→            const isDone = stepOrder.indexOf(step) > i;
   292→            return (
   293→              <React.Fragment key={s}>
   294→                {i > 0 && <ChevronRight className="w-3 h-3 text-gray-300" />}
   295→                <div className={`flex items-center gap-1.5 text-xs font-semibold px-2 py-1 rounded-lg transition-all ${
   296→                  isActive ? 'bg-green-100 text-green-700' : isDone ? 'text-green-500' : 'text-gray-400'
   297→                }`}>
   298→                  <div className={`w-5 h-5 rounded-full flex items-center justify-center text-[10px] ${
   299→                    isActive ? 'bg-green-600 text-white' : isDone ? 'bg-green-200 text-green-600' : 'bg-gray-200 text-gray-500'
   300→                  }`}>
   301→                    {isDone ? '✓' : i + 1}
   302→                  </div>
   303→                  {labels[i]}
   304→                </div>
   305→              </React.Fragment>
   306→            );
   307→          })}
   308→        </div>
   309→
   310→        {/* Content */}
   311→        <div className="flex-1 overflow-y-auto p-6">
   312→
   313→          {/* ===== STEP: MATCHES ===== */}
   314→          {step === 'matches' && (
   315→            <div className="space-y-4">
   316→              {/* Platform Selection */}
   317→              <div className="space-y-2">
   318→                <p className="text-xs font-semibold text-gray-500 uppercase tracking-wider">Platform</p>
   319→                <div className="flex gap-2">
   320→                  {availablePlatforms.map(p => (
   321→                    <button key={p} onClick={() => setPlatform(p)}
   322→                      className={`flex-1 rounded-xl border-2 p-3 text-center font-semibold text-sm transition-all ${
   323→                        platform === p ? 'border-green-500 bg-green-50 text-green-700' : 'border-gray-200 bg-white text-gray-700 hover:border-gray-300'
   324→                      }`}>
   325→                      {p === 'dream11' ? 'Dream11' : 'My11Circle'}
   326→                    </button>
   327→                  ))}
   328→                </div>
   329→                {availablePlatforms.length === 0 && (
   330→                  <div className="bg-amber-50 border border-amber-200 rounded-xl p-3 text-sm text-amber-700">
   331→                    ⚠️ No connected account found. Link your platform account first via Transfer.
   332→                  </div>
   333→                )}
   334→              </div>
   335→
   336→              {/* Match List */}
   337→              <div className="space-y-2">
   338→                <div className="flex items-center justify-between">
   339→                  <p className="text-xs font-semibold text-gray-500 uppercase tracking-wider">Select Matches</p>
   340→                  <button onClick={selectAllMatches} className="text-xs font-semibold text-green-600 hover:text-green-700">
   341→                    Select All
   342→                  </button>
   343→                </div>
   344→                {matches.length === 0 ? (
   345→                  <p className="text-sm text-gray-400 text-center py-8">No upcoming matches</p>
   346→                ) : (
   347→                  <div className="space-y-2 max-h-80 overflow-y-auto">
   348→                    {matches.map(match => {
   349→                      const selected = selectedMatchIds.has(match.id);
   350→                      return (
   351→                        <button key={match.id} onClick={() => toggleMatch(match.id)}
   352→                          className={`w-full text-left rounded-xl border-2 p-3 transition-all ${
   353→                            selected ? 'border-green-500 bg-green-50' : 'border-gray-200 bg-white hover:border-gray-300'
   354→                          }`}>
   355→                          <div className="flex items-center gap-3">
   356→                            <div className={`w-5 h-5 rounded border-2 flex items-center justify-center shrink-0 ${
   357→                              selected ? 'border-green-500 bg-green-500' : 'border-gray-300'
   358→                            }`}>
   359→                              {selected && <CheckCircle2 className="w-3.5 h-3.5 text-white" />}
   360→                            </div>
   361→                            <div className="flex-1">
   362→                              <p className="font-semibold text-sm text-gray-900">
   363→                                {match.left_team_name} vs {match.right_team_name}
   364→                              </p>
   365→                              <div className="flex items-center gap-2 text-xs text-gray-500 mt-0.5">
   366→                                <span>{match.match_time}</span>
   367→                                {match.lineup_out === 1 && <span className="text-green-600 font-medium">Lineup Out</span>}
   368→                              </div>
   369→                            </div>
   370→                          </div>
   371→                        </button>
   372→                      );
   373→                    })}
   374→                  </div>
   375→                )}
   376→              </div>
   377→            </div>
   378→          )}
   379→
   380→          {/* ===== STEP: TEAMS — EXISTING PLATFORM TEAMS ===== */}
   381→          {step === 'teams' && (
   382→            <div className="space-y-4">
   383→              {/* Loading State */}
   384→              {loadingTeams && (
   385→                <div className="flex flex-col items-center justify-center py-12">
   386→                  <Loader2 className="w-8 h-8 animate-spin text-green-500" />
   387→                  <p className="text-sm text-gray-500 mt-3">Loading existing teams from {platform === 'dream11' ? 'Dream11' : 'My11Circle'}...</p>
   388→                </div>
   389→              )}
   390→
   391→              {/* Error State */}
   392→              {!loadingTeams && teamsError && (
   393→                <div className="bg-red-50 border border-red-200 rounded-xl p-4">
   394→                  <div className="flex items-center gap-2">
   395→                    <AlertCircle className="w-5 h-5 text-red-500 shrink-0" />
   396→                    <div>
   397→                      <p className="text-sm font-semibold text-red-700">
   398→                        {teamsTokenExpired ? 'SESSION EXPIRED' : 'Error Loading Teams'}
   399→                      </p>
   400→                      <p className="text-xs text-red-600 mt-0.5">{teamsError}</p>
   401→                    </div>
   402→                  </div>
   403→                  {teamsTokenExpired && (
   404→                    <p className="text-xs text-red-500 mt-2">Reconnect your {platform === 'dream11' ? 'Dream11' : 'My11Circle'} account via Transfer.</p>
   405→                  )}
   406→                  <button onClick={() => loadPlatformTeams()} className="mt-2 flex items-center gap-1 text-xs font-semibold text-red-600 hover:text-red-700">
   407→                    <RefreshCw className="w-3 h-3" /> Retry
   408→                  </button>
   409→                </div>
   410→              )}
   411→
   412→              {/* No Account */}
   413→              {!loadingTeams && !teamsError && !account?.authToken && (
   414→                <div className="bg-amber-50 border border-amber-200 rounded-xl p-4 text-center">
   415→                  <p className="text-sm font-semibold text-amber-700">No connected account found</p>
   416→                  <p className="text-xs text-amber-600 mt-1">Connect your {platform === 'dream11' ? 'Dream11' : 'My11Circle'} account to load existing teams.</p>
   417→                </div>
   418→              )}
   419→
   420→              {/* Teams List */}
   421→              {!loadingTeams && account?.authToken && (
   422→                <div className="space-y-2">
   423→                  <div className="flex items-center justify-between">
   424→                    <p className="text-xs font-semibold text-gray-500 uppercase tracking-wider">
   425→                      Existing Teams ({platformTeams.length} available)
   426→                    </p>
   427→                    <div className="flex gap-2">
   428→                      <button onClick={selectAllTeams} disabled={platformTeams.length === 0}
   429→                        className="text-xs font-semibold text-green-600 hover:text-green-700 disabled:opacity-40">
   430→                        Select All
   431→                      </button>
   432→                      <button onClick={deselectAllTeams} disabled={selectedTeamIds.size === 0}
   433→                        className="text-xs font-semibold text-gray-500 hover:text-gray-700 disabled:opacity-40">
   434→                        Deselect
   435→                      </button>
   436→                    </div>
   437→                  </div>
   438→
   439→                  {/* Mixed Team Toggle */}
   440→                  <button onClick={() => setMixedTeamMode(!mixedTeamMode)}
   441→                    className={`w-full rounded-xl border-2 p-3 transition-all ${
   442→                      mixedTeamMode ? 'border-purple-500 bg-purple-50' : 'border-gray-200 bg-white hover:border-gray-300'
   443→                    }`}>
   444→                    <div className="flex items-center gap-3">
   445→                      <div className={`w-9 h-9 rounded-lg flex items-center justify-center ${
   446→                        mixedTeamMode ? 'bg-purple-500 text-white' : 'bg-gray-100 text-gray-500'
   447→                      }`}>
   448→                        <Filter className="w-5 h-5" />
   449→                      </div>
   450→                      <div className="flex-1">
   451→                        <p className={`font-semibold text-sm ${mixedTeamMode ? 'text-purple-700' : 'text-gray-900'}`}>Mixed Team Mode</p>
   452→                        <p className="text-xs text-gray-500">
   453→                          {mixedTeamMode ? 'Combining teams from compatible platform accounts' : 'Enable to mix teams from compatible accounts'}
   454→                        </p>
   455→                      </div>
   456→                      <div className={`w-10 h-6 rounded-full relative transition-all ${mixedTeamMode ? 'bg-purple-500' : 'bg-gray-300'}`}>
   457→                        <div className={`absolute top-0.5 w-5 h-5 rounded-full bg-white shadow transition-all ${mixedTeamMode ? 'left-[18px]' : 'left-0.5'}`} />
   458→                      </div>
   459→                    </div>
   460→                  </button>
   461→
   462→                  {/* No Teams */}
   463→                  {platformTeams.length === 0 && !teamsError && (
   464→                    <div className="bg-gray-50 border border-gray-200 rounded-xl p-6 text-center">
   465→                      <p className="text-sm text-gray-500">No existing teams found on this platform account.</p>
   466→                      <p className="text-xs text-gray-400 mt-1">Create teams on {platform === 'dream11' ? 'Dream11' : 'My11Circle'} first, then join contests.</p>
   467→                    </div>
   468→                  )}
   469→
   470→                  {/* Team Cards */}
   471→                  {platformTeams.length > 0 && (
   472→                    <div className="space-y-2 max-h-72 overflow-y-auto">
   473→                      {platformTeams.map(team => {
   474→                        const selected = selectedTeamIds.has(team.id);
   475→                        return (
   476→                          <button key={team.id} onClick={() => toggleTeam(team.id)}
   477→                            className={`w-full text-left rounded-xl border-2 p-3 transition-all ${
   478→                              selected ? 'border-green-500 bg-green-50' : 'border-gray-200 bg-white hover:border-gray-300'
   479→                            }`}>
   480→                            <div className="flex items-center gap-3">
   481→                              <div className={`w-5 h-5 rounded border-2 flex items-center justify-center shrink-0 ${
   482→                                selected ? 'border-green-500 bg-green-500' : 'border-gray-300'
   483→                              }`}>
   484→                                {selected && <CheckCircle2 className="w-3.5 h-3.5 text-white" />}
   485→                              </div>
   486→                              <div className="flex-1">
   487→                                <p className="font-semibold text-sm text-gray-900">{team.name}</p>
   488→                                <div className="flex items-center gap-2 text-xs text-gray-500 mt-0.5">
   489→                                  <span>{team.playerCount || 11} Players</span>
   490→                                  {team.captain?.name && <span>C: {team.captain.name}</span>}
   491→                                  {team.viceCaptain?.name && <span>VC: {team.viceCaptain.name}</span>}
