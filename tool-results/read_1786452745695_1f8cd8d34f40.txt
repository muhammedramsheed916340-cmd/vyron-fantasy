   145→// Countdown timer component
   146→function CountdownTimer({ matchTime }: { matchTime: string }) {
   147→  const [timeLeft, setTimeLeft] = useState('')
   148→
   149→  useEffect(() => {
   150→    const calculateTime = () => {
   151→      const now = new Date().getTime()
   152→      const matchDate = new Date(matchTime).getTime()
   153→      const diff = matchDate - now
   154→
   155→      if (diff <= 0) {
   156→        setTimeLeft('Started')
   157→        return
   158→      }
   159→
   160→      const days = Math.floor(diff / (1000 * 60 * 60 * 24))
   161→      const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60))
   162→      const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60))
   163→      const seconds = Math.floor((diff % (1000 * 60)) / 1000)
   164→
   165→      if (days > 0) {
   166→        setTimeLeft(`${days}d ${hours}h ${minutes}m ${seconds}s`)
   167→      } else {
   168→        setTimeLeft(`${hours}h ${minutes}m ${seconds}s`)
   169→      }
   170→    }
   171→
   172→    calculateTime()
   173→    const interval = setInterval(calculateTime, 1000)
   174→    return () => clearInterval(interval)
   175→  }, [matchTime])
   176→
   177→  return <span className="font-medium text-sm text-foreground">{timeLeft}</span>
   178→}
