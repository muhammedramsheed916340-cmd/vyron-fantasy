    79→    79→export function getEligiblePlayers(

    88→    88→
    89→    89→    if (mode === 'after') {
    90→    90→      // AFTER LINEUP: ONLY confirmed Playing XI players
    91→    91→      return p.playing === 1;
    92→    92→    }
    93→    93→
    94→    94→    // BEFORE LINEUP: all players eligible
    95→    95→    return true;

   157→   157→/**
   158→   158→ * Get all valid role combinations for an 11-player cricket team.
   159→   159→ */
   160→   160→export function getAllValidCombinations(): RoleCombination[] {
   161→   161→  const combos: RoleCombination[] = [];
   162→   162→  for (let wk = MIN_WK; wk <= MAX_WK; wk++) {
   163→   163→    for (let bat = MIN_BAT; bat <= MAX_BAT; bat++) {
   164→   164→      for (let ar = MIN_AR; ar <= MAX_AR; ar++) {
   165→   165→        const bowl = CRICKET_TEAM_SIZE - wk - bat - ar;
   166→   166→        if (bowl < MIN_BOWL || bowl > MAX_BOWL) continue;
   167→   167→        combos.push({ wk, bat, ar, bowl });
   168→   168→      }
   169→   169→    }
   170→   170→  }
   171→   171→  return combos;
   172→   172→}

   177→   177→export function isCombinationCompatibleWithFixed(
   178→   178→  combo: RoleCombination,
   179→   179→  fixedPlayers: TGPlayer[],
   180→   180→): boolean {
   181→   181→  const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   182→   182→  const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   183→   183→  const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   184→   184→  const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   185→   185→
   186→   186→  // Fixed players must not exceed combination totals
   187→   187→  if (fixedWK > combo.wk) return false;
   188→   188→  if (fixedBat > combo.bat) return false;
   189→   189→  if (fixedAR > combo.ar) return false;
   190→   190→  if (fixedBowl > combo.bowl) return false;
   191→   191→
   192→   192→  // Remaining slots must be fillable
   193→   193→  const remainingPlayers = CRICKET_TEAM_SIZE - fixedPlayers.length;
   194→   194→  const neededWK = combo.wk - fixedWK;
   195→   195→  const neededBat = combo.bat - fixedBat;
   196→   196→  const neededAR = combo.ar - fixedAR;
   197→   197→  const neededBowl = combo.bowl - fixedBowl;
   198→   198→
   199→   199→  return (neededWK + neededBat + neededAR + neededBowl) === remainingPlayers;
   200→   200→}

   205→   205→export function getCompatibleCombinations(
   206→   206→  fixedPlayers: TGPlayer[],
   207→   207→  eligiblePlayers: TGPlayer[],
   208→   208→): RoleCombination[] {
   209→   209→  const all = getAllValidCombinations();
   210→   210→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   211→   211→  const remaining = eligiblePlayers.filter(p => !fixedIds.has(p.pl_id));
   212→   212→
   213→   213→  const remWK = remaining.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   214→   214→  const remBat = remaining.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   215→   215→  const remAR = remaining.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   216→   216→  const remBowl = remaining.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   217→   217→
   218→   218→  return all.filter(combo => {
   219→   219→    if (!isCombinationCompatibleWithFixed(combo, fixedPlayers)) return false;
   220→   220→
   221→   221→    const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   222→   222→    const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   223→   223→    const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   224→   224→    const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   225→   225→
   226→   226→    if (remWK < combo.wk - fixedWK) return false;
   227→   227→    if (remBat < combo.bat - fixedBat) return false;
   228→   228→    if (remAR < combo.ar - fixedAR) return false;
   229→   229→    if (remBowl < combo.bowl - fixedBowl) return false;
   230→   230→
   231→   231→    return true;
   232→   232→  });
   233→   233→}

   236→   236→ * Auto-select the best combination based on player pool and category.

   239→   239→export function autoSelectCombination(
   240→   240→  eligiblePlayers: TGPlayer[],
   241→   241→  category: string,
   242→   242→  fixedPlayers: TGPlayer[] = [],
   243→   243→  teamIndex: number = 0,

   276→   276→/**
   277→   277→ * Validate a manual combination against available players.
   278→   278→ */
   279→   279→export function validateCombination(
   280→   280→  combo: RoleCombination,
   281→   281→  eligiblePlayers: TGPlayer[],
   282→   282→  fixedPlayers: TGPlayer[] = [],
   283→   283→): { valid: boolean; errors: string[] } {
   284→   284→  const errors: string[] = [];
   285→   285→
   286→   286→  if (combo.wk + combo.bat + combo.ar + combo.bowl !== CRICKET_TEAM_SIZE) {
   287→   287→    errors.push(`Total players must be 11 (got ${combo.wk + combo.bat + combo.ar + combo.bowl})`);
   288→   288→  }
   289→   289→
   290→   290→  if (combo.wk < MIN_WK || combo.wk > MAX_WK) errors.push(`WK must be ${MIN_WK}-${MAX_WK}`);
   291→   291→  if (combo.bat < MIN_BAT || combo.bat > MAX_BAT) errors.push(`BAT must be ${MIN_BAT}-${MAX_BAT}`);
   292→   292→  if (combo.ar < MIN_AR || combo.ar > MAX_AR) errors.push(`AR must be ${MIN_AR}-${MAX_AR}`);
   293→   293→  if (combo.bowl < MIN_BOWL || combo.bowl > MAX_BOWL) errors.push(`BOWL must be ${MIN_BOWL}-${MAX_BOWL}`);
   294→   294→
   295→   295→  if (fixedPlayers.length > 0 && !isCombinationCompatibleWithFixed(combo, fixedPlayers)) {
   296→   296→    errors.push('Combination is not compatible with selected fixed players');
   297→   297→  }
   298→   298→
   299→   299→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   300→   300→  const remaining = eligiblePlayers.filter(p => !fixedIds.has(p.pl_id));
   301→   301→
   302→   302→  const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   303→   303→  const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   304→   304→  const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   305→   305→  const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   306→   306→
   307→   307→  const needWK = Math.max(0, combo.wk - fixedWK);
   308→   308→  const needBat = Math.max(0, combo.bat - fixedBat);
   309→   309→  const needAR = Math.max(0, combo.ar - fixedAR);
   310→   310→  const needBowl = Math.max(0, combo.bowl - fixedBowl);
   311→   311→
   312→   312→  const availWK = remaining.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   313→   313→  const availBat = remaining.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   314→   314→  const availAR = remaining.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   315→   315→  const availBowl = remaining.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   316→   316→
   317→   317→  if (availWK < needWK) errors.push(`Need ${needWK} WK but only ${availWK} available`);
   318→   318→  if (availBat < needBat) errors.push(`Need ${needBat} BAT but only ${availBat} available`);
   319→   319→  if (availAR < needAR) errors.push(`Need ${needAR} AR but only ${availAR} available`);
   320→   320→  if (availBowl < needBowl) errors.push(`Need ${needBowl} BOWL but only ${availBowl} available`);
   321→   321→
   322→   322→  return { valid: errors.length === 0, errors };
   323→   323→}

   603→   603→  captain: TGPlayer;

   608→   608→// Cricket team constraints
   609→   609→export const CRICKET_TEAM_SIZE = 11;
   610→   610→export const MIN_WK = 1;
   611→   611→export const MAX_WK = 4;
   612→   612→export const MIN_BAT = 3;
   613→   613→export const MAX_BAT = 6;
   614→   614→export const MIN_AR = 1;
   615→   615→export const MAX_AR = 4;
   616→   616→export const MIN_BOWL = 3;
   617→   617→export const MAX_BOWL = 6;
   618→   618→export const MAX_CREDITS = 100;
   619→   619→export const MAX_FROM_ONE_TEAM = 7;

   621→   621→function seededRandom(seed: number): () => number {
   622→   622→  let s = seed;
   623→   623→  return () => {
   624→   624→    s = (s * 16807 + 0) % 2147483647;
   625→   625→    return (s - 1) / 2147483646;
   626→   626→  };
   627→   627→}
   628→   628→
   629→   629→function shuffleArray<T>(arr: T[], rng: () => number): T[] {
   630→   630→  const result = [...arr];
   631→   631→  for (let i = result.length - 1; i > 0; i--) {
   632→   632→    const j = Math.floor(rng() * (i + 1));
   633→   633→    [result[i], result[j]] = [result[j], result[i]];
   634→   634→  }
   635→   635→  return result;
   636→   636→}

   638→   638→export function generateTeams(
   639→   639→  leftPlayers: TGPlayer[],
   640→   640→  rightPlayers: TGPlayer[],
   641→   641→  category: string,
   642→   642→  count: number,
   643→   643→  seed: number = Date.now(),
   644→   644→  avoidPlayerIds: Set<number> = new Set(),
   645→   645→  combination: RoleCombination | null = null,
   646→   646→): GeneratedTeam[] {
   647→   647→  // Apply lineup-aware filtering: AFTER LINEUP only playing=1, BEFORE LINEUP all
   648→   648→  const allPlayersRaw = [...leftPlayers, ...rightPlayers];
   649→   649→  const allPlayers = getEligiblePlayers(allPlayersRaw, avoidPlayerIds);
   650→   650→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
   651→   651→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
   652→   652→
   653→   653→  // After filtering, rebuild left/right pools for team distribution checks
   654→   654→  const eligibleLeft = allPlayers.filter(p => p.team_name === leftTeamName);
   655→   655→  const eligibleRight = allPlayers.filter(p => p.team_name === rightTeamName);
   656→   656→
   657→   657→  // Separate by role
   658→   658→  const wk = allPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER);
   659→   659→  const bat = allPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN);
   660→   660→  const ar = allPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER);
   661→   661→  const bowl = allPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER);
   662→   662→
   663→   663→  const teams: GeneratedTeam[] = [];
   664→   664→  const rng = seededRandom(seed);
   665→   665→
   666→   666→  // Category-specific strategies
   667→   667→  const isH2H = category === 'H2H';
   668→   668→  const isSL = category === 'SL';
   669→   669→
   670→   670→  for (let t = 0; t < count; t++) {
   671→   671→    let attempts = 0;
   672→   672→    let team: TGPlayer[] | null = null;
   673→   673→
   674→   674→    while (!team && attempts < 200) {
   675→   675→      attempts++;
   676→   676→      const sr = seededRandom(seed + t * 1000 + attempts);
   677→   677→
   678→   678→      // Determine role counts based on category or explicit combination
   679→   679→      let wkCount, batCount, arCount, bowlCount;
   680→   680→
   681→   681→      if (combination) {
   682→   682→        // Use explicit combination (manual or auto-selected)
   683→   683→        wkCount = combination.wk;
   684→   684→        batCount = combination.bat;
   685→   685→        arCount = combination.ar;
   686→   686→        bowlCount = combination.bowl;
   687→   687→      } else if (isH2H) {
   688→   688→        // H2H: More conservative, popular players
   689→   689→        wkCount = MIN_WK + (sr() > 0.5 ? 1 : 0);
   690→   690→        arCount = MIN_AR + (sr() > 0.6 ? 1 : 0);
   691→   691→        batCount = Math.max(MIN_BAT, 5 - wkCount + (sr() > 0.5 ? 1 : 0));
   692→   692→        bowlCount = CRICKET_TEAM_SIZE - wkCount - batCount - arCount;
   693→   693→      } else if (isSL) {
   694→   694→        // SL: Balanced
   695→   695→        wkCount = 1 + (sr() > 0.7 ? 1 : 0);
   696→   696→        arCount = 2 + (sr() > 0.5 ? 1 : 0);
   697→   697→        batCount = 3 + (sr() > 0.5 ? 1 : 0);
   698→   698→        bowlCount = CRICKET_TEAM_SIZE - wkCount - batCount - arCount;
   699→   699→      } else {
   700→   700→        // Mega GL: More differential picks, varied combinations
   701→   701→        const patterns = [
   702→   702→          [1, 3, 2, 5], [1, 4, 1, 5], [1, 3, 3, 4], [1, 4, 2, 4],
   703→   703→          [2, 3, 2, 4], [1, 5, 1, 4], [2, 3, 1, 5], [1, 4, 3, 3],
   704→   704→          [2, 4, 1, 4], [1, 3, 1, 6], [2, 2, 3, 4], [1, 5, 2, 3],
   705→   705→        ];
   706→   706→        const pattern = patterns[Math.floor(sr() * patterns.length)];
   707→   707→        [wkCount, batCount, arCount, bowlCount] = pattern;
   708→   708→      }
   709→   709→
   710→   710→      // Validate counts
   711→   711→      if (wkCount < MIN_WK || wkCount > MAX_WK) continue;
   712→   712→      if (batCount < MIN_BAT || batCount > MAX_BAT) continue;
   713→   713→      if (arCount < MIN_AR || arCount > MAX_AR) continue;
   714→   714→      if (bowlCount < MIN_BOWL || bowlCount > MAX_BOWL) continue;
   715→   715→      if (wkCount + batCount + arCount + bowlCount !== CRICKET_TEAM_SIZE) continue;
   716→   716→
   717→   717→      // Pick players for each role
   718→   718→      const shuffledWK = shuffleArray(wk, sr);
   719→   719→      const shuffledBat = shuffleArray(bat, sr);
   720→   720→      const shuffledAR = shuffleArray(ar, sr);
   721→   721→      const shuffledBowl = shuffleArray(bowl, sr);
   722→   722→
   723→   723→      // For Mega GL, sort by selected_by sometimes to get differential picks
   724→   724→      let selectedWK, selectedBat, selectedAR, selectedBowl;
   725→   725→
   726→   726→      if (!isH2H && sr() > 0.4) {
   727→   727→        // Sort some roles by selection % to mix popular and differential
   728→   728→        const sortFn = (a: TGPlayer, b: TGPlayer) => a.selected_by - b.selected_by;
   729→   729→        selectedWK = shuffleArray([...wk].sort(sortFn), sr).slice(0, wkCount);
   730→   730→        selectedBat = shuffleArray([...bat].sort(sortFn), sr).slice(0, batCount);
   731→   731→        selectedAR = shuffleArray([...ar].sort(sortFn), sr).slice(0, arCount);
   732→   732→        selectedBowl = shuffleArray([...bowl].sort(sortFn), sr).slice(0, bowlCount);
   733→   733→      } else if (isH2H) {
   734→   734→        // H2H: Pick most selected players
   735→   735→        const sortFn = (a: TGPlayer, b: TGPlayer) => b.selected_by - a.selected_by;
   736→   736→        selectedWK = [...wk].sort(sortFn).slice(0, wkCount);
   737→   737→        selectedBat = shuffleArray([...bat].sort(sortFn), sr).slice(0, batCount);
   738→   738→        selectedAR = [...ar].sort(sortFn).slice(0, arCount);
   739→   739→        selectedBowl = shuffleArray([...bowl].sort(sortFn), sr).slice(0, bowlCount);
   740→   740→      } else {
   741→   741→        selectedWK = shuffledWK.slice(0, wkCount);
   742→   742→        selectedBat = shuffledBat.slice(0, batCount);
   743→   743→        selectedAR = shuffledAR.slice(0, arCount);
   744→   744→        selectedBowl = shuffledBowl.slice(0, bowlCount);
   745→   745→      }
   746→   746→
   747→   747→      // Check if we have enough players
   748→   748→      if (selectedWK.length < wkCount || selectedBat.length < batCount ||
   749→   749→          selectedAR.length < arCount || selectedBowl.length < bowlCount) continue;
   750→   750→
   751→   751→      const selected = [...selectedWK, ...selectedBat, ...selectedAR, ...selectedBowl];
   752→   752→
   753→   753→      // Check total credits
   754→   754→      const totalCredits = selected.reduce((sum, p) => sum + p.credits, 0);
   755→   755→      if (totalCredits > MAX_CREDITS) continue;
   756→   756→
   757→   757→      // Check team distribution (max 7 from one team)
   758→   758→      const leftCount = selected.filter(p => p.team_name === leftTeamName).length;
   759→   759→      const rightCount = selected.filter(p => p.team_name === rightTeamName).length;
   760→   760→      if (leftCount > MAX_FROM_ONE_TEAM || rightCount > MAX_FROM_ONE_TEAM) continue;
   761→   761→
   762→   762→      // Ensure at least 1 from each team
   763→   763→      if (leftCount < 1 || rightCount < 1) continue;
   764→   764→
   765→   765→      team = selected;
   766→   766→    }
   767→   767→
   768→   768→    if (team) {
   769→   769→      // Select Captain & Vice-Captain
   770→   770→      // Sort by a combination of points and selection % for C/VC
   771→   771→      const sorted = [...team].sort((a, b) =>
   772→   772→        (b.points * 0.4 + b.selected_by * 0.3 + b.captain_percentage * 0.3) -
   773→   773→        (a.points * 0.4 + a.selected_by * 0.3 + a.captain_percentage * 0.3)
   774→   774→      );
   775→   775→
   776→   776→      const captain = sorted[0];
   777→   777→      const viceCaptain = sorted.length > 1 ? sorted[1] : sorted[0];
   778→   778→
   779→   779→      teams.push({
   780→   780→        id: t + 1,
   781→   781→        captain,
   782→   782→        viceCaptain,
   783→   783→        players: team,
   784→   784→      });
   785→   785→    }
   786→   786→  }
   787→   787→
   788→   788→  return teams;
   789→   789→}

   827→   827→

   829→   829→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   830→   830→  const remainingPool = allPlayers.filter(p => !fixedIds.has(p.pl_id));
   831→   831→

   835→   835→  const remAR = remainingPool.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER);
   836→   836→  const remBowl = remainingPool.filter(p => p.role === PLAYER_ROLES.BOWLER);
   837→   837→
   838→   838→  // Count fixed players by role
   839→   839→  const fixedWKCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   840→   840→  const fixedBatCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   841→   841→  const fixedARCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   842→   842→  const fixedBowlCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   843→   843→

   845→   845→  const remainingCredits = MAX_CREDITS - fixedCredits;
   846→   846→  const neededPlayers = CRICKET_TEAM_SIZE - fixedPlayers.length; // 3
   847→   847→
   848→   848→  const teams: GeneratedTeam[] = [];
   849→   849→
   850→   850→  // Pre-validate: fixed players must have credits <= 100
   851→   851→  if (fixedCredits > MAX_CREDITS) return teams;
   852→   852→
   853→   853→  // Pre-validate: fixed players must respect max 7 from one team
   854→   854→  const fixedLeftCount = fixedPlayers.filter(p => p.team_name === leftTeamName).length;
   855→   855→  const fixedRightCount = fixedPlayers.filter(p => p.team_name === rightTeamName).length;
   856→   856→  if (fixedLeftCount > MAX_FROM_ONE_TEAM || fixedRightCount > MAX_FROM_ONE_TEAM) return teams;
   857→   857→
   858→   858→  // Generate all valid C/VC combinations from the 5C and 5VC options
   859→   859→  // C must be in the team, VC must be in the team, C ≠ VC
   860→   860→  const cVCCombos: { c: TGPlayer; vc: TGPlayer }[] = [];

   862→   862→    for (const vc of viceCaptainOptions) {
   863→   863→      if (c.pl_id !== vc.pl_id) {
   864→   864→        cVCCombos.push({ c, vc });
   865→   865→      }
   866→   866→    }
   867→   867→  }
   868→   868→  // If no valid combos, return empty
   869→   869→  if (cVCCombos.length === 0) return teams;
   870→   870→
   871→   871→  // Category-specific role distribution preferences for remaining players
   872→   872→  const isH2H = category === 'H2H';
   873→   873→  const isSL = category === 'SL';
   874→   874→
   875→   875→  // Pre-compute all valid target role distributions that are compatible with fixed players
   876→   876→  // This avoids wasting attempts on patterns that can never work
   877→   877→  const categoryPatterns: number[][] = isH2H
   878→   878→    ? [] // H2H uses dynamic generation below
   879→   879→    : isSL
   880→   880→    ? [] // SL uses dynamic generation below

  1159→  1159→// ============ Auto Select for Extra Team Generation ============
  1160→  1160→// Uses real player data (selection %, captain %, points, credits, role, team balance)
  1161→  1161→// to intelligently pick 8 FIX, 5 C, and 5 VC

  1169→  1169→// Player scoring for auto-selection: higher = better pick
  1170→  1170→function playerScore(p: TGPlayer): number {
  1171→  1171→  return (
  1172→  1172→    p.selected_by * 0.35 +       // selection % - most important for GL
  1173→  1173→    p.captain_percentage * 0.25 + // captain % - indicates trust
  1174→  1174→    p.points * 0.2 +             // points - performance indicator
  1175→  1175→    p.vice_captain_percentage * 0.1 + // VC %
  1176→  1176→    (p.playing === 1 ? 10 : 0) + // confirmed playing bonus
  1177→  1177→    (p.role === PLAYER_ROLES.ALL_ROUNDER ? 3 : 0) + // AR bonus (flexible)
  1178→  1178→    (p.role === PLAYER_ROLES.WICKET_KEEPER ? 1 : 0) // WK slight bonus
  1179→  1179→  );
  1180→  1180→}
  1181→  1181→
  1182→  1182→export function autoSelectExtraPlayers(
  1183→  1183→  leftPlayers: TGPlayer[],

  1198→  1198→  const sortByScore = (a: TGPlayer, b: TGPlayer) => playerScore(b) - playerScore(a);
  1199→  1199→  const sortedLeft = [...eligibleLeft].sort(sortByScore);
  1200→  1200→  const sortedRight = [...eligibleRight].sort(sortByScore);
  1201→  1201→
  1202→  1202→  // === SELECT 8 FIX PLAYERS ===

  1204→  1204→  const fixedPlayers: TGPlayer[] = [];
  1205→  1205→  const fixedIds = new Set<number>();

  1208→  1208→  const targetRight = Math.min(4, sortedRight.length);
  1209→  1209→
  1210→  1210→  // Pick diverse players ensuring role coverage
  1211→  1211→  const pickDiverse = (pool: TGPlayer[], count: number, existing: TGPlayer[]): TGPlayer[] => {
  1212→  1212→    const picked: TGPlayer[] = [];
  1213→  1213→    const usedIds = new Set(existing.map(p => p.pl_id));

  1216→  1216→      bat: existing.filter(p => p.role === PLAYER_ROLES.BATSMAN).length,

  1220→  1220→

  1222→  1222→    const roleNeeds = [
  1223→  1223→      { role: PLAYER_ROLES.WICKET_KEEPER, need: Math.max(0, 1 - existingRoles.wk), pool: pool.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER && !usedIds.has(p.pl_id)) },
  1224→  1224→      { role: PLAYER_ROLES.ALL_ROUNDER, need: Math.max(0, 1 - existingRoles.ar), pool: pool.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER && !usedIds.has(p.pl_id)) },

  1226→  1226→      { role: PLAYER_ROLES.BOWLER, need: Math.max(0, 2 - existingRoles.bowl), pool: pool.filter(p => p.role === PLAYER_ROLES.BOWLER && !usedIds.has(p.pl_id)) },
  1227→  1227→    ];
  1228→  1228→
  1229→  1229→    for (const rn of roleNeeds) {
  1230→  1230→      for (let i = 0; i < rn.need && picked.length < count; i++) {
  1231→  1231→        const candidate = rn.pool.find(p => !picked.some(pp => pp.pl_id === p.pl_id));
  1232→  1232→        if (candidate) picked.push(candidate);
  1233→  1233→      }
  1234→  1234→    }
  1235→  1235→
  1236→  1236→    // Second pass: fill remaining with top-scored players
  1237→  1237→    const remaining = pool.filter(p => !usedIds.has(p.pl_id) && !picked.some(pp => pp.pl_id === p.pl_id));
  1238→  1238→    for (const p of remaining) {
  1239→  1239→      if (picked.length >= count) break;
  1240→  1240→      picked.push(p);
  1241→  1241→    }
  1242→  1242→
  1243→  1243→    return picked;
  1244→  1244→  };
  1245→  1245→
  1246→  1246→  const leftPicks = pickDiverse(sortedLeft, targetLeft, []);
  1247→  1247→  for (const p of leftPicks) { fixedPlayers.push(p); fixedIds.add(p.pl_id); }
  1248→  1248→
  1249→  1249→  const rightPicks = pickDiverse(sortedRight, targetRight, fixedPlayers);
  1250→  1250→  for (const p of rightPicks) { fixedPlayers.push(p); fixedIds.add(p.pl_id); }
  1251→  1251→
  1252→  1252→  // Fill to 8 if needed
  1253→  1253→  if (fixedPlayers.length < 8) {
  1254→  1254→    const remaining = eligible.filter(p => !fixedIds.has(p.pl_id)).sort(sortByScore);
  1255→  1255→    for (const p of remaining) {
  1256→  1256→      if (fixedPlayers.length >= 8) break;
  1257→  1257→      fixedPlayers.push(p); fixedIds.add(p.pl_id);
  1258→  1258→    }
  1259→  1259→  }
  1260→  1260→
  1261→  1261→  // Trim to 8 if over
  1262→  1262→  while (fixedPlayers.length > 8) {
  1263→  1263→    let minIdx = 0; let minScore = playerScore(fixedPlayers[0]);
  1264→  1264→    for (let i = 1; i < fixedPlayers.length; i++) {
  1265→  1265→      const s = playerScore(fixedPlayers[i]);
  1266→  1266→      if (s < minScore) { minScore = s; minIdx = i; }
  1267→  1267→    }
  1268→  1268→    const removed = fixedPlayers.splice(minIdx, 1)[0];
  1269→  1269→    fixedIds.delete(removed.pl_id);
  1270→  1270→  }
  1271→  1271→

  1274→  1274→    .filter(p => !avoidPlayerIds.has(p.pl_id))
  1275→  1275→    .sort((a, b) => {
  1276→  1276→      const scoreA = a.captain_percentage * 0.4 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
  1277→  1277→      const scoreB = b.captain_percentage * 0.4 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
  1278→  1278→      return scoreB - scoreA;
  1279→  1279→    })
  1280→  1280→    .slice(0, 5);
  1281→  1281→

  1283→  1283→  const vcCandidates: TGPlayer[] = [];
  1284→  1284→  const vcPool = eligible
  1285→  1285→    .filter(p => !avoidPlayerIds.has(p.pl_id))
  1286→  1286→    .sort((a, b) => {

  1288→  1288→      const scoreB = b.vice_captain_percentage * 0.35 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
  1289→  1289→      return scoreB - scoreA;
  1290→  1290→    });
  1291→  1291→
  1292→  1292→  for (const p of vcPool) {
  1293→  1293→    if (vcCandidates.length >= 5) break;

  1295→  1295→  }
  1296→  1296→
  1297→  1297→  return {
  1298→  1298→    fixedPlayers: fixedPlayers.slice(0, 8),
  1299→  1299→    captainOptions: cCandidates.slice(0, 5),
  1300→  1300→    viceCaptainOptions: vcCandidates.slice(0, 5),
  1301→  1301→  };
  1302→  1302→}

  1304→  1304→// Auto-replace a removed player in a list with the next best eligible player
  1305→  1305→export function autoReplacePlayer(
  1306→  1306→  removedPlayer: TGPlayer,
  1307→  1307→  currentList: TGPlayer[],
  1308→  1308→  allPlayers: TGPlayer[],
  1309→  1309→  avoidPlayerIds: Set<number>,
  1310→  1310→  scoringFn: (p: TGPlayer) => number = playerScore,
  1311→  1311→): TGPlayer[] {
  1312→  1312→  const currentIds = new Set(currentList.map(p => p.pl_id));
  1313→  1313→  const eligible = allPlayers.filter(p =>
  1314→  1314→    !currentIds.has(p.pl_id) && !avoidPlayerIds.has(p.pl_id) && p.pl_id !== removedPlayer.pl_id
  1315→  1315→  );
  1316→  1316→  const sorted = eligible.sort((a, b) => scoringFn(b) - scoringFn(a));
  1317→  1317→  const replacement = sorted[0];
  1318→  1318→  if (replacement) return [...currentList, replacement];
  1319→  1319→  return currentList;
  1320→  1320→}
