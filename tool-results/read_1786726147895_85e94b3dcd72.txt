     6→// ============ Platform Name Normalization ============
     7→
     8→/**
     9→ * Normalize platform names to consistent lowercase identifiers.
    10→ * The TG API may return varied casing in fantasy_id_list[].name
    11→ * (e.g., "Dream11", "dream11", "My11Circle", "my11circle").
    12→ * All internal references use lowercase: "dream11", "my11circle".
    13→ */
    14→export function normalizePlatformName(name: string): string {
    15→  return name.toLowerCase().replace(/[\s_-]/g, '');
    16→}

    18→/**
    19→ * Resolve a player's platform-specific ID from their fantasy_id_list.
    20→ * Uses case-insensitive matching to handle API casing variations.
    21→ */
    22→export function resolvePlatformPlayerId(
    23→  player: TGPlayer,
    24→  platform: string,
    25→): number | null {
    26→  if (!player.fantasy_id_list || player.fantasy_id_list.length === 0) return null;
    27→  const target = normalizePlatformName(platform);
    28→  const found = player.fantasy_id_list.find(
    29→    f => normalizePlatformName(f.name) === target
    30→  );
    31→  return found ? found.id : null;
    32→}

    62→// ============ Lineup Utilities ============
    63→
    64→/**
    65→ * Determine lineup mode based on player data.
    66→ * AFTER LINEUP: enough players have playing === 1 to form valid teams.
    67→ *   Requires at least 11 confirmed total AND at least 4 from each team
    68→ *   (due to max-7-from-one-team constraint: 11 - 7 = 4).
    69→ *   If only a few players are confirmed (partial lineup), we stay in
    70→ *   'before' mode so team generation doesn't fail with too few eligible players.
    71→ * BEFORE LINEUP: lineup not fully confirmed yet — all players eligible.
    72→ *
    73→ * BUG FIX: Previously used players.some(p => p.playing === 1) which
    74→ * switched to 'after' mode when even ONE player was confirmed, causing
    75→ * getEligiblePlayers to filter out all non-confirmed players and leaving
    76→ * too few to form valid 11-player teams → 0 teams generated.
    77→ */
    78→export function getLineupMode(players: TGPlayer[]): 'before' | 'after' {
    79→  const confirmed = players.filter(p => p.playing === 1);
    80→
    81→  // Need at least 11 confirmed players total to form a valid team
    82→  if (confirmed.length < 11) return 'before';
    83→
    84→  // Need at least 4 confirmed from each team
    85→  // (max 7 from one team → need at least 11-7=4 from the other)
    86→  const teamNames = [...new Set(players.map(p => p.team_name))];
    87→  for (const team of teamNames) {
    88→    if (team && confirmed.filter(p => p.team_name === team).length < 4) {
    89→      return 'before';
    90→    }
    91→  }
    92→
    93→  return 'after';
    94→}
    95→
    96→/**
    97→ * Filter players based on lineup mode.
    98→ * BEFORE LINEUP: all players are eligible EXCEPT confirmed OUT (playing === -1).
    99→ * AFTER LINEUP: ONLY players with playing === 1 are eligible (Playing XI only, no bench).
   100→ */
   101→export function getEligiblePlayers(
   102→  allPlayers: TGPlayer[],
   103→  avoidPlayerIds: Set<number> = new Set(),
   104→): TGPlayer[] {
   105→  const mode = getLineupMode(allPlayers);
   106→
   107→  return allPlayers.filter(p => {
   108→    // Avoid players are always excluded
   109→    if (avoidPlayerIds.has(p.pl_id)) return false;
   110→
   111→    // Confirmed OUT players are always excluded
   112→    if (p.playing === -1) return false;
   113→
   114→    if (mode === 'after') {
   115→      // AFTER LINEUP: ONLY confirmed Playing XI players (playing === 1)
   116→      // Bench/probable players (playing === 0) are NOT included
   117→      return p.playing === 1;
   118→    }
   119→
   120→    // BEFORE LINEUP: all non-OUT players eligible (playing === 0 or 1)
   121→    return true;
   122→  });
   123→}

   125→/**
   126→ * Check if a specific player is eligible based on lineup mode.
   127→ */
   128→export function isPlayerEligible(
   129→  player: TGPlayer,
   130→  allPlayers: TGPlayer[],
   131→  avoidPlayerIds: Set<number> = new Set(),
   132→): { eligible: boolean; reason?: string } {
   133→  if (avoidPlayerIds.has(player.pl_id)) {
   134→    return { eligible: false, reason: 'AVOID' };
   135→  }
   136→
   137→  const mode = getLineupMode(allPlayers);
   138→
   139→  if (mode === 'after' && player.playing !== 1) {
   140→    return { eligible: false, reason: 'OUT / NOT IN PLAYING XI' };
   141→  }
   142→
   143→  return { eligible: true };
   144→}

   146→/**
   147→ * Validate a team for lineup eligibility.
   148→ * Returns list of invalid players with reasons.
   149→ */
   150→export function validateTeamForLineup(
   151→  team: GeneratedTeam,
   152→  allPlayers: TGPlayer[],
   153→  avoidPlayerIds: Set<number> = new Set(),
   154→): { valid: boolean; invalidPlayers: { player: TGPlayer; reason: string }[] } {
   155→  const mode = getLineupMode(allPlayers);
   156→  const invalidPlayers: { player: TGPlayer; reason: string }[] = [];
   157→
   158→  for (const player of team.players) {
   159→    if (avoidPlayerIds.has(player.pl_id)) {
   160→      invalidPlayers.push({ player, reason: 'AVOID' });
   161→      continue;
   162→    }
   163→
   164→    if (mode === 'after' && player.playing !== 1) {
   165→      invalidPlayers.push({ player, reason: 'OUT / NOT IN PLAYING XI' });
   166→    }
   167→  }
   168→
   169→  return { valid: invalidPlayers.length === 0, invalidPlayers };
   170→}

   183→/**
   184→ * Get all valid role combinations for an 11-player cricket team.
   185→ */
   186→export function getAllValidCombinations(): RoleCombination[] {
   187→  const combos: RoleCombination[] = [];
   188→  for (let wk = MIN_WK; wk <= MAX_WK; wk++) {
   189→    for (let bat = MIN_BAT; bat <= MAX_BAT; bat++) {
   190→      for (let ar = MIN_AR; ar <= MAX_AR; ar++) {
   191→        const bowl = CRICKET_TEAM_SIZE - wk - bat - ar;
   192→        if (bowl < MIN_BOWL || bowl > MAX_BOWL) continue;
   193→        combos.push({ wk, bat, ar, bowl });
   194→      }
   195→    }
   196→  }
   197→  return combos;
   198→}

   200→/**
   201→ * Check if a role combination is compatible with a set of fixed players.
   202→ */
   203→export function isCombinationCompatibleWithFixed(
   204→  combo: RoleCombination,
   205→  fixedPlayers: TGPlayer[],
   206→): boolean {
   207→  const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   208→  const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   209→  const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   210→  const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   211→
   212→  // Fixed players must not exceed combination totals
   213→  if (fixedWK > combo.wk) return false;
   214→  if (fixedBat > combo.bat) return false;
   215→  if (fixedAR > combo.ar) return false;
   216→  if (fixedBowl > combo.bowl) return false;
   217→
   218→  // Remaining slots must be fillable
   219→  const remainingPlayers = CRICKET_TEAM_SIZE - fixedPlayers.length;
   220→  const neededWK = combo.wk - fixedWK;
   221→  const neededBat = combo.bat - fixedBat;
   222→  const neededAR = combo.ar - fixedAR;
   223→  const neededBowl = combo.bowl - fixedBowl;
   224→
   225→  return (neededWK + neededBat + neededAR + neededBowl) === remainingPlayers;
   226→}

   228→/**
   229→ * Get combinations compatible with fixed players and available player pool.
   230→ */
   231→export function getCompatibleCombinations(
   232→  fixedPlayers: TGPlayer[],
   233→  eligiblePlayers: TGPlayer[],
   234→): RoleCombination[] {
   235→  const all = getAllValidCombinations();
   236→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   237→  const remaining = eligiblePlayers.filter(p => !fixedIds.has(p.pl_id));
   238→
   239→  const remWK = remaining.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   240→  const remBat = remaining.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   241→  const remAR = remaining.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   242→  const remBowl = remaining.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   243→
   244→  return all.filter(combo => {
   245→    if (!isCombinationCompatibleWithFixed(combo, fixedPlayers)) return false;
   246→
   247→    const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   248→    const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   249→    const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   250→    const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   251→
   252→    if (remWK < combo.wk - fixedWK) return false;
   253→    if (remBat < combo.bat - fixedBat) return false;
   254→    if (remAR < combo.ar - fixedAR) return false;
   255→    if (remBowl < combo.bowl - fixedBowl) return false;
   256→
   257→    return true;
   258→  });
   259→}

   261→/**
   262→ * Auto-select the best combination based on player pool and category.
   263→ * For multiple teams, rotates combinations intelligently.
   264→ */
   265→export function autoSelectCombination(
   266→  eligiblePlayers: TGPlayer[],
   267→  category: string,
   268→  fixedPlayers: TGPlayer[] = [],
   269→  teamIndex: number = 0,
   270→): RoleCombination {
   271→  const compatible = getCompatibleCombinations(fixedPlayers, eligiblePlayers);
   272→
   273→  if (compatible.length === 0) {
   274→    return { wk: 1, bat: 4, ar: 2, bowl: 4 };
   275→  }
   276→
   277→  // Category-preferred patterns
   278→  const preferredPatterns: RoleCombination[] =
   279→    category === 'H2H'
   280→      ? [{ wk: 1, bat: 5, ar: 2, bowl: 3 }, { wk: 1, bat: 4, ar: 2, bowl: 4 }, { wk: 1, bat: 3, ar: 3, bowl: 4 }]
   281→      : category === 'SL'
   282→      ? [{ wk: 1, bat: 4, ar: 2, bowl: 4 }, { wk: 1, bat: 3, ar: 3, bowl: 4 }, { wk: 2, bat: 3, ar: 2, bowl: 4 }]
   283→      : [ // Mega GL
   284→        { wk: 1, bat: 3, ar: 2, bowl: 5 }, { wk: 1, bat: 4, ar: 1, bowl: 5 },
   285→        { wk: 1, bat: 3, ar: 3, bowl: 4 }, { wk: 1, bat: 4, ar: 2, bowl: 4 },
   286→        { wk: 2, bat: 3, ar: 2, bowl: 4 }, { wk: 1, bat: 5, ar: 1, bowl: 4 },
   287→        { wk: 2, bat: 3, ar: 1, bowl: 5 }, { wk: 1, bat: 4, ar: 3, bowl: 3 },
   288→        { wk: 2, bat: 4, ar: 1, bowl: 4 }, { wk: 1, bat: 3, ar: 1, bowl: 6 },
   289→        { wk: 2, bat: 2, ar: 3, bowl: 4 }, { wk: 1, bat: 5, ar: 2, bowl: 3 },
   290→      ];
   291→
   292→  const preferredCompatible = preferredPatterns.filter(p =>
   293→    compatible.some(c => c.wk === p.wk && c.bat === p.bat && c.ar === p.ar && c.bowl === p.bowl)
   294→  );
   295→
   296→  const pool = preferredCompatible.length > 0 ? preferredCompatible : compatible;
   297→  const index = teamIndex % pool.length;
   298→
   299→  return pool[index];
   300→}

   302→/**
   303→ * Validate a manual combination against available players.
   304→ */
   305→export function validateCombination(
   306→  combo: RoleCombination,
   307→  eligiblePlayers: TGPlayer[],
   308→  fixedPlayers: TGPlayer[] = [],
   309→): { valid: boolean; errors: string[] } {
   310→  const errors: string[] = [];
   311→
   312→  if (combo.wk + combo.bat + combo.ar + combo.bowl !== CRICKET_TEAM_SIZE) {
   313→    errors.push(`Total players must be 11 (got ${combo.wk + combo.bat + combo.ar + combo.bowl})`);
   314→  }
   315→
   316→  if (combo.wk < MIN_WK || combo.wk > MAX_WK) errors.push(`WK must be ${MIN_WK}-${MAX_WK}`);
   317→  if (combo.bat < MIN_BAT || combo.bat > MAX_BAT) errors.push(`BAT must be ${MIN_BAT}-${MAX_BAT}`);
   318→  if (combo.ar < MIN_AR || combo.ar > MAX_AR) errors.push(`AR must be ${MIN_AR}-${MAX_AR}`);
   319→  if (combo.bowl < MIN_BOWL || combo.bowl > MAX_BOWL) errors.push(`BOWL must be ${MIN_BOWL}-${MAX_BOWL}`);
   320→
   321→  if (fixedPlayers.length > 0 && !isCombinationCompatibleWithFixed(combo, fixedPlayers)) {
   322→    errors.push('Combination is not compatible with selected fixed players');
   323→  }
   324→
   325→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   326→  const remaining = eligiblePlayers.filter(p => !fixedIds.has(p.pl_id));
   327→
   328→  const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   329→  const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   330→  const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   331→  const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   332→
   333→  const needWK = Math.max(0, combo.wk - fixedWK);
   334→  const needBat = Math.max(0, combo.bat - fixedBat);
   335→  const needAR = Math.max(0, combo.ar - fixedAR);
   336→  const needBowl = Math.max(0, combo.bowl - fixedBowl);
   337→
   338→  const availWK = remaining.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   339→  const availBat = remaining.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   340→  const availAR = remaining.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   341→  const availBowl = remaining.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   342→
   343→  if (availWK < needWK) errors.push(`Need ${needWK} WK but only ${availWK} available`);
   344→  if (availBat < needBat) errors.push(`Need ${needBat} BAT but only ${availBat} available`);
   345→  if (availAR < needAR) errors.push(`Need ${needAR} AR but only ${availAR} available`);
   346→  if (availBowl < needBowl) errors.push(`Need ${needBowl} BOWL but only ${availBowl} available`);
   347→
   348→  return { valid: errors.length === 0, errors };
   349→}

   429→function decryptField(encrypted: string): string {
   430→  try {
   431→    const bytes = CryptoJS.AES.decrypt(encrypted, ENCRYPTION_KEY);
   432→    const decrypted = bytes.toString(CryptoJS.enc.Utf8);
   433→    if (!decrypted) return encrypted;
   434→    return decrypted;
   435→  } catch {
   436→    return encrypted;
   437→  }
   438→}

   512→export async function fetchMatches(sport: string = 'cricket'): Promise<TGMatch[]> {
   513→  try {
   514→    const response = await fetch(`${TG_API_BASE}/fantasy/matches/${sport}`, {
   515→      next: { revalidate: 60 },
   516→    });
   517→
   518→    if (!response.ok) {
   519→      throw new Error(`Failed to fetch matches: ${response.status}`);
   520→    }
   521→
   522→    const data = await response.json();
   523→
   524→    if (data.status !== 'success' || !Array.isArray(data.data)) {
   525→      return [];
   526→    }
   527→
   528→    const matches: TGMatch[] = [];
   529→    for (const encryptedMatch of data.data) {
   530→      const match = decryptMatchData(encryptedMatch);
   531→      if (match) {
   532→        matches.push(match);
   533→      }
   534→    }
   535→
   536→    return matches;
   537→  } catch (error) {
   538→    console.error('Error fetching matches:', error);
   539→    return [];
   540→  }

   659→/**
   660→ * Create a unique signature for a team based on sorted player IDs + C/VC IDs.
   661→ * Used for deduplication — two teams with identical signatures are duplicates.
   662→ */
   663→export function makeTeamSignature(players: TGPlayer[], captain: TGPlayer, viceCaptain: TGPlayer): string {
   664→  const playerIds = players.map(p => p.pl_id).sort((a, b) => a - b).join(',');
   665→  return `${playerIds}|C${captain.pl_id}|VC${viceCaptain.pl_id}`;

   668→/**
   669→ * Deduplicate and validate a list of generated teams.
   670→ * Removes exact duplicate player combinations.
   671→ * Separates valid and invalid teams based on lineup eligibility.
   672→ *
   673→ * Returns: { valid, invalid, duplicateCount, invalidLineupCount }
   674→ */
   675→export function deduplicateAndValidateTeams(
   676→  teams: GeneratedTeam[],
   677→  allPlayers: TGPlayer[],
   678→  avoidPlayerIds: Set<number> = new Set(),
   679→): {
   680→  valid: GeneratedTeam[];
   681→  invalid: GeneratedTeam[];
   682→  duplicateCount: number;
   683→  invalidLineupCount: number;
   684→} {
   685→  const seen = new Set<string>();
   686→  const valid: GeneratedTeam[] = [];
   687→  const invalid: GeneratedTeam[] = [];
   688→  let duplicateCount = 0;
   689→  let invalidLineupCount = 0;
   690→
   691→  for (let i = 0; i < teams.length; i++) {
   692→    const team = teams[i];
   693→    const sig = makeTeamSignature(team.players, team.captain, team.viceCaptain);
   694→
   695→    // Skip duplicates
   696→    if (seen.has(sig)) {
   697→      duplicateCount++;
   698→      continue;
   699→    }
   700→    seen.add(sig);
   701→
   702→    // Check lineup validity
   703→    const validation = validateTeamForLineup(team, allPlayers, avoidPlayerIds);
   704→    if (!validation.valid) {
   705→      invalidLineupCount++;
   706→      invalid.push({ ...team, id: invalid.length + 1 });
   707→    } else {
   708→      valid.push({ ...team, id: valid.length + 1 });
   709→    }
   710→  }
   711→
   712→  return { valid, invalid, duplicateCount, invalidLineupCount };
   713→}

   728→function seededRandom(seed: number): () => number {
   729→  let s = seed;
   730→  return () => {
   731→    s = (s * 16807 + 0) % 2147483647;
   732→    return (s - 1) / 2147483646;
   733→  };
   734→}

   736→function shuffleArray<T>(arr: T[], rng: () => number): T[] {
   737→  const result = [...arr];
   738→  for (let i = result.length - 1; i > 0; i--) {
   739→    const j = Math.floor(rng() * (i + 1));
   740→    [result[i], result[j]] = [result[j], result[i]];
   741→  }
   742→  return result;
   743→}

   745→// ============ PRO PLAYER SELECTION ENGINE ============
   746→// Each team tries to rank in the top 1-100.
   747→// Strategy: Must-have players + smart scoring + limited differential.
   748→
   749→/**
   750→ * Score a player for team selection (higher = better pick).
   751→ * Used for the MAIN generation engine (not just autoSelect).
   752→ * This is the PRO scoring that replaces the broken sort+shuffle approach.
   753→ */
   754→function playerScoreForGen(p: TGPlayer, category: string): number {
   755→  if (category === 'H2H') {
   756→    // H2H: Safest possible — selection % is king
   757→    return (
   758→      p.selected_by * 0.50 +
   759→      p.captain_percentage * 0.25 +
   760→      p.points * 0.15 +
   761→      p.vice_captain_percentage * 0.05 +
   762→      (p.playing === 1 ? 12 : 0) +
   763→      (p.role === PLAYER_ROLES.ALL_ROUNDER ? 3 : 0)
   764→    );
   765→  } else if (category === 'SL') {
   766→    // SL: Balanced — form + selection + captain history
   767→    return (
   768→      p.selected_by * 0.35 +
   769→      p.captain_percentage * 0.25 +
   770→      p.points * 0.25 +
   771→      p.vice_captain_percentage * 0.10 +
   772→      (p.playing === 1 ? 12 : 0) +
   773→      (p.role === PLAYER_ROLES.ALL_ROUNDER ? 3 : 0)
   774→    );
   775→  } else {
   776→    // Mega GL: Still prioritize core quality, differential comes from C/VC rotation
   777→    // and 2-3 "punt" players — NOT from making the whole team weak
   778→    return (
   779→      p.selected_by * 0.30 +
   780→      p.captain_percentage * 0.20 +
   781→      p.points * 0.25 +
   782→      p.vice_captain_percentage * 0.15 +
   783→      (p.playing === 1 ? 12 : 0) +
   784→      (p.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0) +
   785→      (p.role === PLAYER_ROLES.WICKET_KEEPER ? 1 : 0)
   786→    );
   787→  }
   788→}

   790→/**
   791→ * Smart select: Picks top players by score, with controlled randomness.
   792→ * - Keeps top `keepTop` players fixed (highest scored).
   793→ * - Randomly selects from the remaining pool for variety.
   794→ * - This FIXES the critical bug where shuffleArray after sort destroyed all ordering.
   795→ */
   796→function smartSelect(
   797→  pool: TGPlayer[],
   798→  count: number,
   799→  scoreFn: (p: TGPlayer) => number,
   800→  rng: () => number,
   801→  keepTopRatio: number = 0.7, // Keep top 70% of picks as best-scored
   802→): TGPlayer[] {
   803→  if (pool.length <= count) return [...pool];
   804→
   805→  const sorted = [...pool].sort((a, b) => scoreFn(b) - scoreFn(a));
   806→  const keepTop = Math.max(1, Math.floor(count * keepTopRatio));
   807→  const randomSlots = count - keepTop;
   808→
   809→  // Top players — always include these (they're the best picks)
   810→  const top = sorted.slice(0, keepTop);
   811→  const topIds = new Set(top.map(p => p.pl_id));
   812→
   813→  // Remaining pool (excluding already picked top players)
   814→  const remaining = sorted.filter(p => !topIds.has(p.pl_id));
   815→
   816→  // Random selection from remaining for variety/diversification
   817→  const shuffled = shuffleArray(remaining, rng);
   818→  const randomPicks = shuffled.slice(0, randomSlots);
   819→
   820→  return [...top, ...randomPicks];
   821→}

   823→/**
   824→ * Identify must-have players — players selected by >70% of users.
   825→ * These should be in EVERY team. Missing them = guaranteed bad rank.
   826→ */
   827→function getMustHavePlayers(
   828→  pool: TGPlayer[],
   829→  threshold: number = 70,
   830→): TGPlayer[] {
   831→  return pool.filter(p => p.selected_by >= threshold && p.playing === 1);

   834→/**
   835→ * Score an entire team's quality (higher = better team).
   836→ * Used to reject low-quality teams and retry.
   837→ */
   838→function scoreTeamQuality(team: TGPlayer[], category: string): number {
   839→  let score = 0;
   840→  const scoreFn = (p: TGPlayer) => playerScoreForGen(p, category);
   841→
   842→  for (const p of team) {
   843→    score += scoreFn(p);
   844→    // Bonus for must-have players (selected_by > 70%)
   845→    if (p.selected_by >= 70) score += 8;
   846→    // Bonus for confirmed playing
   847→    if (p.playing === 1) score += 3;
   848→    // Penalty for very low selection (< 10%) unless Mega GL
   849→    if (p.selected_by < 10 && category !== 'MEGA GL') score -= 5;
   850→  }
   851→
   852→  // Bonus for role diversity
   853→  const roles = new Set(team.map(p => p.role));
   854→  score += roles.size * 3;
   855→
   856→  // Bonus for team balance (close to 6-5 or 5-6 split)
   857→  const teamNames = [...new Set(team.map(p => p.team_name))];
   858→  if (teamNames.length === 2) {
   859→    const c1 = team.filter(p => p.team_name === teamNames[0]).length;
   860→    const c2 = team.filter(p => p.team_name === teamNames[1]).length;
   861→    const balance = Math.min(c1, c2); // Closer to 5.5 is better
   862→    score += balance * 2;
   863→  }
   864→
   865→  return score;
   866→}

   868→/**
   869→ * Category-specific C/VC scoring functions.
   870→ * Each category has a genuinely different strategy for selecting Captain and Vice-Captain.
   871→ *
   872→ * Mega GL: Prioritize differential captains (low selection % but high points).
   873→ *          Wider C/VC rotation across teams — different C/VC per team.
   874→ * SL:      Prioritize consistent/form players. Higher weight on selection % and captain %.
   875→ *          More stable C/VC — top 2 players usually.
   876→ * H2H:     Prioritize most-selected, safest players. Highest weight on selection %.
   877→ *          Most conservative C/VC — always the safest picks.
   878→ */
   879→function scoreCaptain(player: TGPlayer, category: string): number {
   880→  if (category === 'H2H') {
   881→    // H2H: Safest picks — highest selection % and captain % weighted most
   882→    return player.selected_by * 0.45 + player.captain_percentage * 0.35 + player.points * 0.2;
   883→  } else if (category === 'SL') {
   884→    // SL: Balanced — form players with good captain record
   885→    return player.captain_percentage * 0.35 + player.selected_by * 0.35 + player.points * 0.3;
   886→  } else {
   887→    // Mega GL: Differential — rotate captains, some differential (lower selection but higher upside)
   888→    return player.points * 0.4 + player.captain_percentage * 0.3 + player.selected_by * 0.2 + player.vice_captain_percentage * 0.1;
   889→  }
   890→}
   891→
   892→function scoreViceCaptain(player: TGPlayer, category: string): number {
   893→  if (category === 'H2H') {
   894→    return player.selected_by * 0.4 + player.vice_captain_percentage * 0.35 + player.points * 0.25;
   895→  } else if (category === 'SL') {
   896→    return player.vice_captain_percentage * 0.35 + player.selected_by * 0.35 + player.points * 0.3;
   897→  } else {
   898→    // Mega GL: Wider VC rotation
   899→    return player.points * 0.35 + player.vice_captain_percentage * 0.3 + player.selected_by * 0.2 + player.captain_percentage * 0.15;
   900→  }
   901→}
   902→
   903→/**
   904→ * Select Captain and Vice-Captain based on category strategy.
   905→ * Mega GL: Rotates C/VC across team members (wider differential).
   906→ * SL: Picks the top 2 most reliable.
   907→ * H2H: Picks the safest 2.
   908→ */
   909→function selectCaptainViceCaptain(
   910→  team: TGPlayer[],
   911→  category: string,
   912→  teamIndex: number,
   913→): { captain: TGPlayer; viceCaptain: TGPlayer } {
   914→  const isMegaGL = category !== 'H2H' && category !== 'SL';
   915→
   916→  if (isMegaGL) {
   917→    // Mega GL: Sort by captain score, then rotate C/VC position based on teamIndex
   918→    // This gives wider C/VC distribution across teams
   919→    const sorted = [...team].sort((a, b) => scoreCaptain(b, category) - scoreCaptain(a, category));
   920→    // Rotate: sometimes pick #1 as C and #2 as VC, sometimes #1 as C and #3 as VC, etc.
   921→    const cIdx = 0; // Always pick top scorer as C
   922→    let vcIdx = 1 + (teamIndex % Math.min(4, sorted.length - 1)); // Rotate VC among top 2-5
   923→    if (vcIdx >= sorted.length) vcIdx = 1;
   924→    return { captain: sorted[cIdx], viceCaptain: sorted[vcIdx] };
   925→  } else {
   926→    // H2H and SL: Pick top 2 by category-specific scoring
   927→    const sorted = [...team].sort((a, b) => scoreCaptain(b, category) - scoreCaptain(a, category));
   928→    return { captain: sorted[0], viceCaptain: sorted.length > 1 ? sorted[1] : sorted[0] };
   929→  }
   930→}

  1041→        [wkCount, batCount, arCount, bowlCount] = pattern;
  1042→      }

  1618→            combo = tryCombo;
  1619→            foundValidCombo = true;
  1620→            break;
  1621→          }
  1622→        }
  1623→        if (!foundValidCombo) {
  1624→          // Fallback: use category-specific C/VC scoring
  1625→          const { captain: fbC, viceCaptain: fbVC } = selectCaptainViceCaptain(fullTeam, category, t);
  1626→          combo = { c: fbC, vc: fbVC };
  1627→        }
  1628→      }
  1629→
  1630→      // Uniqueness check: skip if this team signature already exists
  1631→      const signature = makeTeamSignature(fullTeam, combo.c, combo.vc);
  1632→      if (teamSignatures.has(signature)) continue;
  1633→
  1634→      team = fullTeam;
  1635→      teamC = combo.c;
  1636→      teamVC = combo.vc;
  1637→      teamSignatures.add(signature);
  1638→    }
  1639→
  1640→    if (team && teamC && teamVC) {
  1641→      teams.push({
  1642→        id: teams.length + 1,
  1643→        captain: teamC,
  1644→        viceCaptain: teamVC,
  1645→        players: team,
  1646→      });
  1647→    } else {
  1648→      // If we failed to generate a unique team for this slot,
  1649→      // try extra attempts with different seeds before giving up
  1650→      let extraAttempts = 0;
  1651→      const MAX_EXTRA = 200;
  1652→      while (teams.length < count && extraAttempts < MAX_EXTRA && totalAttempts < MAX_TOTAL_ATTEMPTS) {
  1653→        extraAttempts++;
  1654→        totalAttempts++;
  1655→        const extraSeed = seed + count * 1000 + extraAttempts * 37;
  1656→        const sr = seededRandom(extraSeed);
  1657→
  1658→        // Pick distribution (or use explicit combination)
  1659→        let targetWK: number, targetBat: number, targetAR: number, targetBowl: number;
  1660→        if (combination) {
  1661→          targetWK = combination.wk;
  1662→          targetBat = combination.bat;
  1663→          targetAR = combination.ar;
  1664→          targetBowl = combination.bowl;
  1665→        } else if (validDistributions.length === 1) {
  1666→          [targetWK, targetBat, targetAR, targetBowl] = validDistributions[0];
  1667→        } else {
  1668→          const idx = Math.floor(sr() * validDistributions.length);
  1669→          [targetWK, targetBat, targetAR, targetBowl] = validDistributions[idx];
  1670→        }
  1671→
  1672→        const needWK = Math.max(0, targetWK - fixedWKCount);
  1673→        const needBat = Math.max(0, targetBat - fixedBatCount);
  1674→        const needAR = Math.max(0, targetAR - fixedARCount);
  1675→        const needBowl = Math.max(0, targetBowl - fixedBowlCount);
  1676→
  1677→        if (remWK.length < needWK || remBat.length < needBat ||
  1678→            remAR.length < needAR || remBowl.length < needBowl) continue;
  1679→
  1680→        const sortFn = (a: TGPlayer, b: TGPlayer) => a.selected_by - b.selected_by;
  1681→        const pickedWK = shuffleArray([...remWK].sort(sortFn), sr).slice(0, needWK);
  1682→        const pickedBat = shuffleArray([...remBat].sort(sortFn), sr).slice(0, needBat);
  1683→        const pickedAR = shuffleArray([...remAR].sort(sortFn), sr).slice(0, needAR);
  1684→        const pickedBowl = shuffleArray([...remBowl].sort(sortFn), sr).slice(0, needBowl);
  1685→
  1686→        if (pickedWK.length < needWK || pickedBat.length < needBat ||
  1687→            pickedAR.length < needAR || pickedBowl.length < needBowl) continue;
  1688→
  1689→        const remaining = [...pickedWK, ...pickedBat, ...pickedAR, ...pickedBowl];
  1690→        const fullTeam = [...fixedPlayers, ...remaining];
  1691→
  1692→        const uniqueIds = new Set(fullTeam.map(p => p.pl_id));

  1719→            const { captain: fbC, viceCaptain: fbVC } = selectCaptainViceCaptain(fullTeam, category, t + extraAttempts);

  1737→  }
  1738→
  1739→  // Build debug info
  1740→  const strategyUsed = combination

  1745→

  1748→

  1752→// ============ Auto Select for Extra Team Generation ============
  1753→// Uses real player data (selection %, captain %, points, credits, role, team balance)
  1754→// to intelligently pick 8 FIX, 5 C, and 5 VC
  1755→
  1756→export interface AutoSelectResult {
  1757→  fixedPlayers: TGPlayer[];
  1758→  captainOptions: TGPlayer[];
  1759→  viceCaptainOptions: TGPlayer[];
  1760→}
  1761→
  1762→// Player scoring for auto-selection: higher = better pick
  1763→function playerScore(p: TGPlayer): number {
  1764→  return (
  1765→    p.selected_by * 0.35 +       // selection % - most important for GL
  1766→    p.captain_percentage * 0.25 + // captain % - indicates trust
  1767→    p.points * 0.2 +             // points - performance indicator
  1768→    p.vice_captain_percentage * 0.1 + // VC %
  1769→    (p.playing === 1 ? 10 : 0) + // confirmed playing bonus
  1770→    (p.role === PLAYER_ROLES.ALL_ROUNDER ? 3 : 0) + // AR bonus (flexible)
  1771→    (p.role === PLAYER_ROLES.WICKET_KEEPER ? 1 : 0) // WK slight bonus
  1772→  );
  1773→}
  1774→
  1775→export function autoSelectExtraPlayers(
  1776→  leftPlayers: TGPlayer[],
  1777→  rightPlayers: TGPlayer[],
  1778→  avoidPlayerIds: Set<number> = new Set(),
  1779→): AutoSelectResult {
  1780→  const allPlayersRaw = [...leftPlayers, ...rightPlayers];
  1781→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
  1782→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
  1783→
  1784→  // Use lineup-aware eligibility filter
  1785→  const eligible = getEligiblePlayers(allPlayersRaw, avoidPlayerIds);
  1786→
  1787→  const eligibleLeft = eligible.filter(p => p.team_name === leftTeamName);
  1788→  const eligibleRight = eligible.filter(p => p.team_name === rightTeamName);
  1789→
  1790→  // Sort by score descending
  1791→  const sortByScore = (a: TGPlayer, b: TGPlayer) => playerScore(b) - playerScore(a);
  1792→  const sortedLeft = [...eligibleLeft].sort(sortByScore);
  1793→  const sortedRight = [...eligibleRight].sort(sortByScore);
  1794→
  1795→  // === SELECT 8 FIX PLAYERS ===
  1796→  // Strategy: balanced 4-5 from each team, role-diverse
  1797→  const fixedPlayers: TGPlayer[] = [];
  1798→  const fixedIds = new Set<number>();
  1799→
  1800→  const targetLeft = Math.min(4, sortedLeft.length);
  1801→  const targetRight = Math.min(4, sortedRight.length);
  1802→
  1803→  // Pick diverse players ensuring role coverage
  1804→  const pickDiverse = (pool: TGPlayer[], count: number, existing: TGPlayer[]): TGPlayer[] => {
  1805→    const picked: TGPlayer[] = [];
  1806→    const usedIds = new Set(existing.map(p => p.pl_id));
  1807→    const existingRoles = {
  1808→      wk: existing.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length,
  1809→      bat: existing.filter(p => p.role === PLAYER_ROLES.BATSMAN).length,
  1810→      ar: existing.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length,
  1811→      bowl: existing.filter(p => p.role === PLAYER_ROLES.BOWLER).length,
  1812→    };
  1813→
  1814→    // First pass: fill role gaps
  1815→    const roleNeeds = [
  1816→      { role: PLAYER_ROLES.WICKET_KEEPER, need: Math.max(0, 1 - existingRoles.wk), pool: pool.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER && !usedIds.has(p.pl_id)) },
  1817→      { role: PLAYER_ROLES.ALL_ROUNDER, need: Math.max(0, 1 - existingRoles.ar), pool: pool.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER && !usedIds.has(p.pl_id)) },
  1818→      { role: PLAYER_ROLES.BATSMAN, need: Math.max(0, 2 - existingRoles.bat), pool: pool.filter(p => p.role === PLAYER_ROLES.BATSMAN && !usedIds.has(p.pl_id)) },
  1819→      { role: PLAYER_ROLES.BOWLER, need: Math.max(0, 2 - existingRoles.bowl), pool: pool.filter(p => p.role === PLAYER_ROLES.BOWLER && !usedIds.has(p.pl_id)) },
  1820→    ];
  1821→
  1822→    for (const rn of roleNeeds) {
  1823→      for (let i = 0; i < rn.need && picked.length < count; i++) {
  1824→        const candidate = rn.pool.find(p => !picked.some(pp => pp.pl_id === p.pl_id));
  1825→        if (candidate) picked.push(candidate);
  1826→      }
  1827→    }
  1828→
  1829→    // Second pass: fill remaining with top-scored players
  1830→    const remaining = pool.filter(p => !usedIds.has(p.pl_id) && !picked.some(pp => pp.pl_id === p.pl_id));
  1831→    for (const p of remaining) {
  1832→      if (picked.length >= count) break;
  1833→      picked.push(p);
  1834→    }
  1835→
  1836→    return picked;
  1837→  };
  1838→
  1839→  const leftPicks = pickDiverse(sortedLeft, targetLeft, []);
  1840→  for (const p of leftPicks) { fixedPlayers.push(p); fixedIds.add(p.pl_id); }
  1841→
  1842→  const rightPicks = pickDiverse(sortedRight, targetRight, fixedPlayers);
  1843→  for (const p of rightPicks) { fixedPlayers.push(p); fixedIds.add(p.pl_id); }
  1844→
  1845→  // Fill to 8 if needed
  1846→  if (fixedPlayers.length < 8) {
  1847→    const remaining = eligible.filter(p => !fixedIds.has(p.pl_id)).sort(sortByScore);
  1848→    for (const p of remaining) {
  1849→      if (fixedPlayers.length >= 8) break;
  1850→      fixedPlayers.push(p); fixedIds.add(p.pl_id);
  1851→    }
  1852→  }
  1853→
  1854→  // Trim to 8 if over
  1855→  while (fixedPlayers.length > 8) {
  1856→    let minIdx = 0; let minScore = playerScore(fixedPlayers[0]);
  1857→    for (let i = 1; i < fixedPlayers.length; i++) {
  1858→      const s = playerScore(fixedPlayers[i]);
  1859→      if (s < minScore) { minScore = s; minIdx = i; }
  1860→    }
  1861→    const removed = fixedPlayers.splice(minIdx, 1)[0];
  1862→    fixedIds.delete(removed.pl_id);
  1863→  }
  1864→
  1865→  // === SELECT 5 CAPTAIN OPTIONS ===
  1866→  const cCandidates = eligible
  1867→    .filter(p => !avoidPlayerIds.has(p.pl_id))
  1868→    .sort((a, b) => {
  1869→      const scoreA = a.captain_percentage * 0.4 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
  1870→      const scoreB = b.captain_percentage * 0.4 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
  1871→      return scoreB - scoreA;
  1872→    })
  1873→    .slice(0, 5);
  1874→
  1875→  // === SELECT 5 VICE CAPTAIN OPTIONS ===
  1876→  const vcCandidates: TGPlayer[] = [];
  1877→  const vcPool = eligible
  1878→    .filter(p => !avoidPlayerIds.has(p.pl_id))
  1879→    .sort((a, b) => {
  1880→      const scoreA = a.vice_captain_percentage * 0.35 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
  1881→      const scoreB = b.vice_captain_percentage * 0.35 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
  1882→      return scoreB - scoreA;
  1883→    });
  1884→
  1885→  for (const p of vcPool) {
  1886→    if (vcCandidates.length >= 5) break;
  1887→    vcCandidates.push(p);
  1888→  }
  1889→
  1890→  return {
  1891→    fixedPlayers: fixedPlayers.slice(0, 8),
  1892→    captainOptions: cCandidates.slice(0, 5),
  1893→    viceCaptainOptions: vcCandidates.slice(0, 5),
  1894→  };
  1895→}

  1897→// Auto-replace a removed player in a list with the next best eligible player
  1898→export function autoReplacePlayer(
  1899→  removedPlayer: TGPlayer,
  1900→  currentList: TGPlayer[],
  1901→  allPlayers: TGPlayer[],
  1902→  avoidPlayerIds: Set<number>,
  1903→  scoringFn: (p: TGPlayer) => number = playerScore,
  1904→): TGPlayer[] {
  1905→  const currentIds = new Set(currentList.map(p => p.pl_id));
  1906→  const eligible = allPlayers.filter(p =>
  1907→    !currentIds.has(p.pl_id) && !avoidPlayerIds.has(p.pl_id) && p.pl_id !== removedPlayer.pl_id
  1908→  );
  1909→  const sorted = eligible.sort((a, b) => scoringFn(b) - scoringFn(a));
  1910→  const replacement = sorted[0];
  1911→  if (replacement) return [...currentList, replacement];
  1912→  return currentList;
  1913→}
