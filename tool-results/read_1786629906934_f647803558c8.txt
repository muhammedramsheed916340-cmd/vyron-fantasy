   146→// Countdown timer component
   147→function CountdownTimer({ matchTime }: { matchTime: string }) {
   148→  const [timeLeft, setTimeLeft] = useState('')
   149→
   150→  useEffect(() => {
   151→    const calculateTime = () => {
   152→      const now = new Date().getTime()
   153→      const matchDate = new Date(matchTime).getTime()
   154→      const diff = matchDate - now
   155→
   156→      if (diff <= 0) {
   157→        setTimeLeft('Started')
   158→        return
   159→      }
   160→
   161→      const days = Math.floor(diff / (1000 * 60 * 60 * 24))
   162→      const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60))
   163→      const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60))
   164→      const seconds = Math.floor((diff % (1000 * 60)) / 1000)
   165→
   166→      if (days > 0) {
   167→        setTimeLeft(`${days}d ${hours}h ${minutes}m ${seconds}s`)
   168→      } else {
   169→        setTimeLeft(`${hours}h ${minutes}m ${seconds}s`)
   170→      }
   171→    }
   172→
   173→    calculateTime()
   174→    const interval = setInterval(calculateTime, 1000)
   175→    return () => clearInterval(interval)
   176→  }, [matchTime])
   177→
   178→  return <span className="font-medium text-sm text-foreground">{timeLeft}</span>
   179→}
