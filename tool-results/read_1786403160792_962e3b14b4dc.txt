    36→    36→/**

    38→    38→ * AFTER LINEUP: any player has playing === 1 (confirmed in Playing XI)

    40→    40→ */
    41→    41→export function getLineupMode(players: TGPlayer[]): 'before' | 'after' {
    42→    42→  const hasConfirmedPlaying = players.some(p => p.playing === 1);
    43→    43→  return hasConfirmedPlaying ? 'after' : 'before';
    44→    44→}

    46→    46→/**
    47→    47→ * Filter players based on lineup mode.
    48→    48→ * BEFORE LINEUP: all players are eligible (no one confirmed OUT yet).
    49→    49→ * AFTER LINEUP: ONLY players with playing === 1 are eligible.
    50→    50→ */
    51→    51→export function getEligiblePlayers(
    52→    52→  allPlayers: TGPlayer[],
    53→    53→  avoidPlayerIds: Set<number> = new Set(),
    54→    54→): TGPlayer[] {
    55→    55→  const mode = getLineupMode(allPlayers);
    56→    56→
    57→    57→  return allPlayers.filter(p => {
    58→    58→    // Avoid players are always excluded
    59→    59→    if (avoidPlayerIds.has(p.pl_id)) return false;
    60→    60→
    61→    61→    if (mode === 'after') {
    62→    62→      // AFTER LINEUP: ONLY confirmed Playing XI players
    63→    63→      return p.playing === 1;
    64→    64→    }
    65→    65→
    66→    66→    // BEFORE LINEUP: all players eligible
    67→    67→    return true;
    68→    68→  });
    69→    69→}

   129→   129→/**
   130→   130→ * Get all valid role combinations for an 11-player cricket team.
   131→   131→ */
   132→   132→export function getAllValidCombinations(): RoleCombination[] {
   133→   133→  const combos: RoleCombination[] = [];
   134→   134→  for (let wk = MIN_WK; wk <= MAX_WK; wk++) {
   135→   135→    for (let bat = MIN_BAT; bat <= MAX_BAT; bat++) {
   136→   136→      for (let ar = MIN_AR; ar <= MAX_AR; ar++) {
   137→   137→        const bowl = CRICKET_TEAM_SIZE - wk - bat - ar;
   138→   138→        if (bowl < MIN_BOWL || bowl > MAX_BOWL) continue;
   139→   139→        combos.push({ wk, bat, ar, bowl });
   140→   140→      }
   141→   141→    }
   142→   142→  }
   143→   143→  return combos;
   144→   144→}

   149→   149→export function isCombinationCompatibleWithFixed(
   150→   150→  combo: RoleCombination,
   151→   151→  fixedPlayers: TGPlayer[],
   152→   152→): boolean {
   153→   153→  const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   154→   154→  const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   155→   155→  const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   156→   156→  const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   157→   157→
   158→   158→  // Fixed players must not exceed combination totals
   159→   159→  if (fixedWK > combo.wk) return false;
   160→   160→  if (fixedBat > combo.bat) return false;
   161→   161→  if (fixedAR > combo.ar) return false;
   162→   162→  if (fixedBowl > combo.bowl) return false;
   163→   163→
   164→   164→  // Remaining slots must be fillable
   165→   165→  const remainingPlayers = CRICKET_TEAM_SIZE - fixedPlayers.length;
   166→   166→  const neededWK = combo.wk - fixedWK;
   167→   167→  const neededBat = combo.bat - fixedBat;
   168→   168→  const neededAR = combo.ar - fixedAR;
   169→   169→  const neededBowl = combo.bowl - fixedBowl;
   170→   170→
   171→   171→  return (neededWK + neededBat + neededAR + neededBowl) === remainingPlayers;
   172→   172→}

   177→   177→export function getCompatibleCombinations(
   178→   178→  fixedPlayers: TGPlayer[],
   179→   179→  eligiblePlayers: TGPlayer[],
   180→   180→): RoleCombination[] {
   181→   181→  const all = getAllValidCombinations();
   182→   182→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   183→   183→  const remaining = eligiblePlayers.filter(p => !fixedIds.has(p.pl_id));
   184→   184→
   185→   185→  const remWK = remaining.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   186→   186→  const remBat = remaining.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   187→   187→  const remAR = remaining.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   188→   188→  const remBowl = remaining.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   189→   189→
   190→   190→  return all.filter(combo => {
   191→   191→    if (!isCombinationCompatibleWithFixed(combo, fixedPlayers)) return false;
   192→   192→
   193→   193→    const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   194→   194→    const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   195→   195→    const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   196→   196→    const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   197→   197→
   198→   198→    if (remWK < combo.wk - fixedWK) return false;
   199→   199→    if (remBat < combo.bat - fixedBat) return false;
   200→   200→    if (remAR < combo.ar - fixedAR) return false;
   201→   201→    if (remBowl < combo.bowl - fixedBowl) return false;
   202→   202→
   203→   203→    return true;
   204→   204→  });
   205→   205→}

   211→   211→export function autoSelectCombination(

   239→   239→    compatible.some(c => c.wk === p.wk && c.bat === p.bat && c.ar === p.ar && c.bowl === p.bowl)

   248→   248→/**
   249→   249→ * Validate a manual combination against available players.
   250→   250→ */
   251→   251→export function validateCombination(
   252→   252→  combo: RoleCombination,
   253→   253→  eligiblePlayers: TGPlayer[],
   254→   254→  fixedPlayers: TGPlayer[] = [],
   255→   255→): { valid: boolean; errors: string[] } {
   256→   256→  const errors: string[] = [];
   257→   257→
   258→   258→  if (combo.wk + combo.bat + combo.ar + combo.bowl !== CRICKET_TEAM_SIZE) {
   259→   259→    errors.push(`Total players must be 11 (got ${combo.wk + combo.bat + combo.ar + combo.bowl})`);
   260→   260→  }
   261→   261→
   262→   262→  if (combo.wk < MIN_WK || combo.wk > MAX_WK) errors.push(`WK must be ${MIN_WK}-${MAX_WK}`);
   263→   263→  if (combo.bat < MIN_BAT || combo.bat > MAX_BAT) errors.push(`BAT must be ${MIN_BAT}-${MAX_BAT}`);
   264→   264→  if (combo.ar < MIN_AR || combo.ar > MAX_AR) errors.push(`AR must be ${MIN_AR}-${MAX_AR}`);
   265→   265→  if (combo.bowl < MIN_BOWL || combo.bowl > MAX_BOWL) errors.push(`BOWL must be ${MIN_BOWL}-${MAX_BOWL}`);
   266→   266→
   267→   267→  if (fixedPlayers.length > 0 && !isCombinationCompatibleWithFixed(combo, fixedPlayers)) {
   268→   268→    errors.push('Combination is not compatible with selected fixed players');
   269→   269→  }
   270→   270→
   271→   271→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   272→   272→  const remaining = eligiblePlayers.filter(p => !fixedIds.has(p.pl_id));
   273→   273→
   274→   274→  const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   275→   275→  const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   276→   276→  const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   277→   277→  const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   278→   278→
   279→   279→  const needWK = Math.max(0, combo.wk - fixedWK);
   280→   280→  const needBat = Math.max(0, combo.bat - fixedBat);
   281→   281→  const needAR = Math.max(0, combo.ar - fixedAR);
   282→   282→  const needBowl = Math.max(0, combo.bowl - fixedBowl);
   283→   283→
   284→   284→  const availWK = remaining.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   285→   285→  const availBat = remaining.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   286→   286→  const availAR = remaining.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   287→   287→  const availBowl = remaining.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   288→   288→
   289→   289→  if (availWK < needWK) errors.push(`Need ${needWK} WK but only ${availWK} available`);
   290→   290→  if (availBat < needBat) errors.push(`Need ${needBat} BAT but only ${availBat} available`);
   291→   291→  if (availAR < needAR) errors.push(`Need ${needAR} AR but only ${availAR} available`);
   292→   292→  if (availBowl < needBowl) errors.push(`Need ${needBowl} BOWL but only ${availBowl} available`);
   293→   293→
   294→   294→  return { valid: errors.length === 0, errors };
   295→   295→}

   580→   580→// Cricket team constraints
   581→   581→export const CRICKET_TEAM_SIZE = 11;
   582→   582→export const MIN_WK = 1;
   583→   583→export const MAX_WK = 4;
   584→   584→export const MIN_BAT = 3;

   586→   586→export const MIN_AR = 1;
   587→   587→export const MAX_AR = 4;

   593→   593→function seededRandom(seed: number): () => number {
   594→   594→  let s = seed;
   595→   595→  return () => {
   596→   596→    s = (s * 16807 + 0) % 2147483647;
   597→   597→    return (s - 1) / 2147483646;
   598→   598→  };
   599→   599→}
   600→   600→
   601→   601→function shuffleArray<T>(arr: T[], rng: () => number): T[] {
   602→   602→  const result = [...arr];
   603→   603→  for (let i = result.length - 1; i > 0; i--) {
   604→   604→    const j = Math.floor(rng() * (i + 1));
   605→   605→    [result[i], result[j]] = [result[j], result[i]];
   606→   606→  }
   607→   607→  return result;
   608→   608→}
   609→   609→
   610→   610→export function generateTeams(
   611→   611→  leftPlayers: TGPlayer[],
   612→   612→  rightPlayers: TGPlayer[],
   613→   613→  category: string,
   614→   614→  count: number,
   615→   615→  seed: number = Date.now(),
   616→   616→  avoidPlayerIds: Set<number> = new Set(),
   617→   617→  combination: RoleCombination | null = null,
   618→   618→): GeneratedTeam[] {
   619→   619→  // Apply lineup-aware filtering: AFTER LINEUP only playing=1, BEFORE LINEUP all
   620→   620→  const allPlayersRaw = [...leftPlayers, ...rightPlayers];
   621→   621→  const allPlayers = getEligiblePlayers(allPlayersRaw, avoidPlayerIds);
   622→   622→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
   623→   623→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
   624→   624→
   625→   625→  // After filtering, rebuild left/right pools for team distribution checks
   626→   626→  const eligibleLeft = allPlayers.filter(p => p.team_name === leftTeamName);
   627→   627→  const eligibleRight = allPlayers.filter(p => p.team_name === rightTeamName);
   628→   628→
   629→   629→  // Separate by role
   630→   630→  const wk = allPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER);
   631→   631→  const bat = allPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN);
   632→   632→  const ar = allPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER);
   633→   633→  const bowl = allPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER);
   634→   634→
   635→   635→  const teams: GeneratedTeam[] = [];
   636→   636→  const rng = seededRandom(seed);
   637→   637→
   638→   638→  // Category-specific strategies
   639→   639→  const isH2H = category === 'H2H';
   640→   640→  const isSL = category === 'SL';
   641→   641→
   642→   642→  for (let t = 0; t < count; t++) {
   643→   643→    let attempts = 0;
   644→   644→    let team: TGPlayer[] | null = null;
   645→   645→
   646→   646→    while (!team && attempts < 200) {
   647→   647→      attempts++;
   648→   648→      const sr = seededRandom(seed + t * 1000 + attempts);
   649→   649→
   650→   650→      // Determine role counts based on category or explicit combination
   651→   651→      let wkCount, batCount, arCount, bowlCount;
   652→   652→
   653→   653→      if (combination) {
   654→   654→        // Use explicit combination (manual or auto-selected)
   655→   655→        wkCount = combination.wk;
   656→   656→        batCount = combination.bat;
   657→   657→        arCount = combination.ar;
   658→   658→        bowlCount = combination.bowl;
   659→   659→      } else if (isH2H) {
   660→   660→        // H2H: More conservative, popular players
   661→   661→        wkCount = MIN_WK + (sr() > 0.5 ? 1 : 0);
   662→   662→        arCount = MIN_AR + (sr() > 0.6 ? 1 : 0);
   663→   663→        batCount = Math.max(MIN_BAT, 5 - wkCount + (sr() > 0.5 ? 1 : 0));
   664→   664→        bowlCount = CRICKET_TEAM_SIZE - wkCount - batCount - arCount;
   665→   665→      } else if (isSL) {
   666→   666→        // SL: Balanced
   667→   667→        wkCount = 1 + (sr() > 0.7 ? 1 : 0);
   668→   668→        arCount = 2 + (sr() > 0.5 ? 1 : 0);
   669→   669→        batCount = 3 + (sr() > 0.5 ? 1 : 0);
   670→   670→        bowlCount = CRICKET_TEAM_SIZE - wkCount - batCount - arCount;
   671→   671→      } else {
   672→   672→        // Mega GL: More differential picks, varied combinations
   673→   673→        const patterns = [
   674→   674→          [1, 3, 2, 5], [1, 4, 1, 5], [1, 3, 3, 4], [1, 4, 2, 4],
   675→   675→          [2, 3, 2, 4], [1, 5, 1, 4], [2, 3, 1, 5], [1, 4, 3, 3],
   676→   676→          [2, 4, 1, 4], [1, 3, 1, 6], [2, 2, 3, 4], [1, 5, 2, 3],
   677→   677→        ];
   678→   678→        const pattern = patterns[Math.floor(sr() * patterns.length)];
   679→   679→        [wkCount, batCount, arCount, bowlCount] = pattern;
   680→   680→      }
   681→   681→
   682→   682→      // Validate counts
   683→   683→      if (wkCount < MIN_WK || wkCount > MAX_WK) continue;
   684→   684→      if (batCount < MIN_BAT || batCount > MAX_BAT) continue;
   685→   685→      if (arCount < MIN_AR || arCount > MAX_AR) continue;
   686→   686→      if (bowlCount < MIN_BOWL || bowlCount > MAX_BOWL) continue;
   687→   687→      if (wkCount + batCount + arCount + bowlCount !== CRICKET_TEAM_SIZE) continue;
   688→   688→
   689→   689→      // Pick players for each role
   690→   690→      const shuffledWK = shuffleArray(wk, sr);
   691→   691→      const shuffledBat = shuffleArray(bat, sr);
   692→   692→      const shuffledAR = shuffleArray(ar, sr);
   693→   693→      const shuffledBowl = shuffleArray(bowl, sr);
   694→   694→
   695→   695→      // For Mega GL, sort by selected_by sometimes to get differential picks
   696→   696→      let selectedWK, selectedBat, selectedAR, selectedBowl;
   697→   697→
   698→   698→      if (!isH2H && sr() > 0.4) {
   699→   699→        // Sort some roles by selection % to mix popular and differential
   700→   700→        const sortFn = (a: TGPlayer, b: TGPlayer) => a.selected_by - b.selected_by;
   701→   701→        selectedWK = shuffleArray([...wk].sort(sortFn), sr).slice(0, wkCount);
   702→   702→        selectedBat = shuffleArray([...bat].sort(sortFn), sr).slice(0, batCount);
   703→   703→        selectedAR = shuffleArray([...ar].sort(sortFn), sr).slice(0, arCount);
   704→   704→        selectedBowl = shuffleArray([...bowl].sort(sortFn), sr).slice(0, bowlCount);
   705→   705→      } else if (isH2H) {
   706→   706→        // H2H: Pick most selected players
   707→   707→        const sortFn = (a: TGPlayer, b: TGPlayer) => b.selected_by - a.selected_by;
   708→   708→        selectedWK = [...wk].sort(sortFn).slice(0, wkCount);
   709→   709→        selectedBat = shuffleArray([...bat].sort(sortFn), sr).slice(0, batCount);
   710→   710→        selectedAR = [...ar].sort(sortFn).slice(0, arCount);
   711→   711→        selectedBowl = shuffleArray([...bowl].sort(sortFn), sr).slice(0, bowlCount);
   712→   712→      } else {
   713→   713→        selectedWK = shuffledWK.slice(0, wkCount);
   714→   714→        selectedBat = shuffledBat.slice(0, batCount);
   715→   715→        selectedAR = shuffledAR.slice(0, arCount);
   716→   716→        selectedBowl = shuffledBowl.slice(0, bowlCount);
   717→   717→      }
   718→   718→
   719→   719→      // Check if we have enough players
   720→   720→      if (selectedWK.length < wkCount || selectedBat.length < batCount ||
   721→   721→          selectedAR.length < arCount || selectedBowl.length < bowlCount) continue;
   722→   722→
   723→   723→      const selected = [...selectedWK, ...selectedBat, ...selectedAR, ...selectedBowl];
   724→   724→
   725→   725→      // Check total credits
   726→   726→      const totalCredits = selected.reduce((sum, p) => sum + p.credits, 0);
   727→   727→      if (totalCredits > MAX_CREDITS) continue;
   728→   728→
   729→   729→      // Check team distribution (max 7 from one team)
   730→   730→      const leftCount = selected.filter(p => p.team_name === leftTeamName).length;
   731→   731→      const rightCount = selected.filter(p => p.team_name === rightTeamName).length;
   732→   732→      if (leftCount > MAX_FROM_ONE_TEAM || rightCount > MAX_FROM_ONE_TEAM) continue;
   733→   733→
   734→   734→      // Ensure at least 1 from each team
   735→   735→      if (leftCount < 1 || rightCount < 1) continue;
   736→   736→
   737→   737→      team = selected;
   738→   738→    }
   739→   739→
   740→   740→    if (team) {
   741→   741→      // Select Captain & Vice-Captain
   742→   742→      // Sort by a combination of points and selection % for C/VC
   743→   743→      const sorted = [...team].sort((a, b) =>
   744→   744→        (b.points * 0.4 + b.selected_by * 0.3 + b.captain_percentage * 0.3) -
   745→   745→        (a.points * 0.4 + a.selected_by * 0.3 + a.captain_percentage * 0.3)
   746→   746→      );
   747→   747→
   748→   748→      const captain = sorted[0];
   749→   749→      const viceCaptain = sorted.length > 1 ? sorted[1] : sorted[0];
   750→   750→
   751→   751→      teams.push({
   752→   752→        id: t + 1,
   753→   753→        captain,
   754→   754→        viceCaptain,
   755→   755→        players: team,
   756→   756→      });
   757→   757→    }
   758→   758→  }
   759→   759→
   760→   760→  return teams;
   761→   761→}

   791→   791→    combination = null,

   799→   799→

   801→   801→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   802→   802→  const remainingPool = allPlayers.filter(p => !fixedIds.has(p.pl_id));
   803→   803→

   808→   808→  const remBowl = remainingPool.filter(p => p.role === PLAYER_ROLES.BOWLER);
   809→   809→
   810→   810→  // Count fixed players by role
   811→   811→  const fixedWKCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   812→   812→  const fixedBatCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   813→   813→  const fixedARCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   814→   814→  const fixedBowlCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   815→   815→
   816→   816→  const fixedCredits = fixedPlayers.reduce((sum, p) => sum + p.credits, 0);
   817→   817→  const remainingCredits = MAX_CREDITS - fixedCredits;
   818→   818→  const neededPlayers = CRICKET_TEAM_SIZE - fixedPlayers.length; // 3
   819→   819→
   820→   820→  const teams: GeneratedTeam[] = [];
   821→   821→
   822→   822→  // Pre-validate: fixed players must have credits <= 100
   823→   823→  if (fixedCredits > MAX_CREDITS) return teams;
   824→   824→
   825→   825→  // Pre-validate: fixed players must respect max 7 from one team
   826→   826→  const fixedLeftCount = fixedPlayers.filter(p => p.team_name === leftTeamName).length;
   827→   827→  const fixedRightCount = fixedPlayers.filter(p => p.team_name === rightTeamName).length;
   828→   828→  if (fixedLeftCount > MAX_FROM_ONE_TEAM || fixedRightCount > MAX_FROM_ONE_TEAM) return teams;
   829→   829→
   830→   830→  // Generate all valid C/VC combinations from the 5C and 5VC options
   831→   831→  // C must be in the team, VC must be in the team, C ≠ VC
   832→   832→  const cVCCombos: { c: TGPlayer; vc: TGPlayer }[] = [];
   833→   833→  for (const c of captainOptions) {
   834→   834→    for (const vc of viceCaptainOptions) {
   835→   835→      if (c.pl_id !== vc.pl_id) {
   836→   836→        cVCCombos.push({ c, vc });
   837→   837→      }
   838→   838→    }
   839→   839→  }
   840→   840→  // If no valid combos, return empty
   841→   841→  if (cVCCombos.length === 0) return teams;
   842→   842→
   843→   843→  // Category-specific role distribution preferences for remaining players
   844→   844→  const isH2H = category === 'H2H';
   845→   845→  const isSL = category === 'SL';
   846→   846→
   847→   847→  // Pre-compute all valid target role distributions that are compatible with fixed players
   848→   848→  // This avoids wasting attempts on patterns that can never work
   849→   849→  const categoryPatterns: number[][] = isH2H
   850→   850→    ? [] // H2H uses dynamic generation below
   851→   851→    : isSL
   852→   852→    ? [] // SL uses dynamic generation below
   853→   853→    : [
   854→   854→        [1, 3, 2, 5], [1, 4, 1, 5], [1, 3, 3, 4], [1, 4, 2, 4],
   855→   855→        [2, 3, 2, 4], [1, 5, 1, 4], [2, 3, 1, 5], [1, 4, 3, 3],
   856→   856→        [2, 4, 1, 4], [1, 3, 1, 6], [2, 2, 3, 4], [1, 5, 2, 3],
   857→   857→      ];
   858→   858→
   859→   859→  // Build list of valid target distributions: [WK, BAT, AR, BOWL]
   860→   860→  const validDistributions: number[][] = [];
   861→   861→  for (let wk = MIN_WK; wk <= MAX_WK; wk++) {
   862→   862→    for (let bat = MIN_BAT; bat <= MAX_BAT; bat++) {
   863→   863→      for (let ar = MIN_AR; ar <= MAX_AR; ar++) {
   864→   864→        const bowl = CRICKET_TEAM_SIZE - wk - bat - ar;
   865→   865→        if (bowl < MIN_BOWL || bowl > MAX_BOWL) continue;
   866→   866→        // Check compatibility with fixed players
   867→   867→        const nWK = Math.max(0, wk - fixedWKCount);
   868→   868→        const nBat = Math.max(0, bat - fixedBatCount);

  1131→  1131→// ============ Auto Select for Extra Team Generation ============
  1132→  1132→// Uses real player data (selection %, captain %, points, credits, role, team balance)
  1133→  1133→// to intelligently pick 8 FIX, 5 C, and 5 VC

  1141→  1141→// Player scoring for auto-selection: higher = better pick
  1142→  1142→function playerScore(p: TGPlayer): number {
  1143→  1143→  return (
  1144→  1144→    p.selected_by * 0.35 +       // selection % - most important for GL
  1145→  1145→    p.captain_percentage * 0.25 + // captain % - indicates trust
  1146→  1146→    p.points * 0.2 +             // points - performance indicator
  1147→  1147→    p.vice_captain_percentage * 0.1 + // VC %
  1148→  1148→    (p.playing === 1 ? 10 : 0) + // confirmed playing bonus
  1149→  1149→    (p.role === PLAYER_ROLES.ALL_ROUNDER ? 3 : 0) + // AR bonus (flexible)
  1150→  1150→    (p.role === PLAYER_ROLES.WICKET_KEEPER ? 1 : 0) // WK slight bonus
  1151→  1151→  );
  1152→  1152→}
  1153→  1153→
  1154→  1154→export function autoSelectExtraPlayers(
  1155→  1155→  leftPlayers: TGPlayer[],

  1198→  1198→      { role: PLAYER_ROLES.BOWLER, need: Math.max(0, 2 - existingRoles.bowl), pool: pool.filter(p => p.role === PLAYER_ROLES.BOWLER && !usedIds.has(p.pl_id)) },

  1216→  1216→  };

  1226→  1226→    const remaining = eligible.filter(p => !fixedIds.has(p.pl_id)).sort(sortByScore);
  1227→  1227→    for (const p of remaining) {
  1228→  1228→      if (fixedPlayers.length >= 8) break;
  1229→  1229→      fixedPlayers.push(p); fixedIds.add(p.pl_id);
  1230→  1230→    }
  1231→  1231→  }
  1232→  1232→
  1233→  1233→  // Trim to 8 if over
  1234→  1234→  while (fixedPlayers.length > 8) {
  1235→  1235→    let minIdx = 0; let minScore = playerScore(fixedPlayers[0]);
  1236→  1236→    for (let i = 1; i < fixedPlayers.length; i++) {
  1237→  1237→      const s = playerScore(fixedPlayers[i]);
  1238→  1238→      if (s < minScore) { minScore = s; minIdx = i; }
  1239→  1239→    }
  1240→  1240→    const removed = fixedPlayers.splice(minIdx, 1)[0];
  1241→  1241→    fixedIds.delete(removed.pl_id);
  1242→  1242→  }
  1243→  1243→
  1244→  1244→  // === SELECT 5 CAPTAIN OPTIONS ===
  1245→  1245→  const cCandidates = eligible
  1246→  1246→    .filter(p => !avoidPlayerIds.has(p.pl_id))
  1247→  1247→    .sort((a, b) => {
  1248→  1248→      const scoreA = a.captain_percentage * 0.4 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
  1249→  1249→      const scoreB = b.captain_percentage * 0.4 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
  1250→  1250→      return scoreB - scoreA;
  1251→  1251→    })
  1252→  1252→    .slice(0, 5);
  1253→  1253→
  1254→  1254→  // === SELECT 5 VICE CAPTAIN OPTIONS ===
  1255→  1255→  const vcCandidates: TGPlayer[] = [];
  1256→  1256→  const vcPool = eligible
  1257→  1257→    .filter(p => !avoidPlayerIds.has(p.pl_id))
  1258→  1258→    .sort((a, b) => {
  1259→  1259→      const scoreA = a.vice_captain_percentage * 0.35 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
  1260→  1260→      const scoreB = b.vice_captain_percentage * 0.35 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
  1261→  1261→      return scoreB - scoreA;
  1262→  1262→    });
  1263→  1263→
  1264→  1264→  for (const p of vcPool) {
  1265→  1265→    if (vcCandidates.length >= 5) break;
  1266→  1266→    vcCandidates.push(p);
  1267→  1267→  }
  1268→  1268→
  1269→  1269→  return {
  1270→  1270→    fixedPlayers: fixedPlayers.slice(0, 8),
  1271→  1271→    captainOptions: cCandidates.slice(0, 5),
  1272→  1272→    viceCaptainOptions: vcCandidates.slice(0, 5),
  1273→  1273→  };
  1274→  1274→}

  1276→  1276→// Auto-replace a removed player in a list with the next best eligible player
  1277→  1277→export function autoReplacePlayer(
  1278→  1278→  removedPlayer: TGPlayer,
  1279→  1279→  currentList: TGPlayer[],
  1280→  1280→  allPlayers: TGPlayer[],
  1281→  1281→  avoidPlayerIds: Set<number>,
  1282→  1282→  scoringFn: (p: TGPlayer) => number = playerScore,
  1283→  1283→): TGPlayer[] {
  1284→  1284→  const currentIds = new Set(currentList.map(p => p.pl_id));
  1285→  1285→  const eligible = allPlayers.filter(p =>
  1286→  1286→    !currentIds.has(p.pl_id) && !avoidPlayerIds.has(p.pl_id) && p.pl_id !== removedPlayer.pl_id
  1287→  1287→  );
  1288→  1288→  const sorted = eligible.sort((a, b) => scoringFn(b) - scoringFn(a));
  1289→  1289→  const replacement = sorted[0];
  1290→  1290→  if (replacement) return [...currentList, replacement];
  1291→  1291→  return currentList;
  1292→  1292→}
