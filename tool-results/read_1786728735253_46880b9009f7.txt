     8→/**
     9→ * Normalize platform names to consistent lowercase identifiers.
    10→ * The TG API may return varied casing in fantasy_id_list[].name
    11→ * (e.g., "Dream11", "dream11", "My11Circle", "my11circle").
    12→ * All internal references use lowercase: "dream11", "my11circle".

    14→export function normalizePlatformName(name: string): string {
    15→  return name.toLowerCase().replace(/[\s_-]/g, '');

    18→/**
    19→ * Resolve a player's platform-specific ID from their fantasy_id_list.
    20→ * Uses case-insensitive matching to handle API casing variations.
    21→ */
    22→export function resolvePlatformPlayerId(
    23→  player: TGPlayer,
    24→  platform: string,
    25→): number | null {
    26→  if (!player.fantasy_id_list || player.fantasy_id_list.length === 0) return null;
    27→  const target = normalizePlatformName(platform);
    28→  const found = player.fantasy_id_list.find(
    29→    f => normalizePlatformName(f.name) === target
    30→  );
    31→  return found ? found.id : null;
    32→}

    62→// ============ Lineup Utilities ============
    63→
    64→/**
    65→ * Determine lineup mode based on player data and lineup status.
    66→ *
    67→ * AFTER LINEUP: lineup is confirmed — only playing===1 players are eligible.
    68→ *   Determined by either:
    69→ *   a) lineupStatus >= 1 (API says lineup is out), OR
    70→ *   b) enough players have playing === 1 to form valid teams (11+ total, 3+ per team)
    71→ *
    72→ * BEFORE LINEUP: lineup not yet confirmed — all non-OUT players are eligible.
    73→ *
    74→ * lineupStatus values (from TG API):
    75→ *   0 = lineup not yet out
    76→ *   1 = lineup confirmed / playing XI announced
    77→ *
    78→ * BUG FIX: Previously required 4 confirmed per team, which was too strict for
    79→ * partial lineups (e.g., 3 confirmed from one team). Relaxed to 3 per team.
    80→ * Also added lineupStatus parameter so API's lineup_status field is respected
    81→ * even when individual player playing values haven't fully propagated yet.
    82→ */
    83→export function getLineupMode(players: TGPlayer[], lineupStatus: number = 0): 'before' | 'after' {
    84→  const confirmed = players.filter(p => p.playing === 1);
    85→
    86→  // If API says lineup is confirmed, trust it — but still need minimum viable pool
    87→  if (lineupStatus >= 1) {
    88→    // Need at least 11 confirmed players to form ANY valid team
    89→    if (confirmed.length >= 11) {
    90→      // Check per-team minimum: need at least 3 from each team
    91→      // (relaxed from 4 to handle partial lineups where one team has fewer confirmed)
    92→      const teamNames = [...new Set(players.map(p => p.team_name))];
    93→      let allTeamsViable = true;
    94→      for (const team of teamNames) {
    95→        if (team && confirmed.filter(p => p.team_name === team).length < 3) {
    96→          allTeamsViable = false;
    97→          break;
    98→        }
    99→      }
   100→      if (allTeamsViable) return 'after';
   101→
   102→      // Even with <3 per team, if lineupStatus says out and we have 11+ confirmed,
   103→      // still go 'after' — the generation will handle the constraints
   104→      console.warn(`[LINEUP] lineupStatus=${lineupStatus} but per-team distribution too skewed. Falling back to 'before' to allow bench players for team distribution.`);
   105→      return 'before';
   106→    }
   107→
   108→    // lineupStatus says out but < 11 confirmed players — data might be incomplete
   109→    // Stay in 'before' to include probable players (playing===0) as well
   110→    console.warn(`[LINEUP] lineupStatus=${lineupStatus} but only ${confirmed.length} confirmed players. Staying 'before' — lineup data may be incomplete.`);
   111→    return 'before';
   112→  }
   113→
   114→  // No explicit lineup status — infer from player data
   115→  // Need at least 11 confirmed players total to form a valid team
   116→  if (confirmed.length < 11) return 'before';
   117→
   118→  // Need at least 3 confirmed from each team (relaxed from 4)
   119→  const teamNames = [...new Set(players.map(p => p.team_name))];
   120→  for (const team of teamNames) {
   121→    if (team && confirmed.filter(p => p.team_name === team).length < 3) {
   122→      return 'before';
   123→    }
   124→  }
   125→
   126→  return 'after';
   127→}

   129→/**
   130→ * Filter players based on lineup mode.
   131→ * BEFORE LINEUP: all players are eligible EXCEPT confirmed OUT (playing === -1).
   132→ * AFTER LINEUP: ONLY players with playing === 1 are eligible (Playing XI only, no bench).
   133→ */
   134→export function getEligiblePlayers(
   135→  allPlayers: TGPlayer[],
   136→  avoidPlayerIds: Set<number> = new Set(),
   137→  lineupStatus: number = 0,
   138→): TGPlayer[] {
   139→  const mode = getLineupMode(allPlayers, lineupStatus);
   140→
   141→  return allPlayers.filter(p => {
   142→    // Avoid players are always excluded
   143→    if (avoidPlayerIds.has(p.pl_id)) return false;
   144→
   145→    // Confirmed OUT players are always excluded
   146→    if (p.playing === -1) return false;
   147→
   148→    if (mode === 'after') {
   149→      // AFTER LINEUP: ONLY confirmed Playing XI players (playing === 1)
   150→      // Bench/probable players (playing === 0) are NOT included
   151→      return p.playing === 1;
   152→    }
   153→
   154→    // BEFORE LINEUP: all non-OUT players eligible (playing === 0 or 1)
   155→    return true;
   156→  });
   157→}

   159→/**
   160→ * Check if a specific player is eligible based on lineup mode.

   162→export function isPlayerEligible(
   163→  player: TGPlayer,
   164→  allPlayers: TGPlayer[],
   165→  avoidPlayerIds: Set<number> = new Set(),
   166→  lineupStatus: number = 0,
   167→): { eligible: boolean; reason?: string } {
   168→  if (avoidPlayerIds.has(player.pl_id)) {
   169→    return { eligible: false, reason: 'AVOID' };
   170→  }
   171→
   172→  const mode = getLineupMode(allPlayers, lineupStatus);
   173→
   174→  if (mode === 'after' && player.playing !== 1) {
   175→    return { eligible: false, reason: 'OUT / NOT IN PLAYING XI' };
   176→  }
   177→
   178→  return { eligible: true };
   179→}

   185→export function validateTeamForLineup(
   186→  team: GeneratedTeam,
   187→  allPlayers: TGPlayer[],
   188→  avoidPlayerIds: Set<number> = new Set(),
   189→  lineupStatus: number = 0,
   190→): { valid: boolean; invalidPlayers: { player: TGPlayer; reason: string }[] } {
   191→  const mode = getLineupMode(allPlayers, lineupStatus);
   192→  const invalidPlayers: { player: TGPlayer; reason: string }[] = [];
   193→
   194→  for (const player of team.players) {
   195→    if (avoidPlayerIds.has(player.pl_id)) {
   196→      invalidPlayers.push({ player, reason: 'AVOID' });
   197→      continue;
   198→    }
   199→
   200→    if (mode === 'after' && player.playing !== 1) {
   201→      invalidPlayers.push({ player, reason: 'OUT / NOT IN PLAYING XI' });
   202→    }
   203→  }
   204→
   205→  return { valid: invalidPlayers.length === 0, invalidPlayers };
   206→}

   219→/**
   220→ * Get all valid role combinations for an 11-player cricket team.
   221→ */
   222→export function getAllValidCombinations(): RoleCombination[] {
   223→  const combos: RoleCombination[] = [];
   224→  for (let wk = MIN_WK; wk <= MAX_WK; wk++) {
   225→    for (let bat = MIN_BAT; bat <= MAX_BAT; bat++) {
   226→      for (let ar = MIN_AR; ar <= MAX_AR; ar++) {
   227→        const bowl = CRICKET_TEAM_SIZE - wk - bat - ar;
   228→        if (bowl < MIN_BOWL || bowl > MAX_BOWL) continue;
   229→        combos.push({ wk, bat, ar, bowl });
   230→      }
   231→    }
   232→  }
   233→  return combos;
   234→}

   236→/**
   237→ * Check if a role combination is compatible with a set of fixed players.
   238→ */
   239→export function isCombinationCompatibleWithFixed(
   240→  combo: RoleCombination,
   241→  fixedPlayers: TGPlayer[],
   242→): boolean {
   243→  const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   244→  const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   245→  const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   246→  const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   247→
   248→  // Fixed players must not exceed combination totals
   249→  if (fixedWK > combo.wk) return false;
   250→  if (fixedBat > combo.bat) return false;
   251→  if (fixedAR > combo.ar) return false;
   252→  if (fixedBowl > combo.bowl) return false;
   253→
   254→  // Remaining slots must be fillable
   255→  const remainingPlayers = CRICKET_TEAM_SIZE - fixedPlayers.length;
   256→  const neededWK = combo.wk - fixedWK;
   257→  const neededBat = combo.bat - fixedBat;
   258→  const neededAR = combo.ar - fixedAR;
   259→  const neededBowl = combo.bowl - fixedBowl;
   260→
   261→  return (neededWK + neededBat + neededAR + neededBowl) === remainingPlayers;
   262→}

   264→/**
   265→ * Get combinations compatible with fixed players and available player pool.
   266→ */
   267→export function getCompatibleCombinations(
   268→  fixedPlayers: TGPlayer[],
   269→  eligiblePlayers: TGPlayer[],
   270→): RoleCombination[] {
   271→  const all = getAllValidCombinations();
   272→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   273→  const remaining = eligiblePlayers.filter(p => !fixedIds.has(p.pl_id));
   274→
   275→  const remWK = remaining.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   276→  const remBat = remaining.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   277→  const remAR = remaining.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   278→  const remBowl = remaining.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   279→
   280→  return all.filter(combo => {
   281→    if (!isCombinationCompatibleWithFixed(combo, fixedPlayers)) return false;
   282→
   283→    const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   284→    const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   285→    const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   286→    const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   287→
   288→    if (remWK < combo.wk - fixedWK) return false;
   289→    if (remBat < combo.bat - fixedBat) return false;
   290→    if (remAR < combo.ar - fixedAR) return false;
   291→    if (remBowl < combo.bowl - fixedBowl) return false;
   292→
   293→    return true;
   294→  });
   295→}

   297→/**
   298→ * Auto-select the best combination based on player pool and category.
   299→ * For multiple teams, rotates combinations intelligently.
   300→ */
   301→export function autoSelectCombination(
   302→  eligiblePlayers: TGPlayer[],
   303→  category: string,
   304→  fixedPlayers: TGPlayer[] = [],
   305→  teamIndex: number = 0,
   306→): RoleCombination {
   307→  const compatible = getCompatibleCombinations(fixedPlayers, eligiblePlayers);
   308→
   309→  if (compatible.length === 0) {
   310→    return { wk: 1, bat: 4, ar: 2, bowl: 4 };
   311→  }
   312→
   313→  // Category-preferred patterns
   314→  const preferredPatterns: RoleCombination[] =
   315→    category === 'H2H'
   316→      ? [{ wk: 1, bat: 5, ar: 2, bowl: 3 }, { wk: 1, bat: 4, ar: 2, bowl: 4 }, { wk: 1, bat: 3, ar: 3, bowl: 4 }]
   317→      : category === 'SL'
   318→      ? [{ wk: 1, bat: 4, ar: 2, bowl: 4 }, { wk: 1, bat: 3, ar: 3, bowl: 4 }, { wk: 2, bat: 3, ar: 2, bowl: 4 }]
   319→      : [ // Mega GL
   320→        { wk: 1, bat: 3, ar: 2, bowl: 5 }, { wk: 1, bat: 4, ar: 1, bowl: 5 },
   321→        { wk: 1, bat: 3, ar: 3, bowl: 4 }, { wk: 1, bat: 4, ar: 2, bowl: 4 },
   322→        { wk: 2, bat: 3, ar: 2, bowl: 4 }, { wk: 1, bat: 5, ar: 1, bowl: 4 },
   323→        { wk: 2, bat: 3, ar: 1, bowl: 5 }, { wk: 1, bat: 4, ar: 3, bowl: 3 },
   324→        { wk: 2, bat: 4, ar: 1, bowl: 4 }, { wk: 1, bat: 3, ar: 1, bowl: 6 },
   325→        { wk: 2, bat: 2, ar: 3, bowl: 4 }, { wk: 1, bat: 5, ar: 2, bowl: 3 },
   326→      ];
   327→
   328→  const preferredCompatible = preferredPatterns.filter(p =>
   329→    compatible.some(c => c.wk === p.wk && c.bat === p.bat && c.ar === p.ar && c.bowl === p.bowl)
   330→  );
   331→
   332→  const pool = preferredCompatible.length > 0 ? preferredCompatible : compatible;
   333→  const index = teamIndex % pool.length;
   334→
   335→  return pool[index];
   336→}

   338→/**
   339→ * Validate a manual combination against available players.
   340→ */
   341→export function validateCombination(
   342→  combo: RoleCombination,
   343→  eligiblePlayers: TGPlayer[],
   344→  fixedPlayers: TGPlayer[] = [],
   345→): { valid: boolean; errors: string[] } {
   346→  const errors: string[] = [];
   347→
   348→  if (combo.wk + combo.bat + combo.ar + combo.bowl !== CRICKET_TEAM_SIZE) {
   349→    errors.push(`Total players must be 11 (got ${combo.wk + combo.bat + combo.ar + combo.bowl})`);
   350→  }
   351→
   352→  if (combo.wk < MIN_WK || combo.wk > MAX_WK) errors.push(`WK must be ${MIN_WK}-${MAX_WK}`);
   353→  if (combo.bat < MIN_BAT || combo.bat > MAX_BAT) errors.push(`BAT must be ${MIN_BAT}-${MAX_BAT}`);
   354→  if (combo.ar < MIN_AR || combo.ar > MAX_AR) errors.push(`AR must be ${MIN_AR}-${MAX_AR}`);
   355→  if (combo.bowl < MIN_BOWL || combo.bowl > MAX_BOWL) errors.push(`BOWL must be ${MIN_BOWL}-${MAX_BOWL}`);
   356→
   357→  if (fixedPlayers.length > 0 && !isCombinationCompatibleWithFixed(combo, fixedPlayers)) {
   358→    errors.push('Combination is not compatible with selected fixed players');
   359→  }
   360→
   361→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   362→  const remaining = eligiblePlayers.filter(p => !fixedIds.has(p.pl_id));
   363→
   364→  const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   365→  const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   366→  const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   367→  const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   368→
   369→  const needWK = Math.max(0, combo.wk - fixedWK);
   370→  const needBat = Math.max(0, combo.bat - fixedBat);
   371→  const needAR = Math.max(0, combo.ar - fixedAR);
   372→  const needBowl = Math.max(0, combo.bowl - fixedBowl);
   373→
   374→  const availWK = remaining.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   375→  const availBat = remaining.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   376→  const availAR = remaining.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   377→  const availBowl = remaining.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   378→
   379→  if (availWK < needWK) errors.push(`Need ${needWK} WK but only ${availWK} available`);
   380→  if (availBat < needBat) errors.push(`Need ${needBat} BAT but only ${availBat} available`);
   381→  if (availAR < needAR) errors.push(`Need ${needAR} AR but only ${availAR} available`);
   382→  if (availBowl < needBowl) errors.push(`Need ${needBowl} BOWL but only ${availBowl} available`);
   383→
   384→  return { valid: errors.length === 0, errors };
   385→}

   466→  try {
   467→    const bytes = CryptoJS.AES.decrypt(encrypted, ENCRYPTION_KEY);
   468→    const decrypted = bytes.toString(CryptoJS.enc.Utf8);
   469→    if (!decrypted) return encrypted;
   470→    return decrypted;
   471→  } catch {
   472→    return encrypted;
   473→  }
   474→}

   700→/**
   701→ * Create a unique signature for a team based on sorted player IDs + C/VC IDs.
   702→ * Used for deduplication — two teams with identical signatures are duplicates.
   703→ */
   704→export function makeTeamSignature(players: TGPlayer[], captain: TGPlayer, viceCaptain: TGPlayer): string {
   705→  const playerIds = players.map(p => p.pl_id).sort((a, b) => a - b).join(',');
   706→  return `${playerIds}|C${captain.pl_id}|VC${viceCaptain.pl_id}`;
   707→}

   709→/**
   710→ * Deduplicate and validate a list of generated teams.
   711→ * Removes exact duplicate player combinations.
   712→ * Separates valid and invalid teams based on lineup eligibility.
   713→ *
   714→ * Returns: { valid, invalid, duplicateCount, invalidLineupCount }
   715→ */
   716→export function deduplicateAndValidateTeams(
   717→  teams: GeneratedTeam[],
   718→  allPlayers: TGPlayer[],
   719→  avoidPlayerIds: Set<number> = new Set(),
   720→  lineupStatus: number = 0,
   721→): {
   722→  valid: GeneratedTeam[];
   723→  invalid: GeneratedTeam[];
   724→  duplicateCount: number;
   725→  invalidLineupCount: number;
   726→} {
   727→  const seen = new Set<string>();
   728→  const valid: GeneratedTeam[] = [];
   729→  const invalid: GeneratedTeam[] = [];
   730→  let duplicateCount = 0;
   731→  let invalidLineupCount = 0;
   732→
   733→  for (let i = 0; i < teams.length; i++) {
   734→    const team = teams[i];
   735→    const sig = makeTeamSignature(team.players, team.captain, team.viceCaptain);
   736→
   737→    // Skip duplicates
   738→    if (seen.has(sig)) {
   739→      duplicateCount++;
   740→      continue;
   741→    }
   742→    seen.add(sig);
   743→
   744→    // Check lineup validity
   745→    const validation = validateTeamForLineup(team, allPlayers, avoidPlayerIds, lineupStatus);
   746→    if (!validation.valid) {
   747→      invalidLineupCount++;
   748→      invalid.push({ ...team, id: invalid.length + 1 });
   749→    } else {
   750→      valid.push({ ...team, id: valid.length + 1 });
   751→    }
   752→  }
   753→
   754→  return { valid, invalid, duplicateCount, invalidLineupCount };
   755→}

   770→function seededRandom(seed: number): () => number {
   771→  let s = seed;
   772→  return () => {
   773→    s = (s * 16807 + 0) % 2147483647;
   774→    return (s - 1) / 2147483646;
   775→  };
   776→}

   778→function shuffleArray<T>(arr: T[], rng: () => number): T[] {
   779→  const result = [...arr];
   780→  for (let i = result.length - 1; i > 0; i--) {
   781→    const j = Math.floor(rng() * (i + 1));
   782→    [result[i], result[j]] = [result[j], result[i]];
   783→  }
   784→  return result;
   785→}

   787→// ============ PRO PLAYER SELECTION ENGINE ============
   788→// Each team tries to rank in the top 1-100.
   789→// Strategy: Must-have players + smart scoring + limited differential.
   790→
   791→/**
   792→ * Score a player for team selection (higher = better pick).
   793→ * Used for the MAIN generation engine (not just autoSelect).
   794→ * This is the PRO scoring that replaces the broken sort+shuffle approach.
   795→ */
   796→function playerScoreForGen(p: TGPlayer, category: string): number {
   797→  if (category === 'H2H') {
   798→    // H2H: Safest possible — selection % is king
   799→    return (
   800→      p.selected_by * 0.50 +
   801→      p.captain_percentage * 0.25 +
   802→      p.points * 0.15 +
   803→      p.vice_captain_percentage * 0.05 +
   804→      (p.playing === 1 ? 12 : 0) +
   805→      (p.role === PLAYER_ROLES.ALL_ROUNDER ? 3 : 0)
   806→    );
   807→  } else if (category === 'SL') {
   808→    // SL: Balanced — form + selection + captain history
   809→    return (
   810→      p.selected_by * 0.35 +
   811→      p.captain_percentage * 0.25 +
   812→      p.points * 0.25 +
   813→      p.vice_captain_percentage * 0.10 +
   814→      (p.playing === 1 ? 12 : 0) +
   815→      (p.role === PLAYER_ROLES.ALL_ROUNDER ? 3 : 0)
   816→    );
   817→  } else {
   818→    // Mega GL: Still prioritize core quality, differential comes from C/VC rotation
   819→    // and 2-3 "punt" players — NOT from making the whole team weak
   820→    return (
   821→      p.selected_by * 0.30 +
   822→      p.captain_percentage * 0.20 +
   823→      p.points * 0.25 +
   824→      p.vice_captain_percentage * 0.15 +
   825→      (p.playing === 1 ? 12 : 0) +
   826→      (p.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0) +
   827→      (p.role === PLAYER_ROLES.WICKET_KEEPER ? 1 : 0)
   828→    );
   829→  }
   830→}
   831→
   832→/**
   833→ * Smart select: Picks top players by score, with controlled randomness.
   834→ * - Keeps top `keepTop` players fixed (highest scored).
   835→ * - Randomly selects from the remaining pool for variety.
   836→ * - This FIXES the critical bug where shuffleArray after sort destroyed all ordering.
   837→ */
   838→function smartSelect(
   839→  pool: TGPlayer[],
   840→  count: number,
   841→  scoreFn: (p: TGPlayer) => number,
   842→  rng: () => number,
   843→  keepTopRatio: number = 0.7, // Keep top 70% of picks as best-scored
   844→): TGPlayer[] {
   845→  if (pool.length <= count) return [...pool];
   846→
   847→  const sorted = [...pool].sort((a, b) => scoreFn(b) - scoreFn(a));
   848→  const keepTop = Math.max(1, Math.floor(count * keepTopRatio));
   849→  const randomSlots = count - keepTop;
   850→
   851→  // Top players — always include these (they're the best picks)
   852→  const top = sorted.slice(0, keepTop);
   853→  const topIds = new Set(top.map(p => p.pl_id));
   854→
   855→  // Remaining pool (excluding already picked top players)
   856→  const remaining = sorted.filter(p => !topIds.has(p.pl_id));
   857→
   858→  // Random selection from remaining for variety/diversification
   859→  const shuffled = shuffleArray(remaining, rng);
   860→  const randomPicks = shuffled.slice(0, randomSlots);
   861→
   862→  return [...top, ...randomPicks];
   863→}

   865→/**
   866→ * Identify must-have players — players selected by >70% of users.
   867→ * These should be in EVERY team. Missing them = guaranteed bad rank.
   868→ */
   869→function getMustHavePlayers(
   870→  pool: TGPlayer[],
   871→  threshold: number = 70,
   872→): TGPlayer[] {
   873→  return pool.filter(p => p.selected_by >= threshold && p.playing === 1);

   876→/**
   877→ * Score an entire team's quality (higher = better team).
   878→ * Used to reject low-quality teams and retry.
   879→ */
   880→function scoreTeamQuality(team: TGPlayer[], category: string): number {
   881→  let score = 0;
   882→  const scoreFn = (p: TGPlayer) => playerScoreForGen(p, category);
   883→
   884→  for (const p of team) {
   885→    score += scoreFn(p);
   886→    // Bonus for must-have players (selected_by > 70%)
   887→    if (p.selected_by >= 70) score += 8;
   888→    // Bonus for confirmed playing
   889→    if (p.playing === 1) score += 3;
   890→    // Penalty for very low selection (< 10%) unless Mega GL
   891→    if (p.selected_by < 10 && category !== 'MEGA GL') score -= 5;
   892→  }
   893→
   894→  // Bonus for role diversity
   895→  const roles = new Set(team.map(p => p.role));
   896→  score += roles.size * 3;
   897→
   898→  // Bonus for team balance (close to 6-5 or 5-6 split)
   899→  const teamNames = [...new Set(team.map(p => p.team_name))];
   900→  if (teamNames.length === 2) {
   901→    const c1 = team.filter(p => p.team_name === teamNames[0]).length;
   902→    const c2 = team.filter(p => p.team_name === teamNames[1]).length;
   903→    const balance = Math.min(c1, c2); // Closer to 5.5 is better
   904→    score += balance * 2;
   905→  }
   906→
   907→  return score;
   908→}

   910→/**
   911→ * Category-specific C/VC scoring functions.
   912→ * Each category has a genuinely different strategy for selecting Captain and Vice-Captain.
   913→ *
   914→ * Mega GL: Prioritize differential captains (low selection % but high points).
   915→ *          Wider C/VC rotation across teams — different C/VC per team.
   916→ * SL:      Prioritize consistent/form players. Higher weight on selection % and captain %.
   917→ *          More stable C/VC — top 2 players usually.
   918→ * H2H:     Prioritize most-selected, safest players. Highest weight on selection %.
   919→ *          Most conservative C/VC — always the safest picks.
   920→ */
   921→function scoreCaptain(player: TGPlayer, category: string): number {
   922→  if (category === 'H2H') {
   923→    // H2H: Safest picks — highest selection % and captain % weighted most
   924→    return player.selected_by * 0.45 + player.captain_percentage * 0.35 + player.points * 0.2;
   925→  } else if (category === 'SL') {
   926→    // SL: Balanced — form players with good captain record
   927→    return player.captain_percentage * 0.35 + player.selected_by * 0.35 + player.points * 0.3;
   928→  } else {
   929→    // Mega GL: Differential — rotate captains, some differential (lower selection but higher upside)
   930→    return player.points * 0.4 + player.captain_percentage * 0.3 + player.selected_by * 0.2 + player.vice_captain_percentage * 0.1;
   931→  }
   932→}
   933→
   934→function scoreViceCaptain(player: TGPlayer, category: string): number {
   935→  if (category === 'H2H') {
   936→    return player.selected_by * 0.4 + player.vice_captain_percentage * 0.35 + player.points * 0.25;
   937→  } else if (category === 'SL') {
   938→    return player.vice_captain_percentage * 0.35 + player.selected_by * 0.35 + player.points * 0.3;
   939→  } else {
   940→    // Mega GL: Wider VC rotation
   941→    return player.points * 0.35 + player.vice_captain_percentage * 0.3 + player.selected_by * 0.2 + player.captain_percentage * 0.15;
   942→  }
   943→}
   944→
   945→/**
   946→ * Select Captain and Vice-Captain based on category strategy.
   947→ * Mega GL: Rotates C/VC across team members (wider differential).
   948→ * SL: Picks the top 2 most reliable.
   949→ * H2H: Picks the safest 2.
   950→ */
   951→function selectCaptainViceCaptain(
   952→  team: TGPlayer[],
   953→  category: string,
   954→  teamIndex: number,
   955→): { captain: TGPlayer; viceCaptain: TGPlayer } {
   956→  const isMegaGL = category !== 'H2H' && category !== 'SL';
   957→
   958→  if (isMegaGL) {
   959→    // Mega GL: Sort by captain score, then rotate C/VC position based on teamIndex
   960→    // This gives wider C/VC distribution across teams
   961→    const sorted = [...team].sort((a, b) => scoreCaptain(b, category) - scoreCaptain(a, category));
   962→    // Rotate: sometimes pick #1 as C and #2 as VC, sometimes #1 as C and #3 as VC, etc.
   963→    const cIdx = 0; // Always pick top scorer as C
   964→    let vcIdx = 1 + (teamIndex % Math.min(4, sorted.length - 1)); // Rotate VC among top 2-5
   965→    if (vcIdx >= sorted.length) vcIdx = 1;
   966→    return { captain: sorted[cIdx], viceCaptain: sorted[vcIdx] };
   967→  } else {
   968→    // H2H and SL: Pick top 2 by category-specific scoring
   969→    const sorted = [...team].sort((a, b) => scoreCaptain(b, category) - scoreCaptain(a, category));
   970→    return { captain: sorted[0], viceCaptain: sorted.length > 1 ? sorted[1] : sorted[0] };
   971→  }
   972→}

  1797→// ============ Auto Select for Extra Team Generation ============
  1798→// Uses real player data (selection %, captain %, points, credits, role, team balance)
  1799→// to intelligently pick 8 FIX, 5 C, and 5 VC
  1800→
  1801→export interface AutoSelectResult {
  1802→  fixedPlayers: TGPlayer[];
  1803→  captainOptions: TGPlayer[];
  1804→  viceCaptainOptions: TGPlayer[];
  1805→}
  1806→
  1807→// Player scoring for auto-selection: higher = better pick
  1808→function playerScore(p: TGPlayer): number {
  1809→  return (
  1810→    p.selected_by * 0.35 +       // selection % - most important for GL
  1811→    p.captain_percentage * 0.25 + // captain % - indicates trust
  1812→    p.points * 0.2 +             // points - performance indicator
  1813→    p.vice_captain_percentage * 0.1 + // VC %
  1814→    (p.playing === 1 ? 10 : 0) + // confirmed playing bonus
  1815→    (p.role === PLAYER_ROLES.ALL_ROUNDER ? 3 : 0) + // AR bonus (flexible)
  1816→    (p.role === PLAYER_ROLES.WICKET_KEEPER ? 1 : 0) // WK slight bonus
  1817→  );
  1818→}
  1819→
  1820→export function autoSelectExtraPlayers(
  1821→  leftPlayers: TGPlayer[],
  1822→  rightPlayers: TGPlayer[],
  1823→  avoidPlayerIds: Set<number> = new Set(),
  1824→): AutoSelectResult {
  1825→  const allPlayersRaw = [...leftPlayers, ...rightPlayers];
  1826→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
  1827→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
  1828→
  1829→  // Use lineup-aware eligibility filter
  1830→  const eligible = getEligiblePlayers(allPlayersRaw, avoidPlayerIds);
  1831→
  1832→  const eligibleLeft = eligible.filter(p => p.team_name === leftTeamName);
  1833→  const eligibleRight = eligible.filter(p => p.team_name === rightTeamName);
  1834→
  1835→  // Sort by score descending
  1836→  const sortByScore = (a: TGPlayer, b: TGPlayer) => playerScore(b) - playerScore(a);
  1837→  const sortedLeft = [...eligibleLeft].sort(sortByScore);
  1838→  const sortedRight = [...eligibleRight].sort(sortByScore);
  1839→
  1840→  // === SELECT 8 FIX PLAYERS ===
  1841→  // Strategy: balanced 4-5 from each team, role-diverse
  1842→  const fixedPlayers: TGPlayer[] = [];
  1843→  const fixedIds = new Set<number>();
  1844→
  1845→  const targetLeft = Math.min(4, sortedLeft.length);
  1846→  const targetRight = Math.min(4, sortedRight.length);
  1847→
  1848→  // Pick diverse players ensuring role coverage
  1849→  const pickDiverse = (pool: TGPlayer[], count: number, existing: TGPlayer[]): TGPlayer[] => {
  1850→    const picked: TGPlayer[] = [];
  1851→    const usedIds = new Set(existing.map(p => p.pl_id));
  1852→    const existingRoles = {
  1853→      wk: existing.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length,
  1854→      bat: existing.filter(p => p.role === PLAYER_ROLES.BATSMAN).length,
  1855→      ar: existing.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length,
  1856→      bowl: existing.filter(p => p.role === PLAYER_ROLES.BOWLER).length,
  1857→    };
  1858→
  1859→    // First pass: fill role gaps
  1860→    const roleNeeds = [
  1861→      { role: PLAYER_ROLES.WICKET_KEEPER, need: Math.max(0, 1 - existingRoles.wk), pool: pool.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER && !usedIds.has(p.pl_id)) },
  1862→      { role: PLAYER_ROLES.ALL_ROUNDER, need: Math.max(0, 1 - existingRoles.ar), pool: pool.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER && !usedIds.has(p.pl_id)) },
  1863→      { role: PLAYER_ROLES.BATSMAN, need: Math.max(0, 2 - existingRoles.bat), pool: pool.filter(p => p.role === PLAYER_ROLES.BATSMAN && !usedIds.has(p.pl_id)) },
  1864→      { role: PLAYER_ROLES.BOWLER, need: Math.max(0, 2 - existingRoles.bowl), pool: pool.filter(p => p.role === PLAYER_ROLES.BOWLER && !usedIds.has(p.pl_id)) },
  1865→    ];
  1866→
  1867→    for (const rn of roleNeeds) {
  1868→      for (let i = 0; i < rn.need && picked.length < count; i++) {
  1869→        const candidate = rn.pool.find(p => !picked.some(pp => pp.pl_id === p.pl_id));
  1870→        if (candidate) picked.push(candidate);
  1871→      }
  1872→    }
  1873→
  1874→    // Second pass: fill remaining with top-scored players
  1875→    const remaining = pool.filter(p => !usedIds.has(p.pl_id) && !picked.some(pp => pp.pl_id === p.pl_id));
  1876→    for (const p of remaining) {
  1877→      if (picked.length >= count) break;
  1878→      picked.push(p);
  1879→    }
  1880→
  1881→    return picked;
  1882→  };
  1883→
  1884→  const leftPicks = pickDiverse(sortedLeft, targetLeft, []);
  1885→  for (const p of leftPicks) { fixedPlayers.push(p); fixedIds.add(p.pl_id); }
  1886→
  1887→  const rightPicks = pickDiverse(sortedRight, targetRight, fixedPlayers);
  1888→  for (const p of rightPicks) { fixedPlayers.push(p); fixedIds.add(p.pl_id); }
  1889→
  1890→  // Fill to 8 if needed
  1891→  if (fixedPlayers.length < 8) {
  1892→    const remaining = eligible.filter(p => !fixedIds.has(p.pl_id)).sort(sortByScore);
  1893→    for (const p of remaining) {
  1894→      if (fixedPlayers.length >= 8) break;
  1895→      fixedPlayers.push(p); fixedIds.add(p.pl_id);
  1896→    }
  1897→  }
  1898→
  1899→  // Trim to 8 if over
  1900→  while (fixedPlayers.length > 8) {
  1901→    let minIdx = 0; let minScore = playerScore(fixedPlayers[0]);
  1902→    for (let i = 1; i < fixedPlayers.length; i++) {
  1903→      const s = playerScore(fixedPlayers[i]);
  1904→      if (s < minScore) { minScore = s; minIdx = i; }
  1905→    }
  1906→    const removed = fixedPlayers.splice(minIdx, 1)[0];
  1907→    fixedIds.delete(removed.pl_id);
  1908→  }
  1909→
  1910→  // === SELECT 5 CAPTAIN OPTIONS ===
  1911→  const cCandidates = eligible
  1912→    .filter(p => !avoidPlayerIds.has(p.pl_id))
  1913→    .sort((a, b) => {
  1914→      const scoreA = a.captain_percentage * 0.4 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
  1915→      const scoreB = b.captain_percentage * 0.4 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
  1916→      return scoreB - scoreA;
  1917→    })
  1918→    .slice(0, 5);
  1919→
  1920→  // === SELECT 5 VICE CAPTAIN OPTIONS ===
  1921→  const vcCandidates: TGPlayer[] = [];
  1922→  const vcPool = eligible
  1923→    .filter(p => !avoidPlayerIds.has(p.pl_id))
  1924→    .sort((a, b) => {
  1925→      const scoreA = a.vice_captain_percentage * 0.35 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
  1926→      const scoreB = b.vice_captain_percentage * 0.35 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
  1927→      return scoreB - scoreA;
  1928→    });
  1929→
  1930→  for (const p of vcPool) {
  1931→    if (vcCandidates.length >= 5) break;
  1932→    vcCandidates.push(p);
  1933→  }
  1934→
  1935→  return {
  1936→    fixedPlayers: fixedPlayers.slice(0, 8),
  1937→    captainOptions: cCandidates.slice(0, 5),
  1938→    viceCaptainOptions: vcCandidates.slice(0, 5),
  1939→  };
  1940→}
  1941→
  1942→// Auto-replace a removed player in a list with the next best eligible player
  1943→export function autoReplacePlayer(
  1944→  removedPlayer: TGPlayer,
  1945→  currentList: TGPlayer[],
  1946→  allPlayers: TGPlayer[],
  1947→  avoidPlayerIds: Set<number>,
  1948→  scoringFn: (p: TGPlayer) => number = playerScore,
  1949→): TGPlayer[] {
  1950→  const currentIds = new Set(currentList.map(p => p.pl_id));
  1951→  const eligible = allPlayers.filter(p =>
  1952→    !currentIds.has(p.pl_id) && !avoidPlayerIds.has(p.pl_id) && p.pl_id !== removedPlayer.pl_id
  1953→  );
  1954→  const sorted = eligible.sort((a, b) => scoringFn(b) - scoringFn(a));
  1955→  const replacement = sorted[0];
  1956→  if (replacement) return [...currentList, replacement];
  1957→  return currentList;
  1958→}
