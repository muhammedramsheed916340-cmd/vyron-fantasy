   112→function decryptField(encrypted: string): string {
   113→  try {
   114→    const bytes = CryptoJS.AES.decrypt(encrypted, ENCRYPTION_KEY);
   115→    const decrypted = bytes.toString(CryptoJS.enc.Utf8);
   116→    if (!decrypted) return encrypted;
   117→    return decrypted;
   118→  } catch {
   119→    return encrypted;
   120→  }
   121→}

   308→// ============ Team Generation Algorithm ============

   310→export interface GeneratedTeam {
   311→  id: number;
   312→  captain: TGPlayer;
   313→  viceCaptain: TGPlayer;
   314→  players: TGPlayer[];
   315→}
   316→
   317→// Cricket team constraints
   318→const CRICKET_TEAM_SIZE = 11;
   319→const MIN_WK = 1;
   320→const MAX_WK = 4;
   321→const MIN_BAT = 3;
   322→const MAX_BAT = 6;
   323→const MIN_AR = 1;
   324→const MAX_AR = 4;
   325→const MIN_BOWL = 3;
   326→const MAX_BOWL = 6;
   327→const MAX_CREDITS = 100;
   328→const MAX_FROM_ONE_TEAM = 7;
   329→
   330→function seededRandom(seed: number): () => number {
   331→  let s = seed;
   332→  return () => {
   333→    s = (s * 16807 + 0) % 2147483647;
   334→    return (s - 1) / 2147483646;
   335→  };
   336→}
   337→
   338→function shuffleArray<T>(arr: T[], rng: () => number): T[] {
   339→  const result = [...arr];
   340→  for (let i = result.length - 1; i > 0; i--) {
   341→    const j = Math.floor(rng() * (i + 1));
   342→    [result[i], result[j]] = [result[j], result[i]];
   343→  }
   344→  return result;
   345→}

   347→export function generateTeams(
   348→  leftPlayers: TGPlayer[],
   349→  rightPlayers: TGPlayer[],
   350→  category: string,
   351→  count: number,
   352→  seed: number = Date.now()
   353→): GeneratedTeam[] {
   354→  const allPlayers = [...leftPlayers, ...rightPlayers];
   355→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
   356→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
   357→
   358→  // Separate by role
   359→  const wk = allPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER);
   360→  const bat = allPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN);
   361→  const ar = allPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER);
   362→  const bowl = allPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER);
   363→
   364→  const teams: GeneratedTeam[] = [];
   365→  const rng = seededRandom(seed);
   366→
   367→  // Category-specific strategies
   368→  const isH2H = category === 'H2H';
   369→  const isSL = category === 'SL';
   370→
   371→  for (let t = 0; t < count; t++) {
   372→    let attempts = 0;
   373→    let team: TGPlayer[] | null = null;
   374→
   375→    while (!team && attempts < 200) {
   376→      attempts++;
   377→      const sr = seededRandom(seed + t * 1000 + attempts);
   378→
   379→      // Determine role counts based on category
   380→      let wkCount, batCount, arCount, bowlCount;
   381→
   382→      if (isH2H) {
   383→        // H2H: More conservative, popular players
   384→        wkCount = MIN_WK + (sr() > 0.5 ? 1 : 0);
   385→        arCount = MIN_AR + (sr() > 0.6 ? 1 : 0);
   386→        batCount = Math.max(MIN_BAT, 5 - wkCount + (sr() > 0.5 ? 1 : 0));
   387→        bowlCount = CRICKET_TEAM_SIZE - wkCount - batCount - arCount;
   388→      } else if (isSL) {
   389→        // SL: Balanced
   390→        wkCount = 1 + (sr() > 0.7 ? 1 : 0);
   391→        arCount = 2 + (sr() > 0.5 ? 1 : 0);
   392→        batCount = 3 + (sr() > 0.5 ? 1 : 0);
   393→        bowlCount = CRICKET_TEAM_SIZE - wkCount - batCount - arCount;
   394→      } else {
   395→        // Mega GL: More differential picks, varied combinations
   396→        const patterns = [
   397→          [1, 3, 2, 5], [1, 4, 1, 5], [1, 3, 3, 4], [1, 4, 2, 4],
   398→          [2, 3, 2, 4], [1, 5, 1, 4], [2, 3, 1, 5], [1, 4, 3, 3],
   399→          [2, 4, 1, 4], [1, 3, 1, 6], [2, 2, 3, 4], [1, 5, 2, 3],
   400→        ];
   401→        const pattern = patterns[Math.floor(sr() * patterns.length)];
   402→        [wkCount, batCount, arCount, bowlCount] = pattern;
   403→      }
   404→
   405→      // Validate counts
   406→      if (wkCount < MIN_WK || wkCount > MAX_WK) continue;
   407→      if (batCount < MIN_BAT || batCount > MAX_BAT) continue;
   408→      if (arCount < MIN_AR || arCount > MAX_AR) continue;
   409→      if (bowlCount < MIN_BOWL || bowlCount > MAX_BOWL) continue;
   410→      if (wkCount + batCount + arCount + bowlCount !== CRICKET_TEAM_SIZE) continue;
   411→
   412→      // Pick players for each role
   413→      const shuffledWK = shuffleArray(wk, sr);
   414→      const shuffledBat = shuffleArray(bat, sr);
   415→      const shuffledAR = shuffleArray(ar, sr);
   416→      const shuffledBowl = shuffleArray(bowl, sr);
   417→
   418→      // For Mega GL, sort by selected_by sometimes to get differential picks
   419→      let selectedWK, selectedBat, selectedAR, selectedBowl;
   420→
   421→      if (!isH2H && sr() > 0.4) {
   422→        // Sort some roles by selection % to mix popular and differential
   423→        const sortFn = (a: TGPlayer, b: TGPlayer) => a.selected_by - b.selected_by;
   424→        selectedWK = shuffleArray([...wk].sort(sortFn), sr).slice(0, wkCount);
   425→        selectedBat = shuffleArray([...bat].sort(sortFn), sr).slice(0, batCount);
   426→        selectedAR = shuffleArray([...ar].sort(sortFn), sr).slice(0, arCount);
   427→        selectedBowl = shuffleArray([...bowl].sort(sortFn), sr).slice(0, bowlCount);
   428→      } else if (isH2H) {
   429→        // H2H: Pick most selected players
   430→        const sortFn = (a: TGPlayer, b: TGPlayer) => b.selected_by - a.selected_by;
   431→        selectedWK = [...wk].sort(sortFn).slice(0, wkCount);
   432→        selectedBat = shuffleArray([...bat].sort(sortFn), sr).slice(0, batCount);
   433→        selectedAR = [...ar].sort(sortFn).slice(0, arCount);
   434→        selectedBowl = shuffleArray([...bowl].sort(sortFn), sr).slice(0, bowlCount);
   435→      } else {
   436→        selectedWK = shuffledWK.slice(0, wkCount);
   437→        selectedBat = shuffledBat.slice(0, batCount);
   438→        selectedAR = shuffledAR.slice(0, arCount);
   439→        selectedBowl = shuffledBowl.slice(0, bowlCount);
   440→      }
   441→
   442→      // Check if we have enough players
   443→      if (selectedWK.length < wkCount || selectedBat.length < batCount ||
   444→          selectedAR.length < arCount || selectedBowl.length < bowlCount) continue;
   445→
   446→      const selected = [...selectedWK, ...selectedBat, ...selectedAR, ...selectedBowl];
   447→
   448→      // Check total credits
   449→      const totalCredits = selected.reduce((sum, p) => sum + p.credits, 0);
   450→      if (totalCredits > MAX_CREDITS) continue;
   451→
   452→      // Check team distribution (max 7 from one team)
   453→      const leftCount = selected.filter(p => p.team_name === leftTeamName).length;
   454→      const rightCount = selected.filter(p => p.team_name === rightTeamName).length;
   455→      if (leftCount > MAX_FROM_ONE_TEAM || rightCount > MAX_FROM_ONE_TEAM) continue;
   456→
   457→      // Ensure at least 1 from each team
   458→      if (leftCount < 1 || rightCount < 1) continue;
   459→
   460→      team = selected;
   461→    }
   462→
   463→    if (team) {
   464→      // Select Captain & Vice-Captain
   465→      // Sort by a combination of points and selection % for C/VC
   466→      const sorted = [...team].sort((a, b) =>
   467→        (b.points * 0.4 + b.selected_by * 0.3 + b.captain_percentage * 0.3) -
   468→        (a.points * 0.4 + a.selected_by * 0.3 + a.captain_percentage * 0.3)
   469→      );
   470→
   471→      const captain = sorted[0];
   472→      const viceCaptain = sorted.length > 1 ? sorted[1] : sorted[0];
   473→
   474→      teams.push({
   475→        id: t + 1,
   476→        captain,
   477→        viceCaptain,
   478→        players: team,
   479→      });
   480→    }
   481→  }
   482→
   483→  return teams;
   484→}

   490→export interface ExtraTeamGenInput {
   491→  fixedPlayers: TGPlayer[];      // Exactly 8 fixed players
   492→  captainOptions: TGPlayer[];    // Exactly 5 captain options
   493→  viceCaptainOptions: TGPlayer[]; // Exactly 5 vice captain options
   494→  leftPlayers: TGPlayer[];       // Full left team player pool
   495→  rightPlayers: TGPlayer[];      // Full right team player pool
   496→  category: string;              // 'Mega GL' | 'SL' | 'H2H'
   497→  count: number;                 // Number of teams to generate
   498→  seed?: number;

   501→export function generateExtraTeams(input: ExtraTeamGenInput): GeneratedTeam[] {
   502→  const {
   503→    fixedPlayers,
   504→    captainOptions,
   505→    viceCaptainOptions,
   506→    leftPlayers,
   507→    rightPlayers,
   508→    category,
   509→    count,
   510→    seed = Date.now(),
   511→  } = input;
   512→
   513→  const allPlayers = [...leftPlayers, ...rightPlayers];
   514→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
   515→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
   516→
   517→  // Pool of players excluding the 8 fixed ones (use pl_id for identity)
   518→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   519→  const remainingPool = allPlayers.filter(p => !fixedIds.has(p.pl_id));
   520→
   521→  // Separate remaining pool by role
   522→  const remWK = remainingPool.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER);
   523→  const remBat = remainingPool.filter(p => p.role === PLAYER_ROLES.BATSMAN);
   524→  const remAR = remainingPool.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER);
   525→  const remBowl = remainingPool.filter(p => p.role === PLAYER_ROLES.BOWLER);
   526→
   527→  // Count fixed players by role
   528→  const fixedWKCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   529→  const fixedBatCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   530→  const fixedARCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   531→  const fixedBowlCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   532→
   533→  const fixedCredits = fixedPlayers.reduce((sum, p) => sum + p.credits, 0);
   534→  const remainingCredits = MAX_CREDITS - fixedCredits;
   535→  const neededPlayers = CRICKET_TEAM_SIZE - fixedPlayers.length; // 3
   536→
   537→  const teams: GeneratedTeam[] = [];
   538→
   539→  // Pre-validate: fixed players must have credits <= 100
   540→  if (fixedCredits > MAX_CREDITS) return teams;
   541→
   542→  // Pre-validate: fixed players must respect max 7 from one team
   543→  const fixedLeftCount = fixedPlayers.filter(p => p.team_name === leftTeamName).length;
   544→  const fixedRightCount = fixedPlayers.filter(p => p.team_name === rightTeamName).length;
   545→  if (fixedLeftCount > MAX_FROM_ONE_TEAM || fixedRightCount > MAX_FROM_ONE_TEAM) return teams;
   546→
   547→  // Generate all valid C/VC combinations from the 5C and 5VC options
   548→  // C must be in the team, VC must be in the team, C ≠ VC
   549→  const cVCCombos: { c: TGPlayer; vc: TGPlayer }[] = [];
   550→  for (const c of captainOptions) {
   551→    for (const vc of viceCaptainOptions) {
   552→      if (c.pl_id !== vc.pl_id) {
   553→        cVCCombos.push({ c, vc });
   554→      }
   555→    }
   556→  }
   557→  // If no valid combos, return empty
   558→  if (cVCCombos.length === 0) return teams;
   559→
   560→  // Category-specific role distribution preferences for remaining players
   561→  const isH2H = category === 'H2H';
   562→  const isSL = category === 'SL';
   563→
   564→  // Pre-compute all valid target role distributions that are compatible with fixed players
   565→  // This avoids wasting attempts on patterns that can never work
   566→  const categoryPatterns: number[][] = isH2H
   567→    ? [] // H2H uses dynamic generation below
   568→    : isSL
   569→    ? [] // SL uses dynamic generation below
   570→    : [
   571→        [1, 3, 2, 5], [1, 4, 1, 5], [1, 3, 3, 4], [1, 4, 2, 4],
   572→        [2, 3, 2, 4], [1, 5, 1, 4], [2, 3, 1, 5], [1, 4, 3, 3],
   573→        [2, 4, 1, 4], [1, 3, 1, 6], [2, 2, 3, 4], [1, 5, 2, 3],
   574→      ];
   575→
   576→  // Build list of valid target distributions: [WK, BAT, AR, BOWL]
   577→  const validDistributions: number[][] = [];
   578→  for (let wk = MIN_WK; wk <= MAX_WK; wk++) {
   579→    for (let bat = MIN_BAT; bat <= MAX_BAT; bat++) {
   580→      for (let ar = MIN_AR; ar <= MAX_AR; ar++) {
   581→        const bowl = CRICKET_TEAM_SIZE - wk - bat - ar;
   582→        if (bowl < MIN_BOWL || bowl > MAX_BOWL) continue;
   583→        // Check compatibility with fixed players
   584→        const nWK = Math.max(0, wk - fixedWKCount);
   585→        const nBat = Math.max(0, bat - fixedBatCount);
   586→        const nAR = Math.max(0, ar - fixedARCount);
   587→        const nBowl = Math.max(0, bowl - fixedBowlCount);
   588→        if (nWK + nBat + nAR + nBowl !== neededPlayers) continue;
   589→        // Check we have enough remaining players for each role
   590→        if (remWK.length < nWK || remBat.length < nBat || remAR.length < nAR || remBowl.length < nBowl) continue;
   591→        // For Mega GL / SL / H2H: prefer category-specific patterns but include all valid ones
   592→        validDistributions.push([wk, bat, ar, bowl]);
   593→      }
   594→    }
   595→  }
   596→
   597→  // Sort: prefer category-specific patterns first, then others
   598→  if (categoryPatterns.length > 0) {
   599→    const patternSet = new Set(categoryPatterns.map(p => p.join(',')));
   600→    validDistributions.sort((a, b) => {
   601→      const aInCat = patternSet.has(a.join(',')) ? 0 : 1;
   602→      const bInCat = patternSet.has(b.join(',')) ? 0 : 1;
   603→      return aInCat - bInCat;
   604→    });
   605→  }
   606→
   607→  if (validDistributions.length === 0) {
   608→    return teams;
   609→  }
   610→
   611→  // Team signature function for uniqueness check across generated teams
   612→  // Uses sorted player IDs + C/VC IDs to create a unique fingerprint
   613→  const teamSignatures = new Set<string>();
   614→  const makeTeamSignature = (players: TGPlayer[], c: TGPlayer, vc: TGPlayer): string => {
   615→    const playerIds = players.map(p => p.pl_id).sort((a, b) => a - b).join(',');
   616→    return `${playerIds}|C${c.pl_id}|VC${vc.pl_id}`;
   617→  };
   618→
   619→  // Generation loop with retry: keep trying until we have `count` unique teams
   620→  // or we exhaust the maximum total attempts
   621→  const MAX_TOTAL_ATTEMPTS = count * 500; // generous budget
   622→  let totalAttempts = 0;
   623→
   624→  for (let t = 0; t < count && totalAttempts < MAX_TOTAL_ATTEMPTS; t++) {
   625→    let attempts = 0;
   626→    let team: TGPlayer[] | null = null;
   627→    let teamC: TGPlayer | null = null;
   628→    let teamVC: TGPlayer | null = null;
   629→
   630→    while (!team && attempts < 500) {
   631→      attempts++;
   632→      totalAttempts++;
   633→      const sr = seededRandom(seed + t * 1000 + attempts);
   634→
   635→      // Pick a valid target distribution
   636→      let targetWK: number, targetBat: number, targetAR: number, targetBowl: number;
   637→
   638→      if (validDistributions.length === 1) {
   639→        [targetWK, targetBat, targetAR, targetBowl] = validDistributions[0];
   640→      } else {
   641→        // Bias towards category-preferred patterns (first half of sorted list)
   642→        const idx = sr() < 0.7
   643→          ? Math.floor(sr() * Math.ceil(validDistributions.length / 2)) // 70% chance from preferred half
   644→          : Math.floor(sr() * validDistributions.length); // 30% from any
   645→        [targetWK, targetBat, targetAR, targetBowl] = validDistributions[Math.min(idx, validDistributions.length - 1)];
   646→      }
   647→
   648→      // How many more of each role do we need?
   649→      const needWK = Math.max(0, targetWK - fixedWKCount);
   650→      const needBat = Math.max(0, targetBat - fixedBatCount);
   651→      const needAR = Math.max(0, targetAR - fixedARCount);
   652→      const needBowl = Math.max(0, targetBowl - fixedBowlCount);
   653→
   654→      // Check if we have enough remaining players for each role
   655→      if (remWK.length < needWK || remBat.length < needBat ||
   656→          remAR.length < needAR || remBowl.length < needBowl) continue;
   657→
   658→      // Pick remaining players for each role
   659→      let pickedWK: TGPlayer[], pickedBat: TGPlayer[], pickedAR: TGPlayer[], pickedBowl: TGPlayer[];
   660→
   661→      if (!isH2H && sr() > 0.4) {
   662→        const sortFn = (a: TGPlayer, b: TGPlayer) => a.selected_by - b.selected_by;
   663→        pickedWK = shuffleArray([...remWK].sort(sortFn), sr).slice(0, needWK);
   664→        pickedBat = shuffleArray([...remBat].sort(sortFn), sr).slice(0, needBat);
   665→        pickedAR = shuffleArray([...remAR].sort(sortFn), sr).slice(0, needAR);
   666→        pickedBowl = shuffleArray([...remBowl].sort(sortFn), sr).slice(0, needBowl);
   667→      } else if (isH2H) {
   668→        const sortFn = (a: TGPlayer, b: TGPlayer) => b.selected_by - a.selected_by;
   669→        pickedWK = [...remWK].sort(sortFn).slice(0, needWK);
   670→        pickedBat = shuffleArray([...remBat].sort(sortFn), sr).slice(0, needBat);
   671→        pickedAR = [...remAR].sort(sortFn).slice(0, needAR);
   672→        pickedBowl = shuffleArray([...remBowl].sort(sortFn), sr).slice(0, needBowl);
   673→      } else {
   674→        pickedWK = shuffleArray(remWK, sr).slice(0, needWK);
   675→        pickedBat = shuffleArray(remBat, sr).slice(0, needBat);
   676→        pickedAR = shuffleArray(remAR, sr).slice(0, needAR);
   677→        pickedBowl = shuffleArray(remBowl, sr).slice(0, needBowl);
   678→      }
   679→
   680→      if (pickedWK.length < needWK || pickedBat.length < needBat ||
   681→          pickedAR.length < needAR || pickedBowl.length < needBowl) continue;
   682→
   683→      const remaining = [...pickedWK, ...pickedBat, ...pickedAR, ...pickedBowl];
   684→      const fullTeam = [...fixedPlayers, ...remaining];
   685→
   686→      // Check no duplicates (shouldn't happen but safety check)
   687→      const uniqueIds = new Set(fullTeam.map(p => p.pl_id));
   688→      if (uniqueIds.size !== fullTeam.length) continue;
   689→
   690→      // Check total credits
   691→      const totalCredits = fullTeam.reduce((sum, p) => sum + p.credits, 0);
   692→      if (totalCredits > MAX_CREDITS) continue;
   693→
   694→      // Check team distribution
   695→      const leftCount = fullTeam.filter(p => p.team_name === leftTeamName).length;
   696→      const rightCount = fullTeam.filter(p => p.team_name === rightTeamName).length;
   697→      if (leftCount > MAX_FROM_ONE_TEAM || rightCount > MAX_FROM_ONE_TEAM) continue;
   698→      if (leftCount < 1 || rightCount < 1) continue;
   699→
   700→      // Select C/VC combination for this team
   701→      // Round-robin through the valid combos, also try to ensure C and VC are in the team
   702→      const comboIndex = (t + attempts) % cVCCombos.length;
   703→      let combo = cVCCombos[comboIndex];
   704→
   705→      // Ensure C and VC are in this team
   706→      const teamPlayerIds = new Set(fullTeam.map(p => p.pl_id));
   707→      if (!teamPlayerIds.has(combo.c.pl_id) || !teamPlayerIds.has(combo.vc.pl_id)) {
   708→        // Try to find a valid combo where both C and VC are in the team
   709→        let foundValidCombo = false;
   710→        for (let ci = 0; ci < cVCCombos.length; ci++) {
   711→          const tryCombo = cVCCombos[(comboIndex + ci) % cVCCombos.length];
   712→          if (teamPlayerIds.has(tryCombo.c.pl_id) && teamPlayerIds.has(tryCombo.vc.pl_id)) {
   713→            combo = tryCombo;
   714→            foundValidCombo = true;
   715→            break;
   716→          }
   717→        }
   718→        if (!foundValidCombo) {
   719→          // Fallback: use top players in team as C/VC
   720→          const sorted = [...fullTeam].sort((a, b) =>
   721→            (b.points * 0.4 + b.selected_by * 0.3 + b.captain_percentage * 0.3) -
   722→            (a.points * 0.4 + a.selected_by * 0.3 + a.captain_percentage * 0.3)
   723→          );
   724→          combo = { c: sorted[0], vc: sorted.length > 1 ? sorted[1] : sorted[0] };
   725→        }
   726→      }
   727→
   728→      // Uniqueness check: skip if this team signature already exists
   729→      const signature = makeTeamSignature(fullTeam, combo.c, combo.vc);
   730→      if (teamSignatures.has(signature)) continue;
   731→
   732→      team = fullTeam;
   733→      teamC = combo.c;
   734→      teamVC = combo.vc;
   735→      teamSignatures.add(signature);
   736→    }
   737→
   738→    if (team && teamC && teamVC) {
   739→      teams.push({
   740→        id: teams.length + 1,
   741→        captain: teamC,
   742→        viceCaptain: teamVC,
   743→        players: team,
   744→      });
   745→    } else {
   746→      // If we failed to generate a unique team for this slot,
   747→      // try extra attempts with different seeds before giving up
   748→      let extraAttempts = 0;
   749→      const MAX_EXTRA = 200;
   750→      while (teams.length < count && extraAttempts < MAX_EXTRA && totalAttempts < MAX_TOTAL_ATTEMPTS) {
   751→        extraAttempts++;
   752→        totalAttempts++;
   753→        const extraSeed = seed + count * 1000 + extraAttempts * 37;
   754→        const sr = seededRandom(extraSeed);
   755→
   756→        // Pick distribution
   757→        let targetWK: number, targetBat: number, targetAR: number, targetBowl: number;
   758→        if (validDistributions.length === 1) {
   759→          [targetWK, targetBat, targetAR, targetBowl] = validDistributions[0];
   760→        } else {
   761→          const idx = Math.floor(sr() * validDistributions.length);
   762→          [targetWK, targetBat, targetAR, targetBowl] = validDistributions[idx];
   763→        }
   764→
   765→        const needWK = Math.max(0, targetWK - fixedWKCount);
   766→        const needBat = Math.max(0, targetBat - fixedBatCount);
   767→        const needAR = Math.max(0, targetAR - fixedARCount);
   768→        const needBowl = Math.max(0, targetBowl - fixedBowlCount);
   769→
   770→        if (remWK.length < needWK || remBat.length < needBat ||
   771→            remAR.length < needAR || remBowl.length < needBowl) continue;
   772→
   773→        const sortFn = (a: TGPlayer, b: TGPlayer) => a.selected_by - b.selected_by;
   774→        const pickedWK = shuffleArray([...remWK].sort(sortFn), sr).slice(0, needWK);
   775→        const pickedBat = shuffleArray([...remBat].sort(sortFn), sr).slice(0, needBat);
   776→        const pickedAR = shuffleArray([...remAR].sort(sortFn), sr).slice(0, needAR);
   777→        const pickedBowl = shuffleArray([...remBowl].sort(sortFn), sr).slice(0, needBowl);
   778→
   779→        if (pickedWK.length < needWK || pickedBat.length < needBat ||
   780→            pickedAR.length < needAR || pickedBowl.length < needBowl) continue;
   781→
   782→        const remaining = [...pickedWK, ...pickedBat, ...pickedAR, ...pickedBowl];
   783→        const fullTeam = [...fixedPlayers, ...remaining];
   784→
   785→        const uniqueIds = new Set(fullTeam.map(p => p.pl_id));
   786→        if (uniqueIds.size !== fullTeam.length) continue;
   787→
   788→        const totalCredits = fullTeam.reduce((sum, p) => sum + p.credits, 0);
   789→        if (totalCredits > MAX_CREDITS) continue;
   790→
   791→        const leftCount = fullTeam.filter(p => p.team_name === leftTeamName).length;
   792→        const rightCount = fullTeam.filter(p => p.team_name === rightTeamName).length;
   793→        if (leftCount > MAX_FROM_ONE_TEAM || rightCount > MAX_FROM_ONE_TEAM) continue;
   794→        if (leftCount < 1 || rightCount < 1) continue;
   795→
   796→        // C/VC selection
   797→        const comboIndex = extraAttempts % cVCCombos.length;
   798→        let combo = cVCCombos[comboIndex];
   799→        const teamPlayerIds = new Set(fullTeam.map(p => p.pl_id));
   800→        if (!teamPlayerIds.has(combo.c.pl_id) || !teamPlayerIds.has(combo.vc.pl_id)) {
   801→          let foundValidCombo = false;
   802→          for (let ci = 0; ci < cVCCombos.length; ci++) {
   803→            const tryCombo = cVCCombos[(comboIndex + ci) % cVCCombos.length];
   804→            if (teamPlayerIds.has(tryCombo.c.pl_id) && teamPlayerIds.has(tryCombo.vc.pl_id)) {
   805→              combo = tryCombo;
   806→              foundValidCombo = true;
   807→              break;
   808→            }
   809→          }
   810→          if (!foundValidCombo) {
   811→            const sorted = [...fullTeam].sort((a, b) =>
   812→              (b.points * 0.4 + b.selected_by * 0.3 + b.captain_percentage * 0.3) -
   813→              (a.points * 0.4 + a.selected_by * 0.3 + a.captain_percentage * 0.3)
   814→            );
   815→            combo = { c: sorted[0], vc: sorted.length > 1 ? sorted[1] : sorted[0] };
   816→          }
   817→        }
   818→
   819→        const signature = makeTeamSignature(fullTeam, combo.c, combo.vc);
   820→        if (teamSignatures.has(signature)) continue;
   821→        teamSignatures.add(signature);
   822→
   823→        teams.push({
   824→          id: teams.length + 1,
   825→          captain: combo.c,
   826→          viceCaptain: combo.vc,
   827→          players: fullTeam,
   828→        });
   829→        break; // Successfully added a team, move on
   830→      }
   831→    }
   832→  }
   833→
   834→  return teams;
   835→}

   837→// ============ Auto Select for Extra Team Generation ============
   838→// Uses real player data (selection %, captain %, points, credits, role, team balance)
   839→// to intelligently pick 8 FIX, 5 C, and 5 VC
   840→
   841→export interface AutoSelectResult {
   842→  fixedPlayers: TGPlayer[];
   843→  captainOptions: TGPlayer[];
   844→  viceCaptainOptions: TGPlayer[];
   845→}
   846→
   847→// Player scoring for auto-selection: higher = better pick
   848→function playerScore(p: TGPlayer): number {
   849→  return (
   850→    p.selected_by * 0.35 +       // selection % - most important for GL
   851→    p.captain_percentage * 0.25 + // captain % - indicates trust
   852→    p.points * 0.2 +             // points - performance indicator
   853→    p.vice_captain_percentage * 0.1 + // VC %
   854→    (p.playing === 1 ? 10 : 0) + // confirmed playing bonus
   855→    (p.role === PLAYER_ROLES.ALL_ROUNDER ? 3 : 0) + // AR bonus (flexible)
   856→    (p.role === PLAYER_ROLES.WICKET_KEEPER ? 1 : 0) // WK slight bonus
   857→  );
   858→}
   859→
   860→export function autoSelectExtraPlayers(
   861→  leftPlayers: TGPlayer[],
   862→  rightPlayers: TGPlayer[],
   863→  avoidPlayerIds: Set<number> = new Set(),
   864→): AutoSelectResult {
   865→  const allPlayers = [...leftPlayers, ...rightPlayers];
   866→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
   867→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
   868→
   869→  // Filter out avoided players and unconfirmed players when lineup is out
   870→  const eligible = allPlayers.filter(p => {
   871→    if (avoidPlayerIds.has(p.pl_id)) return false;
   872→    if (p.playing === 0) {
   873→      const hasConfirmed = allPlayers.some(ap => ap.playing === 1);
   874→      if (hasConfirmed) return false;
   875→    }
   876→    return true;
   877→  });
   878→
   879→  const eligibleLeft = eligible.filter(p => p.team_name === leftTeamName);
   880→  const eligibleRight = eligible.filter(p => p.team_name === rightTeamName);
   881→
   882→  // Sort by score descending
   883→  const sortByScore = (a: TGPlayer, b: TGPlayer) => playerScore(b) - playerScore(a);
   884→  const sortedLeft = [...eligibleLeft].sort(sortByScore);
   885→  const sortedRight = [...eligibleRight].sort(sortByScore);
   886→
   887→  // === SELECT 8 FIX PLAYERS ===
   888→  // Strategy: balanced 4-5 from each team, role-diverse
   889→  const fixedPlayers: TGPlayer[] = [];
   890→  const fixedIds = new Set<number>();
   891→
   892→  const targetLeft = Math.min(4, sortedLeft.length);
   893→  const targetRight = Math.min(4, sortedRight.length);
   894→
   895→  // Pick diverse players ensuring role coverage
   896→  const pickDiverse = (pool: TGPlayer[], count: number, existing: TGPlayer[]): TGPlayer[] => {
   897→    const picked: TGPlayer[] = [];
   898→    const usedIds = new Set(existing.map(p => p.pl_id));
   899→    const existingRoles = {
   900→      wk: existing.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length,
   901→      bat: existing.filter(p => p.role === PLAYER_ROLES.BATSMAN).length,
   902→      ar: existing.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length,
   903→      bowl: existing.filter(p => p.role === PLAYER_ROLES.BOWLER).length,
   904→    };
   905→
   906→    // First pass: fill role gaps
   907→    const roleNeeds = [
   908→      { role: PLAYER_ROLES.WICKET_KEEPER, need: Math.max(0, 1 - existingRoles.wk), pool: pool.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER && !usedIds.has(p.pl_id)) },
   909→      { role: PLAYER_ROLES.ALL_ROUNDER, need: Math.max(0, 1 - existingRoles.ar), pool: pool.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER && !usedIds.has(p.pl_id)) },
   910→      { role: PLAYER_ROLES.BATSMAN, need: Math.max(0, 2 - existingRoles.bat), pool: pool.filter(p => p.role === PLAYER_ROLES.BATSMAN && !usedIds.has(p.pl_id)) },
   911→      { role: PLAYER_ROLES.BOWLER, need: Math.max(0, 2 - existingRoles.bowl), pool: pool.filter(p => p.role === PLAYER_ROLES.BOWLER && !usedIds.has(p.pl_id)) },
   912→    ];
   913→
   914→    for (const rn of roleNeeds) {
   915→      for (let i = 0; i < rn.need && picked.length < count; i++) {
   916→        const candidate = rn.pool.find(p => !picked.some(pp => pp.pl_id === p.pl_id));
   917→        if (candidate) picked.push(candidate);
   918→      }
   919→    }
   920→
   921→    // Second pass: fill remaining with top-scored players
   922→    const remaining = pool.filter(p => !usedIds.has(p.pl_id) && !picked.some(pp => pp.pl_id === p.pl_id));
   923→    for (const p of remaining) {
   924→      if (picked.length >= count) break;
   925→      picked.push(p);
   926→    }
   927→
   928→    return picked;
   929→  };
   930→
   931→  const leftPicks = pickDiverse(sortedLeft, targetLeft, []);
   932→  for (const p of leftPicks) { fixedPlayers.push(p); fixedIds.add(p.pl_id); }
   933→
   934→  const rightPicks = pickDiverse(sortedRight, targetRight, fixedPlayers);
   935→  for (const p of rightPicks) { fixedPlayers.push(p); fixedIds.add(p.pl_id); }
   936→
   937→  // Fill to 8 if needed
   938→  if (fixedPlayers.length < 8) {
   939→    const remaining = eligible.filter(p => !fixedIds.has(p.pl_id)).sort(sortByScore);
   940→    for (const p of remaining) {
   941→      if (fixedPlayers.length >= 8) break;
   942→      fixedPlayers.push(p); fixedIds.add(p.pl_id);
   943→    }
   944→  }
   945→
   946→  // Trim to 8 if over
   947→  while (fixedPlayers.length > 8) {
   948→    let minIdx = 0; let minScore = playerScore(fixedPlayers[0]);
   949→    for (let i = 1; i < fixedPlayers.length; i++) {
   950→      const s = playerScore(fixedPlayers[i]);
   951→      if (s < minScore) { minScore = s; minIdx = i; }
   952→    }
   953→    const removed = fixedPlayers.splice(minIdx, 1)[0];
   954→    fixedIds.delete(removed.pl_id);
   955→  }
   956→
   957→  // === SELECT 5 CAPTAIN OPTIONS ===
   958→  const cCandidates = eligible
   959→    .filter(p => !avoidPlayerIds.has(p.pl_id))
   960→    .sort((a, b) => {
   961→      const scoreA = a.captain_percentage * 0.4 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
   962→      const scoreB = b.captain_percentage * 0.4 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
   963→      return scoreB - scoreA;
   964→    })
   965→    .slice(0, 5);
   966→
   967→  // === SELECT 5 VICE CAPTAIN OPTIONS ===
   968→  const vcCandidates: TGPlayer[] = [];
   969→  const vcPool = eligible
   970→    .filter(p => !avoidPlayerIds.has(p.pl_id))
   971→    .sort((a, b) => {
   972→      const scoreA = a.vice_captain_percentage * 0.35 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
   973→      const scoreB = b.vice_captain_percentage * 0.35 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
   974→      return scoreB - scoreA;
   975→    });
   976→
   977→  for (const p of vcPool) {
   978→    if (vcCandidates.length >= 5) break;
   979→    vcCandidates.push(p);
   980→  }
   981→
   982→  return {
   983→    fixedPlayers: fixedPlayers.slice(0, 8),
   984→    captainOptions: cCandidates.slice(0, 5),
   985→    viceCaptainOptions: vcCandidates.slice(0, 5),
   986→  };
   987→}

   989→// Auto-replace a removed player in a list with the next best eligible player
   990→export function autoReplacePlayer(
   991→  removedPlayer: TGPlayer,
   992→  currentList: TGPlayer[],
