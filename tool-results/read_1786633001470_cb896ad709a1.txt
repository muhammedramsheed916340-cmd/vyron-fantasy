   235→  const handleSessionExpired = useCallback(async () => {

   249→    if (!acc?.mobileNumber) {

   255→      });

   259→    console.log('[JOIN] Session expired — USER initiated refresh for', platform, acc.mobileNumber.slice(0, 4) + '****');

   261→

   263→    otpRequestInProgressRef.current = true;
   264→    lastOtpTimestampRef.current = Date.now();

   266→    // Step 1: Send OTP

   270→      const refreshResult = await initiateSessionRefresh(platform, acc.mobileNumber);

   275→          active: true,
   276→          step: 'failed',
   277→          message: refreshResult.message || 'Failed to send OTP. Please try again.',
   278→        });
   279→        otpRequestInProgressRef.current = false;
   280→        return;
   281→      }
   282→
   283→      // Step 2: Wait for user to enter OTP
   284→      console.log('[JOIN] OTP sent — waiting for user to enter OTP');
   285→      setSessionRefreshState({
   286→        active: true,
   287→        step: 'entering_otp',
   288→        otpState: refreshResult.state,
   289→        challenge: refreshResult.challenge,
   290→        reasonCode: refreshResult.reasonCode,
   291→        message: `OTP sent to ${acc.mobileNumber}. Enter the OTP to refresh your session.`,
   292→      });
   293→    } catch (err) {
   294→      console.error('[JOIN] Session refresh error:', err);
   295→      setSessionRefreshState({
   296→        active: true,
   297→        step: 'failed',
   298→        message: 'Unexpected error during session refresh.',
   299→      });
   300→    } finally {
   301→      // Clear the in-progress flag after sending phase (the verify phase is separate)
   302→      // We keep it set during entering_otp to prevent double-send
   303→      // It will be cleared after verify completes or user cancels
   304→    }
   305→  }, [platform, effectiveAccount, OTP_COOLDOWN_MS]);

   498→      loadTeamsInProgressRef.current = false;
