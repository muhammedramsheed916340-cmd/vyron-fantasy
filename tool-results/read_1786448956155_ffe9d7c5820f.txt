     1→'use client';
     2→
     3→import React, { useState, useCallback, useMemo } from 'react';
     4→import {
     5→  Trophy, X, CheckCircle2, Loader2, ChevronRight, ChevronLeft,
     6→  AlertCircle, Users, DollarSign, Filter, RefreshCw,
     7→} from 'lucide-react';
     8→import {
     9→  JCMatch, JCContest, JCTeam, JCJoinItem, JCProgress,
    10→  JCContestFetchResult,
    11→  parseContest, formatCurrency, getContestTypeColor,
    12→  getExistingPlatformTeams, getPlatformContests, normalizeContests,
    13→  buildJoinItems, getJoinKey,
    14→} from '@/lib/join-contest-service';
    15→
    16→// ============ Step Enum ============
    17→type Step = 'matches' | 'teams' | 'contests' | 'joining' | 'result';
    18→
    19→// ============ Contest Error State ============
    20→interface ContestErrorState {
    21→  matchId: string;
    22→  error: string;
    23→  errorType: 'auth' | 'network' | 'parse' | 'invalid_match' | 'api_fail';
    24→}
    25→
    26→// ============ Props ============
    27→interface JoinContestDialogProps {
    28→  open: boolean;
    29→  onClose: () => void;
    30→  matches: JCMatch[];
    31→  fantasyAccounts: Record<string, { authToken: string; mobileNumber: string; my11circleChallenge?: string | null }>;
    32→}
    33→
    34→// ============ Main Dialog ============
    35→export default function JoinContestDialog({
    36→  open, onClose, matches, fantasyAccounts,
    37→}: JoinContestDialogProps) {
    38→  // Step state
    39→  const [step, setStep] = useState<Step>('matches');
    40→  const [platform, setPlatform] = useState<string>('dream11');
    41→
    42→  // Match selection
    43→  const [selectedMatchIds, setSelectedMatchIds] = useState<Set<string | number>>(new Set());
    44→
    45→  // Team selection — EXISTING PLATFORM TEAMS
    46→  const [platformTeams, setPlatformTeams] = useState<JCTeam[]>([]);
    47→  const [selectedTeamIds, setSelectedTeamIds] = useState<Set<string | number>>(new Set());
    48→  const [mixedTeamMode, setMixedTeamMode] = useState(false);
    49→  const [loadingTeams, setLoadingTeams] = useState(false);
    50→  const [teamsError, setTeamsError] = useState<string | null>(null);
    51→  const [teamsTokenExpired, setTeamsTokenExpired] = useState(false);
    52→
    53→  // Contest selection — with proper error states
    54→  const [contestsMap, setContestsMap] = useState<Map<string, JCContest[]>>(new Map());
    55→  const [selectedContestIds, setSelectedContestIds] = useState<Set<string>>(new Set());
    56→  const [loadingContests, setLoadingContests] = useState(false);
    57→  const [contestErrors, setContestErrors] = useState<ContestErrorState[]>([]);
    58→  const [contestTokenExpired, setContestTokenExpired] = useState(false);
    59→
    60→  // Join progress
    61→  const [joinItems, setJoinItems] = useState<JCJoinItem[]>([]);
    62→  const [progress, setProgress] = useState<JCProgress>({ current: 0, total: 0, status: 'idle' });
    63→
    64→  // Result
    65→  const [resultItems, setResultItems] = useState<JCJoinItem[]>([]);
    66→
    67→  // Available platforms from linked accounts
    68→  const availablePlatforms = useMemo(() => {
    69→    const platforms: string[] = [];
    70→    if (fantasyAccounts.dream11?.authToken) platforms.push('dream11');
    71→    if (fantasyAccounts.my11circle?.authToken) platforms.push('my11circle');
    72→    return platforms;
    73→  }, [fantasyAccounts]);
    74→
    75→  const account = fantasyAccounts[platform];
    76→
    77→  // Reset on close
    78→  const handleClose = () => {
    79→    setStep('matches');
    80→    setSelectedMatchIds(new Set());
    81→    setPlatformTeams([]);
    82→    setSelectedTeamIds(new Set());
    83→    setMixedTeamMode(false);
    84→    setLoadingTeams(false);
    85→    setTeamsError(null);
    86→    setTeamsTokenExpired(false);
    87→    setContestsMap(new Map());
    88→    setSelectedContestIds(new Set());
    89→    setLoadingContests(false);
    90→    setContestErrors([]);
    91→    setContestTokenExpired(false);
    92→    setJoinItems([]);
    93→    setProgress({ current: 0, total: 0, status: 'idle' });
    94→    setResultItems([]);
    95→    onClose();
    96→  };
    97→
    98→  // ============ Match Selection ============
    99→  const toggleMatch = (id: string | number) => {
   100→    setSelectedMatchIds(prev => {
   101→      const next = new Set(prev);
   102→      next.has(id) ? next.delete(id) : next.add(id);
   103→      return next;
   104→    });
   105→  };
   106→
   107→  const selectAllMatches = () => {
   108→    setSelectedMatchIds(new Set(matches.map(m => m.id)));
   109→  };
   110→
   111→  // ============ Load Existing Platform Teams ============
   112→  const loadPlatformTeams = useCallback(async (selectedPlatform?: string) => {
   113→    const plat = selectedPlatform || platform;
   114→    const acc = fantasyAccounts[plat];
   115→    if (!acc?.authToken) {
   116→      setTeamsError('No connected account found. Connect your platform account to load existing teams.');
   117→      setPlatformTeams([]);
   118→      return;
   119→    }
   120→
   121→    setLoadingTeams(true);
   122→    setTeamsError(null);
   123→    setTeamsTokenExpired(false);
   124→    setSelectedTeamIds(new Set());
   125→
   126→    const allTeams: JCTeam[] = [];
   127→    let firstError: string | null = null;
   128→    let anyTokenExpired = false;
   129→
   130→    for (const matchId of selectedMatchIds) {
   131→      const result = await getExistingPlatformTeams(plat, matchId, acc.authToken);
   132→      if (result.teams.length > 0) {
   133→        allTeams.push(...result.teams);
   134→      }
   135→      if (result.tokenExpired) {
   136→        anyTokenExpired = true;
   137→        firstError = result.error || 'Session expired';
   138→      } else if (result.error && !firstError) {
   139→        firstError = result.error;
   140→      }
   141→    }
   142→
   143→    setPlatformTeams(allTeams);
   144→    setTeamsError(allTeams.length === 0 ? firstError : null);
   145→    setTeamsTokenExpired(anyTokenExpired);
   146→    setLoadingTeams(false);
   147→  }, [selectedMatchIds, platform, fantasyAccounts]);
   148→
   149→  // ============ Team Selection ============
   150→  const toggleTeam = (id: string | number) => {
   151→    setSelectedTeamIds(prev => {
   152→      const next = new Set(prev);
   153→      next.has(id) ? next.delete(id) : next.add(id);
   154→      return next;
   155→    });
   156→  };
   157→
   158→  const selectAllTeams = () => {
   159→    setSelectedTeamIds(new Set(platformTeams.map(t => t.id)));
   160→  };
   161→
   162→  const deselectAllTeams = () => {
   163→    setSelectedTeamIds(new Set());
   164→  };
   165→
   166→  // ============ Contest Loading — with proper error differentiation ============
   167→  const loadContests = useCallback(async () => {
   168→    if (!account?.authToken) return;
   169→    setLoadingContests(true);
   170→    setContestErrors([]);
   171→    setContestTokenExpired(false);
   172→    setSelectedContestIds(new Set());
   173→
   174→    const newMap = new Map<string, JCContest[]>();
   175→    const errors: ContestErrorState[] = [];
   176→    let anyTokenExpired = false;
   177→
   178→    for (const matchId of selectedMatchIds) {
   179→      const match = matches.find(m => m.id === matchId);
   180→      const platformMatchId = String(matchId);
   181→
   182→      console.log('[JOIN CONTEST] Selected match:', match?.left_team_name, 'vs', match?.right_team_name);
   183→      console.log('[JOIN CONTEST] Platform:', platform);
   184→      console.log('[JOIN CONTEST] Platform Match ID:', platformMatchId);
   185→      console.log('[JOIN CONTEST] Account ID:', account.mobileNumber || 'unknown');
   186→      console.log('[JOIN CONTEST] Sport Index:', match?.sport_index ?? 0);
   187→
   188→      const result: JCContestFetchResult = await getPlatformContests(
   189→        platform,
   190→        platformMatchId,
   191→        account.authToken,
   192→        match?.sport_index ?? 0,
   193→      );
   194→
   195→      console.log('[JOIN CONTEST] Contest count:', result.contests.length);
   196→      console.log('[JOIN CONTEST] Error type:', result.errorType);
   197→
   198→      if (result.errorType === 'auth') {
   199→        anyTokenExpired = true;
   200→        errors.push({
   201→          matchId: String(matchId),
   202→          error: result.error || 'Session expired',
   203→          errorType: 'auth',
   204→        });
   205→        // Still set empty array for this match
   206→        newMap.set(String(matchId), []);
   207→      } else if (result.errorType && result.errorType !== 'none') {
   208→        errors.push({
   209→          matchId: String(matchId),
   210→          error: result.error || 'Failed to load contests',
   211→          errorType: result.errorType,
   212→        });
   213→        newMap.set(String(matchId), []);
   214→      } else {
   215→        // Success (possibly 0 contests — that's a valid state)
   216→        newMap.set(String(matchId), result.contests);
   217→      }
   218→
   219→      // Validate: contest.matchId should match our matchId
   220→      for (const contest of result.contests) {
   221→        if (contest.matchId && contest.matchId !== platformMatchId) {
   222→          console.warn('[JOIN CONTEST] Contest matchId mismatch:', contest.matchId, '!==', platformMatchId, 'for contest:', contest.id);
   223→        }
   224→      }
   225→
   226→      console.log('[JOIN CONTEST] Normalized contests for match', platformMatchId, ':', result.contests.length);
   227→    }
   228→
   229→    setContestsMap(newMap);
   230→    setContestErrors(errors);
   231→    setContestTokenExpired(anyTokenExpired);
   232→    setLoadingContests(false);
   233→  }, [selectedMatchIds, platform, account, matches]);
   234→
   235→  const toggleContest = (id: string) => {
   236→    setSelectedContestIds(prev => {
   237→      const next = new Set(prev);
   238→      next.has(id) ? next.delete(id) : next.add(id);
   239→      return next;
   240→    });
   241→  };
   242→
   243→  const selectAllContests = () => {
   244→    const allIds: string[] = [];
   245→    for (const contests of contestsMap.values()) {
   246→      for (const c of contests) {
   247→        if (c.joinAvailable) allIds.push(c.id);
   248→      }
   249→    }
   250→    setSelectedContestIds(new Set(allIds));
   251→  };
   252→
   253→  const deselectAllContests = () => {
   254→    setSelectedContestIds(new Set());
   255→  };
   256→
   257→  // Total contest count across all matches
   258→  const totalContestCount = useMemo(() => {
   259→    let count = 0;
   260→    for (const contests of contestsMap.values()) {
   261→      count += contests.length;
   262→    }
   263→    return count;
   264→  }, [contestsMap]);
   265→
   266→  // Total joinable contest count
   267→  const totalJoinableCount = useMemo(() => {
   268→    let count = 0;
   269→    for (const contests of contestsMap.values()) {
   270→      count += contests.filter(c => c.joinAvailable).length;
   271→    }
   272→    return count;
   273→  }, [contestsMap]);
   274→
   275→  // ============ Join Execution ============
   276→  const executeJoin = useCallback(async () => {
   277→    const selectedMatches = matches.filter(m => selectedMatchIds.has(m.id));
   278→    const selectedContestsForJoin = new Map<string, JCContest[]>();
   279→    for (const [matchId, contests] of contestsMap.entries()) {
   280→      selectedContestsForJoin.set(matchId, contests.filter(c => selectedContestIds.has(c.id)));
   281→    }
   282→
   283→    const items = buildJoinItems(selectedMatches, selectedContestsForJoin, selectedTeamIds, platformTeams, platform);
   284→
   285→    if (items.length === 0) {
   286→      setStep('result');
   287→      return;
   288→    }
   289→
   290→    setJoinItems(items);
   291→    setProgress({ current: 0, total: items.length, status: 'joining' });
   292→    setStep('joining');
   293→
   294→    const updatedItems = [...items];
   295→
   296→    for (let i = 0; i < items.length; i++) {
   297→      const item = items[i];
   298→      setProgress(prev => ({ ...prev, current: i + 1 }));
   299→      setJoinItems(prev => prev.map((it, idx) =>
   300→        idx === i ? { ...it, status: 'processing' } : it
   301→      ));
   302→
   303→      try {
   304→        const match = matches.find(m => String(m.id) === String(item.matchId));
   305→        const res = await fetch('/api/fantasy/join-contest', {
   306→          method: 'POST',
   307→          headers: { 'Content-Type': 'application/json' },
   308→          body: JSON.stringify({
   309→            fantasyApp: item.platform || platform,
   310→            matchId: item.matchId,
   311→            authToken: account?.authToken,
   312→            teamId: item.teamId,    // REAL platform team ID
   313→            contestId: item.contestId,
   314→            sportIndex: match?.sport_index ?? 0,
   315→          }),
   316→        });
   317→        const data = await res.json();
   318→
   319→        const status = data.status === 'success' ? 'success'
   320→          : data.status === 'already_joined' ? 'already_joined' : 'fail';
   321→
   322→        updatedItems[i] = { ...updatedItems[i], status, message: data.message };
   323→        setJoinItems([...updatedItems]);
   324→      } catch {
   325→        updatedItems[i] = { ...updatedItems[i], status: 'fail', message: 'Network error' };
   326→        setJoinItems([...updatedItems]);
   327→      }
   328→
   329→      // Rate limit between joins
   330→      await new Promise(r => setTimeout(r, 300));
   331→    }
   332→
   333→    setResultItems([...updatedItems]);
   334→    setProgress(prev => ({ ...prev, status: 'done' }));
   335→    setStep('result');
   336→  }, [matches, selectedMatchIds, contestsMap, selectedContestIds, selectedTeamIds, platformTeams, platform, account]);
   337→
   338→  // ============ Result Stats ============
   339→  const resultStats = useMemo(() => {
   340→    const items = resultItems.length > 0 ? resultItems : joinItems;
   341→    return {
   342→      total: items.length,
   343→      success: items.filter(i => i.status === 'success').length,
   344→      alreadyJoined: items.filter(i => i.status === 'already_joined').length,
   345→      fail: items.filter(i => i.status === 'fail').length,
   346→    };
   347→  }, [resultItems, joinItems]);
   348→
   349→  if (!open) return null;
   350→
   351→  // ============ RENDER ============
   352→  return (
   353→    <div className="fixed inset-0 z-50 bg-black/60 flex items-center justify-center p-4" onClick={handleClose}>
   354→      <div className="bg-white rounded-2xl shadow-2xl w-full max-w-2xl max-h-[90vh] overflow-hidden flex flex-col" onClick={e => e.stopPropagation()}>
   355→        {/* Header */}
   356→        <div className="flex items-center justify-between px-6 py-4 border-b bg-gradient-to-r from-green-600 to-emerald-600 text-white">
   357→          <div className="flex items-center gap-3">
   358→            <Trophy className="w-6 h-6" />
   359→            <h2 className="text-lg font-bold">Join Contest</h2>
   360→          </div>
   361→          <button onClick={handleClose} className="p-1 hover:bg-white/20 rounded-lg transition-all">
   362→            <X className="w-5 h-5" />
   363→          </button>
   364→        </div>
   365→
   366→        {/* Step Indicator */}
   367→        <div className="flex items-center gap-1 px-6 py-3 bg-gray-50 border-b">
   368→          {(['matches', 'teams', 'contests', 'joining', 'result'] as Step[]).map((s, i) => {
   369→            const labels = ['Match', 'Teams', 'Contests', 'Join', 'Result'];
   370→            const stepOrder = ['matches', 'teams', 'contests', 'joining', 'result'];
   371→            const isActive = step === s;
   372→            const isDone = stepOrder.indexOf(step) > i;
   373→            return (
   374→              <React.Fragment key={s}>
   375→                {i > 0 && <ChevronRight className="w-3 h-3 text-gray-300" />}
   376→                <div className={`flex items-center gap-1.5 text-xs font-semibold px-2 py-1 rounded-lg transition-all ${
   377→                  isActive ? 'bg-green-100 text-green-700' : isDone ? 'text-green-500' : 'text-gray-400'
   378→                }`}>
   379→                  <div className={`w-5 h-5 rounded-full flex items-center justify-center text-[10px] ${
   380→                    isActive ? 'bg-green-600 text-white' : isDone ? 'bg-green-200 text-green-600' : 'bg-gray-200 text-gray-500'
   381→                  }`}>
   382→                    {isDone ? '✓' : i + 1}
   383→                  </div>
   384→                  {labels[i]}
   385→                </div>
   386→              </React.Fragment>
   387→            );
   388→          })}
   389→        </div>
   390→
   391→        {/* Content */}
   392→        <div className="flex-1 overflow-y-auto p-6">
   393→
   394→          {/* ===== STEP: MATCHES ===== */}
   395→          {step === 'matches' && (
   396→            <div className="space-y-4">
   397→              {/* Platform Selection */}
   398→              <div className="space-y-2">
   399→                <p className="text-xs font-semibold text-gray-500 uppercase tracking-wider">Platform</p>
   400→                <div className="flex gap-2">
   401→                  {availablePlatforms.map(p => (
   402→                    <button key={p} onClick={() => setPlatform(p)}
   403→                      className={`flex-1 rounded-xl border-2 p-3 text-center font-semibold text-sm transition-all ${
   404→                        platform === p ? 'border-green-500 bg-green-50 text-green-700' : 'border-gray-200 bg-white text-gray-700 hover:border-gray-300'
   405→                      }`}>
   406→                      {p === 'dream11' ? 'Dream11' : 'My11Circle'}
   407→                    </button>
   408→                  ))}
   409→                </div>
   410→                {availablePlatforms.length === 0 && (
   411→                  <div className="bg-amber-50 border border-amber-200 rounded-xl p-3 text-sm text-amber-700">
   412→                    ⚠️ No connected account found. Link your platform account first via Transfer.
   413→                  </div>
   414→                )}
   415→              </div>
   416→
   417→              {/* Match List */}
   418→              <div className="space-y-2">
   419→                <div className="flex items-center justify-between">
   420→                  <p className="text-xs font-semibold text-gray-500 uppercase tracking-wider">Select Matches</p>
   421→                  <button onClick={selectAllMatches} className="text-xs font-semibold text-green-600 hover:text-green-700">
   422→                    Select All
   423→                  </button>
   424→                </div>
   425→                {matches.length === 0 ? (
   426→                  <p className="text-sm text-gray-400 text-center py-8">No upcoming matches</p>
   427→                ) : (
   428→                  <div className="space-y-2 max-h-80 overflow-y-auto">
   429→                    {matches.map(match => {
   430→                      const selected = selectedMatchIds.has(match.id);
   431→                      return (
   432→                        <button key={match.id} onClick={() => toggleMatch(match.id)}
   433→                          className={`w-full text-left rounded-xl border-2 p-3 transition-all ${
   434→                            selected ? 'border-green-500 bg-green-50' : 'border-gray-200 bg-white hover:border-gray-300'
   435→                          }`}>
   436→                          <div className="flex items-center gap-3">
   437→                            <div className={`w-5 h-5 rounded border-2 flex items-center justify-center shrink-0 ${
   438→                              selected ? 'border-green-500 bg-green-500' : 'border-gray-300'
   439→                            }`}>
   440→                              {selected && <CheckCircle2 className="w-3.5 h-3.5 text-white" />}
   441→                            </div>
   442→                            <div className="flex-1">
   443→                              <p className="font-semibold text-sm text-gray-900">
   444→                                {match.left_team_name} vs {match.right_team_name}
   445→                              </p>
   446→                              <div className="flex items-center gap-2 text-xs text-gray-500 mt-0.5">
   447→                                <span>{match.match_time}</span>
   448→                                {match.lineup_out === 1 && <span className="text-green-600 font-medium">Lineup Out</span>}
   449→                              </div>
   450→                            </div>
   451→                          </div>
   452→                        </button>
   453→                      );
   454→                    })}
   455→                  </div>
   456→                )}
   457→              </div>
   458→            </div>
   459→          )}
   460→
   461→          {/* ===== STEP: TEAMS — EXISTING PLATFORM TEAMS ===== */}
   462→          {step === 'teams' && (
   463→            <div className="space-y-4">
   464→              {/* Loading State */}
   465→              {loadingTeams && (
   466→                <div className="flex flex-col items-center justify-center py-12">
   467→                  <Loader2 className="w-8 h-8 animate-spin text-green-500" />
   468→                  <p className="text-sm text-gray-500 mt-3">Loading existing teams from {platform === 'dream11' ? 'Dream11' : 'My11Circle'}...</p>
   469→                </div>
   470→              )}
   471→
   472→              {/* Error State */}
   473→              {!loadingTeams && teamsError && (
   474→                <div className="bg-red-50 border border-red-200 rounded-xl p-4">
   475→                  <div className="flex items-center gap-2">
   476→                    <AlertCircle className="w-5 h-5 text-red-500 shrink-0" />
   477→                    <div>
   478→                      <p className="text-sm font-semibold text-red-700">
   479→                        {teamsTokenExpired ? 'SESSION EXPIRED' : 'Error Loading Teams'}
   480→                      </p>
   481→                      <p className="text-xs text-red-600 mt-0.5">{teamsError}</p>
   482→                    </div>
   483→                  </div>
   484→                  {teamsTokenExpired && (
   485→                    <p className="text-xs text-red-500 mt-2">Reconnect your {platform === 'dream11' ? 'Dream11' : 'My11Circle'} account via Transfer.</p>
   486→                  )}
   487→                  <button onClick={() => loadPlatformTeams()} className="mt-2 flex items-center gap-1 text-xs font-semibold text-red-600 hover:text-red-700">
   488→                    <RefreshCw className="w-3 h-3" /> Retry
   489→                  </button>
   490→                </div>
   491→              )}
   492→
   493→              {/* No Account */}
   494→              {!loadingTeams && !teamsError && !account?.authToken && (
   495→                <div className="bg-amber-50 border border-amber-200 rounded-xl p-4 text-center">
   496→                  <p className="text-sm font-semibold text-amber-700">No connected account found</p>
   497→                  <p className="text-xs text-amber-600 mt-1">Connect your {platform === 'dream11' ? 'Dream11' : 'My11Circle'} account to load existing teams.</p>
   498→                </div>
   499→              )}
   500→
   501→              {/* Teams List */}
   502→              {!loadingTeams && account?.authToken && (
   503→                <div className="space-y-2">
   504→                  <div className="flex items-center justify-between">
   505→                    <p className="text-xs font-semibold text-gray-500 uppercase tracking-wider">
   506→                      Existing Teams ({platformTeams.length} available)
   507→                    </p>
   508→                    <div className="flex gap-2">
   509→                      <button onClick={selectAllTeams} disabled={platformTeams.length === 0}
   510→                        className="text-xs font-semibold text-green-600 hover:text-green-700 disabled:opacity-40">
   511→                        Select All
   512→                      </button>
   513→                      <button onClick={deselectAllTeams} disabled={selectedTeamIds.size === 0}
   514→                        className="text-xs font-semibold text-gray-500 hover:text-gray-700 disabled:opacity-40">
   515→                        Deselect
