     6→// ============ Platform Name Normalization ============
     7→
     8→/**
     9→ * Normalize platform names to consistent lowercase identifiers.
    10→ * The TG API may return varied casing in fantasy_id_list[].name
    11→ * (e.g., "Dream11", "dream11", "My11Circle", "my11circle").
    12→ * All internal references use lowercase: "dream11", "my11circle".
    13→ */
    14→export function normalizePlatformName(name: string): string {
    15→  return name.toLowerCase().replace(/[\s_-]/g, '');
    16→}

    18→/**
    19→ * Resolve a player's platform-specific ID from their fantasy_id_list.
    20→ * Uses case-insensitive matching to handle API casing variations.
    21→ */
    22→export function resolvePlatformPlayerId(
    23→  player: TGPlayer,
    24→  platform: string,
    25→): number | null {
    26→  if (!player.fantasy_id_list || player.fantasy_id_list.length === 0) return null;
    27→  const target = normalizePlatformName(platform);
    28→  const found = player.fantasy_id_list.find(
    29→    f => normalizePlatformName(f.name) === target
    30→  );
    31→  return found ? found.id : null;
    32→}

    62→// ============ Lineup Utilities ============
    63→
    64→/**
    65→ * Determine lineup mode based on player data.
    66→ * AFTER LINEUP: enough players have playing === 1 to form valid teams.
    67→ *   Requires at least 11 confirmed total AND at least 4 from each team
    68→ *   (due to max-7-from-one-team constraint: 11 - 7 = 4).
    69→ *   If only a few players are confirmed (partial lineup), we stay in
    70→ *   'before' mode so team generation doesn't fail with too few eligible players.
    71→ * BEFORE LINEUP: lineup not fully confirmed yet — all players eligible.
    72→ *
    73→ * BUG FIX: Previously used players.some(p => p.playing === 1) which
    74→ * switched to 'after' mode when even ONE player was confirmed, causing
    75→ * getEligiblePlayers to filter out all non-confirmed players and leaving
    76→ * too few to form valid 11-player teams → 0 teams generated.
    77→ */
    78→export function getLineupMode(players: TGPlayer[]): 'before' | 'after' {
    79→  const confirmed = players.filter(p => p.playing === 1);
    80→
    81→  // Need at least 11 confirmed players total to form a valid team
    82→  if (confirmed.length < 11) return 'before';
    83→
    84→  // Need at least 4 confirmed from each team
    85→  // (max 7 from one team → need at least 11-7=4 from the other)
    86→  const teamNames = [...new Set(players.map(p => p.team_name))];
    87→  for (const team of teamNames) {
    88→    if (team && confirmed.filter(p => p.team_name === team).length < 4) {
    89→      return 'before';
    90→    }
    91→  }
    92→
    93→  return 'after';
    94→}
    95→
    96→/**
    97→ * Filter players based on lineup mode.
    98→ * BEFORE LINEUP: all players are eligible EXCEPT confirmed OUT (playing === -1).
    99→ * AFTER LINEUP: ONLY players with playing === 1 are eligible (Playing XI only, no bench).
   100→ */
   101→export function getEligiblePlayers(
   102→  allPlayers: TGPlayer[],
   103→  avoidPlayerIds: Set<number> = new Set(),
   104→): TGPlayer[] {
   105→  const mode = getLineupMode(allPlayers);
   106→
   107→  return allPlayers.filter(p => {
   108→    // Avoid players are always excluded
   109→    if (avoidPlayerIds.has(p.pl_id)) return false;
   110→
   111→    // Confirmed OUT players are always excluded
   112→    if (p.playing === -1) return false;
   113→
   114→    if (mode === 'after') {
   115→      // AFTER LINEUP: ONLY confirmed Playing XI players (playing === 1)
   116→      // Bench/probable players (playing === 0) are NOT included
   117→      return p.playing === 1;
   118→    }
   119→
   120→    // BEFORE LINEUP: all non-OUT players eligible (playing === 0 or 1)
   121→    return true;
   122→  });
   123→}

   125→/**
   126→ * Check if a specific player is eligible based on lineup mode.
   127→ */
   128→export function isPlayerEligible(
   129→  player: TGPlayer,
   130→  allPlayers: TGPlayer[],
   131→  avoidPlayerIds: Set<number> = new Set(),
   132→): { eligible: boolean; reason?: string } {
   133→  if (avoidPlayerIds.has(player.pl_id)) {
   134→    return { eligible: false, reason: 'AVOID' };
   135→  }
   136→
   137→  const mode = getLineupMode(allPlayers);
   138→
   139→  if (mode === 'after' && player.playing !== 1) {
   140→    return { eligible: false, reason: 'OUT / NOT IN PLAYING XI' };
   141→  }
   142→
   143→  return { eligible: true };
   144→}

   146→/**
   147→ * Validate a team for lineup eligibility.
   148→ * Returns list of invalid players with reasons.
   149→ */
   150→export function validateTeamForLineup(
   151→  team: GeneratedTeam,
   152→  allPlayers: TGPlayer[],
   153→  avoidPlayerIds: Set<number> = new Set(),
   154→): { valid: boolean; invalidPlayers: { player: TGPlayer; reason: string }[] } {
   155→  const mode = getLineupMode(allPlayers);
   156→  const invalidPlayers: { player: TGPlayer; reason: string }[] = [];
   157→
   158→  for (const player of team.players) {
   159→    if (avoidPlayerIds.has(player.pl_id)) {
   160→      invalidPlayers.push({ player, reason: 'AVOID' });
   161→      continue;
   162→    }
   163→
   164→    if (mode === 'after' && player.playing !== 1) {
   165→      invalidPlayers.push({ player, reason: 'OUT / NOT IN PLAYING XI' });
   166→    }
   167→  }
   168→
   169→  return { valid: invalidPlayers.length === 0, invalidPlayers };
   170→}

   183→/**
   184→ * Get all valid role combinations for an 11-player cricket team.
   185→ */
   186→export function getAllValidCombinations(): RoleCombination[] {
   187→  const combos: RoleCombination[] = [];
   188→  for (let wk = MIN_WK; wk <= MAX_WK; wk++) {
   189→    for (let bat = MIN_BAT; bat <= MAX_BAT; bat++) {
   190→      for (let ar = MIN_AR; ar <= MAX_AR; ar++) {
   191→        const bowl = CRICKET_TEAM_SIZE - wk - bat - ar;
   192→        if (bowl < MIN_BOWL || bowl > MAX_BOWL) continue;
   193→        combos.push({ wk, bat, ar, bowl });
   194→      }
   195→    }
   196→  }
   197→  return combos;
   198→}

   200→/**
   201→ * Check if a role combination is compatible with a set of fixed players.
   202→ */
   203→export function isCombinationCompatibleWithFixed(
   204→  combo: RoleCombination,
   205→  fixedPlayers: TGPlayer[],
   206→): boolean {
   207→  const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   208→  const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   209→  const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   210→  const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   211→
   212→  // Fixed players must not exceed combination totals
   213→  if (fixedWK > combo.wk) return false;
   214→  if (fixedBat > combo.bat) return false;
   215→  if (fixedAR > combo.ar) return false;
   216→  if (fixedBowl > combo.bowl) return false;
   217→
   218→  // Remaining slots must be fillable
   219→  const remainingPlayers = CRICKET_TEAM_SIZE - fixedPlayers.length;
   220→  const neededWK = combo.wk - fixedWK;
   221→  const neededBat = combo.bat - fixedBat;
   222→  const neededAR = combo.ar - fixedAR;
   223→  const neededBowl = combo.bowl - fixedBowl;
   224→
   225→  return (neededWK + neededBat + neededAR + neededBowl) === remainingPlayers;
   226→}

   228→/**
   229→ * Get combinations compatible with fixed players and available player pool.
   230→ */
   231→export function getCompatibleCombinations(
   232→  fixedPlayers: TGPlayer[],
   233→  eligiblePlayers: TGPlayer[],
   234→): RoleCombination[] {
   235→  const all = getAllValidCombinations();
   236→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   237→  const remaining = eligiblePlayers.filter(p => !fixedIds.has(p.pl_id));
   238→
   239→  const remWK = remaining.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   240→  const remBat = remaining.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   241→  const remAR = remaining.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   242→  const remBowl = remaining.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   243→
   244→  return all.filter(combo => {
   245→    if (!isCombinationCompatibleWithFixed(combo, fixedPlayers)) return false;
   246→
   247→    const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   248→    const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   249→    const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   250→    const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   251→
   252→    if (remWK < combo.wk - fixedWK) return false;
   253→    if (remBat < combo.bat - fixedBat) return false;
   254→    if (remAR < combo.ar - fixedAR) return false;
   255→    if (remBowl < combo.bowl - fixedBowl) return false;
   256→
   257→    return true;
   258→  });
   259→}

   261→/**
   262→ * Auto-select the best combination based on player pool and category.
   263→ * For multiple teams, rotates combinations intelligently.
   264→ */
   265→export function autoSelectCombination(
   266→  eligiblePlayers: TGPlayer[],
   267→  category: string,
   268→  fixedPlayers: TGPlayer[] = [],
   269→  teamIndex: number = 0,
   270→): RoleCombination {
   271→  const compatible = getCompatibleCombinations(fixedPlayers, eligiblePlayers);
   272→
   273→  if (compatible.length === 0) {
   274→    return { wk: 1, bat: 4, ar: 2, bowl: 4 };
   275→  }
   276→
   277→  // Category-preferred patterns
   278→  const preferredPatterns: RoleCombination[] =
   279→    category === 'H2H'
   280→      ? [{ wk: 1, bat: 5, ar: 2, bowl: 3 }, { wk: 1, bat: 4, ar: 2, bowl: 4 }, { wk: 1, bat: 3, ar: 3, bowl: 4 }]
   281→      : category === 'SL'
   282→      ? [{ wk: 1, bat: 4, ar: 2, bowl: 4 }, { wk: 1, bat: 3, ar: 3, bowl: 4 }, { wk: 2, bat: 3, ar: 2, bowl: 4 }]
   283→      : [ // Mega GL
   284→        { wk: 1, bat: 3, ar: 2, bowl: 5 }, { wk: 1, bat: 4, ar: 1, bowl: 5 },
   285→        { wk: 1, bat: 3, ar: 3, bowl: 4 }, { wk: 1, bat: 4, ar: 2, bowl: 4 },
   286→        { wk: 2, bat: 3, ar: 2, bowl: 4 }, { wk: 1, bat: 5, ar: 1, bowl: 4 },
   287→        { wk: 2, bat: 3, ar: 1, bowl: 5 }, { wk: 1, bat: 4, ar: 3, bowl: 3 },
   288→        { wk: 2, bat: 4, ar: 1, bowl: 4 }, { wk: 1, bat: 3, ar: 1, bowl: 6 },
   289→        { wk: 2, bat: 2, ar: 3, bowl: 4 }, { wk: 1, bat: 5, ar: 2, bowl: 3 },
   290→      ];
   291→
   292→  const preferredCompatible = preferredPatterns.filter(p =>
   293→    compatible.some(c => c.wk === p.wk && c.bat === p.bat && c.ar === p.ar && c.bowl === p.bowl)
   294→  );
   295→
   296→  const pool = preferredCompatible.length > 0 ? preferredCompatible : compatible;
   297→  const index = teamIndex % pool.length;
   298→
   299→  return pool[index];
   300→}

   302→/**
   303→ * Validate a manual combination against available players.
   304→ */
   305→export function validateCombination(
   306→  combo: RoleCombination,
   307→  eligiblePlayers: TGPlayer[],
   308→  fixedPlayers: TGPlayer[] = [],
   309→): { valid: boolean; errors: string[] } {
   310→  const errors: string[] = [];
   311→
   312→  if (combo.wk + combo.bat + combo.ar + combo.bowl !== CRICKET_TEAM_SIZE) {
   313→    errors.push(`Total players must be 11 (got ${combo.wk + combo.bat + combo.ar + combo.bowl})`);
   314→  }
   315→
   316→  if (combo.wk < MIN_WK || combo.wk > MAX_WK) errors.push(`WK must be ${MIN_WK}-${MAX_WK}`);
   317→  if (combo.bat < MIN_BAT || combo.bat > MAX_BAT) errors.push(`BAT must be ${MIN_BAT}-${MAX_BAT}`);
   318→  if (combo.ar < MIN_AR || combo.ar > MAX_AR) errors.push(`AR must be ${MIN_AR}-${MAX_AR}`);
   319→  if (combo.bowl < MIN_BOWL || combo.bowl > MAX_BOWL) errors.push(`BOWL must be ${MIN_BOWL}-${MAX_BOWL}`);
   320→
   321→  if (fixedPlayers.length > 0 && !isCombinationCompatibleWithFixed(combo, fixedPlayers)) {
   322→    errors.push('Combination is not compatible with selected fixed players');
   323→  }
   324→
   325→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   326→  const remaining = eligiblePlayers.filter(p => !fixedIds.has(p.pl_id));
   327→
   328→  const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   329→  const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   330→  const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   331→  const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   332→
   333→  const needWK = Math.max(0, combo.wk - fixedWK);
   334→  const needBat = Math.max(0, combo.bat - fixedBat);
   335→  const needAR = Math.max(0, combo.ar - fixedAR);
   336→  const needBowl = Math.max(0, combo.bowl - fixedBowl);
   337→
   338→  const availWK = remaining.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   339→  const availBat = remaining.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   340→  const availAR = remaining.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   341→  const availBowl = remaining.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   342→
   343→  if (availWK < needWK) errors.push(`Need ${needWK} WK but only ${availWK} available`);
   344→  if (availBat < needBat) errors.push(`Need ${needBat} BAT but only ${availBat} available`);
   345→  if (availAR < needAR) errors.push(`Need ${needAR} AR but only ${availAR} available`);
   346→  if (availBowl < needBowl) errors.push(`Need ${needBowl} BOWL but only ${availBowl} available`);
   347→
   348→  return { valid: errors.length === 0, errors };
   349→}

   429→function decryptField(encrypted: string): string {
   430→  try {
   431→    const bytes = CryptoJS.AES.decrypt(encrypted, ENCRYPTION_KEY);
   432→    const decrypted = bytes.toString(CryptoJS.enc.Utf8);
   433→    if (!decrypted) return encrypted;
   434→    return decrypted;
   435→  } catch {
   436→    return encrypted;
   437→  }
   438→}

   512→export async function fetchMatches(sport: string = 'cricket'): Promise<TGMatch[]> {
   513→  try {
   514→    const response = await fetch(`${TG_API_BASE}/fantasy/matches/${sport}`, {
   515→      next: { revalidate: 60 },
   516→    });
   517→
   518→    if (!response.ok) {
   519→      throw new Error(`Failed to fetch matches: ${response.status}`);
   520→    }
   521→
   522→    const data = await response.json();
   523→
   524→    if (data.status !== 'success' || !Array.isArray(data.data)) {
   525→      return [];
   526→    }
   527→
   528→    const matches: TGMatch[] = [];
   529→    for (const encryptedMatch of data.data) {
   530→      const match = decryptMatchData(encryptedMatch);
   531→      if (match) {
   532→        matches.push(match);
   533→      }
   534→    }
   535→
   536→    return matches;
   537→  } catch (error) {
   538→    console.error('Error fetching matches:', error);
   539→    return [];
   540→  }

   664→/**
   665→ * Create a unique signature for a team based on sorted player IDs + C/VC IDs.
   666→ * Used for deduplication — two teams with identical signatures are duplicates.
   667→ */
   668→export function makeTeamSignature(players: TGPlayer[], captain: TGPlayer, viceCaptain: TGPlayer): string {
   669→  const playerIds = players.map(p => p.pl_id).sort((a, b) => a - b).join(',');
   670→  return `${playerIds}|C${captain.pl_id}|VC${viceCaptain.pl_id}`;

   673→/**
   674→ * Deduplicate and validate a list of generated teams.
   675→ * Removes exact duplicate player combinations.
   676→ * Separates valid and invalid teams based on lineup eligibility.
   677→ *
   678→ * Returns: { valid, invalid, duplicateCount, invalidLineupCount }
   679→ */
   680→export function deduplicateAndValidateTeams(
   681→  teams: GeneratedTeam[],
   682→  allPlayers: TGPlayer[],
   683→  avoidPlayerIds: Set<number> = new Set(),
   684→): {
   685→  valid: GeneratedTeam[];
   686→  invalid: GeneratedTeam[];
   687→  duplicateCount: number;
   688→  invalidLineupCount: number;
   689→} {
   690→  const seen = new Set<string>();
   691→  const valid: GeneratedTeam[] = [];
   692→  const invalid: GeneratedTeam[] = [];
   693→  let duplicateCount = 0;
   694→  let invalidLineupCount = 0;
   695→
   696→  for (let i = 0; i < teams.length; i++) {
   697→    const team = teams[i];
   698→    const sig = makeTeamSignature(team.players, team.captain, team.viceCaptain);
   699→
   700→    // Skip duplicates
   701→    if (seen.has(sig)) {
   702→      duplicateCount++;
   703→      continue;
   704→    }
   705→    seen.add(sig);
   706→
   707→    // Check lineup validity
   708→    const validation = validateTeamForLineup(team, allPlayers, avoidPlayerIds);
   709→    if (!validation.valid) {
   710→      invalidLineupCount++;
   711→      invalid.push({ ...team, id: invalid.length + 1 });
   712→    } else {
   713→      valid.push({ ...team, id: valid.length + 1 });
   714→    }
   715→  }
   716→
   717→  return { valid, invalid, duplicateCount, invalidLineupCount };
   718→}

   733→function seededRandom(seed: number): () => number {
   734→  let s = seed;
   735→  return () => {
   736→    s = (s * 16807 + 0) % 2147483647;
   737→    return (s - 1) / 2147483646;
   738→  };
   739→}

   741→function shuffleArray<T>(arr: T[], rng: () => number): T[] {
   742→  const result = [...arr];
   743→  for (let i = result.length - 1; i > 0; i--) {
   744→    const j = Math.floor(rng() * (i + 1));
   745→    [result[i], result[j]] = [result[j], result[i]];
   746→  }
   747→  return result;
   748→}

   750→// ============ PRO PLAYER SELECTION ENGINE ============
   751→// Each team tries to rank in the top 1-100.
   752→// Strategy: Must-have players + smart scoring + limited differential.
   753→
   754→/**
   755→ * Score a player for team selection (higher = better pick).
   756→ * Used for the MAIN generation engine (not just autoSelect).
   757→ * This is the PRO scoring that replaces the broken sort+shuffle approach.
   758→ */
   759→function playerScoreForGen(p: TGPlayer, category: string): number {
   760→  if (category === 'H2H') {
   761→    // H2H: Safest possible — selection % is king
   762→    return (
   763→      p.selected_by * 0.50 +
   764→      p.captain_percentage * 0.25 +
   765→      p.points * 0.15 +
   766→      p.vice_captain_percentage * 0.05 +
   767→      (p.playing === 1 ? 12 : 0) +
   768→      (p.role === PLAYER_ROLES.ALL_ROUNDER ? 3 : 0)
   769→    );
   770→  } else if (category === 'SL') {
   771→    // SL: Balanced — form + selection + captain history
   772→    return (
   773→      p.selected_by * 0.35 +
   774→      p.captain_percentage * 0.25 +
   775→      p.points * 0.25 +
   776→      p.vice_captain_percentage * 0.10 +
   777→      (p.playing === 1 ? 12 : 0) +
   778→      (p.role === PLAYER_ROLES.ALL_ROUNDER ? 3 : 0)
   779→    );
   780→  } else {
   781→    // Mega GL: Still prioritize core quality, differential comes from C/VC rotation
   782→    // and 2-3 "punt" players — NOT from making the whole team weak
   783→    return (
   784→      p.selected_by * 0.30 +
   785→      p.captain_percentage * 0.20 +
   786→      p.points * 0.25 +
   787→      p.vice_captain_percentage * 0.15 +
   788→      (p.playing === 1 ? 12 : 0) +
   789→      (p.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0) +
   790→      (p.role === PLAYER_ROLES.WICKET_KEEPER ? 1 : 0)
   791→    );
   792→  }
   793→}
   794→
   795→/**
   796→ * Smart select: Picks top players by score, with controlled randomness.
   797→ * - Keeps top `keepTop` players fixed (highest scored).
   798→ * - Randomly selects from the remaining pool for variety.
   799→ * - This FIXES the critical bug where shuffleArray after sort destroyed all ordering.
   800→ */
   801→function smartSelect(
   802→  pool: TGPlayer[],
   803→  count: number,
   804→  scoreFn: (p: TGPlayer) => number,
   805→  rng: () => number,
   806→  keepTopRatio: number = 0.7, // Keep top 70% of picks as best-scored
   807→): TGPlayer[] {
   808→  if (pool.length <= count) return [...pool];
   809→
   810→  const sorted = [...pool].sort((a, b) => scoreFn(b) - scoreFn(a));
   811→  const keepTop = Math.max(1, Math.floor(count * keepTopRatio));
   812→  const randomSlots = count - keepTop;
   813→
   814→  // Top players — always include these (they're the best picks)
   815→  const top = sorted.slice(0, keepTop);
   816→  const topIds = new Set(top.map(p => p.pl_id));
   817→
   818→  // Remaining pool (excluding already picked top players)
   819→  const remaining = sorted.filter(p => !topIds.has(p.pl_id));
   820→
   821→  // Random selection from remaining for variety/diversification
   822→  const shuffled = shuffleArray(remaining, rng);
   823→  const randomPicks = shuffled.slice(0, randomSlots);
   824→
   825→  return [...top, ...randomPicks];
   826→}

   828→/**
   829→ * Identify must-have players — players selected by >70% of users.
   830→ * These should be in EVERY team. Missing them = guaranteed bad rank.
   831→ */
   832→function getMustHavePlayers(
   833→  pool: TGPlayer[],
   834→  threshold: number = 70,
   835→): TGPlayer[] {
   836→  return pool.filter(p => p.selected_by >= threshold && p.playing === 1);

   839→/**
   840→ * Score an entire team's quality (higher = better team).
   841→ * Used to reject low-quality teams and retry.
   842→ */
   843→function scoreTeamQuality(team: TGPlayer[], category: string): number {
   844→  let score = 0;
   845→  const scoreFn = (p: TGPlayer) => playerScoreForGen(p, category);
   846→
   847→  for (const p of team) {
   848→    score += scoreFn(p);
   849→    // Bonus for must-have players (selected_by > 70%)
   850→    if (p.selected_by >= 70) score += 8;
   851→    // Bonus for confirmed playing
   852→    if (p.playing === 1) score += 3;
   853→    // Penalty for very low selection (< 10%) unless Mega GL
   854→    if (p.selected_by < 10 && category !== 'MEGA GL') score -= 5;
   855→  }
   856→
   857→  // Bonus for role diversity
   858→  const roles = new Set(team.map(p => p.role));
   859→  score += roles.size * 3;
   860→
   861→  // Bonus for team balance (close to 6-5 or 5-6 split)
   862→  const teamNames = [...new Set(team.map(p => p.team_name))];
   863→  if (teamNames.length === 2) {
   864→    const c1 = team.filter(p => p.team_name === teamNames[0]).length;
   865→    const c2 = team.filter(p => p.team_name === teamNames[1]).length;
   866→    const balance = Math.min(c1, c2); // Closer to 5.5 is better
   867→    score += balance * 2;
   868→  }
   869→
   870→  return score;
   871→}

   873→/**
   874→ * Category-specific C/VC scoring functions.
   875→ * Each category has a genuinely different strategy for selecting Captain and Vice-Captain.
   876→ *
   877→ * Mega GL: Prioritize differential captains (low selection % but high points).
   878→ *          Wider C/VC rotation across teams — different C/VC per team.
   879→ * SL:      Prioritize consistent/form players. Higher weight on selection % and captain %.
   880→ *          More stable C/VC — top 2 players usually.
   881→ * H2H:     Prioritize most-selected, safest players. Highest weight on selection %.
   882→ *          Most conservative C/VC — always the safest picks.
   883→ */
   884→function scoreCaptain(player: TGPlayer, category: string): number {
   885→  if (category === 'H2H') {
   886→    // H2H: Safest picks — highest selection % and captain % weighted most
   887→    return player.selected_by * 0.45 + player.captain_percentage * 0.35 + player.points * 0.2;
   888→  } else if (category === 'SL') {
   889→    // SL: Balanced — form players with good captain record
   890→    return player.captain_percentage * 0.35 + player.selected_by * 0.35 + player.points * 0.3;
   891→  } else {
   892→    // Mega GL: Differential — rotate captains, some differential (lower selection but higher upside)
   893→    return player.points * 0.4 + player.captain_percentage * 0.3 + player.selected_by * 0.2 + player.vice_captain_percentage * 0.1;
   894→  }
   895→}
   896→
   897→function scoreViceCaptain(player: TGPlayer, category: string): number {
   898→  if (category === 'H2H') {
   899→    return player.selected_by * 0.4 + player.vice_captain_percentage * 0.35 + player.points * 0.25;
   900→  } else if (category === 'SL') {
   901→    return player.vice_captain_percentage * 0.35 + player.selected_by * 0.35 + player.points * 0.3;
   902→  } else {
   903→    // Mega GL: Wider VC rotation
   904→    return player.points * 0.35 + player.vice_captain_percentage * 0.3 + player.selected_by * 0.2 + player.captain_percentage * 0.15;
   905→  }
   906→}
   907→
   908→/**
   909→ * Select Captain and Vice-Captain based on category strategy.
   910→ * Mega GL: Rotates C/VC across team members (wider differential).
   911→ * SL: Picks the top 2 most reliable.
   912→ * H2H: Picks the safest 2.
   913→ */
   914→function selectCaptainViceCaptain(
   915→  team: TGPlayer[],
   916→  category: string,
   917→  teamIndex: number,
   918→): { captain: TGPlayer; viceCaptain: TGPlayer } {
   919→  const isMegaGL = category !== 'H2H' && category !== 'SL';
   920→
   921→  if (isMegaGL) {
   922→    // Mega GL: Sort by captain score, then rotate C/VC position based on teamIndex
   923→    // This gives wider C/VC distribution across teams
   924→    const sorted = [...team].sort((a, b) => scoreCaptain(b, category) - scoreCaptain(a, category));
   925→    // Rotate: sometimes pick #1 as C and #2 as VC, sometimes #1 as C and #3 as VC, etc.
   926→    const cIdx = 0; // Always pick top scorer as C
   927→    let vcIdx = 1 + (teamIndex % Math.min(4, sorted.length - 1)); // Rotate VC among top 2-5
   928→    if (vcIdx >= sorted.length) vcIdx = 1;
   929→    return { captain: sorted[cIdx], viceCaptain: sorted[vcIdx] };
   930→  } else {
   931→    // H2H and SL: Pick top 2 by category-specific scoring
   932→    const sorted = [...team].sort((a, b) => scoreCaptain(b, category) - scoreCaptain(a, category));
   933→    return { captain: sorted[0], viceCaptain: sorted.length > 1 ? sorted[1] : sorted[0] };
   934→  }
   935→}

  1475→        const nAR = Math.max(0, ar - fixedARCount);

  1626→          }

  1628→        if (!foundValidCombo) {
  1629→          // Fallback: use category-specific C/VC scoring
  1630→          const { captain: fbC, viceCaptain: fbVC } = selectCaptainViceCaptain(fullTeam, category, t);

  1633→      }
  1634→

  1638→

  1640→      teamC = combo.c;
  1641→      teamVC = combo.vc;

  1643→    }
  1644→
  1645→    if (team && teamC && teamVC) {
  1646→      teams.push({
  1647→        id: teams.length + 1,
  1648→        captain: teamC,
  1649→        viceCaptain: teamVC,
  1650→        players: team,

  1652→    } else {
  1653→      // If we failed to generate a unique team for this slot,
  1654→      // try extra attempts with different seeds before giving up
  1655→      let extraAttempts = 0;
  1656→      const MAX_EXTRA = 200;

  1658→        extraAttempts++;
  1659→        totalAttempts++;
  1660→        const extraSeed = seed + count * 1000 + extraAttempts * 37;
  1661→        const sr = seededRandom(extraSeed);
  1662→
  1663→        // Pick distribution (or use explicit combination)
  1664→        let targetWK: number, targetBat: number, targetAR: number, targetBowl: number;
  1665→        if (combination) {
  1666→          targetWK = combination.wk;
  1667→          targetBat = combination.bat;
  1668→          targetAR = combination.ar;
  1669→          targetBowl = combination.bowl;
  1670→        } else if (validDistributions.length === 1) {
  1671→          [targetWK, targetBat, targetAR, targetBowl] = validDistributions[0];
  1672→        } else {
  1673→          const idx = Math.floor(sr() * validDistributions.length);
  1674→          [targetWK, targetBat, targetAR, targetBowl] = validDistributions[idx];
  1675→        }
  1676→
  1677→        const needWK = Math.max(0, targetWK - fixedWKCount);
  1678→        const needBat = Math.max(0, targetBat - fixedBatCount);

  1680→        const needBowl = Math.max(0, targetBowl - fixedBowlCount);
  1681→
  1682→        if (remWK.length < needWK || remBat.length < needBat ||
  1683→            remAR.length < needAR || remBowl.length < needBowl) continue;
  1684→
  1685→        const sortFn = (a: TGPlayer, b: TGPlayer) => a.selected_by - b.selected_by;
  1686→        const pickedWK = shuffleArray([...remWK].sort(sortFn), sr).slice(0, needWK);
  1687→        const pickedBat = shuffleArray([...remBat].sort(sortFn), sr).slice(0, needBat);
  1688→        const pickedAR = shuffleArray([...remAR].sort(sortFn), sr).slice(0, needAR);

  1692→            pickedAR.length < needAR || pickedBowl.length < needBowl) continue;
  1693→
  1694→        const remaining = [...pickedWK, ...pickedBat, ...pickedAR, ...pickedBowl];

  1745→  const strategyUsed = combination

  1757→// ============ Auto Select for Extra Team Generation ============
  1758→// Uses real player data (selection %, captain %, points, credits, role, team balance)
  1759→// to intelligently pick 8 FIX, 5 C, and 5 VC
  1760→
  1761→export interface AutoSelectResult {
  1762→  fixedPlayers: TGPlayer[];
  1763→  captainOptions: TGPlayer[];
  1764→  viceCaptainOptions: TGPlayer[];
  1765→}
  1766→
  1767→// Player scoring for auto-selection: higher = better pick
  1768→function playerScore(p: TGPlayer): number {
  1769→  return (
  1770→    p.selected_by * 0.35 +       // selection % - most important for GL
  1771→    p.captain_percentage * 0.25 + // captain % - indicates trust
  1772→    p.points * 0.2 +             // points - performance indicator
  1773→    p.vice_captain_percentage * 0.1 + // VC %
  1774→    (p.playing === 1 ? 10 : 0) + // confirmed playing bonus
  1775→    (p.role === PLAYER_ROLES.ALL_ROUNDER ? 3 : 0) + // AR bonus (flexible)
  1776→    (p.role === PLAYER_ROLES.WICKET_KEEPER ? 1 : 0) // WK slight bonus
  1777→  );
  1778→}
  1779→
  1780→export function autoSelectExtraPlayers(
  1781→  leftPlayers: TGPlayer[],
  1782→  rightPlayers: TGPlayer[],
  1783→  avoidPlayerIds: Set<number> = new Set(),
  1784→): AutoSelectResult {
  1785→  const allPlayersRaw = [...leftPlayers, ...rightPlayers];
  1786→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
  1787→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
  1788→
  1789→  // Use lineup-aware eligibility filter
  1790→  const eligible = getEligiblePlayers(allPlayersRaw, avoidPlayerIds);
  1791→
  1792→  const eligibleLeft = eligible.filter(p => p.team_name === leftTeamName);
  1793→  const eligibleRight = eligible.filter(p => p.team_name === rightTeamName);
  1794→
  1795→  // Sort by score descending
  1796→  const sortByScore = (a: TGPlayer, b: TGPlayer) => playerScore(b) - playerScore(a);
  1797→  const sortedLeft = [...eligibleLeft].sort(sortByScore);
  1798→  const sortedRight = [...eligibleRight].sort(sortByScore);
  1799→
  1800→  // === SELECT 8 FIX PLAYERS ===
  1801→  // Strategy: balanced 4-5 from each team, role-diverse
  1802→  const fixedPlayers: TGPlayer[] = [];
  1803→  const fixedIds = new Set<number>();
  1804→
  1805→  const targetLeft = Math.min(4, sortedLeft.length);
  1806→  const targetRight = Math.min(4, sortedRight.length);
  1807→
  1808→  // Pick diverse players ensuring role coverage
  1809→  const pickDiverse = (pool: TGPlayer[], count: number, existing: TGPlayer[]): TGPlayer[] => {
  1810→    const picked: TGPlayer[] = [];
  1811→    const usedIds = new Set(existing.map(p => p.pl_id));
  1812→    const existingRoles = {
  1813→      wk: existing.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length,
  1814→      bat: existing.filter(p => p.role === PLAYER_ROLES.BATSMAN).length,
  1815→      ar: existing.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length,
  1816→      bowl: existing.filter(p => p.role === PLAYER_ROLES.BOWLER).length,
  1817→    };
  1818→
  1819→    // First pass: fill role gaps
  1820→    const roleNeeds = [
  1821→      { role: PLAYER_ROLES.WICKET_KEEPER, need: Math.max(0, 1 - existingRoles.wk), pool: pool.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER && !usedIds.has(p.pl_id)) },
  1822→      { role: PLAYER_ROLES.ALL_ROUNDER, need: Math.max(0, 1 - existingRoles.ar), pool: pool.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER && !usedIds.has(p.pl_id)) },
  1823→      { role: PLAYER_ROLES.BATSMAN, need: Math.max(0, 2 - existingRoles.bat), pool: pool.filter(p => p.role === PLAYER_ROLES.BATSMAN && !usedIds.has(p.pl_id)) },
  1824→      { role: PLAYER_ROLES.BOWLER, need: Math.max(0, 2 - existingRoles.bowl), pool: pool.filter(p => p.role === PLAYER_ROLES.BOWLER && !usedIds.has(p.pl_id)) },
  1825→    ];
  1826→
  1827→    for (const rn of roleNeeds) {
  1828→      for (let i = 0; i < rn.need && picked.length < count; i++) {
  1829→        const candidate = rn.pool.find(p => !picked.some(pp => pp.pl_id === p.pl_id));
  1830→        if (candidate) picked.push(candidate);
  1831→      }
  1832→    }
  1833→
  1834→    // Second pass: fill remaining with top-scored players
  1835→    const remaining = pool.filter(p => !usedIds.has(p.pl_id) && !picked.some(pp => pp.pl_id === p.pl_id));
  1836→    for (const p of remaining) {
  1837→      if (picked.length >= count) break;
  1838→      picked.push(p);
  1839→    }
  1840→
  1841→    return picked;
  1842→  };
  1843→
  1844→  const leftPicks = pickDiverse(sortedLeft, targetLeft, []);
  1845→  for (const p of leftPicks) { fixedPlayers.push(p); fixedIds.add(p.pl_id); }
  1846→
  1847→  const rightPicks = pickDiverse(sortedRight, targetRight, fixedPlayers);
  1848→  for (const p of rightPicks) { fixedPlayers.push(p); fixedIds.add(p.pl_id); }
  1849→
  1850→  // Fill to 8 if needed
  1851→  if (fixedPlayers.length < 8) {
  1852→    const remaining = eligible.filter(p => !fixedIds.has(p.pl_id)).sort(sortByScore);
  1853→    for (const p of remaining) {
  1854→      if (fixedPlayers.length >= 8) break;
  1855→      fixedPlayers.push(p); fixedIds.add(p.pl_id);
  1856→    }
  1857→  }
  1858→
  1859→  // Trim to 8 if over
  1860→  while (fixedPlayers.length > 8) {
  1861→    let minIdx = 0; let minScore = playerScore(fixedPlayers[0]);
  1862→    for (let i = 1; i < fixedPlayers.length; i++) {
  1863→      const s = playerScore(fixedPlayers[i]);
  1864→      if (s < minScore) { minScore = s; minIdx = i; }
  1865→    }
  1866→    const removed = fixedPlayers.splice(minIdx, 1)[0];
  1867→    fixedIds.delete(removed.pl_id);
  1868→  }
  1869→
  1870→  // === SELECT 5 CAPTAIN OPTIONS ===
  1871→  const cCandidates = eligible
  1872→    .filter(p => !avoidPlayerIds.has(p.pl_id))
  1873→    .sort((a, b) => {
  1874→      const scoreA = a.captain_percentage * 0.4 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
  1875→      const scoreB = b.captain_percentage * 0.4 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
  1876→      return scoreB - scoreA;
  1877→    })
  1878→    .slice(0, 5);
  1879→
  1880→  // === SELECT 5 VICE CAPTAIN OPTIONS ===
  1881→  const vcCandidates: TGPlayer[] = [];
  1882→  const vcPool = eligible
  1883→    .filter(p => !avoidPlayerIds.has(p.pl_id))
  1884→    .sort((a, b) => {
  1885→      const scoreA = a.vice_captain_percentage * 0.35 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
  1886→      const scoreB = b.vice_captain_percentage * 0.35 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
  1887→      return scoreB - scoreA;
  1888→    });
  1889→
  1890→  for (const p of vcPool) {
  1891→    if (vcCandidates.length >= 5) break;
  1892→    vcCandidates.push(p);
  1893→  }
  1894→
  1895→  return {
  1896→    fixedPlayers: fixedPlayers.slice(0, 8),
  1897→    captainOptions: cCandidates.slice(0, 5),
  1898→    viceCaptainOptions: vcCandidates.slice(0, 5),
  1899→  };
  1900→}

  1902→// Auto-replace a removed player in a list with the next best eligible player
  1903→export function autoReplacePlayer(
  1904→  removedPlayer: TGPlayer,
  1905→  currentList: TGPlayer[],
  1906→  allPlayers: TGPlayer[],
  1907→  avoidPlayerIds: Set<number>,
  1908→  scoringFn: (p: TGPlayer) => number = playerScore,
  1909→): TGPlayer[] {
  1910→  const currentIds = new Set(currentList.map(p => p.pl_id));
  1911→  const eligible = allPlayers.filter(p =>
  1912→    !currentIds.has(p.pl_id) && !avoidPlayerIds.has(p.pl_id) && p.pl_id !== removedPlayer.pl_id
  1913→  );
  1914→  const sorted = eligible.sort((a, b) => scoringFn(b) - scoringFn(a));
  1915→  const replacement = sorted[0];
  1916→  if (replacement) return [...currentList, replacement];
  1917→  return currentList;
  1918→}
