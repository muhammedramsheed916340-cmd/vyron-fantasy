   317→   317→// Cricket team constraints
   318→   318→const CRICKET_TEAM_SIZE = 11;
   319→   319→const MIN_WK = 1;
   320→   320→const MAX_WK = 4;
   321→   321→const MIN_BAT = 3;
   322→   322→const MAX_BAT = 6;
   323→   323→const MIN_AR = 1;
   324→   324→const MAX_AR = 4;
   325→   325→const MIN_BOWL = 3;
   326→   326→const MAX_BOWL = 6;
   327→   327→const MAX_CREDITS = 100;
   328→   328→const MAX_FROM_ONE_TEAM = 7;
   329→   329→
   330→   330→function seededRandom(seed: number): () => number {
   331→   331→  let s = seed;
   332→   332→  return () => {
   333→   333→    s = (s * 16807 + 0) % 2147483647;
   334→   334→    return (s - 1) / 2147483646;
   335→   335→  };
   336→   336→}
   337→   337→
   338→   338→function shuffleArray<T>(arr: T[], rng: () => number): T[] {
   339→   339→  const result = [...arr];
   340→   340→  for (let i = result.length - 1; i > 0; i--) {
   341→   341→    const j = Math.floor(rng() * (i + 1));
   342→   342→    [result[i], result[j]] = [result[j], result[i]];
   343→   343→  }
   344→   344→  return result;
   345→   345→}

   347→   347→export function generateTeams(
   348→   348→  leftPlayers: TGPlayer[],
   349→   349→  rightPlayers: TGPlayer[],
   350→   350→  category: string,
   351→   351→  count: number,
   352→   352→  seed: number = Date.now()
   353→   353→): GeneratedTeam[] {
   354→   354→  const allPlayers = [...leftPlayers, ...rightPlayers];
   355→   355→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
   356→   356→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
   357→   357→
   358→   358→  // Separate by role
   359→   359→  const wk = allPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER);
   360→   360→  const bat = allPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN);
   361→   361→  const ar = allPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER);
   362→   362→  const bowl = allPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER);
   363→   363→
   364→   364→  const teams: GeneratedTeam[] = [];
   365→   365→  const rng = seededRandom(seed);
   366→   366→
   367→   367→  // Category-specific strategies
   368→   368→  const isH2H = category === 'H2H';
   369→   369→  const isSL = category === 'SL';
   370→   370→
   371→   371→  for (let t = 0; t < count; t++) {
   372→   372→    let attempts = 0;
   373→   373→    let team: TGPlayer[] | null = null;
   374→   374→
   375→   375→    while (!team && attempts < 200) {
   376→   376→      attempts++;
   377→   377→      const sr = seededRandom(seed + t * 1000 + attempts);
   378→   378→
   379→   379→      // Determine role counts based on category
   380→   380→      let wkCount, batCount, arCount, bowlCount;
   381→   381→
   382→   382→      if (isH2H) {
   383→   383→        // H2H: More conservative, popular players
   384→   384→        wkCount = MIN_WK + (sr() > 0.5 ? 1 : 0);
   385→   385→        arCount = MIN_AR + (sr() > 0.6 ? 1 : 0);
   386→   386→        batCount = Math.max(MIN_BAT, 5 - wkCount + (sr() > 0.5 ? 1 : 0));
   387→   387→        bowlCount = CRICKET_TEAM_SIZE - wkCount - batCount - arCount;
   388→   388→      } else if (isSL) {
   389→   389→        // SL: Balanced
   390→   390→        wkCount = 1 + (sr() > 0.7 ? 1 : 0);
   391→   391→        arCount = 2 + (sr() > 0.5 ? 1 : 0);
   392→   392→        batCount = 3 + (sr() > 0.5 ? 1 : 0);
   393→   393→        bowlCount = CRICKET_TEAM_SIZE - wkCount - batCount - arCount;
   394→   394→      } else {
   395→   395→        // Mega GL: More differential picks, varied combinations
   396→   396→        const patterns = [
   397→   397→          [1, 3, 2, 5], [1, 4, 1, 5], [1, 3, 3, 4], [1, 4, 2, 4],
   398→   398→          [2, 3, 2, 4], [1, 5, 1, 4], [2, 3, 1, 5], [1, 4, 3, 3],
   399→   399→          [2, 4, 1, 4], [1, 3, 1, 6], [2, 2, 3, 4], [1, 5, 2, 3],
   400→   400→        ];
   401→   401→        const pattern = patterns[Math.floor(sr() * patterns.length)];
   402→   402→        [wkCount, batCount, arCount, bowlCount] = pattern;
   403→   403→      }
   404→   404→
   405→   405→      // Validate counts
   406→   406→      if (wkCount < MIN_WK || wkCount > MAX_WK) continue;
   407→   407→      if (batCount < MIN_BAT || batCount > MAX_BAT) continue;
   408→   408→      if (arCount < MIN_AR || arCount > MAX_AR) continue;
   409→   409→      if (bowlCount < MIN_BOWL || bowlCount > MAX_BOWL) continue;
   410→   410→      if (wkCount + batCount + arCount + bowlCount !== CRICKET_TEAM_SIZE) continue;
   411→   411→
   412→   412→      // Pick players for each role
   413→   413→      const shuffledWK = shuffleArray(wk, sr);
   414→   414→      const shuffledBat = shuffleArray(bat, sr);
   415→   415→      const shuffledAR = shuffleArray(ar, sr);
   416→   416→      const shuffledBowl = shuffleArray(bowl, sr);
   417→   417→
   418→   418→      // For Mega GL, sort by selected_by sometimes to get differential picks
   419→   419→      let selectedWK, selectedBat, selectedAR, selectedBowl;
   420→   420→
   421→   421→      if (!isH2H && sr() > 0.4) {
   422→   422→        // Sort some roles by selection % to mix popular and differential
   423→   423→        const sortFn = (a: TGPlayer, b: TGPlayer) => a.selected_by - b.selected_by;
   424→   424→        selectedWK = shuffleArray([...wk].sort(sortFn), sr).slice(0, wkCount);
   425→   425→        selectedBat = shuffleArray([...bat].sort(sortFn), sr).slice(0, batCount);
   426→   426→        selectedAR = shuffleArray([...ar].sort(sortFn), sr).slice(0, arCount);
   427→   427→        selectedBowl = shuffleArray([...bowl].sort(sortFn), sr).slice(0, bowlCount);
   428→   428→      } else if (isH2H) {
   429→   429→        // H2H: Pick most selected players
   430→   430→        const sortFn = (a: TGPlayer, b: TGPlayer) => b.selected_by - a.selected_by;
   431→   431→        selectedWK = [...wk].sort(sortFn).slice(0, wkCount);
   432→   432→        selectedBat = shuffleArray([...bat].sort(sortFn), sr).slice(0, batCount);
   433→   433→        selectedAR = [...ar].sort(sortFn).slice(0, arCount);
   434→   434→        selectedBowl = shuffleArray([...bowl].sort(sortFn), sr).slice(0, bowlCount);
   435→   435→      } else {
   436→   436→        selectedWK = shuffledWK.slice(0, wkCount);
   437→   437→        selectedBat = shuffledBat.slice(0, batCount);
   438→   438→        selectedAR = shuffledAR.slice(0, arCount);
   439→   439→        selectedBowl = shuffledBowl.slice(0, bowlCount);
   440→   440→      }
   441→   441→
   442→   442→      // Check if we have enough players
   443→   443→      if (selectedWK.length < wkCount || selectedBat.length < batCount ||
   444→   444→          selectedAR.length < arCount || selectedBowl.length < bowlCount) continue;
   445→   445→
   446→   446→      const selected = [...selectedWK, ...selectedBat, ...selectedAR, ...selectedBowl];
   447→   447→
   448→   448→      // Check total credits
   449→   449→      const totalCredits = selected.reduce((sum, p) => sum + p.credits, 0);
   450→   450→      if (totalCredits > MAX_CREDITS) continue;
   451→   451→
   452→   452→      // Check team distribution (max 7 from one team)
   453→   453→      const leftCount = selected.filter(p => p.team_name === leftTeamName).length;
   454→   454→      const rightCount = selected.filter(p => p.team_name === rightTeamName).length;
   455→   455→      if (leftCount > MAX_FROM_ONE_TEAM || rightCount > MAX_FROM_ONE_TEAM) continue;
   456→   456→
   457→   457→      // Ensure at least 1 from each team
   458→   458→      if (leftCount < 1 || rightCount < 1) continue;
   459→   459→
   460→   460→      team = selected;
   461→   461→    }
   462→   462→
   463→   463→    if (team) {
   464→   464→      // Select Captain & Vice-Captain
   465→   465→      // Sort by a combination of points and selection % for C/VC
   466→   466→      const sorted = [...team].sort((a, b) =>
   467→   467→        (b.points * 0.4 + b.selected_by * 0.3 + b.captain_percentage * 0.3) -
   468→   468→        (a.points * 0.4 + a.selected_by * 0.3 + a.captain_percentage * 0.3)
   469→   469→      );
   470→   470→
   471→   471→      const captain = sorted[0];
   472→   472→      const viceCaptain = sorted.length > 1 ? sorted[1] : sorted[0];
   473→   473→
   474→   474→      teams.push({
   475→   475→        id: t + 1,
   476→   476→        captain,
   477→   477→        viceCaptain,
   478→   478→        players: team,
   479→   479→      });
   480→   480→    }
   481→   481→  }
   482→   482→
   483→   483→  return teams;
   484→   484→}

   486→   486→// ============ Extra Team Generation Algorithm ============
   487→   487→// 8 Fixed Players + remaining players filled by engine
   488→   488→// 5 Captain options + 5 Vice Captain options distributed across teams
   489→   489→
   490→   490→export interface ExtraTeamGenInput {
   491→   491→  fixedPlayers: TGPlayer[];      // Exactly 8 fixed players
   492→   492→  captainOptions: TGPlayer[];    // Exactly 5 captain options
   493→   493→  viceCaptainOptions: TGPlayer[]; // Exactly 5 vice captain options
   494→   494→  leftPlayers: TGPlayer[];       // Full left team player pool
   495→   495→  rightPlayers: TGPlayer[];      // Full right team player pool
   496→   496→  category: string;              // 'Mega GL' | 'SL' | 'H2H'
   497→   497→  count: number;                 // Number of teams to generate
   498→   498→  seed?: number;
   499→   499→}
   500→   500→
   501→   501→export function generateExtraTeams(input: ExtraTeamGenInput): GeneratedTeam[] {
   502→   502→  const {
   503→   503→    fixedPlayers,
   504→   504→    captainOptions,
   505→   505→    viceCaptainOptions,
   506→   506→    leftPlayers,
   507→   507→    rightPlayers,
   508→   508→    category,
   509→   509→    count,
   510→   510→    seed = Date.now(),
   511→   511→  } = input;
   512→   512→
   513→   513→  const allPlayers = [...leftPlayers, ...rightPlayers];
   514→   514→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
   515→   515→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
   516→   516→
   517→   517→  // Pool of players excluding the 8 fixed ones (use pl_id for identity)
   518→   518→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   519→   519→  const remainingPool = allPlayers.filter(p => !fixedIds.has(p.pl_id));
   520→   520→
   521→   521→  // Separate remaining pool by role
   522→   522→  const remWK = remainingPool.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER);
   523→   523→  const remBat = remainingPool.filter(p => p.role === PLAYER_ROLES.BATSMAN);
   524→   524→  const remAR = remainingPool.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER);
   525→   525→  const remBowl = remainingPool.filter(p => p.role === PLAYER_ROLES.BOWLER);
   526→   526→
   527→   527→  // Count fixed players by role
   528→   528→  const fixedWKCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   529→   529→  const fixedBatCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   530→   530→  const fixedARCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   531→   531→  const fixedBowlCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   532→   532→
   533→   533→  const fixedCredits = fixedPlayers.reduce((sum, p) => sum + p.credits, 0);
   534→   534→  const remainingCredits = MAX_CREDITS - fixedCredits;
   535→   535→  const neededPlayers = CRICKET_TEAM_SIZE - fixedPlayers.length; // 3
   536→   536→
   537→   537→  const teams: GeneratedTeam[] = [];
   538→   538→
   539→   539→  // Pre-validate: fixed players must have credits <= 100
   540→   540→  if (fixedCredits > MAX_CREDITS) return teams;
   541→   541→
   542→   542→  // Pre-validate: fixed players must respect max 7 from one team
   543→   543→  const fixedLeftCount = fixedPlayers.filter(p => p.team_name === leftTeamName).length;
   544→   544→  const fixedRightCount = fixedPlayers.filter(p => p.team_name === rightTeamName).length;
   545→   545→  if (fixedLeftCount > MAX_FROM_ONE_TEAM || fixedRightCount > MAX_FROM_ONE_TEAM) return teams;
   546→   546→
   547→   547→  // Generate all valid C/VC combinations from the 5C and 5VC options
   548→   548→  // C must be in the team, VC must be in the team, C ≠ VC
   549→   549→  const cVCCombos: { c: TGPlayer; vc: TGPlayer }[] = [];
   550→   550→  for (const c of captainOptions) {
   551→   551→    for (const vc of viceCaptainOptions) {
   552→   552→      if (c.pl_id !== vc.pl_id) {
   553→   553→        cVCCombos.push({ c, vc });
   554→   554→      }
   555→   555→    }
   556→   556→  }
   557→   557→  // If no valid combos, return empty
   558→   558→  if (cVCCombos.length === 0) return teams;
   559→   559→
   560→   560→  // Category-specific role distribution preferences for remaining players
   561→   561→  const isH2H = category === 'H2H';
   562→   562→  const isSL = category === 'SL';
   563→   563→
   564→   564→  for (let t = 0; t < count; t++) {
   565→   565→    let attempts = 0;
   566→   566→    let team: TGPlayer[] | null = null;
   567→   567→    let teamC: TGPlayer | null = null;
   568→   568→    let teamVC: TGPlayer | null = null;
   569→   569→
   570→   570→    while (!team && attempts < 500) {
   571→   571→      attempts++;
   572→   572→      const sr = seededRandom(seed + t * 1000 + attempts);
   573→   573→
   574→   574→      // We need `neededPlayers` more players (3)
   575→   575→      // Determine target role counts for the full 11-player team
   576→   576→      let targetWK, targetBat, targetAR, targetBowl;
   577→   577→
   578→   578→      if (isH2H) {
   579→   579→        targetWK = MIN_WK + (sr() > 0.5 ? 1 : 0);
   580→   580→        targetAR = MIN_AR + (sr() > 0.6 ? 1 : 0);
   581→   581→        targetBat = Math.max(MIN_BAT, 5 - targetWK + (sr() > 0.5 ? 1 : 0));
   582→   582→        targetBowl = CRICKET_TEAM_SIZE - targetWK - targetBat - targetAR;
   583→   583→      } else if (isSL) {
   584→   584→        targetWK = 1 + (sr() > 0.7 ? 1 : 0);
   585→   585→        targetAR = 2 + (sr() > 0.5 ? 1 : 0);
   586→   586→        targetBat = 3 + (sr() > 0.5 ? 1 : 0);
   587→   587→        targetBowl = CRICKET_TEAM_SIZE - targetWK - targetBat - targetAR;
   588→   588→      } else {
   589→   589→        const patterns = [
   590→   590→          [1, 3, 2, 5], [1, 4, 1, 5], [1, 3, 3, 4], [1, 4, 2, 4],
   591→   591→          [2, 3, 2, 4], [1, 5, 1, 4], [2, 3, 1, 5], [1, 4, 3, 3],
   592→   592→          [2, 4, 1, 4], [1, 3, 1, 6], [2, 2, 3, 4], [1, 5, 2, 3],
   593→   593→        ];
   594→   594→        const pattern = patterns[Math.floor(sr() * patterns.length)];
   595→   595→        [targetWK, targetBat, targetAR, targetBowl] = pattern;
   596→   596→      }
   597→   597→
   598→   598→      // Validate total counts
   599→   599→      if (targetWK < MIN_WK || targetWK > MAX_WK) continue;
   600→   600→      if (targetBat < MIN_BAT || targetBat > MAX_BAT) continue;
   601→   601→      if (targetAR < MIN_AR || targetAR > MAX_AR) continue;
   602→   602→      if (targetBowl < MIN_BOWL || targetBowl > MAX_BOWL) continue;
   603→   603→      if (targetWK + targetBat + targetAR + targetBowl !== CRICKET_TEAM_SIZE) continue;
   604→   604→
   605→   605→      // How many more of each role do we need?
   606→   606→      const needWK = Math.max(0, targetWK - fixedWKCount);
   607→   607→      const needBat = Math.max(0, targetBat - fixedBatCount);
   608→   608→      const needAR = Math.max(0, targetAR - fixedARCount);
   609→   609→      const needBowl = Math.max(0, targetBowl - fixedBowlCount);
   610→   610→
   611→   611→      // Total needed must equal neededPlayers (3)
   612→   612→      if (needWK + needBat + needAR + needBowl !== neededPlayers) continue;
   613→   613→
   614→   614→      // Check if we have enough remaining players for each role
   615→   615→      if (remWK.length < needWK || remBat.length < needBat ||
   616→   616→          remAR.length < needAR || remBowl.length < needBowl) continue;
   617→   617→
   618→   618→      // Pick remaining players for each role
   619→   619→      let pickedWK: TGPlayer[], pickedBat: TGPlayer[], pickedAR: TGPlayer[], pickedBowl: TGPlayer[];
   620→   620→
   621→   621→      if (!isH2H && sr() > 0.4) {
   622→   622→        const sortFn = (a: TGPlayer, b: TGPlayer) => a.selected_by - b.selected_by;
   623→   623→        pickedWK = shuffleArray([...remWK].sort(sortFn), sr).slice(0, needWK);
   624→   624→        pickedBat = shuffleArray([...remBat].sort(sortFn), sr).slice(0, needBat);
   625→   625→        pickedAR = shuffleArray([...remAR].sort(sortFn), sr).slice(0, needAR);
   626→   626→        pickedBowl = shuffleArray([...remBowl].sort(sortFn), sr).slice(0, needBowl);
   627→   627→      } else if (isH2H) {
   628→   628→        const sortFn = (a: TGPlayer, b: TGPlayer) => b.selected_by - a.selected_by;
   629→   629→        pickedWK = [...remWK].sort(sortFn).slice(0, needWK);
   630→   630→        pickedBat = shuffleArray([...remBat].sort(sortFn), sr).slice(0, needBat);
   631→   631→        pickedAR = [...remAR].sort(sortFn).slice(0, needAR);
   632→   632→        pickedBowl = shuffleArray([...remBowl].sort(sortFn), sr).slice(0, needBowl);
   633→   633→      } else {
   634→   634→        pickedWK = shuffleArray(remWK, sr).slice(0, needWK);
   635→   635→        pickedBat = shuffleArray(remBat, sr).slice(0, needBat);
   636→   636→        pickedAR = shuffleArray(remAR, sr).slice(0, needAR);
   637→   637→        pickedBowl = shuffleArray(remBowl, sr).slice(0, needBowl);
   638→   638→      }
   639→   639→
   640→   640→      if (pickedWK.length < needWK || pickedBat.length < needBat ||
   641→   641→          pickedAR.length < needAR || pickedBowl.length < needBowl) continue;
   642→   642→
   643→   643→      const remaining = [...pickedWK, ...pickedBat, ...pickedAR, ...pickedBowl];
   644→   644→      const fullTeam = [...fixedPlayers, ...remaining];
   645→   645→
   646→   646→      // Check no duplicates (shouldn't happen but safety check)
   647→   647→      const uniqueIds = new Set(fullTeam.map(p => p.pl_id));
   648→   648→      if (uniqueIds.size !== fullTeam.length) continue;
   649→   649→
   650→   650→      // Check total credits
   651→   651→      const totalCredits = fullTeam.reduce((sum, p) => sum + p.credits, 0);
   652→   652→      if (totalCredits > MAX_CREDITS) continue;
   653→   653→
   654→   654→      // Check team distribution
   655→   655→      const leftCount = fullTeam.filter(p => p.team_name === leftTeamName).length;
   656→   656→      const rightCount = fullTeam.filter(p => p.team_name === rightTeamName).length;
   657→   657→      if (leftCount > MAX_FROM_ONE_TEAM || rightCount > MAX_FROM_ONE_TEAM) continue;
   658→   658→      if (leftCount < 1 || rightCount < 1) continue;
   659→   659→
   660→   660→      // Select C/VC combination for this team
   661→   661→      // Round-robin through the valid combos, also try to ensure C and VC are in the team
   662→   662→      const comboIndex = (t + attempts) % cVCCombos.length;
   663→   663→      let combo = cVCCombos[comboIndex];
   664→   664→
   665→   665→      // Ensure C and VC are in this team
   666→   666→      const teamPlayerIds = new Set(fullTeam.map(p => p.pl_id));
   667→   667→      if (!teamPlayerIds.has(combo.c.pl_id) || !teamPlayerIds.has(combo.vc.pl_id)) {
   668→   668→        // Try to find a valid combo where both C and VC are in the team
   669→   669→        let foundValidCombo = false;
   670→   670→        for (let ci = 0; ci < cVCCombos.length; ci++) {
   671→   671→          const tryCombo = cVCCombos[(comboIndex + ci) % cVCCombos.length];
   672→   672→          if (teamPlayerIds.has(tryCombo.c.pl_id) && teamPlayerIds.has(tryCombo.vc.pl_id)) {
   673→   673→            combo = tryCombo;
   674→   674→            foundValidCombo = true;
   675→   675→            break;
   676→   676→          }
   677→   677→        }
   678→   678→        if (!foundValidCombo) {
   679→   679→          // Fallback: use top players in team as C/VC
   680→   680→          const sorted = [...fullTeam].sort((a, b) =>
   681→   681→            (b.points * 0.4 + b.selected_by * 0.3 + b.captain_percentage * 0.3) -
   682→   682→            (a.points * 0.4 + a.selected_by * 0.3 + a.captain_percentage * 0.3)
   683→   683→          );
   684→   684→          combo = { c: sorted[0], vc: sorted.length > 1 ? sorted[1] : sorted[0] };
   685→   685→        }
   686→   686→      }
   687→   687→
   688→   688→      team = fullTeam;
   689→   689→      teamC = combo.c;
   690→   690→      teamVC = combo.vc;
   691→   691→    }
   692→   692→
   693→   693→    if (team && teamC && teamVC) {
   694→   694→      teams.push({
   695→   695→        id: t + 1,
   696→   696→        captain: teamC,
   697→   697→        viceCaptain: teamVC,
   698→   698→        players: team,
   699→   699→      });
   700→   700→    }
   701→   701→  }
   702→   702→
   703→   703→  return teams;
   704→   704→}

   706→   706→// ============ Auto Select for Extra Team Generation ============
   707→   707→// Uses real player data (selection %, captain %, points, credits, role, team balance)
   708→   708→// to intelligently pick 8 FIX, 5 C, and 5 VC
   709→   709→
   710→   710→export interface AutoSelectResult {
   711→   711→  fixedPlayers: TGPlayer[];
   712→   712→  captainOptions: TGPlayer[];
   713→   713→  viceCaptainOptions: TGPlayer[];
   714→   714→}
   715→   715→
   716→   716→// Player scoring for auto-selection: higher = better pick
   717→   717→function playerScore(p: TGPlayer): number {
   718→   718→  return (
   719→   719→    p.selected_by * 0.35 +       // selection % - most important for GL
   720→   720→    p.captain_percentage * 0.25 + // captain % - indicates trust
   721→   721→    p.points * 0.2 +             // points - performance indicator
   722→   722→    p.vice_captain_percentage * 0.1 + // VC %
   723→   723→    (p.playing === 1 ? 10 : 0) + // confirmed playing bonus
   724→   724→    (p.role === PLAYER_ROLES.ALL_ROUNDER ? 3 : 0) + // AR bonus (flexible)
   725→   725→    (p.role === PLAYER_ROLES.WICKET_KEEPER ? 1 : 0) // WK slight bonus
   726→   726→  );
   727→   727→}
   728→   728→
   729→   729→export function autoSelectExtraPlayers(
   730→   730→  leftPlayers: TGPlayer[],
   731→   731→  rightPlayers: TGPlayer[],
   732→   732→  avoidPlayerIds: Set<number> = new Set(),
   733→   733→): AutoSelectResult {
   734→   734→  const allPlayers = [...leftPlayers, ...rightPlayers];
   735→   735→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
   736→   736→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
   737→   737→
   738→   738→  // Filter out avoided players and unconfirmed players when lineup is out
   739→   739→  const eligible = allPlayers.filter(p => {
   740→   740→    if (avoidPlayerIds.has(p.pl_id)) return false;
   741→   741→    if (p.playing === 0) {
   742→   742→      const hasConfirmed = allPlayers.some(ap => ap.playing === 1);
   743→   743→      if (hasConfirmed) return false;
   744→   744→    }
   745→   745→    return true;
   746→   746→  });
   747→   747→
   748→   748→  const eligibleLeft = eligible.filter(p => p.team_name === leftTeamName);
   749→   749→  const eligibleRight = eligible.filter(p => p.team_name === rightTeamName);
   750→   750→
   751→   751→  // Sort by score descending
   752→   752→  const sortByScore = (a: TGPlayer, b: TGPlayer) => playerScore(b) - playerScore(a);
   753→   753→  const sortedLeft = [...eligibleLeft].sort(sortByScore);
   754→   754→  const sortedRight = [...eligibleRight].sort(sortByScore);
   755→   755→
   756→   756→  // === SELECT 8 FIX PLAYERS ===
   757→   757→  // Strategy: balanced 4-5 from each team, role-diverse
   758→   758→  const fixedPlayers: TGPlayer[] = [];
   759→   759→  const fixedIds = new Set<number>();
   760→   760→
   761→   761→  const targetLeft = Math.min(4, sortedLeft.length);
   762→   762→  const targetRight = Math.min(4, sortedRight.length);
   763→   763→
   764→   764→  // Pick diverse players ensuring role coverage
   765→   765→  const pickDiverse = (pool: TGPlayer[], count: number, existing: TGPlayer[]): TGPlayer[] => {
   766→   766→    const picked: TGPlayer[] = [];
   767→   767→    const usedIds = new Set(existing.map(p => p.pl_id));
   768→   768→    const existingRoles = {
   769→   769→      wk: existing.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length,
   770→   770→      bat: existing.filter(p => p.role === PLAYER_ROLES.BATSMAN).length,
   771→   771→      ar: existing.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length,
   772→   772→      bowl: existing.filter(p => p.role === PLAYER_ROLES.BOWLER).length,
   773→   773→    };
   774→   774→
   775→   775→    // First pass: fill role gaps
   776→   776→    const roleNeeds = [
   777→   777→      { role: PLAYER_ROLES.WICKET_KEEPER, need: Math.max(0, 1 - existingRoles.wk), pool: pool.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER && !usedIds.has(p.pl_id)) },
   778→   778→      { role: PLAYER_ROLES.ALL_ROUNDER, need: Math.max(0, 1 - existingRoles.ar), pool: pool.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER && !usedIds.has(p.pl_id)) },
   779→   779→      { role: PLAYER_ROLES.BATSMAN, need: Math.max(0, 2 - existingRoles.bat), pool: pool.filter(p => p.role === PLAYER_ROLES.BATSMAN && !usedIds.has(p.pl_id)) },
   780→   780→      { role: PLAYER_ROLES.BOWLER, need: Math.max(0, 2 - existingRoles.bowl), pool: pool.filter(p => p.role === PLAYER_ROLES.BOWLER && !usedIds.has(p.pl_id)) },
   781→   781→    ];
   782→   782→
   783→   783→    for (const rn of roleNeeds) {
   784→   784→      for (let i = 0; i < rn.need && picked.length < count; i++) {
   785→   785→        const candidate = rn.pool.find(p => !picked.some(pp => pp.pl_id === p.pl_id));
   786→   786→        if (candidate) picked.push(candidate);
   787→   787→      }
   788→   788→    }
   789→   789→
   790→   790→    // Second pass: fill remaining with top-scored players
   791→   791→    const remaining = pool.filter(p => !usedIds.has(p.pl_id) && !picked.some(pp => pp.pl_id === p.pl_id));
   792→   792→    for (const p of remaining) {
   793→   793→      if (picked.length >= count) break;
   794→   794→      picked.push(p);
   795→   795→    }
   796→   796→
   797→   797→    return picked;
   798→   798→  };
   799→   799→
   800→   800→  const leftPicks = pickDiverse(sortedLeft, targetLeft, []);
   801→   801→  for (const p of leftPicks) { fixedPlayers.push(p); fixedIds.add(p.pl_id); }
   802→   802→
   803→   803→  const rightPicks = pickDiverse(sortedRight, targetRight, fixedPlayers);
   804→   804→  for (const p of rightPicks) { fixedPlayers.push(p); fixedIds.add(p.pl_id); }
   805→   805→
   806→   806→  // Fill to 8 if needed
   807→   807→  if (fixedPlayers.length < 8) {
   808→   808→    const remaining = eligible.filter(p => !fixedIds.has(p.pl_id)).sort(sortByScore);
   809→   809→    for (const p of remaining) {
   810→   810→      if (fixedPlayers.length >= 8) break;
   811→   811→      fixedPlayers.push(p); fixedIds.add(p.pl_id);
   812→   812→    }
   813→   813→  }
   814→   814→
   815→   815→  // Trim to 8 if over
   816→   816→  while (fixedPlayers.length > 8) {
   817→   817→    let minIdx = 0; let minScore = playerScore(fixedPlayers[0]);
   818→   818→    for (let i = 1; i < fixedPlayers.length; i++) {
   819→   819→      const s = playerScore(fixedPlayers[i]);
   820→   820→      if (s < minScore) { minScore = s; minIdx = i; }
   821→   821→    }
   822→   822→    const removed = fixedPlayers.splice(minIdx, 1)[0];
   823→   823→    fixedIds.delete(removed.pl_id);
   824→   824→  }
   825→   825→
   826→   826→  // === SELECT 5 CAPTAIN OPTIONS ===
   827→   827→  const cCandidates = eligible
   828→   828→    .filter(p => !avoidPlayerIds.has(p.pl_id))
   829→   829→    .sort((a, b) => {
   830→   830→      const scoreA = a.captain_percentage * 0.4 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
   831→   831→      const scoreB = b.captain_percentage * 0.4 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
   832→   832→      return scoreB - scoreA;
   833→   833→    })
   834→   834→    .slice(0, 5);
   835→   835→
   836→   836→  // === SELECT 5 VICE CAPTAIN OPTIONS ===
   837→   837→  const vcCandidates: TGPlayer[] = [];
   838→   838→  const vcPool = eligible
   839→   839→    .filter(p => !avoidPlayerIds.has(p.pl_id))
   840→   840→    .sort((a, b) => {
   841→   841→      const scoreA = a.vice_captain_percentage * 0.35 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
   842→   842→      const scoreB = b.vice_captain_percentage * 0.35 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
   843→   843→      return scoreB - scoreA;
   844→   844→    });
   845→   845→
   846→   846→  for (const p of vcPool) {
   847→   847→    if (vcCandidates.length >= 5) break;
   848→   848→    vcCandidates.push(p);
   849→   849→  }
   850→   850→
   851→   851→  return {
   852→   852→    fixedPlayers: fixedPlayers.slice(0, 8),
   853→   853→    captainOptions: cCandidates.slice(0, 5),
   854→   854→    viceCaptainOptions: vcCandidates.slice(0, 5),
   855→   855→  };
   856→   856→}

   858→   858→// Auto-replace a removed player in a list with the next best eligible player
   859→   859→export function autoReplacePlayer(
   860→   860→  removedPlayer: TGPlayer,
   861→   861→  currentList: TGPlayer[],
   862→   862→  allPlayers: TGPlayer[],
   863→   863→  avoidPlayerIds: Set<number>,
   864→   864→  scoringFn: (p: TGPlayer) => number = playerScore,
   865→   865→): TGPlayer[] {
   866→   866→  const currentIds = new Set(currentList.map(p => p.pl_id));
   867→   867→  const eligible = allPlayers.filter(p =>
   868→   868→    !currentIds.has(p.pl_id) && !avoidPlayerIds.has(p.pl_id) && p.pl_id !== removedPlayer.pl_id
   869→   869→  );
   870→   870→  const sorted = eligible.sort((a, b) => scoringFn(b) - scoringFn(a));
   871→   871→  const replacement = sorted[0];
   872→   872→  if (replacement) return [...currentList, replacement];
   873→   873→  return currentList;
   874→   874→}
