     6→// ============ Platform Name Normalization ============

     8→/**
     9→ * Normalize platform names to consistent lowercase identifiers.
    10→ * The TG API may return varied casing in fantasy_id_list[].name
    11→ * (e.g., "Dream11", "dream11", "My11Circle", "my11circle").
    12→ * All internal references use lowercase: "dream11", "my11circle".

    14→export function normalizePlatformName(name: string): string {
    15→  return name.toLowerCase().replace(/[\s_-]/g, '');

    18→/**
    19→ * Resolve a player's platform-specific ID from their fantasy_id_list.
    20→ * Uses case-insensitive matching to handle API casing variations.
    21→ */
    22→export function resolvePlatformPlayerId(
    23→  player: TGPlayer,
    24→  platform: string,
    25→): number | null {
    26→  if (!player.fantasy_id_list || player.fantasy_id_list.length === 0) return null;
    27→  const target = normalizePlatformName(platform);
    28→  const found = player.fantasy_id_list.find(
    29→    f => normalizePlatformName(f.name) === target
    30→  );
    31→  return found ? found.id : null;
    32→}

    62→// ============ Lineup Utilities ============
    63→
    64→/**
    65→ * Determine lineup mode based on player data.
    66→ * AFTER LINEUP: enough players have playing === 1 to form valid teams.
    67→ *   Requires at least 11 confirmed total AND at least 4 from each team
    68→ *   (due to max-7-from-one-team constraint: 11 - 7 = 4).
    69→ *   If only a few players are confirmed (partial lineup), we stay in
    70→ *   'before' mode so team generation doesn't fail with too few eligible players.
    71→ * BEFORE LINEUP: lineup not fully confirmed yet — all players eligible.
    72→ *
    73→ * BUG FIX: Previously used players.some(p => p.playing === 1) which
    74→ * switched to 'after' mode when even ONE player was confirmed, causing
    75→ * getEligiblePlayers to filter out all non-confirmed players and leaving
    76→ * too few to form valid 11-player teams → 0 teams generated.
    77→ */
    78→export function getLineupMode(players: TGPlayer[]): 'before' | 'after' {
    79→  const confirmed = players.filter(p => p.playing === 1);
    80→
    81→  // Need at least 11 confirmed players total to form a valid team
    82→  if (confirmed.length < 11) return 'before';
    83→
    84→  // Need at least 4 confirmed from each team
    85→  // (max 7 from one team → need at least 11-7=4 from the other)
    86→  const teamNames = [...new Set(players.map(p => p.team_name))];
    87→  for (const team of teamNames) {
    88→    if (team && confirmed.filter(p => p.team_name === team).length < 4) {
    89→      return 'before';
    90→    }
    91→  }
    92→
    93→  return 'after';
    94→}
    95→
    96→/**
    97→ * Filter players based on lineup mode.
    98→ * BEFORE LINEUP: all players are eligible (no one confirmed OUT yet).
    99→ * AFTER LINEUP: ONLY players with playing === 1 are eligible.
   100→ */
   101→export function getEligiblePlayers(
   102→  allPlayers: TGPlayer[],
   103→  avoidPlayerIds: Set<number> = new Set(),
   104→): TGPlayer[] {
   105→  const mode = getLineupMode(allPlayers);
   106→
   107→  return allPlayers.filter(p => {
   108→    // Avoid players are always excluded
   109→    if (avoidPlayerIds.has(p.pl_id)) return false;
   110→
   111→    if (mode === 'after') {
   112→      // AFTER LINEUP: ONLY confirmed Playing XI players
   113→      return p.playing === 1;
   114→    }
   115→
   116→    // BEFORE LINEUP: all players eligible
   117→    return true;
   118→  });
   119→}

   121→/**
   122→ * Check if a specific player is eligible based on lineup mode.

   124→export function isPlayerEligible(
   125→  player: TGPlayer,
   126→  allPlayers: TGPlayer[],
   127→  avoidPlayerIds: Set<number> = new Set(),
   128→): { eligible: boolean; reason?: string } {
   129→  if (avoidPlayerIds.has(player.pl_id)) {
   130→    return { eligible: false, reason: 'AVOID' };
   131→  }
   132→
   133→  const mode = getLineupMode(allPlayers);
   134→
   135→  if (mode === 'after' && player.playing !== 1) {
   136→    return { eligible: false, reason: 'OUT / NOT IN PLAYING XI' };
   137→  }
   138→
   139→  return { eligible: true };
   140→}

   142→/**
   143→ * Validate a team for lineup eligibility.
   144→ * Returns list of invalid players with reasons.
   145→ */
   146→export function validateTeamForLineup(
   147→  team: GeneratedTeam,
   148→  allPlayers: TGPlayer[],
   149→  avoidPlayerIds: Set<number> = new Set(),
   150→): { valid: boolean; invalidPlayers: { player: TGPlayer; reason: string }[] } {
   151→  const mode = getLineupMode(allPlayers);
   152→  const invalidPlayers: { player: TGPlayer; reason: string }[] = [];
   153→
   154→  for (const player of team.players) {
   155→    if (avoidPlayerIds.has(player.pl_id)) {
   156→      invalidPlayers.push({ player, reason: 'AVOID' });
   157→      continue;
   158→    }
   159→
   160→    if (mode === 'after' && player.playing !== 1) {
   161→      invalidPlayers.push({ player, reason: 'OUT / NOT IN PLAYING XI' });
   162→    }
   163→  }
   164→
   165→  return { valid: invalidPlayers.length === 0, invalidPlayers };
   166→}

   179→/**
   180→ * Get all valid role combinations for an 11-player cricket team.
   181→ */
   182→export function getAllValidCombinations(): RoleCombination[] {
   183→  const combos: RoleCombination[] = [];
   184→  for (let wk = MIN_WK; wk <= MAX_WK; wk++) {
   185→    for (let bat = MIN_BAT; bat <= MAX_BAT; bat++) {
   186→      for (let ar = MIN_AR; ar <= MAX_AR; ar++) {
   187→        const bowl = CRICKET_TEAM_SIZE - wk - bat - ar;
   188→        if (bowl < MIN_BOWL || bowl > MAX_BOWL) continue;
   189→        combos.push({ wk, bat, ar, bowl });
   190→      }
   191→    }
   192→  }
   193→  return combos;
   194→}

   196→/**
   197→ * Check if a role combination is compatible with a set of fixed players.
   198→ */
   199→export function isCombinationCompatibleWithFixed(
   200→  combo: RoleCombination,
   201→  fixedPlayers: TGPlayer[],
   202→): boolean {
   203→  const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   204→  const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   205→  const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   206→  const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   207→
   208→  // Fixed players must not exceed combination totals
   209→  if (fixedWK > combo.wk) return false;
   210→  if (fixedBat > combo.bat) return false;
   211→  if (fixedAR > combo.ar) return false;
   212→  if (fixedBowl > combo.bowl) return false;
   213→
   214→  // Remaining slots must be fillable
   215→  const remainingPlayers = CRICKET_TEAM_SIZE - fixedPlayers.length;
   216→  const neededWK = combo.wk - fixedWK;
   217→  const neededBat = combo.bat - fixedBat;
   218→  const neededAR = combo.ar - fixedAR;
   219→  const neededBowl = combo.bowl - fixedBowl;
   220→
   221→  return (neededWK + neededBat + neededAR + neededBowl) === remainingPlayers;
   222→}

   224→/**
   225→ * Get combinations compatible with fixed players and available player pool.
   226→ */
   227→export function getCompatibleCombinations(
   228→  fixedPlayers: TGPlayer[],
   229→  eligiblePlayers: TGPlayer[],
   230→): RoleCombination[] {
   231→  const all = getAllValidCombinations();
   232→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   233→  const remaining = eligiblePlayers.filter(p => !fixedIds.has(p.pl_id));
   234→
   235→  const remWK = remaining.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   236→  const remBat = remaining.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   237→  const remAR = remaining.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   238→  const remBowl = remaining.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   239→
   240→  return all.filter(combo => {
   241→    if (!isCombinationCompatibleWithFixed(combo, fixedPlayers)) return false;
   242→
   243→    const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   244→    const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   245→    const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   246→    const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   247→
   248→    if (remWK < combo.wk - fixedWK) return false;
   249→    if (remBat < combo.bat - fixedBat) return false;
   250→    if (remAR < combo.ar - fixedAR) return false;
   251→    if (remBowl < combo.bowl - fixedBowl) return false;
   252→
   253→    return true;
   254→  });
   255→}

   257→/**
   258→ * Auto-select the best combination based on player pool and category.
   259→ * For multiple teams, rotates combinations intelligently.
   260→ */
   261→export function autoSelectCombination(
   262→  eligiblePlayers: TGPlayer[],
   263→  category: string,
   264→  fixedPlayers: TGPlayer[] = [],
   265→  teamIndex: number = 0,
   266→): RoleCombination {
   267→  const compatible = getCompatibleCombinations(fixedPlayers, eligiblePlayers);
   268→
   269→  if (compatible.length === 0) {
   270→    return { wk: 1, bat: 4, ar: 2, bowl: 4 };
   271→  }
   272→
   273→  // Category-preferred patterns
   274→  const preferredPatterns: RoleCombination[] =
   275→    category === 'H2H'
   276→      ? [{ wk: 1, bat: 5, ar: 2, bowl: 3 }, { wk: 1, bat: 4, ar: 2, bowl: 4 }, { wk: 1, bat: 3, ar: 3, bowl: 4 }]
   277→      : category === 'SL'
   278→      ? [{ wk: 1, bat: 4, ar: 2, bowl: 4 }, { wk: 1, bat: 3, ar: 3, bowl: 4 }, { wk: 2, bat: 3, ar: 2, bowl: 4 }]
   279→      : [ // Mega GL
   280→        { wk: 1, bat: 3, ar: 2, bowl: 5 }, { wk: 1, bat: 4, ar: 1, bowl: 5 },
   281→        { wk: 1, bat: 3, ar: 3, bowl: 4 }, { wk: 1, bat: 4, ar: 2, bowl: 4 },
   282→        { wk: 2, bat: 3, ar: 2, bowl: 4 }, { wk: 1, bat: 5, ar: 1, bowl: 4 },
   283→        { wk: 2, bat: 3, ar: 1, bowl: 5 }, { wk: 1, bat: 4, ar: 3, bowl: 3 },
   284→        { wk: 2, bat: 4, ar: 1, bowl: 4 }, { wk: 1, bat: 3, ar: 1, bowl: 6 },
   285→        { wk: 2, bat: 2, ar: 3, bowl: 4 }, { wk: 1, bat: 5, ar: 2, bowl: 3 },
   286→      ];
   287→
   288→  const preferredCompatible = preferredPatterns.filter(p =>
   289→    compatible.some(c => c.wk === p.wk && c.bat === p.bat && c.ar === p.ar && c.bowl === p.bowl)
   290→  );
   291→
   292→  const pool = preferredCompatible.length > 0 ? preferredCompatible : compatible;
   293→  const index = teamIndex % pool.length;
   294→
   295→  return pool[index];
   296→}

   298→/**
   299→ * Validate a manual combination against available players.
   300→ */
   301→export function validateCombination(
   302→  combo: RoleCombination,
   303→  eligiblePlayers: TGPlayer[],
   304→  fixedPlayers: TGPlayer[] = [],
   305→): { valid: boolean; errors: string[] } {
   306→  const errors: string[] = [];
   307→
   308→  if (combo.wk + combo.bat + combo.ar + combo.bowl !== CRICKET_TEAM_SIZE) {
   309→    errors.push(`Total players must be 11 (got ${combo.wk + combo.bat + combo.ar + combo.bowl})`);
   310→  }
   311→
   312→  if (combo.wk < MIN_WK || combo.wk > MAX_WK) errors.push(`WK must be ${MIN_WK}-${MAX_WK}`);
   313→  if (combo.bat < MIN_BAT || combo.bat > MAX_BAT) errors.push(`BAT must be ${MIN_BAT}-${MAX_BAT}`);
   314→  if (combo.ar < MIN_AR || combo.ar > MAX_AR) errors.push(`AR must be ${MIN_AR}-${MAX_AR}`);
   315→  if (combo.bowl < MIN_BOWL || combo.bowl > MAX_BOWL) errors.push(`BOWL must be ${MIN_BOWL}-${MAX_BOWL}`);
   316→
   317→  if (fixedPlayers.length > 0 && !isCombinationCompatibleWithFixed(combo, fixedPlayers)) {
   318→    errors.push('Combination is not compatible with selected fixed players');
   319→  }
   320→
   321→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   322→  const remaining = eligiblePlayers.filter(p => !fixedIds.has(p.pl_id));
   323→
   324→  const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   325→  const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   326→  const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   327→  const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   328→
   329→  const needWK = Math.max(0, combo.wk - fixedWK);
   330→  const needBat = Math.max(0, combo.bat - fixedBat);
   331→  const needAR = Math.max(0, combo.ar - fixedAR);
   332→  const needBowl = Math.max(0, combo.bowl - fixedBowl);
   333→
   334→  const availWK = remaining.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   335→  const availBat = remaining.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   336→  const availAR = remaining.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   337→  const availBowl = remaining.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   338→
   339→  if (availWK < needWK) errors.push(`Need ${needWK} WK but only ${availWK} available`);
   340→  if (availBat < needBat) errors.push(`Need ${needBat} BAT but only ${availBat} available`);
   341→  if (availAR < needAR) errors.push(`Need ${needAR} AR but only ${availAR} available`);
   342→  if (availBowl < needBowl) errors.push(`Need ${needBowl} BOWL but only ${availBowl} available`);
   343→
   344→  return { valid: errors.length === 0, errors };
   345→}

   425→function decryptField(encrypted: string): string {
   426→  try {
   427→    const bytes = CryptoJS.AES.decrypt(encrypted, ENCRYPTION_KEY);
   428→    const decrypted = bytes.toString(CryptoJS.enc.Utf8);
   429→    if (!decrypted) return encrypted;
   430→    return decrypted;
   431→  } catch {
   432→    return encrypted;
   433→  }
   434→}

   508→export async function fetchMatches(sport: string = 'cricket'): Promise<TGMatch[]> {
   509→  try {
   510→    const response = await fetch(`${TG_API_BASE}/fantasy/matches/${sport}`, {
   511→      next: { revalidate: 60 },
   512→    });
   513→
   514→    if (!response.ok) {
   515→      throw new Error(`Failed to fetch matches: ${response.status}`);
   516→    }
   517→
   518→    const data = await response.json();
   519→
   520→    if (data.status !== 'success' || !Array.isArray(data.data)) {
   521→      return [];
   522→    }
   523→
   524→    const matches: TGMatch[] = [];
   525→    for (const encryptedMatch of data.data) {
   526→      const match = decryptMatchData(encryptedMatch);
   527→      if (match) {
   528→        matches.push(match);
   529→      }
   530→    }
   531→
   532→    return matches;
   533→  } catch (error) {
   534→    console.error('Error fetching matches:', error);
   535→    return [];
   536→  }
   537→}

   545→    if (!response.ok) {

   561→        const player = decryptPlayerData(ep);
   562→        if (player) leftPlayers.push(player);

   570→        if (player) rightPlayers.push(player);

   573→

   576→      id: raw.id || '',
   577→      left_team_name: decryptField(raw.left_team_name || ''),
   578→      right_team_name: decryptField(raw.right_team_name || ''),
   579→      left_team_image: decryptField(raw.left_team_image || ''),
   580→      right_team_image: decryptField(raw.right_team_image || ''),
   581→      match_time: decryptField(raw.match_time || ''),
   582→      match_type: raw.match_type || 'normal',
   583→      sport_index: raw.sport_index ?? 0,
   584→      lineup_status: raw.lineup_status ?? 0,
   585→      toss: raw.toss || 'NA',
   586→      left_team_players: leftPlayers,
   587→      right_team_players: rightPlayers,
   588→      fantasy_version: raw.fantasy_version || [],

   590→  } catch (error) {

   592→    return null;
   593→  }

   630→// Cricket team constraints

   633→export const MAX_WK = 4;

   635→export const MAX_BAT = 6;
   636→export const MIN_AR = 1;
   637→export const MAX_AR = 4;

   639→export const MAX_BOWL = 6;
   640→export const MAX_CREDITS = 100;

   643→function seededRandom(seed: number): () => number {
   644→  let s = seed;
   645→  return () => {
   646→    s = (s * 16807 + 0) % 2147483647;
   647→    return (s - 1) / 2147483646;
   648→  };
   649→}
   650→
   651→function shuffleArray<T>(arr: T[], rng: () => number): T[] {
   652→  const result = [...arr];
   653→  for (let i = result.length - 1; i > 0; i--) {
   654→    const j = Math.floor(rng() * (i + 1));
   655→    [result[i], result[j]] = [result[j], result[i]];
   656→  }
   657→  return result;
   658→}

   660→export function generateTeams(
   661→  leftPlayers: TGPlayer[],
   662→  rightPlayers: TGPlayer[],
   663→  category: string,
   664→  count: number,
   665→  seed: number = Date.now(),
   666→  avoidPlayerIds: Set<number> = new Set(),
   667→  combination: RoleCombination | null = null,
   668→): GeneratedTeam[] {
   669→  // Apply lineup-aware filtering: AFTER LINEUP only playing=1, BEFORE LINEUP all
   670→  const allPlayersRaw = [...leftPlayers, ...rightPlayers];
   671→  const lineupMode = getLineupMode(allPlayersRaw);
   672→  let allPlayers = getEligiblePlayers(allPlayersRaw, avoidPlayerIds);
   673→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
   674→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
   675→
   676→  // After filtering, rebuild left/right pools for team distribution checks
   677→  let eligibleLeft = allPlayers.filter(p => p.team_name === leftTeamName);
   678→  let eligibleRight = allPlayers.filter(p => p.team_name === rightTeamName);
   679→
   680→  // Separate by role
   681→  let wk = allPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER);
   682→  let bat = allPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN);
   683→  let ar = allPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER);
   684→  let bowl = allPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER);
   685→
   686→  // AFTER LINEUP DIAGNOSTIC: Check if we have enough eligible players
   687→  if (lineupMode === 'after') {
   688→    console.log(`[TEAM GEN] AFTER LINEUP — Eligible: ${allPlayers.length}, Left: ${eligibleLeft.length}, Right: ${eligibleRight.length}, WK: ${wk.length}, BAT: ${bat.length}, AR: ${ar.length}, BOWL: ${bowl.length}`);
   689→    const totalCredits = allPlayers.reduce((s, p) => s + p.credits, 0);
   690→    console.log(`[TEAM GEN] Total eligible credits: ${totalCredits}, Avg: ${(totalCredits / allPlayers.length).toFixed(1)}`);
   691→
   692→    // If too few eligible players after lineup, fall back to including probable players
   693→    // (playing === 0 but likely to play — e.g., not explicitly ruled out)
   694→    if (allPlayers.length < 11 || eligibleLeft.length < 1 || eligibleRight.length < 1 ||
   695→        wk.length < MIN_WK || bat.length < MIN_BAT || ar.length < MIN_AR || bowl.length < MIN_BOWL) {
   696→      console.warn(`[TEAM GEN] Not enough eligible players after lineup. Including probable players (playing=0) as fallback.`);
   697→      // Include playing=0 players as fallback (they might still play)
   698→      allPlayers = allPlayersRaw.filter(p => !avoidPlayerIds.has(p.pl_id) && p.playing !== -1);
   699→      eligibleLeft = allPlayers.filter(p => p.team_name === leftTeamName);
   700→      eligibleRight = allPlayers.filter(p => p.team_name === rightTeamName);
   701→      wk = allPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER);
   702→      bat = allPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN);
   703→      ar = allPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER);
   704→      bowl = allPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER);
   705→      console.log(`[TEAM GEN] Fallback — Eligible: ${allPlayers.length}, Left: ${eligibleLeft.length}, Right: ${eligibleRight.length}, WK: ${wk.length}, BAT: ${bat.length}, AR: ${ar.length}, BOWL: ${bowl.length}`);
   706→    }
   707→  }
   708→
   709→  const teams: GeneratedTeam[] = [];
   710→  const rng = seededRandom(seed);
   711→
   712→  // Category-specific strategies
   713→  const isH2H = category === 'H2H';
   714→  const isSL = category === 'SL';
   715→
   716→  for (let t = 0; t < count; t++) {
   717→    let attempts = 0;
   718→    let team: TGPlayer[] | null = null;
   719→
   720→    while (!team && attempts < 200) {
   721→      attempts++;
   722→      const sr = seededRandom(seed + t * 1000 + attempts);
   723→
   724→      // Determine role counts based on category or explicit combination
   725→      let wkCount, batCount, arCount, bowlCount;
   726→
   727→      if (combination) {
   728→        // Use explicit combination (manual or auto-selected)
   729→        wkCount = combination.wk;
   730→        batCount = combination.bat;
   731→        arCount = combination.ar;
   732→        bowlCount = combination.bowl;
   733→      } else if (isH2H) {
   734→        // H2H: More conservative, popular players
   735→        wkCount = MIN_WK + (sr() > 0.5 ? 1 : 0);
   736→        arCount = MIN_AR + (sr() > 0.6 ? 1 : 0);
   737→        batCount = Math.max(MIN_BAT, 5 - wkCount + (sr() > 0.5 ? 1 : 0));
   738→        bowlCount = CRICKET_TEAM_SIZE - wkCount - batCount - arCount;
   739→      } else if (isSL) {
   740→        // SL: Balanced
   741→        wkCount = 1 + (sr() > 0.7 ? 1 : 0);
   742→        arCount = 2 + (sr() > 0.5 ? 1 : 0);
   743→        batCount = 3 + (sr() > 0.5 ? 1 : 0);
   744→        bowlCount = CRICKET_TEAM_SIZE - wkCount - batCount - arCount;
   745→      } else {
   746→        // Mega GL: More differential picks, varied combinations
   747→        const patterns = [
   748→          [1, 3, 2, 5], [1, 4, 1, 5], [1, 3, 3, 4], [1, 4, 2, 4],
   749→          [2, 3, 2, 4], [1, 5, 1, 4], [2, 3, 1, 5], [1, 4, 3, 3],
   750→          [2, 4, 1, 4], [1, 3, 1, 6], [2, 2, 3, 4], [1, 5, 2, 3],
   751→        ];
   752→        const pattern = patterns[Math.floor(sr() * patterns.length)];
   753→        [wkCount, batCount, arCount, bowlCount] = pattern;
   754→      }
   755→
   756→      // Validate counts
   757→      if (wkCount < MIN_WK || wkCount > MAX_WK) continue;
   758→      if (batCount < MIN_BAT || batCount > MAX_BAT) continue;
   759→      if (arCount < MIN_AR || arCount > MAX_AR) continue;
   760→      if (bowlCount < MIN_BOWL || bowlCount > MAX_BOWL) continue;
   761→      if (wkCount + batCount + arCount + bowlCount !== CRICKET_TEAM_SIZE) continue;
   762→
   763→      // Pick players for each role
   764→      const shuffledWK = shuffleArray(wk, sr);
   765→      const shuffledBat = shuffleArray(bat, sr);
   766→      const shuffledAR = shuffleArray(ar, sr);
   767→      const shuffledBowl = shuffleArray(bowl, sr);
   768→
   769→      // For Mega GL, sort by selected_by sometimes to get differential picks
   770→      let selectedWK, selectedBat, selectedAR, selectedBowl;
   771→
   772→      if (!isH2H && sr() > 0.4) {
   773→        // Sort some roles by selection % to mix popular and differential
   774→        const sortFn = (a: TGPlayer, b: TGPlayer) => a.selected_by - b.selected_by;
   775→        selectedWK = shuffleArray([...wk].sort(sortFn), sr).slice(0, wkCount);
   776→        selectedBat = shuffleArray([...bat].sort(sortFn), sr).slice(0, batCount);
   777→        selectedAR = shuffleArray([...ar].sort(sortFn), sr).slice(0, arCount);
   778→        selectedBowl = shuffleArray([...bowl].sort(sortFn), sr).slice(0, bowlCount);
   779→      } else if (isH2H) {
   780→        // H2H: Pick most selected players
   781→        const sortFn = (a: TGPlayer, b: TGPlayer) => b.selected_by - a.selected_by;
   782→        selectedWK = [...wk].sort(sortFn).slice(0, wkCount);
   783→        selectedBat = shuffleArray([...bat].sort(sortFn), sr).slice(0, batCount);
   784→        selectedAR = [...ar].sort(sortFn).slice(0, arCount);
   785→        selectedBowl = shuffleArray([...bowl].sort(sortFn), sr).slice(0, bowlCount);
   786→      } else {
   787→        selectedWK = shuffledWK.slice(0, wkCount);
   788→        selectedBat = shuffledBat.slice(0, batCount);
   789→        selectedAR = shuffledAR.slice(0, arCount);
   790→        selectedBowl = shuffledBowl.slice(0, bowlCount);
   791→      }
   792→
   793→      // Check if we have enough players
   794→      if (selectedWK.length < wkCount || selectedBat.length < batCount ||
   795→          selectedAR.length < arCount || selectedBowl.length < bowlCount) continue;
   796→
   797→      const selected = [...selectedWK, ...selectedBat, ...selectedAR, ...selectedBowl];
   798→
   799→      // Check total credits
   800→      const totalCredits = selected.reduce((sum, p) => sum + p.credits, 0);
   801→      if (totalCredits > MAX_CREDITS) continue;
   802→
   803→      // Check team distribution (max 7 from one team)
   804→      const leftCount = selected.filter(p => p.team_name === leftTeamName).length;
   805→      const rightCount = selected.filter(p => p.team_name === rightTeamName).length;
   806→      if (leftCount > MAX_FROM_ONE_TEAM || rightCount > MAX_FROM_ONE_TEAM) continue;
   807→
   808→      // Ensure at least 1 from each team
   809→      if (leftCount < 1 || rightCount < 1) continue;
   810→
   811→      team = selected;
   812→    }
   813→
   814→    if (team) {
   815→      // Select Captain & Vice-Captain
   816→      // Sort by a combination of points and selection % for C/VC
   817→      const sorted = [...team].sort((a, b) =>
   818→        (b.points * 0.4 + b.selected_by * 0.3 + b.captain_percentage * 0.3) -
   819→        (a.points * 0.4 + a.selected_by * 0.3 + a.captain_percentage * 0.3)
   820→      );
   821→
   822→      const captain = sorted[0];
   823→      const viceCaptain = sorted.length > 1 ? sorted[1] : sorted[0];
   824→
   825→      teams.push({
   826→        id: t + 1,
   827→        captain,
   828→        viceCaptain,
   829→        players: team,
   830→      });
   831→    }
   832→  }
   833→
   834→  // AFTER LINEUP FALLBACK: If 0 teams generated, retry with relaxed constraints
   835→  if (teams.length === 0 && lineupMode === 'after') {
   836→    console.warn(`[TEAM GEN] 0 teams generated after lineup! Retrying with relaxed constraints...`);
   837→
   838→    // Fallback 1: Increase max attempts and allow higher credits
   839→    const RELAXED_MAX_CREDITS = 110; // Allow 10 extra credits
   840→    for (let t = 0; t < count; t++) {
   841→      let attempts = 0;
   842→      let team: TGPlayer[] | null = null;
   843→
   844→      while (!team && attempts < 500) {
   845→        attempts++;
   846→        const sr = seededRandom(seed + t * 2000 + attempts + 99999);
   847→
   848→        let wkCount, batCount, arCount, bowlCount;
   849→        // Try all valid combinations systematically
   850→        const combos = getAllValidCombinations();
   851→        const combo = combos[Math.floor(sr() * combos.length)];
   852→        wkCount = combo.wk;
   853→        batCount = combo.bat;
   854→        arCount = combo.ar;
   855→        bowlCount = combo.bowl;
   856→
   857→        const shuffledWK = shuffleArray(wk, sr);
   858→        const shuffledBat = shuffleArray(bat, sr);
   859→        const shuffledAR = shuffleArray(ar, sr);
   860→        const shuffledBowl = shuffleArray(bowl, sr);
   861→
   862→        const selectedWK = shuffledWK.slice(0, wkCount);
   863→        const selectedBat = shuffledBat.slice(0, batCount);
   864→        const selectedAR = shuffledAR.slice(0, arCount);
   865→        const selectedBowl = shuffledBowl.slice(0, bowlCount);
   866→
   867→        if (selectedWK.length < wkCount || selectedBat.length < batCount ||
   868→            selectedAR.length < arCount || selectedBowl.length < bowlCount) continue;
   869→
   870→        const selected = [...selectedWK, ...selectedBat, ...selectedAR, ...selectedBowl];
   871→        const totalCredits = selected.reduce((sum, p) => sum + p.credits, 0);
   872→        if (totalCredits > RELAXED_MAX_CREDITS) continue;
   873→
   874→        const leftCount = selected.filter(p => p.team_name === leftTeamName).length;
   875→        const rightCount = selected.filter(p => p.team_name === rightTeamName).length;
   876→        // Relaxed: allow up to 8 from one team (instead of 7)
   877→        if (leftCount > 8 || rightCount > 8) continue;
   878→        if (leftCount < 1 || rightCount < 1) continue;
   879→
   880→        team = selected;
   881→      }
   882→
   883→      if (team) {
   884→        const sorted = [...team].sort((a, b) =>
   885→          (b.points * 0.4 + b.selected_by * 0.3 + b.captain_percentage * 0.3) -
   886→          (a.points * 0.4 + a.selected_by * 0.3 + a.captain_percentage * 0.3)
   887→        );
   888→        teams.push({
   889→          id: t + 1,
   890→          captain: sorted[0],
   891→          viceCaptain: sorted.length > 1 ? sorted[1] : sorted[0],
   892→          players: team,
   893→        });
   894→      }
   895→    }
   896→
   897→    if (teams.length > 0) {
   898→      console.log(`[TEAM GEN] Relaxed constraints generated ${teams.length} teams`);
   899→    } else {
   900→      console.error(`[TEAM GEN] Still 0 teams after relaxed constraints. Eligible: ${allPlayers.length}, WK: ${wk.length}, BAT: ${bat.length}, AR: ${ar.length}, BOWL: ${bowl.length}`);
   901→    }
   902→  }
   903→
   904→  return teams;
   905→}

   924→export function generateExtraTeams(input: ExtraTeamGenInput): GeneratedTeam[] {
   925→  const {
   926→    fixedPlayers,
   927→    captainOptions,
   928→    viceCaptainOptions,
   929→    leftPlayers,
   930→    rightPlayers,
   931→    category,
   932→    count,
   933→    seed = Date.now(),
   934→    avoidPlayerIds = new Set(),
   935→    combination = null,
   936→  } = input;
   937→
   938→  // Apply lineup-aware filtering to the full player pool
   939→  const allPlayersRaw = [...leftPlayers, ...rightPlayers];
   940→  const lineupMode = getLineupMode(allPlayersRaw);
   941→  let allPlayers = getEligiblePlayers(allPlayersRaw, avoidPlayerIds);
   942→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
   943→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
   944→
   945→  // AFTER LINEUP FALLBACK: If too few eligible players, include probable players
   946→  if (lineupMode === 'after' && allPlayers.length < 14) {
   947→    console.warn(`[EXTRA TEAM GEN] Only ${allPlayers.length} eligible after lineup, including probable players as fallback.`);
   948→    allPlayers = allPlayersRaw.filter(p => !avoidPlayerIds.has(p.pl_id) && p.playing !== -1);
   949→  }
   950→
   951→  // Pool of players excluding the 8 fixed ones (use pl_id for identity)
   952→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   953→  const remainingPool = allPlayers.filter(p => !fixedIds.has(p.pl_id));
   954→
   955→  // Separate remaining pool by role
   956→  const remWK = remainingPool.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER);
   957→  const remBat = remainingPool.filter(p => p.role === PLAYER_ROLES.BATSMAN);
   958→  const remAR = remainingPool.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER);
   959→  const remBowl = remainingPool.filter(p => p.role === PLAYER_ROLES.BOWLER);
   960→
   961→  // Count fixed players by role
   962→  const fixedWKCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   963→  const fixedBatCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   964→  const fixedARCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   965→  const fixedBowlCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   966→
   967→  const fixedCredits = fixedPlayers.reduce((sum, p) => sum + p.credits, 0);
   968→  const remainingCredits = MAX_CREDITS - fixedCredits;
   969→  const neededPlayers = CRICKET_TEAM_SIZE - fixedPlayers.length; // 3
   970→
   971→  const teams: GeneratedTeam[] = [];
   972→
   973→  // Pre-validate: fixed players must have credits <= 100
   974→  if (fixedCredits > MAX_CREDITS) return teams;
   975→
   976→  // Pre-validate: fixed players must respect max 7 from one team
   977→  const fixedLeftCount = fixedPlayers.filter(p => p.team_name === leftTeamName).length;
   978→  const fixedRightCount = fixedPlayers.filter(p => p.team_name === rightTeamName).length;
   979→  if (fixedLeftCount > MAX_FROM_ONE_TEAM || fixedRightCount > MAX_FROM_ONE_TEAM) return teams;
   980→
   981→  // Generate all valid C/VC combinations from the 5C and 5VC options
   982→  // C must be in the team, VC must be in the team, C ≠ VC
   983→  const cVCCombos: { c: TGPlayer; vc: TGPlayer }[] = [];
   984→  for (const c of captainOptions) {
   985→    for (const vc of viceCaptainOptions) {
   986→      if (c.pl_id !== vc.pl_id) {
   987→        cVCCombos.push({ c, vc });
   988→      }
   989→    }
   990→  }
   991→  // If no valid combos, return empty
   992→  if (cVCCombos.length === 0) return teams;
   993→
   994→  // Category-specific role distribution preferences for remaining players
   995→  const isH2H = category === 'H2H';
   996→  const isSL = category === 'SL';
   997→
   998→  // Pre-compute all valid target role distributions that are compatible with fixed players
   999→  // This avoids wasting attempts on patterns that can never work
  1000→  const categoryPatterns: number[][] = isH2H
  1001→    ? [] // H2H uses dynamic generation below
  1002→    : isSL
  1003→    ? [] // SL uses dynamic generation below
  1004→    : [
  1005→        [1, 3, 2, 5], [1, 4, 1, 5], [1, 3, 3, 4], [1, 4, 2, 4],
  1006→        [2, 3, 2, 4], [1, 5, 1, 4], [2, 3, 1, 5], [1, 4, 3, 3],
  1007→        [2, 4, 1, 4], [1, 3, 1, 6], [2, 2, 3, 4], [1, 5, 2, 3],
  1008→      ];
  1009→
  1010→  // Build list of valid target distributions: [WK, BAT, AR, BOWL]
  1011→  const validDistributions: number[][] = [];
  1012→  for (let wk = MIN_WK; wk <= MAX_WK; wk++) {
  1013→    for (let bat = MIN_BAT; bat <= MAX_BAT; bat++) {
  1014→      for (let ar = MIN_AR; ar <= MAX_AR; ar++) {
  1015→        const bowl = CRICKET_TEAM_SIZE - wk - bat - ar;
  1016→        if (bowl < MIN_BOWL || bowl > MAX_BOWL) continue;
  1017→        // Check compatibility with fixed players
  1018→        const nWK = Math.max(0, wk - fixedWKCount);
  1019→        const nBat = Math.max(0, bat - fixedBatCount);
  1020→        const nAR = Math.max(0, ar - fixedARCount);
  1021→        const nBowl = Math.max(0, bowl - fixedBowlCount);
  1022→        if (nWK + nBat + nAR + nBowl !== neededPlayers) continue;
  1023→        // Check we have enough remaining players for each role
  1024→        if (remWK.length < nWK || remBat.length < nBat || remAR.length < nAR || remBowl.length < nBowl) continue;
  1025→        // For Mega GL / SL / H2H: prefer category-specific patterns but include all valid ones
  1026→        validDistributions.push([wk, bat, ar, bowl]);
  1027→      }
  1028→    }
  1029→  }
  1030→
  1031→  // Sort: prefer category-specific patterns first, then others
  1032→  if (categoryPatterns.length > 0) {
  1033→    const patternSet = new Set(categoryPatterns.map(p => p.join(',')));
  1034→    validDistributions.sort((a, b) => {
  1035→      const aInCat = patternSet.has(a.join(',')) ? 0 : 1;
  1036→      const bInCat = patternSet.has(b.join(',')) ? 0 : 1;
  1037→      return aInCat - bInCat;
  1038→    });
  1039→  }
  1040→
  1041→  if (validDistributions.length === 0) {
  1042→    return teams;
  1043→  }
  1044→

  1046→  // Uses sorted player IDs + C/VC IDs to create a unique fingerprint

  1051→  };

  1054→  // or we exhaust the maximum total attempts
  1055→  const MAX_TOTAL_ATTEMPTS = count * 500; // generous budget
  1056→  let totalAttempts = 0;
  1057→

  1059→    let attempts = 0;

  1061→    let teamC: TGPlayer | null = null;
  1062→    let teamVC: TGPlayer | null = null;

  1065→      attempts++;
  1066→      totalAttempts++;

  1068→

  1070→      let targetWK: number, targetBat: number, targetAR: number, targetBowl: number;

  1073→        // Use explicit combination
  1074→        targetWK = combination.wk;
  1075→        targetBat = combination.bat;
  1076→        targetAR = combination.ar;
  1077→        targetBowl = combination.bowl;

  1079→        [targetWK, targetBat, targetAR, targetBowl] = validDistributions[0];
  1080→      } else {
  1081→        // Bias towards category-preferred patterns (first half of sorted list)
  1082→        const idx = sr() < 0.7
  1083→          ? Math.floor(sr() * Math.ceil(validDistributions.length / 2)) // 70% chance from preferred half
  1084→          : Math.floor(sr() * validDistributions.length); // 30% from any
  1085→        [targetWK, targetBat, targetAR, targetBowl] = validDistributions[Math.min(idx, validDistributions.length - 1)];
  1086→      }
  1087→
  1088→      // How many more of each role do we need?
  1089→      const needWK = Math.max(0, targetWK - fixedWKCount);
  1090→      const needBat = Math.max(0, targetBat - fixedBatCount);
  1091→      const needAR = Math.max(0, targetAR - fixedARCount);
  1092→      const needBowl = Math.max(0, targetBowl - fixedBowlCount);
  1093→

  1095→      if (remWK.length < needWK || remBat.length < needBat ||
  1096→          remAR.length < needAR || remBowl.length < needBowl) continue;
  1097→

  1099→      let pickedWK: TGPlayer[], pickedBat: TGPlayer[], pickedAR: TGPlayer[], pickedBowl: TGPlayer[];
  1100→
  1101→      if (!isH2H && sr() > 0.4) {
  1102→        const sortFn = (a: TGPlayer, b: TGPlayer) => a.selected_by - b.selected_by;
  1103→        pickedWK = shuffleArray([...remWK].sort(sortFn), sr).slice(0, needWK);
  1104→        pickedBat = shuffleArray([...remBat].sort(sortFn), sr).slice(0, needBat);

  1107→      } else if (isH2H) {
  1108→        const sortFn = (a: TGPlayer, b: TGPlayer) => b.selected_by - a.selected_by;
  1109→        pickedWK = [...remWK].sort(sortFn).slice(0, needWK);
  1110→        pickedBat = shuffleArray([...remBat].sort(sortFn), sr).slice(0, needBat);
  1111→        pickedAR = [...remAR].sort(sortFn).slice(0, needAR);
  1112→        pickedBowl = shuffleArray([...remBowl].sort(sortFn), sr).slice(0, needBowl);
  1113→      } else {
  1114→        pickedWK = shuffleArray(remWK, sr).slice(0, needWK);
  1115→        pickedBat = shuffleArray(remBat, sr).slice(0, needBat);
  1116→        pickedAR = shuffleArray(remAR, sr).slice(0, needAR);
  1117→        pickedBowl = shuffleArray(remBowl, sr).slice(0, needBowl);
  1118→      }
  1119→
  1120→      if (pickedWK.length < needWK || pickedBat.length < needBat ||
  1121→          pickedAR.length < needAR || pickedBowl.length < needBowl) continue;
  1122→

  1124→      const fullTeam = [...fixedPlayers, ...remaining];
  1125→

  1127→      const uniqueIds = new Set(fullTeam.map(p => p.pl_id));
  1128→      if (uniqueIds.size !== fullTeam.length) continue;
  1129→

  1131→      const totalCredits = fullTeam.reduce((sum, p) => sum + p.credits, 0);

  1133→

  1135→      const leftCount = fullTeam.filter(p => p.team_name === leftTeamName).length;
  1136→      const rightCount = fullTeam.filter(p => p.team_name === rightTeamName).length;
  1137→      if (leftCount > MAX_FROM_ONE_TEAM || rightCount > MAX_FROM_ONE_TEAM) continue;
  1138→      if (leftCount < 1 || rightCount < 1) continue;
  1139→

  1141→      // Round-robin through the valid combos, also try to ensure C and VC are in the team
  1142→      const comboIndex = (t + attempts) % cVCCombos.length;
  1143→      let combo = cVCCombos[comboIndex];

  1149→        let foundValidCombo = false;

  1151→          const tryCombo = cVCCombos[(comboIndex + ci) % cVCCombos.length];
  1152→          if (teamPlayerIds.has(tryCombo.c.pl_id) && teamPlayerIds.has(tryCombo.vc.pl_id)) {
  1153→            combo = tryCombo;
  1154→            foundValidCombo = true;
  1155→            break;
  1156→          }
  1157→        }
  1158→        if (!foundValidCombo) {
  1159→          // Fallback: use top players in team as C/VC
  1160→          const sorted = [...fullTeam].sort((a, b) =>
  1161→            (b.points * 0.4 + b.selected_by * 0.3 + b.captain_percentage * 0.3) -
  1162→            (a.points * 0.4 + a.selected_by * 0.3 + a.captain_percentage * 0.3)
  1163→          );
  1164→          combo = { c: sorted[0], vc: sorted.length > 1 ? sorted[1] : sorted[0] };
  1165→        }
  1166→      }
  1167→
  1168→      // Uniqueness check: skip if this team signature already exists
  1169→      const signature = makeTeamSignature(fullTeam, combo.c, combo.vc);
  1170→      if (teamSignatures.has(signature)) continue;
  1171→
  1172→      team = fullTeam;
  1173→      teamC = combo.c;
  1174→      teamVC = combo.vc;
  1175→      teamSignatures.add(signature);
  1176→    }
  1177→
  1178→    if (team && teamC && teamVC) {
  1179→      teams.push({
  1180→        id: teams.length + 1,
  1181→        captain: teamC,
  1182→        viceCaptain: teamVC,
  1183→        players: team,
  1184→      });
  1185→    } else {
  1186→      // If we failed to generate a unique team for this slot,
  1187→      // try extra attempts with different seeds before giving up
  1188→      let extraAttempts = 0;
  1189→      const MAX_EXTRA = 200;
  1190→      while (teams.length < count && extraAttempts < MAX_EXTRA && totalAttempts < MAX_TOTAL_ATTEMPTS) {
  1191→        extraAttempts++;
  1192→        totalAttempts++;
  1193→        const extraSeed = seed + count * 1000 + extraAttempts * 37;
  1194→        const sr = seededRandom(extraSeed);
  1195→
  1196→        // Pick distribution (or use explicit combination)
  1197→        let targetWK: number, targetBat: number, targetAR: number, targetBowl: number;
  1198→        if (combination) {
  1199→          targetWK = combination.wk;
  1200→          targetBat = combination.bat;
  1201→          targetAR = combination.ar;
  1202→          targetBowl = combination.bowl;
  1203→        } else if (validDistributions.length === 1) {
  1204→          [targetWK, targetBat, targetAR, targetBowl] = validDistributions[0];
  1205→        } else {
  1206→          const idx = Math.floor(sr() * validDistributions.length);
  1207→          [targetWK, targetBat, targetAR, targetBowl] = validDistributions[idx];
  1208→        }
  1209→
  1210→        const needWK = Math.max(0, targetWK - fixedWKCount);
  1211→        const needBat = Math.max(0, targetBat - fixedBatCount);
  1212→        const needAR = Math.max(0, targetAR - fixedARCount);
  1213→        const needBowl = Math.max(0, targetBowl - fixedBowlCount);
  1214→
  1215→        if (remWK.length < needWK || remBat.length < needBat ||
  1216→            remAR.length < needAR || remBowl.length < needBowl) continue;
  1217→
  1218→        const sortFn = (a: TGPlayer, b: TGPlayer) => a.selected_by - b.selected_by;
  1219→        const pickedWK = shuffleArray([...remWK].sort(sortFn), sr).slice(0, needWK);
  1220→        const pickedBat = shuffleArray([...remBat].sort(sortFn), sr).slice(0, needBat);
  1221→        const pickedAR = shuffleArray([...remAR].sort(sortFn), sr).slice(0, needAR);
  1222→        const pickedBowl = shuffleArray([...remBowl].sort(sortFn), sr).slice(0, needBowl);
  1223→
  1224→        if (pickedWK.length < needWK || pickedBat.length < needBat ||
  1225→            pickedAR.length < needAR || pickedBowl.length < needBowl) continue;
  1226→
  1227→        const remaining = [...pickedWK, ...pickedBat, ...pickedAR, ...pickedBowl];
  1228→        const fullTeam = [...fixedPlayers, ...remaining];
  1229→
  1230→        const uniqueIds = new Set(fullTeam.map(p => p.pl_id));
  1231→        if (uniqueIds.size !== fullTeam.length) continue;
  1232→
  1233→        const totalCredits = fullTeam.reduce((sum, p) => sum + p.credits, 0);
  1234→        if (totalCredits > MAX_CREDITS) continue;
  1235→
  1236→        const leftCount = fullTeam.filter(p => p.team_name === leftTeamName).length;
  1237→        const rightCount = fullTeam.filter(p => p.team_name === rightTeamName).length;
  1238→        if (leftCount > MAX_FROM_ONE_TEAM || rightCount > MAX_FROM_ONE_TEAM) continue;
  1239→        if (leftCount < 1 || rightCount < 1) continue;
  1240→
  1241→        // C/VC selection
  1242→        const comboIndex = extraAttempts % cVCCombos.length;
  1243→        let combo = cVCCombos[comboIndex];
  1244→        const teamPlayerIds = new Set(fullTeam.map(p => p.pl_id));
  1245→        if (!teamPlayerIds.has(combo.c.pl_id) || !teamPlayerIds.has(combo.vc.pl_id)) {
  1246→          let foundValidCombo = false;
  1247→          for (let ci = 0; ci < cVCCombos.length; ci++) {
  1248→            const tryCombo = cVCCombos[(comboIndex + ci) % cVCCombos.length];
  1249→            if (teamPlayerIds.has(tryCombo.c.pl_id) && teamPlayerIds.has(tryCombo.vc.pl_id)) {
  1250→              combo = tryCombo;
  1251→              foundValidCombo = true;
  1252→              break;
  1253→            }
  1254→          }
  1255→          if (!foundValidCombo) {
  1256→            const sorted = [...fullTeam].sort((a, b) =>
  1257→              (b.points * 0.4 + b.selected_by * 0.3 + b.captain_percentage * 0.3) -
  1258→              (a.points * 0.4 + a.selected_by * 0.3 + a.captain_percentage * 0.3)

  1260→            combo = { c: sorted[0], vc: sorted.length > 1 ? sorted[1] : sorted[0] };
  1261→          }
  1262→        }
  1263→
  1264→        const signature = makeTeamSignature(fullTeam, combo.c, combo.vc);
  1265→        if (teamSignatures.has(signature)) continue;
  1266→        teamSignatures.add(signature);
  1267→
  1268→        teams.push({
  1269→          id: teams.length + 1,
  1270→          captain: combo.c,
  1271→          viceCaptain: combo.vc,
  1272→          players: fullTeam,
  1273→        });
  1274→        break; // Successfully added a team, move on
  1275→      }
  1276→    }
  1277→  }
  1278→
  1279→  return teams;
  1280→}

  1282→// ============ Auto Select for Extra Team Generation ============
  1283→// Uses real player data (selection %, captain %, points, credits, role, team balance)
  1284→// to intelligently pick 8 FIX, 5 C, and 5 VC
  1285→
  1286→export interface AutoSelectResult {
  1287→  fixedPlayers: TGPlayer[];
  1288→  captainOptions: TGPlayer[];
  1289→  viceCaptainOptions: TGPlayer[];
  1290→}
  1291→
  1292→// Player scoring for auto-selection: higher = better pick
  1293→function playerScore(p: TGPlayer): number {
  1294→  return (
  1295→    p.selected_by * 0.35 +       // selection % - most important for GL
  1296→    p.captain_percentage * 0.25 + // captain % - indicates trust
  1297→    p.points * 0.2 +             // points - performance indicator
  1298→    p.vice_captain_percentage * 0.1 + // VC %
  1299→    (p.playing === 1 ? 10 : 0) + // confirmed playing bonus
  1300→    (p.role === PLAYER_ROLES.ALL_ROUNDER ? 3 : 0) + // AR bonus (flexible)
  1301→    (p.role === PLAYER_ROLES.WICKET_KEEPER ? 1 : 0) // WK slight bonus
  1302→  );
  1303→}

  1305→export function autoSelectExtraPlayers(
  1306→  leftPlayers: TGPlayer[],
  1307→  rightPlayers: TGPlayer[],
  1308→  avoidPlayerIds: Set<number> = new Set(),
  1309→): AutoSelectResult {
  1310→  const allPlayersRaw = [...leftPlayers, ...rightPlayers];
  1311→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
  1312→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
  1313→
  1314→  // Use lineup-aware eligibility filter
  1315→  const eligible = getEligiblePlayers(allPlayersRaw, avoidPlayerIds);
  1316→
  1317→  const eligibleLeft = eligible.filter(p => p.team_name === leftTeamName);
  1318→  const eligibleRight = eligible.filter(p => p.team_name === rightTeamName);
  1319→
  1320→  // Sort by score descending
  1321→  const sortByScore = (a: TGPlayer, b: TGPlayer) => playerScore(b) - playerScore(a);
  1322→  const sortedLeft = [...eligibleLeft].sort(sortByScore);
  1323→  const sortedRight = [...eligibleRight].sort(sortByScore);
  1324→
  1325→  // === SELECT 8 FIX PLAYERS ===
  1326→  // Strategy: balanced 4-5 from each team, role-diverse
  1327→  const fixedPlayers: TGPlayer[] = [];
  1328→  const fixedIds = new Set<number>();
  1329→
  1330→  const targetLeft = Math.min(4, sortedLeft.length);
  1331→  const targetRight = Math.min(4, sortedRight.length);
  1332→
  1333→  // Pick diverse players ensuring role coverage
  1334→  const pickDiverse = (pool: TGPlayer[], count: number, existing: TGPlayer[]): TGPlayer[] => {
  1335→    const picked: TGPlayer[] = [];
  1336→    const usedIds = new Set(existing.map(p => p.pl_id));
  1337→    const existingRoles = {
  1338→      wk: existing.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length,
  1339→      bat: existing.filter(p => p.role === PLAYER_ROLES.BATSMAN).length,
  1340→      ar: existing.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length,
  1341→      bowl: existing.filter(p => p.role === PLAYER_ROLES.BOWLER).length,
  1342→    };
  1343→
  1344→    // First pass: fill role gaps
  1345→    const roleNeeds = [
  1346→      { role: PLAYER_ROLES.WICKET_KEEPER, need: Math.max(0, 1 - existingRoles.wk), pool: pool.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER && !usedIds.has(p.pl_id)) },
  1347→      { role: PLAYER_ROLES.ALL_ROUNDER, need: Math.max(0, 1 - existingRoles.ar), pool: pool.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER && !usedIds.has(p.pl_id)) },
  1348→      { role: PLAYER_ROLES.BATSMAN, need: Math.max(0, 2 - existingRoles.bat), pool: pool.filter(p => p.role === PLAYER_ROLES.BATSMAN && !usedIds.has(p.pl_id)) },
  1349→      { role: PLAYER_ROLES.BOWLER, need: Math.max(0, 2 - existingRoles.bowl), pool: pool.filter(p => p.role === PLAYER_ROLES.BOWLER && !usedIds.has(p.pl_id)) },
  1350→    ];
  1351→
  1352→    for (const rn of roleNeeds) {
  1353→      for (let i = 0; i < rn.need && picked.length < count; i++) {
  1354→        const candidate = rn.pool.find(p => !picked.some(pp => pp.pl_id === p.pl_id));
  1355→        if (candidate) picked.push(candidate);
  1356→      }
  1357→    }
  1358→
  1359→    // Second pass: fill remaining with top-scored players
  1360→    const remaining = pool.filter(p => !usedIds.has(p.pl_id) && !picked.some(pp => pp.pl_id === p.pl_id));
  1361→    for (const p of remaining) {
  1362→      if (picked.length >= count) break;
  1363→      picked.push(p);
  1364→    }
  1365→
  1366→    return picked;
  1367→  };
  1368→
  1369→  const leftPicks = pickDiverse(sortedLeft, targetLeft, []);
  1370→  for (const p of leftPicks) { fixedPlayers.push(p); fixedIds.add(p.pl_id); }
  1371→
  1372→  const rightPicks = pickDiverse(sortedRight, targetRight, fixedPlayers);
  1373→  for (const p of rightPicks) { fixedPlayers.push(p); fixedIds.add(p.pl_id); }
  1374→
  1375→  // Fill to 8 if needed
  1376→  if (fixedPlayers.length < 8) {
  1377→    const remaining = eligible.filter(p => !fixedIds.has(p.pl_id)).sort(sortByScore);
  1378→    for (const p of remaining) {
  1379→      if (fixedPlayers.length >= 8) break;
  1380→      fixedPlayers.push(p); fixedIds.add(p.pl_id);
  1381→    }
  1382→  }
  1383→
  1384→  // Trim to 8 if over
  1385→  while (fixedPlayers.length > 8) {
  1386→    let minIdx = 0; let minScore = playerScore(fixedPlayers[0]);
  1387→    for (let i = 1; i < fixedPlayers.length; i++) {
  1388→      const s = playerScore(fixedPlayers[i]);
  1389→      if (s < minScore) { minScore = s; minIdx = i; }
  1390→    }
  1391→    const removed = fixedPlayers.splice(minIdx, 1)[0];
  1392→    fixedIds.delete(removed.pl_id);
  1393→  }
  1394→
  1395→  // === SELECT 5 CAPTAIN OPTIONS ===
  1396→  const cCandidates = eligible
  1397→    .filter(p => !avoidPlayerIds.has(p.pl_id))
  1398→    .sort((a, b) => {
  1399→      const scoreA = a.captain_percentage * 0.4 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
  1400→      const scoreB = b.captain_percentage * 0.4 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
  1401→      return scoreB - scoreA;
  1402→    })
  1403→    .slice(0, 5);
  1404→
  1405→  // === SELECT 5 VICE CAPTAIN OPTIONS ===
  1406→  const vcCandidates: TGPlayer[] = [];
  1407→  const vcPool = eligible
  1408→    .filter(p => !avoidPlayerIds.has(p.pl_id))
  1409→    .sort((a, b) => {
  1410→      const scoreA = a.vice_captain_percentage * 0.35 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
  1411→      const scoreB = b.vice_captain_percentage * 0.35 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
  1412→      return scoreB - scoreA;
  1413→    });
  1414→
  1415→  for (const p of vcPool) {
  1416→    if (vcCandidates.length >= 5) break;
  1417→    vcCandidates.push(p);
  1418→  }
  1419→
  1420→  return {
  1421→    fixedPlayers: fixedPlayers.slice(0, 8),
  1422→    captainOptions: cCandidates.slice(0, 5),
  1423→    viceCaptainOptions: vcCandidates.slice(0, 5),
  1424→  };
  1425→}

  1427→// Auto-replace a removed player in a list with the next best eligible player
  1428→export function autoReplacePlayer(
  1429→  removedPlayer: TGPlayer,
  1430→  currentList: TGPlayer[],
  1431→  allPlayers: TGPlayer[],
  1432→  avoidPlayerIds: Set<number>,
  1433→  scoringFn: (p: TGPlayer) => number = playerScore,
  1434→): TGPlayer[] {
  1435→  const currentIds = new Set(currentList.map(p => p.pl_id));
  1436→  const eligible = allPlayers.filter(p =>
