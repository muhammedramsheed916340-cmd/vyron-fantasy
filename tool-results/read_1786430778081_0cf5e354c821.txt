    34→// ============ Lineup Utilities ============
    35→
    36→/**
    37→ * Determine lineup mode based on player data.
    38→ * AFTER LINEUP: any player has playing === 1 (confirmed in Playing XI)
    39→ * BEFORE LINEUP: no player is confirmed yet
    40→ */
    41→export function getLineupMode(players: TGPlayer[]): 'before' | 'after' {
    42→  const hasConfirmedPlaying = players.some(p => p.playing === 1);
    43→  return hasConfirmedPlaying ? 'after' : 'before';
    44→}
    45→
    46→/**
    47→ * Filter players based on lineup mode.
    48→ * BEFORE LINEUP: all players are eligible (no one confirmed OUT yet).
    49→ * AFTER LINEUP: ONLY players with playing === 1 are eligible.
    50→ */
    51→export function getEligiblePlayers(
    52→  allPlayers: TGPlayer[],
    53→  avoidPlayerIds: Set<number> = new Set(),
    54→): TGPlayer[] {
    55→  const mode = getLineupMode(allPlayers);
    56→
    57→  return allPlayers.filter(p => {
    58→    // Avoid players are always excluded
    59→    if (avoidPlayerIds.has(p.pl_id)) return false;
    60→
    61→    if (mode === 'after') {
    62→      // AFTER LINEUP: ONLY confirmed Playing XI players
    63→      return p.playing === 1;
    64→    }
    65→
    66→    // BEFORE LINEUP: all players eligible
    67→    return true;
    68→  });
    69→}

    71→/**
    72→ * Check if a specific player is eligible based on lineup mode.
    73→ */
    74→export function isPlayerEligible(
    75→  player: TGPlayer,
    76→  allPlayers: TGPlayer[],
    77→  avoidPlayerIds: Set<number> = new Set(),
    78→): { eligible: boolean; reason?: string } {
    79→  if (avoidPlayerIds.has(player.pl_id)) {
    80→    return { eligible: false, reason: 'AVOID' };
    81→  }
    82→
    83→  const mode = getLineupMode(allPlayers);
    84→
    85→  if (mode === 'after' && player.playing !== 1) {
    86→    return { eligible: false, reason: 'OUT / NOT IN PLAYING XI' };
    87→  }
    88→
    89→  return { eligible: true };
    90→}

    92→/**
    93→ * Validate a team for lineup eligibility.
    94→ * Returns list of invalid players with reasons.
    95→ */
    96→export function validateTeamForLineup(
    97→  team: GeneratedTeam,
    98→  allPlayers: TGPlayer[],
    99→  avoidPlayerIds: Set<number> = new Set(),
   100→): { valid: boolean; invalidPlayers: { player: TGPlayer; reason: string }[] } {
   101→  const mode = getLineupMode(allPlayers);
   102→  const invalidPlayers: { player: TGPlayer; reason: string }[] = [];
   103→
   104→  for (const player of team.players) {
   105→    if (avoidPlayerIds.has(player.pl_id)) {
   106→      invalidPlayers.push({ player, reason: 'AVOID' });
   107→      continue;
   108→    }
   109→
   110→    if (mode === 'after' && player.playing !== 1) {
   111→      invalidPlayers.push({ player, reason: 'OUT / NOT IN PLAYING XI' });
   112→    }
   113→  }
   114→
   115→  return { valid: invalidPlayers.length === 0, invalidPlayers };
   116→}

   129→/**
   130→ * Get all valid role combinations for an 11-player cricket team.
   131→ */
   132→export function getAllValidCombinations(): RoleCombination[] {
   133→  const combos: RoleCombination[] = [];
   134→  for (let wk = MIN_WK; wk <= MAX_WK; wk++) {
   135→    for (let bat = MIN_BAT; bat <= MAX_BAT; bat++) {
   136→      for (let ar = MIN_AR; ar <= MAX_AR; ar++) {
   137→        const bowl = CRICKET_TEAM_SIZE - wk - bat - ar;
   138→        if (bowl < MIN_BOWL || bowl > MAX_BOWL) continue;
   139→        combos.push({ wk, bat, ar, bowl });
   140→      }
   141→    }
   142→  }
   143→  return combos;
   144→}

   146→/**
   147→ * Check if a role combination is compatible with a set of fixed players.
   148→ */
   149→export function isCombinationCompatibleWithFixed(
   150→  combo: RoleCombination,
   151→  fixedPlayers: TGPlayer[],
   152→): boolean {
   153→  const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   154→  const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   155→  const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   156→  const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   157→
   158→  // Fixed players must not exceed combination totals
   159→  if (fixedWK > combo.wk) return false;
   160→  if (fixedBat > combo.bat) return false;
   161→  if (fixedAR > combo.ar) return false;
   162→  if (fixedBowl > combo.bowl) return false;
   163→
   164→  // Remaining slots must be fillable
   165→  const remainingPlayers = CRICKET_TEAM_SIZE - fixedPlayers.length;
   166→  const neededWK = combo.wk - fixedWK;
   167→  const neededBat = combo.bat - fixedBat;
   168→  const neededAR = combo.ar - fixedAR;
   169→  const neededBowl = combo.bowl - fixedBowl;
   170→
   171→  return (neededWK + neededBat + neededAR + neededBowl) === remainingPlayers;
   172→}

   174→/**
   175→ * Get combinations compatible with fixed players and available player pool.
   176→ */
   177→export function getCompatibleCombinations(
   178→  fixedPlayers: TGPlayer[],
   179→  eligiblePlayers: TGPlayer[],
   180→): RoleCombination[] {
   181→  const all = getAllValidCombinations();
   182→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   183→  const remaining = eligiblePlayers.filter(p => !fixedIds.has(p.pl_id));
   184→
   185→  const remWK = remaining.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   186→  const remBat = remaining.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   187→  const remAR = remaining.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   188→  const remBowl = remaining.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   189→
   190→  return all.filter(combo => {
   191→    if (!isCombinationCompatibleWithFixed(combo, fixedPlayers)) return false;
   192→
   193→    const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   194→    const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   195→    const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   196→    const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   197→
   198→    if (remWK < combo.wk - fixedWK) return false;
   199→    if (remBat < combo.bat - fixedBat) return false;
   200→    if (remAR < combo.ar - fixedAR) return false;
   201→    if (remBowl < combo.bowl - fixedBowl) return false;
   202→
   203→    return true;
   204→  });
   205→}

   207→/**
   208→ * Auto-select the best combination based on player pool and category.
   209→ * For multiple teams, rotates combinations intelligently.
   210→ */
   211→export function autoSelectCombination(
   212→  eligiblePlayers: TGPlayer[],
   213→  category: string,
   214→  fixedPlayers: TGPlayer[] = [],
   215→  teamIndex: number = 0,
   216→): RoleCombination {
   217→  const compatible = getCompatibleCombinations(fixedPlayers, eligiblePlayers);
   218→
   219→  if (compatible.length === 0) {
   220→    return { wk: 1, bat: 4, ar: 2, bowl: 4 };
   221→  }
   222→
   223→  // Category-preferred patterns
   224→  const preferredPatterns: RoleCombination[] =
   225→    category === 'H2H'
   226→      ? [{ wk: 1, bat: 5, ar: 2, bowl: 3 }, { wk: 1, bat: 4, ar: 2, bowl: 4 }, { wk: 1, bat: 3, ar: 3, bowl: 4 }]
   227→      : category === 'SL'
   228→      ? [{ wk: 1, bat: 4, ar: 2, bowl: 4 }, { wk: 1, bat: 3, ar: 3, bowl: 4 }, { wk: 2, bat: 3, ar: 2, bowl: 4 }]
   229→      : [ // Mega GL
   230→        { wk: 1, bat: 3, ar: 2, bowl: 5 }, { wk: 1, bat: 4, ar: 1, bowl: 5 },
   231→        { wk: 1, bat: 3, ar: 3, bowl: 4 }, { wk: 1, bat: 4, ar: 2, bowl: 4 },
   232→        { wk: 2, bat: 3, ar: 2, bowl: 4 }, { wk: 1, bat: 5, ar: 1, bowl: 4 },
   233→        { wk: 2, bat: 3, ar: 1, bowl: 5 }, { wk: 1, bat: 4, ar: 3, bowl: 3 },
   234→        { wk: 2, bat: 4, ar: 1, bowl: 4 }, { wk: 1, bat: 3, ar: 1, bowl: 6 },
   235→        { wk: 2, bat: 2, ar: 3, bowl: 4 }, { wk: 1, bat: 5, ar: 2, bowl: 3 },
   236→      ];
   237→
   238→  const preferredCompatible = preferredPatterns.filter(p =>
   239→    compatible.some(c => c.wk === p.wk && c.bat === p.bat && c.ar === p.ar && c.bowl === p.bowl)
   240→  );
   241→
   242→  const pool = preferredCompatible.length > 0 ? preferredCompatible : compatible;
   243→  const index = teamIndex % pool.length;
   244→
   245→  return pool[index];
   246→}

   248→/**
   249→ * Validate a manual combination against available players.
   250→ */
   251→export function validateCombination(
   252→  combo: RoleCombination,
   253→  eligiblePlayers: TGPlayer[],
   254→  fixedPlayers: TGPlayer[] = [],
   255→): { valid: boolean; errors: string[] } {
   256→  const errors: string[] = [];
   257→
   258→  if (combo.wk + combo.bat + combo.ar + combo.bowl !== CRICKET_TEAM_SIZE) {
   259→    errors.push(`Total players must be 11 (got ${combo.wk + combo.bat + combo.ar + combo.bowl})`);
   260→  }
   261→
   262→  if (combo.wk < MIN_WK || combo.wk > MAX_WK) errors.push(`WK must be ${MIN_WK}-${MAX_WK}`);
   263→  if (combo.bat < MIN_BAT || combo.bat > MAX_BAT) errors.push(`BAT must be ${MIN_BAT}-${MAX_BAT}`);
   264→  if (combo.ar < MIN_AR || combo.ar > MAX_AR) errors.push(`AR must be ${MIN_AR}-${MAX_AR}`);
   265→  if (combo.bowl < MIN_BOWL || combo.bowl > MAX_BOWL) errors.push(`BOWL must be ${MIN_BOWL}-${MAX_BOWL}`);
   266→
   267→  if (fixedPlayers.length > 0 && !isCombinationCompatibleWithFixed(combo, fixedPlayers)) {
   268→    errors.push('Combination is not compatible with selected fixed players');
   269→  }
   270→
   271→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   272→  const remaining = eligiblePlayers.filter(p => !fixedIds.has(p.pl_id));
   273→
   274→  const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   275→  const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   276→  const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   277→  const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   278→
   279→  const needWK = Math.max(0, combo.wk - fixedWK);
   280→  const needBat = Math.max(0, combo.bat - fixedBat);
   281→  const needAR = Math.max(0, combo.ar - fixedAR);
   282→  const needBowl = Math.max(0, combo.bowl - fixedBowl);
   283→
   284→  const availWK = remaining.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   285→  const availBat = remaining.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   286→  const availAR = remaining.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   287→  const availBowl = remaining.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   288→
   289→  if (availWK < needWK) errors.push(`Need ${needWK} WK but only ${availWK} available`);
   290→  if (availBat < needBat) errors.push(`Need ${needBat} BAT but only ${availBat} available`);
   291→  if (availAR < needAR) errors.push(`Need ${needAR} AR but only ${availAR} available`);
   292→  if (availBowl < needBowl) errors.push(`Need ${needBowl} BOWL but only ${availBowl} available`);
   293→
   294→  return { valid: errors.length === 0, errors };
   295→}

   375→function decryptField(encrypted: string): string {
   376→  try {
   377→    const bytes = CryptoJS.AES.decrypt(encrypted, ENCRYPTION_KEY);
   378→    const decrypted = bytes.toString(CryptoJS.enc.Utf8);
   379→    if (!decrypted) return encrypted;
   380→    return decrypted;
   381→  } catch {
   382→    return encrypted;
   383→  }
   384→}

   458→export async function fetchMatches(sport: string = 'cricket'): Promise<TGMatch[]> {
   459→  try {
   460→    const response = await fetch(`${TG_API_BASE}/fantasy/matches/${sport}`, {
   461→      next: { revalidate: 60 },
   462→    });
   463→
   464→    if (!response.ok) {
   465→      throw new Error(`Failed to fetch matches: ${response.status}`);
   466→    }
   467→
   468→    const data = await response.json();
   469→
   470→    if (data.status !== 'success' || !Array.isArray(data.data)) {
   471→      return [];
   472→    }
   473→
   474→    const matches: TGMatch[] = [];
   475→    for (const encryptedMatch of data.data) {
   476→      const match = decryptMatchData(encryptedMatch);
   477→      if (match) {
   478→        matches.push(match);
   479→      }
   480→    }
   481→
   482→    return matches;
   483→  } catch (error) {
   484→    console.error('Error fetching matches:', error);
   485→    return [];
   486→  }
   487→}

   501→    if (data.status !== 'success' || !data.data) {
   502→      return null;
   503→    }
   504→
   505→    const raw = data.data;
   506→
   507→    // Decrypt player lists

   509→    if (Array.isArray(raw.left_team_players)) {
   510→      for (const ep of raw.left_team_players) {
   511→        const player = decryptPlayerData(ep);

   519→        const player = decryptPlayerData(ep);
   520→        if (player) rightPlayers.push(player);
   521→      }
   522→    }
   523→
   524→    return {
   525→      _id: raw._id || '',
   526→      id: raw.id || '',
   527→      left_team_name: decryptField(raw.left_team_name || ''),

   537→      right_team_players: rightPlayers,
   538→      fantasy_version: raw.fantasy_version || [],
   539→    };
   540→  } catch (error) {
   541→    console.error('Error fetching match detail:', error);
   542→    return null;
   543→  }

   551→

   553→      throw new Error(`Failed to fetch promotions: ${response.status}`);
   554→    }
   555→
   556→    const data = await response.json();
   557→
   558→    if (data.status !== 'success' || !Array.isArray(data.data)) {
   559→      return [];
   560→    }
   561→
   562→    return data.data
   563→      .filter((p: TGPromotion) => p.active)

   580→// Cricket team constraints
   581→export const CRICKET_TEAM_SIZE = 11;

   583→export const MAX_WK = 4;
   584→export const MIN_BAT = 3;
   585→export const MAX_BAT = 6;
   586→export const MIN_AR = 1;
   587→export const MAX_AR = 4;
   588→export const MIN_BOWL = 3;
   589→export const MAX_BOWL = 6;
   590→export const MAX_CREDITS = 100;
   591→export const MAX_FROM_ONE_TEAM = 7;

   593→function seededRandom(seed: number): () => number {
   594→  let s = seed;
   595→  return () => {
   596→    s = (s * 16807 + 0) % 2147483647;
   597→    return (s - 1) / 2147483646;
   598→  };
   599→}
   600→
   601→function shuffleArray<T>(arr: T[], rng: () => number): T[] {
   602→  const result = [...arr];
   603→  for (let i = result.length - 1; i > 0; i--) {
   604→    const j = Math.floor(rng() * (i + 1));
   605→    [result[i], result[j]] = [result[j], result[i]];
   606→  }
   607→  return result;
   608→}

   610→export function generateTeams(
   611→  leftPlayers: TGPlayer[],
   612→  rightPlayers: TGPlayer[],
   613→  category: string,
   614→  count: number,
   615→  seed: number = Date.now(),
   616→  avoidPlayerIds: Set<number> = new Set(),
   617→  combination: RoleCombination | null = null,
   618→): GeneratedTeam[] {
   619→  // Apply lineup-aware filtering: AFTER LINEUP only playing=1, BEFORE LINEUP all
   620→  const allPlayersRaw = [...leftPlayers, ...rightPlayers];
   621→  const allPlayers = getEligiblePlayers(allPlayersRaw, avoidPlayerIds);
   622→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
   623→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
   624→
   625→  // After filtering, rebuild left/right pools for team distribution checks
   626→  const eligibleLeft = allPlayers.filter(p => p.team_name === leftTeamName);
   627→  const eligibleRight = allPlayers.filter(p => p.team_name === rightTeamName);
   628→
   629→  // Separate by role
   630→  const wk = allPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER);
   631→  const bat = allPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN);
   632→  const ar = allPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER);
   633→  const bowl = allPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER);
   634→
   635→  const teams: GeneratedTeam[] = [];
   636→  const rng = seededRandom(seed);
   637→
   638→  // Category-specific strategies
   639→  const isH2H = category === 'H2H';
   640→  const isSL = category === 'SL';
   641→
   642→  for (let t = 0; t < count; t++) {
   643→    let attempts = 0;
   644→    let team: TGPlayer[] | null = null;
   645→
   646→    while (!team && attempts < 200) {
   647→      attempts++;
   648→      const sr = seededRandom(seed + t * 1000 + attempts);
   649→
   650→      // Determine role counts based on category or explicit combination
   651→      let wkCount, batCount, arCount, bowlCount;
   652→
   653→      if (combination) {
   654→        // Use explicit combination (manual or auto-selected)
   655→        wkCount = combination.wk;
   656→        batCount = combination.bat;
   657→        arCount = combination.ar;
   658→        bowlCount = combination.bowl;
   659→      } else if (isH2H) {
   660→        // H2H: More conservative, popular players
   661→        wkCount = MIN_WK + (sr() > 0.5 ? 1 : 0);
   662→        arCount = MIN_AR + (sr() > 0.6 ? 1 : 0);
   663→        batCount = Math.max(MIN_BAT, 5 - wkCount + (sr() > 0.5 ? 1 : 0));
   664→        bowlCount = CRICKET_TEAM_SIZE - wkCount - batCount - arCount;
   665→      } else if (isSL) {
   666→        // SL: Balanced
   667→        wkCount = 1 + (sr() > 0.7 ? 1 : 0);
   668→        arCount = 2 + (sr() > 0.5 ? 1 : 0);
   669→        batCount = 3 + (sr() > 0.5 ? 1 : 0);
   670→        bowlCount = CRICKET_TEAM_SIZE - wkCount - batCount - arCount;
   671→      } else {
   672→        // Mega GL: More differential picks, varied combinations
   673→        const patterns = [
   674→          [1, 3, 2, 5], [1, 4, 1, 5], [1, 3, 3, 4], [1, 4, 2, 4],
   675→          [2, 3, 2, 4], [1, 5, 1, 4], [2, 3, 1, 5], [1, 4, 3, 3],
   676→          [2, 4, 1, 4], [1, 3, 1, 6], [2, 2, 3, 4], [1, 5, 2, 3],
   677→        ];
   678→        const pattern = patterns[Math.floor(sr() * patterns.length)];
   679→        [wkCount, batCount, arCount, bowlCount] = pattern;
   680→      }
   681→
   682→      // Validate counts
   683→      if (wkCount < MIN_WK || wkCount > MAX_WK) continue;
   684→      if (batCount < MIN_BAT || batCount > MAX_BAT) continue;
   685→      if (arCount < MIN_AR || arCount > MAX_AR) continue;
   686→      if (bowlCount < MIN_BOWL || bowlCount > MAX_BOWL) continue;
   687→      if (wkCount + batCount + arCount + bowlCount !== CRICKET_TEAM_SIZE) continue;
   688→
   689→      // Pick players for each role
   690→      const shuffledWK = shuffleArray(wk, sr);
   691→      const shuffledBat = shuffleArray(bat, sr);
   692→      const shuffledAR = shuffleArray(ar, sr);
   693→      const shuffledBowl = shuffleArray(bowl, sr);
   694→
   695→      // For Mega GL, sort by selected_by sometimes to get differential picks
   696→      let selectedWK, selectedBat, selectedAR, selectedBowl;
   697→
   698→      if (!isH2H && sr() > 0.4) {
   699→        // Sort some roles by selection % to mix popular and differential
   700→        const sortFn = (a: TGPlayer, b: TGPlayer) => a.selected_by - b.selected_by;
   701→        selectedWK = shuffleArray([...wk].sort(sortFn), sr).slice(0, wkCount);
   702→        selectedBat = shuffleArray([...bat].sort(sortFn), sr).slice(0, batCount);
   703→        selectedAR = shuffleArray([...ar].sort(sortFn), sr).slice(0, arCount);
   704→        selectedBowl = shuffleArray([...bowl].sort(sortFn), sr).slice(0, bowlCount);
   705→      } else if (isH2H) {
   706→        // H2H: Pick most selected players
   707→        const sortFn = (a: TGPlayer, b: TGPlayer) => b.selected_by - a.selected_by;
   708→        selectedWK = [...wk].sort(sortFn).slice(0, wkCount);
   709→        selectedBat = shuffleArray([...bat].sort(sortFn), sr).slice(0, batCount);
   710→        selectedAR = [...ar].sort(sortFn).slice(0, arCount);
   711→        selectedBowl = shuffleArray([...bowl].sort(sortFn), sr).slice(0, bowlCount);
   712→      } else {
   713→        selectedWK = shuffledWK.slice(0, wkCount);
   714→        selectedBat = shuffledBat.slice(0, batCount);
   715→        selectedAR = shuffledAR.slice(0, arCount);
   716→        selectedBowl = shuffledBowl.slice(0, bowlCount);
   717→      }
   718→
   719→      // Check if we have enough players
   720→      if (selectedWK.length < wkCount || selectedBat.length < batCount ||
   721→          selectedAR.length < arCount || selectedBowl.length < bowlCount) continue;
   722→
   723→      const selected = [...selectedWK, ...selectedBat, ...selectedAR, ...selectedBowl];
   724→
   725→      // Check total credits
   726→      const totalCredits = selected.reduce((sum, p) => sum + p.credits, 0);
   727→      if (totalCredits > MAX_CREDITS) continue;
   728→
   729→      // Check team distribution (max 7 from one team)
   730→      const leftCount = selected.filter(p => p.team_name === leftTeamName).length;
   731→      const rightCount = selected.filter(p => p.team_name === rightTeamName).length;
   732→      if (leftCount > MAX_FROM_ONE_TEAM || rightCount > MAX_FROM_ONE_TEAM) continue;
   733→
   734→      // Ensure at least 1 from each team
   735→      if (leftCount < 1 || rightCount < 1) continue;
   736→
   737→      team = selected;
   738→    }
   739→
   740→    if (team) {
   741→      // Select Captain & Vice-Captain
   742→      // Sort by a combination of points and selection % for C/VC
   743→      const sorted = [...team].sort((a, b) =>
   744→        (b.points * 0.4 + b.selected_by * 0.3 + b.captain_percentage * 0.3) -
   745→        (a.points * 0.4 + a.selected_by * 0.3 + a.captain_percentage * 0.3)
   746→      );
   747→
   748→      const captain = sorted[0];
   749→      const viceCaptain = sorted.length > 1 ? sorted[1] : sorted[0];
   750→
   751→      teams.push({
   752→        id: t + 1,
   753→        captain,
   754→        viceCaptain,
   755→        players: team,
   756→      });
   757→    }
   758→  }
   759→
   760→  return teams;
   761→}

   777→  combination?: RoleCombination | null; // Explicit role combination

   780→export function generateExtraTeams(input: ExtraTeamGenInput): GeneratedTeam[] {
   781→  const {
   782→    fixedPlayers,
   783→    captainOptions,
   784→    viceCaptainOptions,
   785→    leftPlayers,
   786→    rightPlayers,
   787→    category,
   788→    count,
   789→    seed = Date.now(),
   790→    avoidPlayerIds = new Set(),
   791→    combination = null,
   792→  } = input;
   793→
   794→  // Apply lineup-aware filtering to the full player pool
   795→  const allPlayersRaw = [...leftPlayers, ...rightPlayers];
   796→  const allPlayers = getEligiblePlayers(allPlayersRaw, avoidPlayerIds);
   797→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
   798→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
   799→
   800→  // Pool of players excluding the 8 fixed ones (use pl_id for identity)
   801→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   802→  const remainingPool = allPlayers.filter(p => !fixedIds.has(p.pl_id));
   803→
   804→  // Separate remaining pool by role
   805→  const remWK = remainingPool.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER);
   806→  const remBat = remainingPool.filter(p => p.role === PLAYER_ROLES.BATSMAN);
   807→  const remAR = remainingPool.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER);
   808→  const remBowl = remainingPool.filter(p => p.role === PLAYER_ROLES.BOWLER);
   809→
   810→  // Count fixed players by role
   811→  const fixedWKCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   812→  const fixedBatCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   813→  const fixedARCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   814→  const fixedBowlCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   815→
   816→  const fixedCredits = fixedPlayers.reduce((sum, p) => sum + p.credits, 0);
   817→  const remainingCredits = MAX_CREDITS - fixedCredits;
   818→  const neededPlayers = CRICKET_TEAM_SIZE - fixedPlayers.length; // 3
   819→
   820→  const teams: GeneratedTeam[] = [];
   821→
   822→  // Pre-validate: fixed players must have credits <= 100
   823→  if (fixedCredits > MAX_CREDITS) return teams;
   824→
   825→  // Pre-validate: fixed players must respect max 7 from one team
   826→  const fixedLeftCount = fixedPlayers.filter(p => p.team_name === leftTeamName).length;
   827→  const fixedRightCount = fixedPlayers.filter(p => p.team_name === rightTeamName).length;
   828→  if (fixedLeftCount > MAX_FROM_ONE_TEAM || fixedRightCount > MAX_FROM_ONE_TEAM) return teams;
   829→
   830→  // Generate all valid C/VC combinations from the 5C and 5VC options
   831→  // C must be in the team, VC must be in the team, C ≠ VC
   832→  const cVCCombos: { c: TGPlayer; vc: TGPlayer }[] = [];
   833→  for (const c of captainOptions) {
   834→    for (const vc of viceCaptainOptions) {
   835→      if (c.pl_id !== vc.pl_id) {
   836→        cVCCombos.push({ c, vc });
   837→      }
   838→    }
   839→  }
   840→  // If no valid combos, return empty
   841→  if (cVCCombos.length === 0) return teams;
   842→
   843→  // Category-specific role distribution preferences for remaining players
   844→  const isH2H = category === 'H2H';
   845→  const isSL = category === 'SL';
   846→
   847→  // Pre-compute all valid target role distributions that are compatible with fixed players
   848→  // This avoids wasting attempts on patterns that can never work
   849→  const categoryPatterns: number[][] = isH2H
   850→    ? [] // H2H uses dynamic generation below
   851→    : isSL
   852→    ? [] // SL uses dynamic generation below
   853→    : [
   854→        [1, 3, 2, 5], [1, 4, 1, 5], [1, 3, 3, 4], [1, 4, 2, 4],
   855→        [2, 3, 2, 4], [1, 5, 1, 4], [2, 3, 1, 5], [1, 4, 3, 3],
   856→        [2, 4, 1, 4], [1, 3, 1, 6], [2, 2, 3, 4], [1, 5, 2, 3],
   857→      ];
   858→
   859→  // Build list of valid target distributions: [WK, BAT, AR, BOWL]
   860→  const validDistributions: number[][] = [];
   861→  for (let wk = MIN_WK; wk <= MAX_WK; wk++) {
   862→    for (let bat = MIN_BAT; bat <= MAX_BAT; bat++) {
   863→      for (let ar = MIN_AR; ar <= MAX_AR; ar++) {
   864→        const bowl = CRICKET_TEAM_SIZE - wk - bat - ar;
   865→        if (bowl < MIN_BOWL || bowl > MAX_BOWL) continue;
   866→        // Check compatibility with fixed players
   867→        const nWK = Math.max(0, wk - fixedWKCount);
   868→        const nBat = Math.max(0, bat - fixedBatCount);
   869→        const nAR = Math.max(0, ar - fixedARCount);
   870→        const nBowl = Math.max(0, bowl - fixedBowlCount);
   871→        if (nWK + nBat + nAR + nBowl !== neededPlayers) continue;
   872→        // Check we have enough remaining players for each role
   873→        if (remWK.length < nWK || remBat.length < nBat || remAR.length < nAR || remBowl.length < nBowl) continue;
   874→        // For Mega GL / SL / H2H: prefer category-specific patterns but include all valid ones
   875→        validDistributions.push([wk, bat, ar, bowl]);
   876→      }
   877→    }
   878→  }
   879→
   880→  // Sort: prefer category-specific patterns first, then others
   881→  if (categoryPatterns.length > 0) {
   882→    const patternSet = new Set(categoryPatterns.map(p => p.join(',')));
   883→    validDistributions.sort((a, b) => {
   884→      const aInCat = patternSet.has(a.join(',')) ? 0 : 1;
   885→      const bInCat = patternSet.has(b.join(',')) ? 0 : 1;
   886→      return aInCat - bInCat;
   887→    });
   888→  }
   889→
   890→  if (validDistributions.length === 0) {
   891→    return teams;
   892→  }
   893→
   894→  // Team signature function for uniqueness check across generated teams
   895→  // Uses sorted player IDs + C/VC IDs to create a unique fingerprint
   896→  const teamSignatures = new Set<string>();
   897→  const makeTeamSignature = (players: TGPlayer[], c: TGPlayer, vc: TGPlayer): string => {
   898→    const playerIds = players.map(p => p.pl_id).sort((a, b) => a - b).join(',');
   899→    return `${playerIds}|C${c.pl_id}|VC${vc.pl_id}`;
   900→  };
   901→
   902→  // Generation loop with retry: keep trying until we have `count` unique teams
   903→  // or we exhaust the maximum total attempts
   904→  const MAX_TOTAL_ATTEMPTS = count * 500; // generous budget
   905→  let totalAttempts = 0;
   906→
   907→  for (let t = 0; t < count && totalAttempts < MAX_TOTAL_ATTEMPTS; t++) {
   908→    let attempts = 0;
   909→    let team: TGPlayer[] | null = null;
   910→    let teamC: TGPlayer | null = null;
   911→    let teamVC: TGPlayer | null = null;
   912→
   913→    while (!team && attempts < 500) {
   914→      attempts++;
   915→      totalAttempts++;
   916→      const sr = seededRandom(seed + t * 1000 + attempts);
   917→
   918→      // Pick a valid target distribution (or use explicit combination)
   919→      let targetWK: number, targetBat: number, targetAR: number, targetBowl: number;
   920→
   921→      if (combination) {
   922→        // Use explicit combination
   923→        targetWK = combination.wk;
   924→        targetBat = combination.bat;
   925→        targetAR = combination.ar;
   926→        targetBowl = combination.bowl;
   927→      } else if (validDistributions.length === 1) {
   928→        [targetWK, targetBat, targetAR, targetBowl] = validDistributions[0];
   929→      } else {
   930→        // Bias towards category-preferred patterns (first half of sorted list)
   931→        const idx = sr() < 0.7
   932→          ? Math.floor(sr() * Math.ceil(validDistributions.length / 2)) // 70% chance from preferred half
   933→          : Math.floor(sr() * validDistributions.length); // 30% from any
   934→        [targetWK, targetBat, targetAR, targetBowl] = validDistributions[Math.min(idx, validDistributions.length - 1)];
   935→      }
   936→
   937→      // How many more of each role do we need?
   938→      const needWK = Math.max(0, targetWK - fixedWKCount);
   939→      const needBat = Math.max(0, targetBat - fixedBatCount);
   940→      const needAR = Math.max(0, targetAR - fixedARCount);
   941→      const needBowl = Math.max(0, targetBowl - fixedBowlCount);
   942→
   943→      // Check if we have enough remaining players for each role
   944→      if (remWK.length < needWK || remBat.length < needBat ||
   945→          remAR.length < needAR || remBowl.length < needBowl) continue;
   946→
   947→      // Pick remaining players for each role
   948→      let pickedWK: TGPlayer[], pickedBat: TGPlayer[], pickedAR: TGPlayer[], pickedBowl: TGPlayer[];
   949→
   950→      if (!isH2H && sr() > 0.4) {
   951→        const sortFn = (a: TGPlayer, b: TGPlayer) => a.selected_by - b.selected_by;
   952→        pickedWK = shuffleArray([...remWK].sort(sortFn), sr).slice(0, needWK);
   953→        pickedBat = shuffleArray([...remBat].sort(sortFn), sr).slice(0, needBat);
   954→        pickedAR = shuffleArray([...remAR].sort(sortFn), sr).slice(0, needAR);
   955→        pickedBowl = shuffleArray([...remBowl].sort(sortFn), sr).slice(0, needBowl);
   956→      } else if (isH2H) {
   957→        const sortFn = (a: TGPlayer, b: TGPlayer) => b.selected_by - a.selected_by;
   958→        pickedWK = [...remWK].sort(sortFn).slice(0, needWK);
   959→        pickedBat = shuffleArray([...remBat].sort(sortFn), sr).slice(0, needBat);
   960→        pickedAR = [...remAR].sort(sortFn).slice(0, needAR);
   961→        pickedBowl = shuffleArray([...remBowl].sort(sortFn), sr).slice(0, needBowl);
   962→      } else {
   963→        pickedWK = shuffleArray(remWK, sr).slice(0, needWK);
   964→        pickedBat = shuffleArray(remBat, sr).slice(0, needBat);
   965→        pickedAR = shuffleArray(remAR, sr).slice(0, needAR);
   966→        pickedBowl = shuffleArray(remBowl, sr).slice(0, needBowl);
   967→      }
   968→
   969→      if (pickedWK.length < needWK || pickedBat.length < needBat ||
   970→          pickedAR.length < needAR || pickedBowl.length < needBowl) continue;
   971→
   972→      const remaining = [...pickedWK, ...pickedBat, ...pickedAR, ...pickedBowl];
   973→      const fullTeam = [...fixedPlayers, ...remaining];
   974→
   975→      // Check no duplicates (shouldn't happen but safety check)
   976→      const uniqueIds = new Set(fullTeam.map(p => p.pl_id));
   977→      if (uniqueIds.size !== fullTeam.length) continue;
   978→
   979→      // Check total credits
   980→      const totalCredits = fullTeam.reduce((sum, p) => sum + p.credits, 0);
   981→      if (totalCredits > MAX_CREDITS) continue;
   982→
   983→      // Check team distribution
   984→      const leftCount = fullTeam.filter(p => p.team_name === leftTeamName).length;
   985→      const rightCount = fullTeam.filter(p => p.team_name === rightTeamName).length;
   986→      if (leftCount > MAX_FROM_ONE_TEAM || rightCount > MAX_FROM_ONE_TEAM) continue;
   987→      if (leftCount < 1 || rightCount < 1) continue;
   988→
   989→      // Select C/VC combination for this team
   990→      // Round-robin through the valid combos, also try to ensure C and VC are in the team
   991→      const comboIndex = (t + attempts) % cVCCombos.length;
   992→      let combo = cVCCombos[comboIndex];
   993→
   994→      // Ensure C and VC are in this team
   995→      const teamPlayerIds = new Set(fullTeam.map(p => p.pl_id));
   996→      if (!teamPlayerIds.has(combo.c.pl_id) || !teamPlayerIds.has(combo.vc.pl_id)) {
   997→        // Try to find a valid combo where both C and VC are in the team
   998→        let foundValidCombo = false;
   999→        for (let ci = 0; ci < cVCCombos.length; ci++) {
  1000→          const tryCombo = cVCCombos[(comboIndex + ci) % cVCCombos.length];
  1001→          if (teamPlayerIds.has(tryCombo.c.pl_id) && teamPlayerIds.has(tryCombo.vc.pl_id)) {
  1002→            combo = tryCombo;
  1003→            foundValidCombo = true;
  1004→            break;
  1005→          }
  1006→        }
  1007→        if (!foundValidCombo) {
  1008→          // Fallback: use top players in team as C/VC
  1009→          const sorted = [...fullTeam].sort((a, b) =>
  1010→            (b.points * 0.4 + b.selected_by * 0.3 + b.captain_percentage * 0.3) -
  1011→            (a.points * 0.4 + a.selected_by * 0.3 + a.captain_percentage * 0.3)
  1012→          );
  1013→          combo = { c: sorted[0], vc: sorted.length > 1 ? sorted[1] : sorted[0] };
  1014→        }
  1015→      }
  1016→
  1017→      // Uniqueness check: skip if this team signature already exists
  1018→      const signature = makeTeamSignature(fullTeam, combo.c, combo.vc);
  1019→      if (teamSignatures.has(signature)) continue;
  1020→
  1021→      team = fullTeam;
  1022→      teamC = combo.c;
  1023→      teamVC = combo.vc;
  1024→      teamSignatures.add(signature);
  1025→    }
  1026→
  1027→    if (team && teamC && teamVC) {
  1028→      teams.push({
  1029→        id: teams.length + 1,
  1030→        captain: teamC,
  1031→        viceCaptain: teamVC,
  1032→        players: team,
  1033→      });
  1034→    } else {
  1035→      // If we failed to generate a unique team for this slot,
  1036→      // try extra attempts with different seeds before giving up
  1037→      let extraAttempts = 0;
  1038→      const MAX_EXTRA = 200;
  1039→      while (teams.length < count && extraAttempts < MAX_EXTRA && totalAttempts < MAX_TOTAL_ATTEMPTS) {
  1040→        extraAttempts++;
  1041→        totalAttempts++;
  1042→        const extraSeed = seed + count * 1000 + extraAttempts * 37;
  1043→        const sr = seededRandom(extraSeed);
  1044→
  1045→        // Pick distribution (or use explicit combination)
  1046→        let targetWK: number, targetBat: number, targetAR: number, targetBowl: number;
  1047→        if (combination) {
  1048→          targetWK = combination.wk;
  1049→          targetBat = combination.bat;
  1050→          targetAR = combination.ar;
  1051→          targetBowl = combination.bowl;
  1052→        } else if (validDistributions.length === 1) {
  1053→          [targetWK, targetBat, targetAR, targetBowl] = validDistributions[0];
  1054→        } else {
  1055→          const idx = Math.floor(sr() * validDistributions.length);
  1056→          [targetWK, targetBat, targetAR, targetBowl] = validDistributions[idx];
  1057→        }
  1058→
  1059→        const needWK = Math.max(0, targetWK - fixedWKCount);
  1060→        const needBat = Math.max(0, targetBat - fixedBatCount);
  1061→        const needAR = Math.max(0, targetAR - fixedARCount);
  1062→        const needBowl = Math.max(0, targetBowl - fixedBowlCount);
  1063→
  1064→        if (remWK.length < needWK || remBat.length < needBat ||
  1065→            remAR.length < needAR || remBowl.length < needBowl) continue;
  1066→
  1067→        const sortFn = (a: TGPlayer, b: TGPlayer) => a.selected_by - b.selected_by;
  1068→        const pickedWK = shuffleArray([...remWK].sort(sortFn), sr).slice(0, needWK);
  1069→        const pickedBat = shuffleArray([...remBat].sort(sortFn), sr).slice(0, needBat);
  1070→        const pickedAR = shuffleArray([...remAR].sort(sortFn), sr).slice(0, needAR);
  1071→        const pickedBowl = shuffleArray([...remBowl].sort(sortFn), sr).slice(0, needBowl);
  1072→
  1073→        if (pickedWK.length < needWK || pickedBat.length < needBat ||
  1074→            pickedAR.length < needAR || pickedBowl.length < needBowl) continue;
  1075→
  1076→        const remaining = [...pickedWK, ...pickedBat, ...pickedAR, ...pickedBowl];
  1077→        const fullTeam = [...fixedPlayers, ...remaining];
  1078→
  1079→        const uniqueIds = new Set(fullTeam.map(p => p.pl_id));
  1080→        if (uniqueIds.size !== fullTeam.length) continue;
  1081→
  1082→        const totalCredits = fullTeam.reduce((sum, p) => sum + p.credits, 0);
  1083→        if (totalCredits > MAX_CREDITS) continue;
  1084→
  1085→        const leftCount = fullTeam.filter(p => p.team_name === leftTeamName).length;
  1086→        const rightCount = fullTeam.filter(p => p.team_name === rightTeamName).length;
  1087→        if (leftCount > MAX_FROM_ONE_TEAM || rightCount > MAX_FROM_ONE_TEAM) continue;
  1088→        if (leftCount < 1 || rightCount < 1) continue;
  1089→
  1090→        // C/VC selection
  1091→        const comboIndex = extraAttempts % cVCCombos.length;
  1092→        let combo = cVCCombos[comboIndex];
  1093→        const teamPlayerIds = new Set(fullTeam.map(p => p.pl_id));
  1094→        if (!teamPlayerIds.has(combo.c.pl_id) || !teamPlayerIds.has(combo.vc.pl_id)) {
  1095→          let foundValidCombo = false;
  1096→          for (let ci = 0; ci < cVCCombos.length; ci++) {
  1097→            const tryCombo = cVCCombos[(comboIndex + ci) % cVCCombos.length];
  1098→            if (teamPlayerIds.has(tryCombo.c.pl_id) && teamPlayerIds.has(tryCombo.vc.pl_id)) {
  1099→              combo = tryCombo;
  1100→              foundValidCombo = true;
  1101→              break;
  1102→            }
  1103→          }
  1104→          if (!foundValidCombo) {
  1105→            const sorted = [...fullTeam].sort((a, b) =>
  1106→              (b.points * 0.4 + b.selected_by * 0.3 + b.captain_percentage * 0.3) -
  1107→              (a.points * 0.4 + a.selected_by * 0.3 + a.captain_percentage * 0.3)
  1108→            );
  1109→            combo = { c: sorted[0], vc: sorted.length > 1 ? sorted[1] : sorted[0] };
  1110→          }
  1111→        }
  1112→
  1113→        const signature = makeTeamSignature(fullTeam, combo.c, combo.vc);
  1114→        if (teamSignatures.has(signature)) continue;
  1115→        teamSignatures.add(signature);
  1116→
  1117→        teams.push({
  1118→          id: teams.length + 1,
  1119→          captain: combo.c,
  1120→          viceCaptain: combo.vc,
  1121→          players: fullTeam,
  1122→        });
  1123→        break; // Successfully added a team, move on
  1124→      }
  1125→    }
  1126→  }
  1127→
  1128→  return teams;
  1129→}

  1131→// ============ Auto Select for Extra Team Generation ============
  1132→// Uses real player data (selection %, captain %, points, credits, role, team balance)
  1133→// to intelligently pick 8 FIX, 5 C, and 5 VC
  1134→
  1135→export interface AutoSelectResult {
  1136→  fixedPlayers: TGPlayer[];
  1137→  captainOptions: TGPlayer[];
  1138→  viceCaptainOptions: TGPlayer[];
  1139→}
  1140→
  1141→// Player scoring for auto-selection: higher = better pick
  1142→function playerScore(p: TGPlayer): number {
  1143→  return (
  1144→    p.selected_by * 0.35 +       // selection % - most important for GL
  1145→    p.captain_percentage * 0.25 + // captain % - indicates trust
  1146→    p.points * 0.2 +             // points - performance indicator
  1147→    p.vice_captain_percentage * 0.1 + // VC %
  1148→    (p.playing === 1 ? 10 : 0) + // confirmed playing bonus
  1149→    (p.role === PLAYER_ROLES.ALL_ROUNDER ? 3 : 0) + // AR bonus (flexible)
  1150→    (p.role === PLAYER_ROLES.WICKET_KEEPER ? 1 : 0) // WK slight bonus
  1151→  );
  1152→}
  1153→
  1154→export function autoSelectExtraPlayers(
  1155→  leftPlayers: TGPlayer[],
  1156→  rightPlayers: TGPlayer[],
  1157→  avoidPlayerIds: Set<number> = new Set(),
  1158→): AutoSelectResult {
  1159→  const allPlayersRaw = [...leftPlayers, ...rightPlayers];
  1160→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
  1161→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
  1162→
  1163→  // Use lineup-aware eligibility filter
  1164→  const eligible = getEligiblePlayers(allPlayersRaw, avoidPlayerIds);
  1165→
  1166→  const eligibleLeft = eligible.filter(p => p.team_name === leftTeamName);
  1167→  const eligibleRight = eligible.filter(p => p.team_name === rightTeamName);
  1168→
  1169→  // Sort by score descending
  1170→  const sortByScore = (a: TGPlayer, b: TGPlayer) => playerScore(b) - playerScore(a);
  1171→  const sortedLeft = [...eligibleLeft].sort(sortByScore);
  1172→  const sortedRight = [...eligibleRight].sort(sortByScore);
  1173→
  1174→  // === SELECT 8 FIX PLAYERS ===
  1175→  // Strategy: balanced 4-5 from each team, role-diverse
  1176→  const fixedPlayers: TGPlayer[] = [];
  1177→  const fixedIds = new Set<number>();
  1178→
  1179→  const targetLeft = Math.min(4, sortedLeft.length);
  1180→  const targetRight = Math.min(4, sortedRight.length);
  1181→
  1182→  // Pick diverse players ensuring role coverage
  1183→  const pickDiverse = (pool: TGPlayer[], count: number, existing: TGPlayer[]): TGPlayer[] => {
  1184→    const picked: TGPlayer[] = [];
  1185→    const usedIds = new Set(existing.map(p => p.pl_id));
  1186→    const existingRoles = {
  1187→      wk: existing.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length,
  1188→      bat: existing.filter(p => p.role === PLAYER_ROLES.BATSMAN).length,
  1189→      ar: existing.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length,
  1190→      bowl: existing.filter(p => p.role === PLAYER_ROLES.BOWLER).length,
  1191→    };
  1192→
  1193→    // First pass: fill role gaps
  1194→    const roleNeeds = [
  1195→      { role: PLAYER_ROLES.WICKET_KEEPER, need: Math.max(0, 1 - existingRoles.wk), pool: pool.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER && !usedIds.has(p.pl_id)) },
  1196→      { role: PLAYER_ROLES.ALL_ROUNDER, need: Math.max(0, 1 - existingRoles.ar), pool: pool.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER && !usedIds.has(p.pl_id)) },
  1197→      { role: PLAYER_ROLES.BATSMAN, need: Math.max(0, 2 - existingRoles.bat), pool: pool.filter(p => p.role === PLAYER_ROLES.BATSMAN && !usedIds.has(p.pl_id)) },
  1198→      { role: PLAYER_ROLES.BOWLER, need: Math.max(0, 2 - existingRoles.bowl), pool: pool.filter(p => p.role === PLAYER_ROLES.BOWLER && !usedIds.has(p.pl_id)) },
  1199→    ];
  1200→
  1201→    for (const rn of roleNeeds) {
  1202→      for (let i = 0; i < rn.need && picked.length < count; i++) {
  1203→        const candidate = rn.pool.find(p => !picked.some(pp => pp.pl_id === p.pl_id));
  1204→        if (candidate) picked.push(candidate);
  1205→      }
  1206→    }
  1207→
  1208→    // Second pass: fill remaining with top-scored players
  1209→    const remaining = pool.filter(p => !usedIds.has(p.pl_id) && !picked.some(pp => pp.pl_id === p.pl_id));
  1210→    for (const p of remaining) {
  1211→      if (picked.length >= count) break;
  1212→      picked.push(p);
  1213→    }
  1214→
  1215→    return picked;
  1216→  };
  1217→
  1218→  const leftPicks = pickDiverse(sortedLeft, targetLeft, []);
  1219→  for (const p of leftPicks) { fixedPlayers.push(p); fixedIds.add(p.pl_id); }
  1220→
  1221→  const rightPicks = pickDiverse(sortedRight, targetRight, fixedPlayers);
  1222→  for (const p of rightPicks) { fixedPlayers.push(p); fixedIds.add(p.pl_id); }
  1223→
  1224→  // Fill to 8 if needed
  1225→  if (fixedPlayers.length < 8) {
  1226→    const remaining = eligible.filter(p => !fixedIds.has(p.pl_id)).sort(sortByScore);
  1227→    for (const p of remaining) {
  1228→      if (fixedPlayers.length >= 8) break;
  1229→      fixedPlayers.push(p); fixedIds.add(p.pl_id);
  1230→    }
  1231→  }
  1232→
  1233→  // Trim to 8 if over
  1234→  while (fixedPlayers.length > 8) {
  1235→    let minIdx = 0; let minScore = playerScore(fixedPlayers[0]);
  1236→    for (let i = 1; i < fixedPlayers.length; i++) {
  1237→      const s = playerScore(fixedPlayers[i]);
  1238→      if (s < minScore) { minScore = s; minIdx = i; }
  1239→    }
  1240→    const removed = fixedPlayers.splice(minIdx, 1)[0];
  1241→    fixedIds.delete(removed.pl_id);
  1242→  }
  1243→
  1244→  // === SELECT 5 CAPTAIN OPTIONS ===
  1245→  const cCandidates = eligible
  1246→    .filter(p => !avoidPlayerIds.has(p.pl_id))
  1247→    .sort((a, b) => {
  1248→      const scoreA = a.captain_percentage * 0.4 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
  1249→      const scoreB = b.captain_percentage * 0.4 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
  1250→      return scoreB - scoreA;
  1251→    })
  1252→    .slice(0, 5);
  1253→
  1254→  // === SELECT 5 VICE CAPTAIN OPTIONS ===
  1255→  const vcCandidates: TGPlayer[] = [];
  1256→  const vcPool = eligible
  1257→    .filter(p => !avoidPlayerIds.has(p.pl_id))
  1258→    .sort((a, b) => {
  1259→      const scoreA = a.vice_captain_percentage * 0.35 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
  1260→      const scoreB = b.vice_captain_percentage * 0.35 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
  1261→      return scoreB - scoreA;
  1262→    });
  1263→
  1264→  for (const p of vcPool) {
  1265→    if (vcCandidates.length >= 5) break;
  1266→    vcCandidates.push(p);
  1267→  }
  1268→
  1269→  return {
  1270→    fixedPlayers: fixedPlayers.slice(0, 8),
  1271→    captainOptions: cCandidates.slice(0, 5),
  1272→    viceCaptainOptions: vcCandidates.slice(0, 5),
  1273→  };
  1274→}

  1276→// Auto-replace a removed player in a list with the next best eligible player
  1277→export function autoReplacePlayer(
  1278→  removedPlayer: TGPlayer,
  1279→  currentList: TGPlayer[],
  1280→  allPlayers: TGPlayer[],
  1281→  avoidPlayerIds: Set<number>,
  1282→  scoringFn: (p: TGPlayer) => number = playerScore,
  1283→): TGPlayer[] {
  1284→  const currentIds = new Set(currentList.map(p => p.pl_id));
  1285→  const eligible = allPlayers.filter(p =>
  1286→    !currentIds.has(p.pl_id) && !avoidPlayerIds.has(p.pl_id) && p.pl_id !== removedPlayer.pl_id
  1287→  );
  1288→  const sorted = eligible.sort((a, b) => scoringFn(b) - scoringFn(a));
  1289→  const replacement = sorted[0];
  1290→  if (replacement) return [...currentList, replacement];
  1291→  return currentList;
  1292→}
