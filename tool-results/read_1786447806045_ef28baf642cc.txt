     1→     1→'use client';
     2→     2→
     3→     3→import React, { useState, useCallback, useMemo } from 'react';
     4→     4→import {
     5→     5→  Trophy, X, CheckCircle2, Loader2, ChevronRight, ChevronLeft,
     6→     6→  AlertCircle, Users, DollarSign, Filter, RefreshCw,
     7→     7→} from 'lucide-react';
     8→     8→import {
     9→     9→  JCMatch, JCContest, JCTeam, JCJoinItem, JCProgress,
    10→    10→  parseContest, formatCurrency, getContestTypeColor,
    11→    11→  getExistingPlatformTeams, buildJoinItems, getJoinKey,
    12→    12→} from '@/lib/join-contest-service';
    13→    13→
    14→    14→// ============ Step Enum ============
    15→    15→type Step = 'matches' | 'teams' | 'contests' | 'joining' | 'result';
    16→    16→
    17→    17→// ============ Props ============
    18→    18→interface JoinContestDialogProps {
    19→    19→  open: boolean;
    20→    20→  onClose: () => void;
    21→    21→  matches: JCMatch[];
    22→    22→  fantasyAccounts: Record<string, { authToken: string; mobileNumber: string; my11circleChallenge?: string | null }>;
    23→    23→}
    24→    24→
    25→    25→// ============ Main Dialog ============
    26→    26→export default function JoinContestDialog({
    27→    27→  open, onClose, matches, fantasyAccounts,
    28→    28→}: JoinContestDialogProps) {
    29→    29→  // Step state
    30→    30→  const [step, setStep] = useState<Step>('matches');
    31→    31→  const [platform, setPlatform] = useState<string>('dream11');
    32→    32→
    33→    33→  // Match selection
    34→    34→  const [selectedMatchIds, setSelectedMatchIds] = useState<Set<string | number>>(new Set());
    35→    35→
    36→    36→  // Team selection — EXISTING PLATFORM TEAMS
    37→    37→  const [platformTeams, setPlatformTeams] = useState<JCTeam[]>([]);
    38→    38→  const [selectedTeamIds, setSelectedTeamIds] = useState<Set<string | number>>(new Set());
    39→    39→  const [mixedTeamMode, setMixedTeamMode] = useState(false);
    40→    40→  const [loadingTeams, setLoadingTeams] = useState(false);
    41→    41→  const [teamsError, setTeamsError] = useState<string | null>(null);
    42→    42→  const [teamsTokenExpired, setTeamsTokenExpired] = useState(false);
    43→    43→
    44→    44→  // Contest selection
    45→    45→  const [contestsMap, setContestsMap] = useState<Map<string, JCContest[]>>(new Map());
    46→    46→  const [selectedContestIds, setSelectedContestIds] = useState<Set<string>>(new Set());
    47→    47→  const [loadingContests, setLoadingContests] = useState(false);
    48→    48→
    49→    49→  // Join progress
    50→    50→  const [joinItems, setJoinItems] = useState<JCJoinItem[]>([]);
    51→    51→  const [progress, setProgress] = useState<JCProgress>({ current: 0, total: 0, status: 'idle' });
    52→    52→
    53→    53→  // Result
    54→    54→  const [resultItems, setResultItems] = useState<JCJoinItem[]>([]);
    55→    55→
    56→    56→  // Available platforms from linked accounts
    57→    57→  const availablePlatforms = useMemo(() => {
    58→    58→    const platforms: string[] = [];
    59→    59→    if (fantasyAccounts.dream11?.authToken) platforms.push('dream11');
    60→    60→    if (fantasyAccounts.my11circle?.authToken) platforms.push('my11circle');
    61→    61→    return platforms;
    62→    62→  }, [fantasyAccounts]);
    63→    63→
    64→    64→  const account = fantasyAccounts[platform];
    65→    65→
    66→    66→  // Reset on close
    67→    67→  const handleClose = () => {
    68→    68→    setStep('matches');
    69→    69→    setSelectedMatchIds(new Set());
    70→    70→    setPlatformTeams([]);
    71→    71→    setSelectedTeamIds(new Set());
    72→    72→    setMixedTeamMode(false);
    73→    73→    setLoadingTeams(false);
    74→    74→    setTeamsError(null);
    75→    75→    setTeamsTokenExpired(false);
    76→    76→    setContestsMap(new Map());
    77→    77→    setSelectedContestIds(new Set());
    78→    78→    setJoinItems([]);
    79→    79→    setProgress({ current: 0, total: 0, status: 'idle' });
    80→    80→    setResultItems([]);
    81→    81→    onClose();
    82→    82→  };
    83→    83→
    84→    84→  // ============ Match Selection ============
    85→    85→  const toggleMatch = (id: string | number) => {
    86→    86→    setSelectedMatchIds(prev => {
    87→    87→      const next = new Set(prev);
    88→    88→      next.has(id) ? next.delete(id) : next.add(id);
    89→    89→      return next;
    90→    90→    });
    91→    91→  };
    92→    92→
    93→    93→  const selectAllMatches = () => {
    94→    94→    setSelectedMatchIds(new Set(matches.map(m => m.id)));
    95→    95→  };
    96→    96→
    97→    97→  // ============ Load Existing Platform Teams ============
    98→    98→  const loadPlatformTeams = useCallback(async (selectedPlatform?: string) => {
    99→    99→    const plat = selectedPlatform || platform;
   100→   100→    const acc = fantasyAccounts[plat];
   101→   101→    if (!acc?.authToken) {
   102→   102→      setTeamsError('No connected account found. Connect your platform account to load existing teams.');
   103→   103→      setPlatformTeams([]);
   104→   104→      return;
   105→   105→    }
   106→   106→
   107→   107→    setLoadingTeams(true);
   108→   108→    setTeamsError(null);
   109→   109→    setTeamsTokenExpired(false);
   110→   110→    setSelectedTeamIds(new Set());
   111→   111→
   112→   112→    const allTeams: JCTeam[] = [];
   113→   113→    let firstError: string | null = null;
   114→   114→    let anyTokenExpired = false;
   115→   115→
   116→   116→    for (const matchId of selectedMatchIds) {
   117→   117→      const result = await getExistingPlatformTeams(plat, matchId, acc.authToken);
   118→   118→      if (result.teams.length > 0) {
   119→   119→        allTeams.push(...result.teams);
   120→   120→      }
   121→   121→      if (result.tokenExpired) {
   122→   122→        anyTokenExpired = true;
   123→   123→        firstError = result.error || 'Session expired';
   124→   124→      } else if (result.error && !firstError) {
   125→   125→        firstError = result.error;
   126→   126→      }
   127→   127→    }
   128→   128→
   129→   129→    setPlatformTeams(allTeams);
   130→   130→    setTeamsError(allTeams.length === 0 ? firstError : null);
   131→   131→    setTeamsTokenExpired(anyTokenExpired);
   132→   132→    setLoadingTeams(false);
   133→   133→  }, [selectedMatchIds, platform, fantasyAccounts]);
   134→   134→
   135→   135→  // ============ Team Selection ============
   136→   136→  const toggleTeam = (id: string | number) => {
   137→   137→    setSelectedTeamIds(prev => {
   138→   138→      const next = new Set(prev);
   139→   139→      next.has(id) ? next.delete(id) : next.add(id);
   140→   140→      return next;
   141→   141→    });
   142→   142→  };
   143→   143→
   144→   144→  const selectAllTeams = () => {
   145→   145→    setSelectedTeamIds(new Set(platformTeams.map(t => t.id)));
   146→   146→  };
   147→   147→
   148→   148→  const deselectAllTeams = () => {
   149→   149→    setSelectedTeamIds(new Set());
   150→   150→  };
   151→   151→
   152→   152→  // ============ Contest Loading ============
   153→   153→  const loadContests = useCallback(async () => {
   154→   154→    if (!account?.authToken) return;
   155→   155→    setLoadingContests(true);
   156→   156→    const newMap = new Map<string, JCContest[]>();
   157→   157→
   158→   158→    for (const matchId of selectedMatchIds) {
   159→   159→      const match = matches.find(m => m.id === matchId);
   160→   160→      try {
   161→   161→        const res = await fetch('/api/fantasy/list-contests', {
   162→   162→          method: 'POST',
   163→   163→          headers: { 'Content-Type': 'application/json' },
   164→   164→          body: JSON.stringify({
   165→   165→            fantasyApp: platform,
   166→   166→            matchId,
   167→   167→            authToken: account.authToken,
   168→   168→            sportIndex: match?.sport_index ?? 0,
   169→   169→          }),
   170→   170→        });
   171→   171→        const data = await res.json();
   172→   172→        if (data.status === 'success' && Array.isArray(data.data?.contests)) {
   173→   173→          newMap.set(String(matchId), data.data.contests.map((c: Record<string, unknown>) => parseContest(c)));
   174→   174→        } else {
   175→   175→          newMap.set(String(matchId), []);
   176→   176→        }
   177→   177→      } catch {
   178→   178→        newMap.set(String(matchId), []);
   179→   179→      }
   180→   180→    }
   181→   181→
   182→   182→    setContestsMap(newMap);
   183→   183→    setLoadingContests(false);
   184→   184→  }, [selectedMatchIds, platform, account, matches]);
   185→   185→
   186→   186→  const toggleContest = (id: string) => {
   187→   187→    setSelectedContestIds(prev => {
   188→   188→      const next = new Set(prev);
   189→   189→      next.has(id) ? next.delete(id) : next.add(id);
   190→   190→      return next;
   191→   191→    });
   192→   192→  };
   193→   193→
   194→   194→  // ============ Join Execution ============
   195→   195→  const executeJoin = useCallback(async () => {
   196→   196→    const selectedMatches = matches.filter(m => selectedMatchIds.has(m.id));
   197→   197→    const selectedContestsForJoin = new Map<string, JCContest[]>();
   198→   198→    for (const [matchId, contests] of contestsMap.entries()) {
   199→   199→      selectedContestsForJoin.set(matchId, contests.filter(c => selectedContestIds.has(c.id)));
   200→   200→    }
   201→   201→
   202→   202→    const items = buildJoinItems(selectedMatches, selectedContestsForJoin, selectedTeamIds, platformTeams, platform);
   203→   203→
   204→   204→    if (items.length === 0) {
   205→   205→      setStep('result');
   206→   206→      return;
   207→   207→    }
   208→   208→
   209→   209→    setJoinItems(items);
   210→   210→    setProgress({ current: 0, total: items.length, status: 'joining' });
   211→   211→    setStep('joining');
   212→   212→
   213→   213→    const updatedItems = [...items];
   214→   214→
   215→   215→    for (let i = 0; i < items.length; i++) {
   216→   216→      const item = items[i];
   217→   217→      setProgress(prev => ({ ...prev, current: i + 1 }));
   218→   218→      setJoinItems(prev => prev.map((it, idx) =>
   219→   219→        idx === i ? { ...it, status: 'processing' } : it
   220→   220→      ));
   221→   221→
   222→   222→      try {
   223→   223→        const match = matches.find(m => String(m.id) === String(item.matchId));
   224→   224→        const res = await fetch('/api/fantasy/join-contest', {
   225→   225→          method: 'POST',
   226→   226→          headers: { 'Content-Type': 'application/json' },
   227→   227→          body: JSON.stringify({
   228→   228→            fantasyApp: item.platform || platform,
   229→   229→            matchId: item.matchId,
   230→   230→            authToken: account?.authToken,
   231→   231→            teamId: item.teamId,    // REAL platform team ID
   232→   232→            contestId: item.contestId,
   233→   233→            sportIndex: match?.sport_index ?? 0,
   234→   234→          }),
   235→   235→        });
   236→   236→        const data = await res.json();
   237→   237→
   238→   238→        const status = data.status === 'success' ? 'success'
   239→   239→          : data.status === 'already_joined' ? 'already_joined' : 'fail';
   240→   240→
   241→   241→        updatedItems[i] = { ...updatedItems[i], status, message: data.message };
   242→   242→        setJoinItems([...updatedItems]);
   243→   243→      } catch {
   244→   244→        updatedItems[i] = { ...updatedItems[i], status: 'fail', message: 'Network error' };
   245→   245→        setJoinItems([...updatedItems]);
   246→   246→      }
   247→   247→
   248→   248→      // Rate limit between joins
   249→   249→      await new Promise(r => setTimeout(r, 300));
   250→   250→    }
   251→   251→
   252→   252→    setResultItems([...updatedItems]);
   253→   253→    setProgress(prev => ({ ...prev, status: 'done' }));
   254→   254→    setStep('result');
   255→   255→  }, [matches, selectedMatchIds, contestsMap, selectedContestIds, selectedTeamIds, platformTeams, platform, account]);
   256→   256→
   257→   257→  // ============ Result Stats ============
   258→   258→  const resultStats = useMemo(() => {
   259→   259→    const items = resultItems.length > 0 ? resultItems : joinItems;
   260→   260→    return {
   261→   261→      total: items.length,
   262→   262→      success: items.filter(i => i.status === 'success').length,
   263→   263→      alreadyJoined: items.filter(i => i.status === 'already_joined').length,
   264→   264→      fail: items.filter(i => i.status === 'fail').length,
   265→   265→    };
   266→   266→  }, [resultItems, joinItems]);
   267→   267→
   268→   268→  if (!open) return null;
   269→   269→
   270→   270→  // ============ RENDER ============
   271→   271→  return (
   272→   272→    <div className="fixed inset-0 z-50 bg-black/60 flex items-center justify-center p-4" onClick={handleClose}>
   273→   273→      <div className="bg-white rounded-2xl shadow-2xl w-full max-w-2xl max-h-[90vh] overflow-hidden flex flex-col" onClick={e => e.stopPropagation()}>
   274→   274→        {/* Header */}
   275→   275→        <div className="flex items-center justify-between px-6 py-4 border-b bg-gradient-to-r from-green-600 to-emerald-600 text-white">
   276→   276→          <div className="flex items-center gap-3">
   277→   277→            <Trophy className="w-6 h-6" />
   278→   278→            <h2 className="text-lg font-bold">Join Contest</h2>
   279→   279→          </div>
   280→   280→          <button onClick={handleClose} className="p-1 hover:bg-white/20 rounded-lg transition-all">
   281→   281→            <X className="w-5 h-5" />
   282→   282→          </button>
   283→   283→        </div>
   284→   284→
   285→   285→        {/* Step Indicator */}
   286→   286→        <div className="flex items-center gap-1 px-6 py-3 bg-gray-50 border-b">
   287→   287→          {(['matches', 'teams', 'contests', 'joining', 'result'] as Step[]).map((s, i) => {
   288→   288→            const labels = ['Match', 'Teams', 'Contests', 'Join', 'Result'];
   289→   289→            const stepOrder = ['matches', 'teams', 'contests', 'joining', 'result'];
   290→   290→            const isActive = step === s;
   291→   291→            const isDone = stepOrder.indexOf(step) > i;
   292→   292→            return (
   293→   293→              <React.Fragment key={s}>
   294→   294→                {i > 0 && <ChevronRight className="w-3 h-3 text-gray-300" />}
   295→   295→                <div className={`flex items-center gap-1.5 text-xs font-semibold px-2 py-1 rounded-lg transition-all ${
   296→   296→                  isActive ? 'bg-green-100 text-green-700' : isDone ? 'text-green-500' : 'text-gray-400'
   297→   297→                }`}>
   298→   298→                  <div className={`w-5 h-5 rounded-full flex items-center justify-center text-[10px] ${
   299→   299→                    isActive ? 'bg-green-600 text-white' : isDone ? 'bg-green-200 text-green-600' : 'bg-gray-200 text-gray-500'
   300→   300→                  }`}>
   301→   301→                    {isDone ? '✓' : i + 1}
   302→   302→                  </div>
   303→   303→                  {labels[i]}
   304→   304→                </div>
   305→   305→              </React.Fragment>
   306→   306→            );
   307→   307→          })}
   308→   308→        </div>
   309→   309→
   310→   310→        {/* Content */}
   311→   311→        <div className="flex-1 overflow-y-auto p-6">
   312→   312→
   313→   313→          {/* ===== STEP: MATCHES ===== */}
   314→   314→          {step === 'matches' && (
   315→   315→            <div className="space-y-4">
   316→   316→              {/* Platform Selection */}
   317→   317→              <div className="space-y-2">
   318→   318→                <p className="text-xs font-semibold text-gray-500 uppercase tracking-wider">Platform</p>
   319→   319→                <div className="flex gap-2">
   320→   320→                  {availablePlatforms.map(p => (
   321→   321→                    <button key={p} onClick={() => setPlatform(p)}
   322→   322→                      className={`flex-1 rounded-xl border-2 p-3 text-center font-semibold text-sm transition-all ${
   323→   323→                        platform === p ? 'border-green-500 bg-green-50 text-green-700' : 'border-gray-200 bg-white text-gray-700 hover:border-gray-300'
   324→   324→                      }`}>
   325→   325→                      {p === 'dream11' ? 'Dream11' : 'My11Circle'}
   326→   326→                    </button>
   327→   327→                  ))}
   328→   328→                </div>
   329→   329→                {availablePlatforms.length === 0 && (
   330→   330→                  <div className="bg-amber-50 border border-amber-200 rounded-xl p-3 text-sm text-amber-700">
   331→   331→                    ⚠️ No connected account found. Link your platform account first via Transfer.
   332→   332→                  </div>
   333→   333→                )}
   334→   334→              </div>
   335→   335→
   336→   336→              {/* Match List */}
   337→   337→              <div className="space-y-2">
   338→   338→                <div className="flex items-center justify-between">
   339→   339→                  <p className="text-xs font-semibold text-gray-500 uppercase tracking-wider">Select Matches</p>
   340→   340→                  <button onClick={selectAllMatches} className="text-xs font-semibold text-green-600 hover:text-green-700">
   341→   341→                    Select All
   342→   342→                  </button>
   343→   343→                </div>
   344→   344→                {matches.length === 0 ? (
   345→   345→                  <p className="text-sm text-gray-400 text-center py-8">No upcoming matches</p>
   346→   346→                ) : (
   347→   347→                  <div className="space-y-2 max-h-80 overflow-y-auto">
   348→   348→                    {matches.map(match => {
   349→   349→                      const selected = selectedMatchIds.has(match.id);
   350→   350→                      return (
   351→   351→                        <button key={match.id} onClick={() => toggleMatch(match.id)}
   352→   352→                          className={`w-full text-left rounded-xl border-2 p-3 transition-all ${
   353→   353→                            selected ? 'border-green-500 bg-green-50' : 'border-gray-200 bg-white hover:border-gray-300'
   354→   354→                          }`}>
   355→   355→                          <div className="flex items-center gap-3">
   356→   356→                            <div className={`w-5 h-5 rounded border-2 flex items-center justify-center shrink-0 ${
   357→   357→                              selected ? 'border-green-500 bg-green-500' : 'border-gray-300'
   358→   358→                            }`}>
   359→   359→                              {selected && <CheckCircle2 className="w-3.5 h-3.5 text-white" />}
   360→   360→                            </div>
   361→   361→                            <div className="flex-1">
   362→   362→                              <p className="font-semibold text-sm text-gray-900">
   363→   363→                                {match.left_team_name} vs {match.right_team_name}
   364→   364→                              </p>
   365→   365→                              <div className="flex items-center gap-2 text-xs text-gray-500 mt-0.5">
   366→   366→                                <span>{match.match_time}</span>
   367→   367→                                {match.lineup_out === 1 && <span className="text-green-600 font-medium">Lineup Out</span>}
   368→   368→                              </div>
   369→   369→                            </div>
   370→   370→                          </div>
   371→   371→                        </button>
   372→   372→                      );
   373→   373→                    })}
   374→   374→                  </div>
   375→   375→                )}
   376→   376→              </div>
   377→   377→            </div>
   378→   378→          )}
   379→   379→
   380→   380→          {/* ===== STEP: TEAMS — EXISTING PLATFORM TEAMS ===== */}
   381→   381→          {step === 'teams' && (
   382→   382→            <div className="space-y-4">
   383→   383→              {/* Loading State */}
   384→   384→              {loadingTeams && (
   385→   385→                <div className="flex flex-col items-center justify-center py-12">
   386→   386→                  <Loader2 className="w-8 h-8 animate-spin text-green-500" />
   387→   387→                  <p className="text-sm text-gray-500 mt-3">Loading existing teams from {platform === 'dream11' ? 'Dream11' : 'My11Circle'}...</p>
   388→   388→                </div>
   389→   389→              )}
   390→   390→
   391→   391→              {/* Error State */}
   392→   392→              {!loadingTeams && teamsError && (
   393→   393→                <div className="bg-red-50 border border-red-200 rounded-xl p-4">
   394→   394→                  <div className="flex items-center gap-2">
   395→   395→                    <AlertCircle className="w-5 h-5 text-red-500 shrink-0" />
   396→   396→                    <div>
   397→   397→                      <p className="text-sm font-semibold text-red-700">
   398→   398→                        {teamsTokenExpired ? 'SESSION EXPIRED' : 'Error Loading Teams'}
   399→   399→                      </p>
   400→   400→                      <p className="text-xs text-red-600 mt-0.5">{teamsError}</p>
   401→   401→                    </div>
   402→   402→                  </div>
   403→   403→                  {teamsTokenExpired && (
   404→   404→                    <p className="text-xs text-red-500 mt-2">Reconnect your {platform === 'dream11' ? 'Dream11' : 'My11Circle'} account via Transfer.</p>
   405→   405→                  )}
   406→   406→                  <button onClick={() => loadPlatformTeams()} className="mt-2 flex items-center gap-1 text-xs font-semibold text-red-600 hover:text-red-700">
   407→   407→                    <RefreshCw className="w-3 h-3" /> Retry
   408→   408→                  </button>
   409→   409→                </div>
   410→   410→              )}
   411→   411→
   412→   412→              {/* No Account */}
   413→   413→              {!loadingTeams && !teamsError && !account?.authToken && (
   414→   414→                <div className="bg-amber-50 border border-amber-200 rounded-xl p-4 text-center">
   415→   415→                  <p className="text-sm font-semibold text-amber-700">No connected account found</p>
   416→   416→                  <p className="text-xs text-amber-600 mt-1">Connect your {platform === 'dream11' ? 'Dream11' : 'My11Circle'} account to load existing teams.</p>
   417→   417→                </div>
   418→   418→              )}
   419→   419→
   420→   420→              {/* Teams List */}
   421→   421→              {!loadingTeams && account?.authToken && (
   422→   422→                <div className="space-y-2">
   423→   423→                  <div className="flex items-center justify-between">
   424→   424→                    <p className="text-xs font-semibold text-gray-500 uppercase tracking-wider">
   425→   425→                      Existing Teams ({platformTeams.length} available)
   426→   426→                    </p>
