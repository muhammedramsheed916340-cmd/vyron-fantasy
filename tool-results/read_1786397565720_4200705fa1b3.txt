   133→   133→// Countdown timer component
   134→   134→function CountdownTimer({ matchTime }: { matchTime: string }) {
   135→   135→  const [timeLeft, setTimeLeft] = useState('')
   136→   136→
   137→   137→  useEffect(() => {
   138→   138→    const calculateTime = () => {
   139→   139→      const now = new Date().getTime()
   140→   140→      const matchDate = new Date(matchTime).getTime()
   141→   141→      const diff = matchDate - now
   142→   142→
   143→   143→      if (diff <= 0) {
   144→   144→        setTimeLeft('Started')
   145→   145→        return
   146→   146→      }
   147→   147→
   148→   148→      const days = Math.floor(diff / (1000 * 60 * 60 * 24))
   149→   149→      const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60))
   150→   150→      const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60))
   151→   151→      const seconds = Math.floor((diff % (1000 * 60)) / 1000)
   152→   152→
   153→   153→      if (days > 0) {
   154→   154→        setTimeLeft(`${days}d ${hours}h ${minutes}m ${seconds}s`)
   155→   155→      } else {
   156→   156→        setTimeLeft(`${hours}h ${minutes}m ${seconds}s`)
   157→   157→      }
   158→   158→    }
   159→   159→
   160→   160→    calculateTime()
   161→   161→    const interval = setInterval(calculateTime, 1000)
   162→   162→    return () => clearInterval(interval)
   163→   163→  }, [matchTime])
   164→   164→
   165→   165→  return <span className="font-medium text-sm text-foreground">{timeLeft}</span>
   166→   166→}
