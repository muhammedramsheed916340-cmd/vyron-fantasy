     1→'use client';
     2→
     3→import React, { useState, useCallback, useMemo } from 'react';
     4→import {
     5→  Trophy, X, CheckCircle2, Loader2, ChevronRight, ChevronLeft,
     6→  AlertCircle, Users, DollarSign, Filter, RefreshCw,
     7→} from 'lucide-react';
     8→import {
     9→  JCMatch, JCContest, JCTeam, JCJoinItem, JCProgress,
    10→  JCContestFetchResult,
    11→  parseContest, formatCurrency, getContestTypeColor,
    12→  getExistingPlatformTeams, getPlatformContests, normalizeContests,
    13→  buildJoinItems, getJoinKey,
    14→} from '@/lib/join-contest-service';
    15→
    16→// ============ Step Enum ============
    17→type Step = 'matches' | 'teams' | 'contests' | 'joining' | 'result';
    18→
    19→// ============ Contest Error State ============
    20→interface ContestErrorState {
    21→  matchId: string;
    22→  error: string;
    23→  errorType: 'auth' | 'network' | 'parse' | 'invalid_match' | 'api_fail';
    24→}
    25→
    26→// ============ Props ============
    27→interface JoinContestDialogProps {
    28→  open: boolean;
    29→  onClose: () => void;
    30→  matches: JCMatch[];
    31→  fantasyAccounts: Record<string, { authToken: string; mobileNumber: string; my11circleChallenge?: string | null }>;
    32→}
    33→
    34→// ============ Main Dialog ============
    35→export default function JoinContestDialog({
    36→  open, onClose, matches, fantasyAccounts,
    37→}: JoinContestDialogProps) {
    38→  // Step state
    39→  const [step, setStep] = useState<Step>('matches');
    40→  const [platform, setPlatform] = useState<string>('dream11');
    41→
    42→  // Match selection
    43→  const [selectedMatchIds, setSelectedMatchIds] = useState<Set<string | number>>(new Set());
    44→
    45→  // Team selection — EXISTING PLATFORM TEAMS
    46→  const [platformTeams, setPlatformTeams] = useState<JCTeam[]>([]);
    47→  const [selectedTeamIds, setSelectedTeamIds] = useState<Set<string | number>>(new Set());
    48→  const [mixedTeamMode, setMixedTeamMode] = useState(false);
    49→  const [loadingTeams, setLoadingTeams] = useState(false);
    50→  const [teamsError, setTeamsError] = useState<string | null>(null);
    51→  const [teamsTokenExpired, setTeamsTokenExpired] = useState(false);
    52→
    53→  // Contest selection — with proper error states
    54→  const [contestsMap, setContestsMap] = useState<Map<string, JCContest[]>>(new Map());
    55→  const [selectedContestIds, setSelectedContestIds] = useState<Set<string>>(new Set());
    56→  const [loadingContests, setLoadingContests] = useState(false);
    57→  const [contestErrors, setContestErrors] = useState<ContestErrorState[]>([]);
    58→  const [contestTokenExpired, setContestTokenExpired] = useState(false);
    59→
    60→  // Join progress
    61→  const [joinItems, setJoinItems] = useState<JCJoinItem[]>([]);
    62→  const [progress, setProgress] = useState<JCProgress>({ current: 0, total: 0, status: 'idle' });
    63→
    64→  // Result
    65→  const [resultItems, setResultItems] = useState<JCJoinItem[]>([]);
    66→
    67→  // Available platforms from linked accounts
    68→  const availablePlatforms = useMemo(() => {
    69→    const platforms: string[] = [];
    70→    if (fantasyAccounts.dream11?.authToken) platforms.push('dream11');
    71→    if (fantasyAccounts.my11circle?.authToken) platforms.push('my11circle');
    72→    return platforms;
    73→  }, [fantasyAccounts]);
    74→
    75→  const account = fantasyAccounts[platform];
    76→
    77→  // Reset on close
    78→  const handleClose = () => {
    79→    setStep('matches');
    80→    setSelectedMatchIds(new Set());
    81→    setPlatformTeams([]);
    82→    setSelectedTeamIds(new Set());
    83→    setMixedTeamMode(false);
    84→    setLoadingTeams(false);
    85→    setTeamsError(null);
    86→    setTeamsTokenExpired(false);
    87→    setContestsMap(new Map());
    88→    setSelectedContestIds(new Set());
    89→    setLoadingContests(false);
    90→    setContestErrors([]);
    91→    setContestTokenExpired(false);
    92→    setJoinItems([]);
    93→    setProgress({ current: 0, total: 0, status: 'idle' });
    94→    setResultItems([]);
    95→    onClose();
    96→  };
    97→
    98→  // ============ Match Selection ============
    99→  const toggleMatch = (id: string | number) => {
   100→    setSelectedMatchIds(prev => {
   101→      const next = new Set(prev);
   102→      next.has(id) ? next.delete(id) : next.add(id);
   103→      return next;
   104→    });
   105→  };
   106→
   107→  const selectAllMatches = () => {
   108→    setSelectedMatchIds(new Set(matches.map(m => m.id)));
   109→  };
   110→
   111→  // ============ Load Existing Platform Teams ============
   112→  const loadPlatformTeams = useCallback(async (selectedPlatform?: string) => {
   113→    const plat = selectedPlatform || platform;
   114→    const acc = fantasyAccounts[plat];
   115→    if (!acc?.authToken) {
   116→      setTeamsError('No connected account found. Connect your platform account to load existing teams.');
   117→      setPlatformTeams([]);
   118→      return;
   119→    }
   120→
   121→    setLoadingTeams(true);
   122→    setTeamsError(null);
   123→    setTeamsTokenExpired(false);
   124→    setSelectedTeamIds(new Set());
   125→
   126→    const allTeams: JCTeam[] = [];
   127→    let firstError: string | null = null;
   128→    let anyTokenExpired = false;
   129→
   130→    for (const matchId of selectedMatchIds) {
   131→      const result = await getExistingPlatformTeams(plat, matchId, acc.authToken);
   132→      if (result.teams.length > 0) {
   133→        allTeams.push(...result.teams);
   134→      }
   135→      if (result.tokenExpired) {
   136→        anyTokenExpired = true;
   137→        firstError = result.error || 'Session expired';
   138→      } else if (result.error && !firstError) {
   139→        firstError = result.error;
   140→      }
   141→    }
   142→
   143→    setPlatformTeams(allTeams);
   144→    setTeamsError(allTeams.length === 0 ? firstError : null);
   145→    setTeamsTokenExpired(anyTokenExpired);
   146→    setLoadingTeams(false);
   147→  }, [selectedMatchIds, platform, fantasyAccounts]);
   148→
   149→  // ============ Team Selection ============
   150→  const toggleTeam = (id: string | number) => {
   151→    setSelectedTeamIds(prev => {
   152→      const next = new Set(prev);
   153→      next.has(id) ? next.delete(id) : next.add(id);
   154→      return next;
   155→    });
   156→  };
   157→
   158→  const selectAllTeams = () => {
   159→    setSelectedTeamIds(new Set(platformTeams.map(t => t.id)));
   160→  };
   161→
   162→  const deselectAllTeams = () => {
   163→    setSelectedTeamIds(new Set());
   164→  };
   165→
   166→  // ============ Contest Loading — with proper error differentiation ============
   167→  const loadContests = useCallback(async () => {
   168→    if (!account?.authToken) return;
   169→    setLoadingContests(true);
   170→    setContestErrors([]);
   171→    setContestTokenExpired(false);
   172→    setSelectedContestIds(new Set());
   173→
   174→    const newMap = new Map<string, JCContest[]>();
   175→    const errors: ContestErrorState[] = [];
   176→    let anyTokenExpired = false;
   177→
   178→    for (const matchId of selectedMatchIds) {
   179→      const match = matches.find(m => m.id === matchId);
   180→      const platformMatchId = String(matchId);
   181→
   182→      console.log('[JOIN CONTEST] Selected match:', match?.left_team_name, 'vs', match?.right_team_name);
   183→      console.log('[JOIN CONTEST] Platform:', platform);
   184→      console.log('[JOIN CONTEST] Platform Match ID:', platformMatchId);
   185→      console.log('[JOIN CONTEST] Account ID:', account.mobileNumber || 'unknown');
   186→      console.log('[JOIN CONTEST] Sport Index:', match?.sport_index ?? 0);
   187→
   188→      const result: JCContestFetchResult = await getPlatformContests(
   189→        platform,
   190→        platformMatchId,
   191→        account.authToken,
   192→        match?.sport_index ?? 0,
   193→        account.my11circleChallenge || undefined,
   194→      );
   195→
   196→      console.log('[JOIN CONTEST] Contest count:', result.contests.length);
   197→      console.log('[JOIN CONTEST] Error type:', result.errorType);
   198→
   199→      if (result.errorType === 'auth') {
   200→        anyTokenExpired = true;
   201→        errors.push({
   202→          matchId: String(matchId),
   203→          error: result.error || 'Session expired',
   204→          errorType: 'auth',
   205→        });
   206→        // Still set empty array for this match
   207→        newMap.set(String(matchId), []);
   208→      } else if (result.errorType && result.errorType !== 'none') {
   209→        errors.push({
   210→          matchId: String(matchId),
   211→          error: result.error || 'Failed to load contests',
   212→          errorType: result.errorType,
   213→        });
   214→        newMap.set(String(matchId), []);
   215→      } else {
   216→        // Success (possibly 0 contests — that's a valid state)
   217→        newMap.set(String(matchId), result.contests);
   218→      }
   219→
   220→      // Validate: contest.matchId should match our matchId
   221→      for (const contest of result.contests) {
   222→        if (contest.matchId && contest.matchId !== platformMatchId) {
   223→          console.warn('[JOIN CONTEST] Contest matchId mismatch:', contest.matchId, '!==', platformMatchId, 'for contest:', contest.id);
   224→        }
   225→      }
   226→
   227→      console.log('[JOIN CONTEST] Normalized contests for match', platformMatchId, ':', result.contests.length);
   228→    }
   229→
   230→    setContestsMap(newMap);
   231→    setContestErrors(errors);
   232→    setContestTokenExpired(anyTokenExpired);
   233→    setLoadingContests(false);
   234→  }, [selectedMatchIds, platform, account, matches]);
   235→
   236→  const toggleContest = (id: string) => {
   237→    setSelectedContestIds(prev => {
   238→      const next = new Set(prev);
   239→      next.has(id) ? next.delete(id) : next.add(id);
   240→      return next;
   241→    });
   242→  };
   243→
   244→  const selectAllContests = () => {
   245→    const allIds: string[] = [];
   246→    for (const contests of contestsMap.values()) {
   247→      for (const c of contests) {
   248→        if (c.joinAvailable) allIds.push(c.id);
   249→      }
   250→    }
   251→    setSelectedContestIds(new Set(allIds));
   252→  };
   253→
   254→  const deselectAllContests = () => {
   255→    setSelectedContestIds(new Set());
   256→  };
   257→
   258→  // Total contest count across all matches
   259→  const totalContestCount = useMemo(() => {
   260→    let count = 0;
   261→    for (const contests of contestsMap.values()) {
   262→      count += contests.length;
   263→    }
   264→    return count;
   265→  }, [contestsMap]);
   266→
   267→  // Total joinable contest count
   268→  const totalJoinableCount = useMemo(() => {
   269→    let count = 0;
   270→    for (const contests of contestsMap.values()) {
   271→      count += contests.filter(c => c.joinAvailable).length;
   272→    }
   273→    return count;
   274→  }, [contestsMap]);
   275→
   276→  // ============ Join Execution ============
   277→  const executeJoin = useCallback(async () => {
   278→    const selectedMatches = matches.filter(m => selectedMatchIds.has(m.id));
   279→    const selectedContestsForJoin = new Map<string, JCContest[]>();
   280→    for (const [matchId, contests] of contestsMap.entries()) {
   281→      selectedContestsForJoin.set(matchId, contests.filter(c => selectedContestIds.has(c.id)));
   282→    }
   283→
   284→    const items = buildJoinItems(selectedMatches, selectedContestsForJoin, selectedTeamIds, platformTeams, platform);
   285→
   286→    if (items.length === 0) {
   287→      setStep('result');
   288→      return;
   289→    }
   290→
   291→    setJoinItems(items);
   292→    setProgress({ current: 0, total: items.length, status: 'joining' });
   293→    setStep('joining');
   294→
   295→    const updatedItems = [...items];
   296→
   297→    for (let i = 0; i < items.length; i++) {
   298→      const item = items[i];
   299→      setProgress(prev => ({ ...prev, current: i + 1 }));
   300→      setJoinItems(prev => prev.map((it, idx) =>
   301→        idx === i ? { ...it, status: 'processing' } : it
   302→      ));
   303→
   304→      try {
   305→        const match = matches.find(m => String(m.id) === String(item.matchId));
   306→        const res = await fetch('/api/fantasy/join-contest', {
   307→          method: 'POST',
   308→          headers: { 'Content-Type': 'application/json' },
   309→          body: JSON.stringify({
   310→            fantasyApp: item.platform || platform,
   311→            matchId: item.matchId,
   312→            authToken: account?.authToken,
   313→            teamId: item.teamId,    // REAL platform team ID
   314→            contestId: item.contestId,
   315→            sportIndex: match?.sport_index ?? 0,
   316→            challenge: account?.my11circleChallenge || undefined,
   317→          }),
   318→        });
   319→        const data = await res.json();
   320→
   321→        const status = data.status === 'success' ? 'success'
   322→          : data.status === 'already_joined' ? 'already_joined' : 'fail';
   323→
   324→        updatedItems[i] = { ...updatedItems[i], status, message: data.message };
   325→        setJoinItems([...updatedItems]);
   326→      } catch {
   327→        updatedItems[i] = { ...updatedItems[i], status: 'fail', message: 'Network error' };
   328→        setJoinItems([...updatedItems]);
   329→      }
   330→
   331→      // Rate limit between joins
   332→      await new Promise(r => setTimeout(r, 300));
   333→    }
   334→
   335→    setResultItems([...updatedItems]);
   336→    setProgress(prev => ({ ...prev, status: 'done' }));
   337→    setStep('result');
   338→  }, [matches, selectedMatchIds, contestsMap, selectedContestIds, selectedTeamIds, platformTeams, platform, account]);
   339→
   340→  // ============ Result Stats ============
   341→  const resultStats = useMemo(() => {
   342→    const items = resultItems.length > 0 ? resultItems : joinItems;
   343→    return {
   344→      total: items.length,
   345→      success: items.filter(i => i.status === 'success').length,
   346→      alreadyJoined: items.filter(i => i.status === 'already_joined').length,
   347→      fail: items.filter(i => i.status === 'fail').length,
   348→    };
   349→  }, [resultItems, joinItems]);
   350→
   351→  if (!open) return null;
   352→
   353→  // ============ RENDER ============
   354→  return (
   355→    <div className="fixed inset-0 z-50 bg-black/60 flex items-center justify-center p-4" onClick={handleClose}>
   356→      <div className="bg-white rounded-2xl shadow-2xl w-full max-w-2xl max-h-[90vh] overflow-hidden flex flex-col" onClick={e => e.stopPropagation()}>
   357→        {/* Header */}
   358→        <div className="flex items-center justify-between px-6 py-4 border-b bg-gradient-to-r from-green-600 to-emerald-600 text-white">
   359→          <div className="flex items-center gap-3">
   360→            <Trophy className="w-6 h-6" />
   361→            <h2 className="text-lg font-bold">Join Contest</h2>
   362→          </div>
   363→          <button onClick={handleClose} className="p-1 hover:bg-white/20 rounded-lg transition-all">
   364→            <X className="w-5 h-5" />
   365→          </button>
   366→        </div>
   367→
   368→        {/* Step Indicator */}
   369→        <div className="flex items-center gap-1 px-6 py-3 bg-gray-50 border-b">
   370→          {(['matches', 'teams', 'contests', 'joining', 'result'] as Step[]).map((s, i) => {
   371→            const labels = ['Match', 'Teams', 'Contests', 'Join', 'Result'];
   372→            const stepOrder = ['matches', 'teams', 'contests', 'joining', 'result'];
   373→            const isActive = step === s;
   374→            const isDone = stepOrder.indexOf(step) > i;
   375→            return (
   376→              <React.Fragment key={s}>
   377→                {i > 0 && <ChevronRight className="w-3 h-3 text-gray-300" />}
   378→                <div className={`flex items-center gap-1.5 text-xs font-semibold px-2 py-1 rounded-lg transition-all ${
   379→                  isActive ? 'bg-green-100 text-green-700' : isDone ? 'text-green-500' : 'text-gray-400'
   380→                }`}>
   381→                  <div className={`w-5 h-5 rounded-full flex items-center justify-center text-[10px] ${
   382→                    isActive ? 'bg-green-600 text-white' : isDone ? 'bg-green-200 text-green-600' : 'bg-gray-200 text-gray-500'
   383→                  }`}>
   384→                    {isDone ? '✓' : i + 1}
   385→                  </div>
   386→                  {labels[i]}
   387→                </div>
   388→              </React.Fragment>
   389→            );
   390→          })}
   391→        </div>
   392→
   393→        {/* Content */}
   394→        <div className="flex-1 overflow-y-auto p-6">
   395→
   396→          {/* ===== STEP: MATCHES ===== */}
   397→          {step === 'matches' && (
   398→            <div className="space-y-4">
   399→              {/* Platform Selection */}
   400→              <div className="space-y-2">
   401→                <p className="text-xs font-semibold text-gray-500 uppercase tracking-wider">Platform</p>
   402→                <div className="flex gap-2">
   403→                  {availablePlatforms.map(p => (
   404→                    <button key={p} onClick={() => setPlatform(p)}
   405→                      className={`flex-1 rounded-xl border-2 p-3 text-center font-semibold text-sm transition-all ${
   406→                        platform === p ? 'border-green-500 bg-green-50 text-green-700' : 'border-gray-200 bg-white text-gray-700 hover:border-gray-300'
   407→                      }`}>
   408→                      {p === 'dream11' ? 'Dream11' : 'My11Circle'}
   409→                    </button>
   410→                  ))}
   411→                </div>
   412→                {availablePlatforms.length === 0 && (
   413→                  <div className="bg-amber-50 border border-amber-200 rounded-xl p-3 text-sm text-amber-700">
   414→                    ⚠️ No connected account found. Link your platform account first via Transfer.
   415→                  </div>
   416→                )}
   417→              </div>
   418→
   419→              {/* Match List */}
   420→              <div className="space-y-2">
   421→                <div className="flex items-center justify-between">
   422→                  <p className="text-xs font-semibold text-gray-500 uppercase tracking-wider">Select Matches</p>
   423→                  <button onClick={selectAllMatches} className="text-xs font-semibold text-green-600 hover:text-green-700">
   424→                    Select All
   425→                  </button>
   426→                </div>
   427→                {matches.length === 0 ? (
   428→                  <p className="text-sm text-gray-400 text-center py-8">No upcoming matches</p>
   429→                ) : (
   430→                  <div className="space-y-2 max-h-80 overflow-y-auto">
   431→                    {matches.map(match => {
   432→                      const selected = selectedMatchIds.has(match.id);
   433→                      return (
   434→                        <button key={match.id} onClick={() => toggleMatch(match.id)}
   435→                          className={`w-full text-left rounded-xl border-2 p-3 transition-all ${
   436→                            selected ? 'border-green-500 bg-green-50' : 'border-gray-200 bg-white hover:border-gray-300'
   437→                          }`}>
   438→                          <div className="flex items-center gap-3">
   439→                            <div className={`w-5 h-5 rounded border-2 flex items-center justify-center shrink-0 ${
   440→                              selected ? 'border-green-500 bg-green-500' : 'border-gray-300'
   441→                            }`}>
   442→                              {selected && <CheckCircle2 className="w-3.5 h-3.5 text-white" />}
   443→                            </div>
   444→                            <div className="flex-1">
   445→                              <p className="font-semibold text-sm text-gray-900">
   446→                                {match.left_team_name} vs {match.right_team_name}
   447→                              </p>
   448→                              <div className="flex items-center gap-2 text-xs text-gray-500 mt-0.5">
   449→                                <span>{match.match_time}</span>
   450→                                {match.lineup_out === 1 && <span className="text-green-600 font-medium">Lineup Out</span>}
   451→                              </div>
   452→                            </div>
   453→                          </div>
   454→                        </button>
   455→                      );
   456→                    })}
   457→                  </div>
   458→                )}
   459→              </div>
   460→            </div>
   461→          )}
   462→
   463→          {/* ===== STEP: TEAMS — EXISTING PLATFORM TEAMS ===== */}
   464→          {step === 'teams' && (
   465→            <div className="space-y-4">
   466→              {/* Loading State */}
   467→              {loadingTeams && (
   468→                <div className="flex flex-col items-center justify-center py-12">
   469→                  <Loader2 className="w-8 h-8 animate-spin text-green-500" />
   470→                  <p className="text-sm text-gray-500 mt-3">Loading existing teams from {platform === 'dream11' ? 'Dream11' : 'My11Circle'}...</p>
   471→                </div>
   472→              )}
   473→
   474→              {/* Error State */}
   475→              {!loadingTeams && teamsError && (
   476→                <div className="bg-red-50 border border-red-200 rounded-xl p-4">
   477→                  <div className="flex items-center gap-2">
   478→                    <AlertCircle className="w-5 h-5 text-red-500 shrink-0" />
   479→                    <div>
   480→                      <p className="text-sm font-semibold text-red-700">
   481→                        {teamsTokenExpired ? 'SESSION EXPIRED' : 'Error Loading Teams'}
   482→                      </p>
   483→                      <p className="text-xs text-red-600 mt-0.5">{teamsError}</p>
   484→                    </div>
   485→                  </div>
   486→                  {teamsTokenExpired && (
   487→                    <p className="text-xs text-red-500 mt-2">Reconnect your {platform === 'dream11' ? 'Dream11' : 'My11Circle'} account via Transfer.</p>
   488→                  )}
   489→                  <button onClick={() => loadPlatformTeams()} className="mt-2 flex items-center gap-1 text-xs font-semibold text-red-600 hover:text-red-700">
   490→                    <RefreshCw className="w-3 h-3" /> Retry
   491→                  </button>
   492→                </div>
   493→              )}
   494→
   495→              {/* No Account */}
   496→              {!loadingTeams && !teamsError && !account?.authToken && (
   497→                <div className="bg-amber-50 border border-amber-200 rounded-xl p-4 text-center">
   498→                  <p className="text-sm font-semibold text-amber-700">No connected account found</p>
   499→                  <p className="text-xs text-amber-600 mt-1">Connect your {platform === 'dream11' ? 'Dream11' : 'My11Circle'} account to load existing teams.</p>
   500→                </div>
   501→              )}
   502→
   503→              {/* Teams List */}
   504→              {!loadingTeams && account?.authToken && (
   505→                <div className="space-y-2">
   506→                  <div className="flex items-center justify-between">
   507→                    <p className="text-xs font-semibold text-gray-500 uppercase tracking-wider">
   508→                      Existing Teams ({platformTeams.length} available)
   509→                    </p>
   510→                    <div className="flex gap-2">
   511→                      <button onClick={selectAllTeams} disabled={platformTeams.length === 0}
   512→                        className="text-xs font-semibold text-green-600 hover:text-green-700 disabled:opacity-40">
   513→                        Select All
   514→                      </button>
