   112→function decryptField(encrypted: string): string {
   113→  try {
   114→    const bytes = CryptoJS.AES.decrypt(encrypted, ENCRYPTION_KEY);
   115→    const decrypted = bytes.toString(CryptoJS.enc.Utf8);
   116→    if (!decrypted) return encrypted;
   117→    return decrypted;
   118→  } catch {
   119→    return encrypted;
   120→  }
   121→}

   308→// ============ Team Generation Algorithm ============

   310→export interface GeneratedTeam {
   311→  id: number;
   312→  captain: TGPlayer;
   313→  viceCaptain: TGPlayer;
   314→  players: TGPlayer[];
   315→}
   316→
   317→// Cricket team constraints
   318→const CRICKET_TEAM_SIZE = 11;
   319→const MIN_WK = 1;
   320→const MAX_WK = 4;
   321→const MIN_BAT = 3;
   322→const MAX_BAT = 6;
   323→const MIN_AR = 1;
   324→const MAX_AR = 4;
   325→const MIN_BOWL = 3;
   326→const MAX_BOWL = 6;
   327→const MAX_CREDITS = 100;
   328→const MAX_FROM_ONE_TEAM = 7;
   329→
   330→function seededRandom(seed: number): () => number {
   331→  let s = seed;
   332→  return () => {
   333→    s = (s * 16807 + 0) % 2147483647;
   334→    return (s - 1) / 2147483646;
   335→  };
   336→}
   337→
   338→function shuffleArray<T>(arr: T[], rng: () => number): T[] {
   339→  const result = [...arr];
   340→  for (let i = result.length - 1; i > 0; i--) {
   341→    const j = Math.floor(rng() * (i + 1));
   342→    [result[i], result[j]] = [result[j], result[i]];
   343→  }
   344→  return result;
   345→}

   347→export function generateTeams(
   348→  leftPlayers: TGPlayer[],
   349→  rightPlayers: TGPlayer[],
   350→  category: string,
   351→  count: number,
   352→  seed: number = Date.now()
   353→): GeneratedTeam[] {
   354→  const allPlayers = [...leftPlayers, ...rightPlayers];
   355→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
   356→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
   357→
   358→  // Separate by role
   359→  const wk = allPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER);
   360→  const bat = allPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN);
   361→  const ar = allPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER);
   362→  const bowl = allPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER);
   363→
   364→  const teams: GeneratedTeam[] = [];
   365→  const rng = seededRandom(seed);
   366→
   367→  // Category-specific strategies
   368→  const isH2H = category === 'H2H';
   369→  const isSL = category === 'SL';
   370→
   371→  for (let t = 0; t < count; t++) {
   372→    let attempts = 0;
   373→    let team: TGPlayer[] | null = null;
   374→
   375→    while (!team && attempts < 200) {
   376→      attempts++;
   377→      const sr = seededRandom(seed + t * 1000 + attempts);
   378→
   379→      // Determine role counts based on category
   380→      let wkCount, batCount, arCount, bowlCount;
   381→
   382→      if (isH2H) {
   383→        // H2H: More conservative, popular players
   384→        wkCount = MIN_WK + (sr() > 0.5 ? 1 : 0);
   385→        arCount = MIN_AR + (sr() > 0.6 ? 1 : 0);
   386→        batCount = Math.max(MIN_BAT, 5 - wkCount + (sr() > 0.5 ? 1 : 0));
   387→        bowlCount = CRICKET_TEAM_SIZE - wkCount - batCount - arCount;
   388→      } else if (isSL) {
   389→        // SL: Balanced
   390→        wkCount = 1 + (sr() > 0.7 ? 1 : 0);
   391→        arCount = 2 + (sr() > 0.5 ? 1 : 0);
   392→        batCount = 3 + (sr() > 0.5 ? 1 : 0);
   393→        bowlCount = CRICKET_TEAM_SIZE - wkCount - batCount - arCount;
   394→      } else {
   395→        // Mega GL: More differential picks, varied combinations
   396→        const patterns = [
   397→          [1, 3, 2, 5], [1, 4, 1, 5], [1, 3, 3, 4], [1, 4, 2, 4],
   398→          [2, 3, 2, 4], [1, 5, 1, 4], [2, 3, 1, 5], [1, 4, 3, 3],
   399→          [2, 4, 1, 4], [1, 3, 1, 6], [2, 2, 3, 4], [1, 5, 2, 3],
   400→        ];
   401→        const pattern = patterns[Math.floor(sr() * patterns.length)];
   402→        [wkCount, batCount, arCount, bowlCount] = pattern;
   403→      }
   404→
   405→      // Validate counts
   406→      if (wkCount < MIN_WK || wkCount > MAX_WK) continue;
   407→      if (batCount < MIN_BAT || batCount > MAX_BAT) continue;
   408→      if (arCount < MIN_AR || arCount > MAX_AR) continue;
   409→      if (bowlCount < MIN_BOWL || bowlCount > MAX_BOWL) continue;
   410→      if (wkCount + batCount + arCount + bowlCount !== CRICKET_TEAM_SIZE) continue;
   411→
   412→      // Pick players for each role
   413→      const shuffledWK = shuffleArray(wk, sr);
   414→      const shuffledBat = shuffleArray(bat, sr);
   415→      const shuffledAR = shuffleArray(ar, sr);
   416→      const shuffledBowl = shuffleArray(bowl, sr);
   417→
   418→      // For Mega GL, sort by selected_by sometimes to get differential picks
   419→      let selectedWK, selectedBat, selectedAR, selectedBowl;
   420→
   421→      if (!isH2H && sr() > 0.4) {
   422→        // Sort some roles by selection % to mix popular and differential
   423→        const sortFn = (a: TGPlayer, b: TGPlayer) => a.selected_by - b.selected_by;
   424→        selectedWK = shuffleArray([...wk].sort(sortFn), sr).slice(0, wkCount);
   425→        selectedBat = shuffleArray([...bat].sort(sortFn), sr).slice(0, batCount);
   426→        selectedAR = shuffleArray([...ar].sort(sortFn), sr).slice(0, arCount);
   427→        selectedBowl = shuffleArray([...bowl].sort(sortFn), sr).slice(0, bowlCount);
   428→      } else if (isH2H) {
   429→        // H2H: Pick most selected players
   430→        const sortFn = (a: TGPlayer, b: TGPlayer) => b.selected_by - a.selected_by;
   431→        selectedWK = [...wk].sort(sortFn).slice(0, wkCount);
   432→        selectedBat = shuffleArray([...bat].sort(sortFn), sr).slice(0, batCount);
   433→        selectedAR = [...ar].sort(sortFn).slice(0, arCount);
   434→        selectedBowl = shuffleArray([...bowl].sort(sortFn), sr).slice(0, bowlCount);
   435→      } else {
   436→        selectedWK = shuffledWK.slice(0, wkCount);
   437→        selectedBat = shuffledBat.slice(0, batCount);
   438→        selectedAR = shuffledAR.slice(0, arCount);
   439→        selectedBowl = shuffledBowl.slice(0, bowlCount);
   440→      }
   441→
   442→      // Check if we have enough players
   443→      if (selectedWK.length < wkCount || selectedBat.length < batCount ||
   444→          selectedAR.length < arCount || selectedBowl.length < bowlCount) continue;
   445→
   446→      const selected = [...selectedWK, ...selectedBat, ...selectedAR, ...selectedBowl];
   447→
   448→      // Check total credits
   449→      const totalCredits = selected.reduce((sum, p) => sum + p.credits, 0);
   450→      if (totalCredits > MAX_CREDITS) continue;
   451→
   452→      // Check team distribution (max 7 from one team)
   453→      const leftCount = selected.filter(p => p.team_name === leftTeamName).length;
   454→      const rightCount = selected.filter(p => p.team_name === rightTeamName).length;
   455→      if (leftCount > MAX_FROM_ONE_TEAM || rightCount > MAX_FROM_ONE_TEAM) continue;
   456→
   457→      // Ensure at least 1 from each team
   458→      if (leftCount < 1 || rightCount < 1) continue;
   459→
   460→      team = selected;
   461→    }
   462→
   463→    if (team) {
   464→      // Select Captain & Vice-Captain
   465→      // Sort by a combination of points and selection % for C/VC
   466→      const sorted = [...team].sort((a, b) =>
   467→        (b.points * 0.4 + b.selected_by * 0.3 + b.captain_percentage * 0.3) -
   468→        (a.points * 0.4 + a.selected_by * 0.3 + a.captain_percentage * 0.3)
   469→      );
   470→
   471→      const captain = sorted[0];
   472→      const viceCaptain = sorted.length > 1 ? sorted[1] : sorted[0];
   473→
   474→      teams.push({
   475→        id: t + 1,
   476→        captain,
   477→        viceCaptain,
   478→        players: team,
   479→      });
   480→    }
   481→  }
   482→
   483→  return teams;
   484→}

   490→export interface ExtraTeamGenInput {
   491→  fixedPlayers: TGPlayer[];      // Exactly 8 fixed players
   492→  captainOptions: TGPlayer[];    // Exactly 5 captain options
   493→  viceCaptainOptions: TGPlayer[]; // Exactly 5 vice captain options
   494→  leftPlayers: TGPlayer[];       // Full left team player pool
   495→  rightPlayers: TGPlayer[];      // Full right team player pool
   496→  category: string;              // 'Mega GL' | 'SL' | 'H2H'
   497→  count: number;                 // Number of teams to generate
   498→  seed?: number;

   501→export function generateExtraTeams(input: ExtraTeamGenInput): GeneratedTeam[] {
   502→  const {
   503→    fixedPlayers,
   504→    captainOptions,
   505→    viceCaptainOptions,
   506→    leftPlayers,
   507→    rightPlayers,
   508→    category,
   509→    count,
   510→    seed = Date.now(),
   511→  } = input;
   512→
   513→  const allPlayers = [...leftPlayers, ...rightPlayers];
   514→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
   515→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
   516→
   517→  // Pool of players excluding the 8 fixed ones (use pl_id for identity)
   518→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   519→  const remainingPool = allPlayers.filter(p => !fixedIds.has(p.pl_id));
   520→
   521→  // Separate remaining pool by role
   522→  const remWK = remainingPool.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER);
   523→  const remBat = remainingPool.filter(p => p.role === PLAYER_ROLES.BATSMAN);
   524→  const remAR = remainingPool.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER);
   525→  const remBowl = remainingPool.filter(p => p.role === PLAYER_ROLES.BOWLER);
   526→
   527→  // Count fixed players by role
   528→  const fixedWKCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   529→  const fixedBatCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   530→  const fixedARCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   531→  const fixedBowlCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   532→
   533→  const fixedCredits = fixedPlayers.reduce((sum, p) => sum + p.credits, 0);
   534→  const remainingCredits = MAX_CREDITS - fixedCredits;
   535→  const neededPlayers = CRICKET_TEAM_SIZE - fixedPlayers.length; // 3
   536→
   537→  const teams: GeneratedTeam[] = [];
   538→
   539→  // Pre-validate: fixed players must have credits <= 100
   540→  if (fixedCredits > MAX_CREDITS) return teams;
   541→
   542→  // Pre-validate: fixed players must respect max 7 from one team
   543→  const fixedLeftCount = fixedPlayers.filter(p => p.team_name === leftTeamName).length;
   544→  const fixedRightCount = fixedPlayers.filter(p => p.team_name === rightTeamName).length;
   545→  if (fixedLeftCount > MAX_FROM_ONE_TEAM || fixedRightCount > MAX_FROM_ONE_TEAM) return teams;
   546→
   547→  // Generate all valid C/VC combinations from the 5C and 5VC options
   548→  // C must be in the team, VC must be in the team, C ≠ VC
   549→  const cVCCombos: { c: TGPlayer; vc: TGPlayer }[] = [];
   550→  for (const c of captainOptions) {
   551→    for (const vc of viceCaptainOptions) {
   552→      if (c.pl_id !== vc.pl_id) {
   553→        cVCCombos.push({ c, vc });
   554→      }
   555→    }
   556→  }
   557→  // If no valid combos, return empty
   558→  if (cVCCombos.length === 0) return teams;
   559→
   560→  // Category-specific role distribution preferences for remaining players
   561→  const isH2H = category === 'H2H';
   562→  const isSL = category === 'SL';
   563→
   564→  for (let t = 0; t < count; t++) {
   565→    let attempts = 0;
   566→    let team: TGPlayer[] | null = null;
   567→    let teamC: TGPlayer | null = null;
   568→    let teamVC: TGPlayer | null = null;
   569→
   570→    while (!team && attempts < 500) {
   571→      attempts++;
   572→      const sr = seededRandom(seed + t * 1000 + attempts);
   573→
   574→      // We need `neededPlayers` more players (3)
   575→      // Determine target role counts for the full 11-player team
   576→      let targetWK, targetBat, targetAR, targetBowl;
   577→
   578→      if (isH2H) {
   579→        targetWK = MIN_WK + (sr() > 0.5 ? 1 : 0);
   580→        targetAR = MIN_AR + (sr() > 0.6 ? 1 : 0);
   581→        targetBat = Math.max(MIN_BAT, 5 - targetWK + (sr() > 0.5 ? 1 : 0));
   582→        targetBowl = CRICKET_TEAM_SIZE - targetWK - targetBat - targetAR;
   583→      } else if (isSL) {
   584→        targetWK = 1 + (sr() > 0.7 ? 1 : 0);
   585→        targetAR = 2 + (sr() > 0.5 ? 1 : 0);
   586→        targetBat = 3 + (sr() > 0.5 ? 1 : 0);
   587→        targetBowl = CRICKET_TEAM_SIZE - targetWK - targetBat - targetAR;
   588→      } else {
   589→        const patterns = [
   590→          [1, 3, 2, 5], [1, 4, 1, 5], [1, 3, 3, 4], [1, 4, 2, 4],
   591→          [2, 3, 2, 4], [1, 5, 1, 4], [2, 3, 1, 5], [1, 4, 3, 3],
   592→          [2, 4, 1, 4], [1, 3, 1, 6], [2, 2, 3, 4], [1, 5, 2, 3],
   593→        ];
   594→        const pattern = patterns[Math.floor(sr() * patterns.length)];
   595→        [targetWK, targetBat, targetAR, targetBowl] = pattern;
   596→      }
   597→
   598→      // Validate total counts
   599→      if (targetWK < MIN_WK || targetWK > MAX_WK) continue;
   600→      if (targetBat < MIN_BAT || targetBat > MAX_BAT) continue;
   601→      if (targetAR < MIN_AR || targetAR > MAX_AR) continue;
   602→      if (targetBowl < MIN_BOWL || targetBowl > MAX_BOWL) continue;
   603→      if (targetWK + targetBat + targetAR + targetBowl !== CRICKET_TEAM_SIZE) continue;
   604→
   605→      // How many more of each role do we need?
   606→      const needWK = Math.max(0, targetWK - fixedWKCount);
   607→      const needBat = Math.max(0, targetBat - fixedBatCount);
   608→      const needAR = Math.max(0, targetAR - fixedARCount);
   609→      const needBowl = Math.max(0, targetBowl - fixedBowlCount);
   610→
   611→      // Total needed must equal neededPlayers (3)
   612→      if (needWK + needBat + needAR + needBowl !== neededPlayers) continue;
   613→
   614→      // Check if we have enough remaining players for each role
   615→      if (remWK.length < needWK || remBat.length < needBat ||
   616→          remAR.length < needAR || remBowl.length < needBowl) continue;
   617→
   618→      // Pick remaining players for each role
   619→      let pickedWK: TGPlayer[], pickedBat: TGPlayer[], pickedAR: TGPlayer[], pickedBowl: TGPlayer[];
   620→
   621→      if (!isH2H && sr() > 0.4) {
   622→        const sortFn = (a: TGPlayer, b: TGPlayer) => a.selected_by - b.selected_by;
   623→        pickedWK = shuffleArray([...remWK].sort(sortFn), sr).slice(0, needWK);
   624→        pickedBat = shuffleArray([...remBat].sort(sortFn), sr).slice(0, needBat);
   625→        pickedAR = shuffleArray([...remAR].sort(sortFn), sr).slice(0, needAR);
   626→        pickedBowl = shuffleArray([...remBowl].sort(sortFn), sr).slice(0, needBowl);
   627→      } else if (isH2H) {
   628→        const sortFn = (a: TGPlayer, b: TGPlayer) => b.selected_by - a.selected_by;
   629→        pickedWK = [...remWK].sort(sortFn).slice(0, needWK);
   630→        pickedBat = shuffleArray([...remBat].sort(sortFn), sr).slice(0, needBat);
   631→        pickedAR = [...remAR].sort(sortFn).slice(0, needAR);
   632→        pickedBowl = shuffleArray([...remBowl].sort(sortFn), sr).slice(0, needBowl);
   633→      } else {
   634→        pickedWK = shuffleArray(remWK, sr).slice(0, needWK);
   635→        pickedBat = shuffleArray(remBat, sr).slice(0, needBat);
   636→        pickedAR = shuffleArray(remAR, sr).slice(0, needAR);
   637→        pickedBowl = shuffleArray(remBowl, sr).slice(0, needBowl);
   638→      }
   639→
   640→      if (pickedWK.length < needWK || pickedBat.length < needBat ||
   641→          pickedAR.length < needAR || pickedBowl.length < needBowl) continue;
   642→
   643→      const remaining = [...pickedWK, ...pickedBat, ...pickedAR, ...pickedBowl];
   644→      const fullTeam = [...fixedPlayers, ...remaining];
   645→
   646→      // Check no duplicates (shouldn't happen but safety check)
   647→      const uniqueIds = new Set(fullTeam.map(p => p.pl_id));
   648→      if (uniqueIds.size !== fullTeam.length) continue;
   649→
   650→      // Check total credits
   651→      const totalCredits = fullTeam.reduce((sum, p) => sum + p.credits, 0);
   652→      if (totalCredits > MAX_CREDITS) continue;
   653→
   654→      // Check team distribution
   655→      const leftCount = fullTeam.filter(p => p.team_name === leftTeamName).length;
   656→      const rightCount = fullTeam.filter(p => p.team_name === rightTeamName).length;
   657→      if (leftCount > MAX_FROM_ONE_TEAM || rightCount > MAX_FROM_ONE_TEAM) continue;
   658→      if (leftCount < 1 || rightCount < 1) continue;
   659→
   660→      // Select C/VC combination for this team
   661→      // Round-robin through the valid combos, also try to ensure C and VC are in the team
   662→      const comboIndex = (t + attempts) % cVCCombos.length;
   663→      let combo = cVCCombos[comboIndex];
   664→
   665→      // Ensure C and VC are in this team
   666→      const teamPlayerIds = new Set(fullTeam.map(p => p.pl_id));
   667→      if (!teamPlayerIds.has(combo.c.pl_id) || !teamPlayerIds.has(combo.vc.pl_id)) {
   668→        // Try to find a valid combo where both C and VC are in the team
   669→        let foundValidCombo = false;
   670→        for (let ci = 0; ci < cVCCombos.length; ci++) {
   671→          const tryCombo = cVCCombos[(comboIndex + ci) % cVCCombos.length];
   672→          if (teamPlayerIds.has(tryCombo.c.pl_id) && teamPlayerIds.has(tryCombo.vc.pl_id)) {
   673→            combo = tryCombo;
   674→            foundValidCombo = true;
   675→            break;
   676→          }
   677→        }
   678→        if (!foundValidCombo) {
   679→          // Fallback: use top players in team as C/VC
   680→          const sorted = [...fullTeam].sort((a, b) =>
   681→            (b.points * 0.4 + b.selected_by * 0.3 + b.captain_percentage * 0.3) -
   682→            (a.points * 0.4 + a.selected_by * 0.3 + a.captain_percentage * 0.3)
   683→          );
   684→          combo = { c: sorted[0], vc: sorted.length > 1 ? sorted[1] : sorted[0] };
   685→        }
   686→      }
   687→
   688→      team = fullTeam;
   689→      teamC = combo.c;
   690→      teamVC = combo.vc;
   691→    }
   692→
   693→    if (team && teamC && teamVC) {
   694→      teams.push({
   695→        id: t + 1,
   696→        captain: teamC,
   697→        viceCaptain: teamVC,
   698→        players: team,
   699→      });
   700→    }
   701→  }
   702→
   703→  return teams;
   704→}

   706→// ============ Auto Select for Extra Team Generation ============
   707→// Uses real player data (selection %, captain %, points, credits, role, team balance)
   708→// to intelligently pick 8 FIX, 5 C, and 5 VC
   709→
   710→export interface AutoSelectResult {
   711→  fixedPlayers: TGPlayer[];
   712→  captainOptions: TGPlayer[];
   713→  viceCaptainOptions: TGPlayer[];
   714→}
   715→
   716→// Player scoring for auto-selection: higher = better pick
   717→function playerScore(p: TGPlayer): number {
   718→  return (
   719→    p.selected_by * 0.35 +       // selection % - most important for GL
   720→    p.captain_percentage * 0.25 + // captain % - indicates trust
   721→    p.points * 0.2 +             // points - performance indicator
   722→    p.vice_captain_percentage * 0.1 + // VC %
   723→    (p.playing === 1 ? 10 : 0) + // confirmed playing bonus
   724→    (p.role === PLAYER_ROLES.ALL_ROUNDER ? 3 : 0) + // AR bonus (flexible)
   725→    (p.role === PLAYER_ROLES.WICKET_KEEPER ? 1 : 0) // WK slight bonus
   726→  );
   727→}
   728→
   729→export function autoSelectExtraPlayers(
   730→  leftPlayers: TGPlayer[],
   731→  rightPlayers: TGPlayer[],
   732→  avoidPlayerIds: Set<number> = new Set(),
   733→): AutoSelectResult {
   734→  const allPlayers = [...leftPlayers, ...rightPlayers];
   735→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
   736→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
   737→
   738→  // Filter out avoided players and unconfirmed players when lineup is out
   739→  const eligible = allPlayers.filter(p => {
   740→    if (avoidPlayerIds.has(p.pl_id)) return false;
   741→    if (p.playing === 0) {
   742→      const hasConfirmed = allPlayers.some(ap => ap.playing === 1);
   743→      if (hasConfirmed) return false;
   744→    }
   745→    return true;
   746→  });
   747→
   748→  const eligibleLeft = eligible.filter(p => p.team_name === leftTeamName);
   749→  const eligibleRight = eligible.filter(p => p.team_name === rightTeamName);
   750→
   751→  // Sort by score descending
   752→  const sortByScore = (a: TGPlayer, b: TGPlayer) => playerScore(b) - playerScore(a);
   753→  const sortedLeft = [...eligibleLeft].sort(sortByScore);
   754→  const sortedRight = [...eligibleRight].sort(sortByScore);
   755→
   756→  // === SELECT 8 FIX PLAYERS ===
   757→  // Strategy: balanced 4-5 from each team, role-diverse
   758→  const fixedPlayers: TGPlayer[] = [];
   759→  const fixedIds = new Set<number>();
   760→
   761→  const targetLeft = Math.min(4, sortedLeft.length);
   762→  const targetRight = Math.min(4, sortedRight.length);
   763→
   764→  // Pick diverse players ensuring role coverage
   765→  const pickDiverse = (pool: TGPlayer[], count: number, existing: TGPlayer[]): TGPlayer[] => {
   766→    const picked: TGPlayer[] = [];
   767→    const usedIds = new Set(existing.map(p => p.pl_id));
   768→    const existingRoles = {
   769→      wk: existing.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length,
   770→      bat: existing.filter(p => p.role === PLAYER_ROLES.BATSMAN).length,
   771→      ar: existing.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length,
   772→      bowl: existing.filter(p => p.role === PLAYER_ROLES.BOWLER).length,
   773→    };
   774→
   775→    // First pass: fill role gaps
   776→    const roleNeeds = [
   777→      { role: PLAYER_ROLES.WICKET_KEEPER, need: Math.max(0, 1 - existingRoles.wk), pool: pool.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER && !usedIds.has(p.pl_id)) },
   778→      { role: PLAYER_ROLES.ALL_ROUNDER, need: Math.max(0, 1 - existingRoles.ar), pool: pool.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER && !usedIds.has(p.pl_id)) },
   779→      { role: PLAYER_ROLES.BATSMAN, need: Math.max(0, 2 - existingRoles.bat), pool: pool.filter(p => p.role === PLAYER_ROLES.BATSMAN && !usedIds.has(p.pl_id)) },
   780→      { role: PLAYER_ROLES.BOWLER, need: Math.max(0, 2 - existingRoles.bowl), pool: pool.filter(p => p.role === PLAYER_ROLES.BOWLER && !usedIds.has(p.pl_id)) },
   781→    ];
   782→
   783→    for (const rn of roleNeeds) {
   784→      for (let i = 0; i < rn.need && picked.length < count; i++) {
   785→        const candidate = rn.pool.find(p => !picked.some(pp => pp.pl_id === p.pl_id));
   786→        if (candidate) picked.push(candidate);
   787→      }
   788→    }
   789→
   790→    // Second pass: fill remaining with top-scored players
   791→    const remaining = pool.filter(p => !usedIds.has(p.pl_id) && !picked.some(pp => pp.pl_id === p.pl_id));
   792→    for (const p of remaining) {
   793→      if (picked.length >= count) break;
   794→      picked.push(p);
   795→    }
   796→
   797→    return picked;
   798→  };
   799→
   800→  const leftPicks = pickDiverse(sortedLeft, targetLeft, []);
   801→  for (const p of leftPicks) { fixedPlayers.push(p); fixedIds.add(p.pl_id); }
   802→
   803→  const rightPicks = pickDiverse(sortedRight, targetRight, fixedPlayers);
   804→  for (const p of rightPicks) { fixedPlayers.push(p); fixedIds.add(p.pl_id); }
   805→
   806→  // Fill to 8 if needed
   807→  if (fixedPlayers.length < 8) {
   808→    const remaining = eligible.filter(p => !fixedIds.has(p.pl_id)).sort(sortByScore);
   809→    for (const p of remaining) {
   810→      if (fixedPlayers.length >= 8) break;
   811→      fixedPlayers.push(p); fixedIds.add(p.pl_id);
   812→    }
   813→  }
   814→
   815→  // Trim to 8 if over
   816→  while (fixedPlayers.length > 8) {
   817→    let minIdx = 0; let minScore = playerScore(fixedPlayers[0]);
   818→    for (let i = 1; i < fixedPlayers.length; i++) {
   819→      const s = playerScore(fixedPlayers[i]);
   820→      if (s < minScore) { minScore = s; minIdx = i; }
   821→    }
   822→    const removed = fixedPlayers.splice(minIdx, 1)[0];
   823→    fixedIds.delete(removed.pl_id);
   824→  }
   825→
   826→  // === SELECT 5 CAPTAIN OPTIONS ===
   827→  const cCandidates = eligible
   828→    .filter(p => !avoidPlayerIds.has(p.pl_id))
   829→    .sort((a, b) => {
   830→      const scoreA = a.captain_percentage * 0.4 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
   831→      const scoreB = b.captain_percentage * 0.4 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
   832→      return scoreB - scoreA;
   833→    })
   834→    .slice(0, 5);
   835→
   836→  // === SELECT 5 VICE CAPTAIN OPTIONS ===
   837→  const vcCandidates: TGPlayer[] = [];
   838→  const vcPool = eligible
   839→    .filter(p => !avoidPlayerIds.has(p.pl_id))
   840→    .sort((a, b) => {
   841→      const scoreA = a.vice_captain_percentage * 0.35 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
   842→      const scoreB = b.vice_captain_percentage * 0.35 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
   843→      return scoreB - scoreA;
   844→    });
   845→
   846→  for (const p of vcPool) {
   847→    if (vcCandidates.length >= 5) break;
   848→    vcCandidates.push(p);
   849→  }
   850→
   851→  return {
   852→    fixedPlayers: fixedPlayers.slice(0, 8),
   853→    captainOptions: cCandidates.slice(0, 5),
   854→    viceCaptainOptions: vcCandidates.slice(0, 5),
   855→  };
   856→}

   858→// Auto-replace a removed player in a list with the next best eligible player
   859→export function autoReplacePlayer(
   860→  removedPlayer: TGPlayer,
   861→  currentList: TGPlayer[],
   862→  allPlayers: TGPlayer[],
   863→  avoidPlayerIds: Set<number>,
   864→  scoringFn: (p: TGPlayer) => number = playerScore,
   865→): TGPlayer[] {
   866→  const currentIds = new Set(currentList.map(p => p.pl_id));
   867→  const eligible = allPlayers.filter(p =>
   868→    !currentIds.has(p.pl_id) && !avoidPlayerIds.has(p.pl_id) && p.pl_id !== removedPlayer.pl_id
   869→  );
   870→  const sorted = eligible.sort((a, b) => scoringFn(b) - scoringFn(a));
   871→  const replacement = sorted[0];
   872→  if (replacement) return [...currentList, replacement];
   873→  return currentList;
   874→}
