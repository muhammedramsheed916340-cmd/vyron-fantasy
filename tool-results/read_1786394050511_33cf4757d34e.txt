   133→// Countdown timer component
   134→function CountdownTimer({ matchTime }: { matchTime: string }) {
   135→  const [timeLeft, setTimeLeft] = useState('')
   136→
   137→  useEffect(() => {
   138→    const calculateTime = () => {
   139→      const now = new Date().getTime()
   140→      const matchDate = new Date(matchTime).getTime()
   141→      const diff = matchDate - now
   142→
   143→      if (diff <= 0) {
   144→        setTimeLeft('Started')
   145→        return
   146→      }
   147→
   148→      const days = Math.floor(diff / (1000 * 60 * 60 * 24))
   149→      const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60))
   150→      const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60))
   151→      const seconds = Math.floor((diff % (1000 * 60)) / 1000)
   152→
   153→      if (days > 0) {
   154→        setTimeLeft(`${days}d ${hours}h ${minutes}m ${seconds}s`)
   155→      } else {
   156→        setTimeLeft(`${hours}h ${minutes}m ${seconds}s`)
   157→      }
   158→    }
   159→
   160→    calculateTime()
   161→    const interval = setInterval(calculateTime, 1000)
   162→    return () => clearInterval(interval)
   163→  }, [matchTime])
   164→
   165→  return <span className="font-medium text-sm text-foreground">{timeLeft}</span>
   166→}

   299→function GeneratedTeamCard({ team, leftTeamName, rightTeamName, teamIndex }: {
   300→  team: GeneratedTeam
   301→  leftTeamName: string
   302→  rightTeamName: string
   303→  teamIndex: number
   304→}) {
   305→  const [expanded, setExpanded] = useState(false)
   306→  const leftPlayers = team.players.filter(p => p.team_name === leftTeamName)

   308→  const totalCredits = team.players.reduce((sum, p) => sum + p.credits, 0)
   309→
   310→  return (
   311→    <div className="bg-white rounded-lg border border-gray-200 shadow-sm overflow-hidden mb-2">
   312→      {/* Header */}

   318→          <span className="text-xs font-bold text-[#5b4b8a]">Team {teamIndex}</span>
   319→          <Badge className="bg-[#00bfa5] text-white text-[9px] border-0 px-1.5">
   320→            C: {team.captain.name}
   321→          </Badge>
   322→          <Badge className="bg-[#5e35b1] text-white text-[9px] border-0 px-1.5">

   326→        <div className="flex items-center gap-2">

   329→          {expanded ? <ChevronUp className="w-3.5 h-3.5 text-gray-400" /> : <ChevronDown className="w-3.5 h-3.5 text-gray-400" />}

   331→      </div>
   332→
   333→      {/* Expanded content */}
   334→      {expanded && (
   335→        <div className="border-t border-gray-100">
   336→          {/* Players by team */}

   338→            <div className="p-2">

   342→                  key={p.pl_id}

   352→                <PlayerRow
   353→                  key={p.pl_id}
   354→                  player={p}
   355→                  isCaptain={p.pl_id === team.captain.pl_id}
   356→                  isViceCaptain={p.pl_id === team.viceCaptain.pl_id}

   358→              ))}
   359→            </div>
   360→          </div>
