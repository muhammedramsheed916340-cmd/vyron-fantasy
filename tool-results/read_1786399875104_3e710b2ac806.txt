   132→// Countdown timer component
   133→function CountdownTimer({ matchTime }: { matchTime: string }) {
   134→  const [timeLeft, setTimeLeft] = useState('')
   135→
   136→  useEffect(() => {
   137→    const calculateTime = () => {
   138→      const now = new Date().getTime()
   139→      const matchDate = new Date(matchTime).getTime()
   140→      const diff = matchDate - now
   141→
   142→      if (diff <= 0) {
   143→        setTimeLeft('Started')
   144→        return
   145→      }
   146→
   147→      const days = Math.floor(diff / (1000 * 60 * 60 * 24))
   148→      const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60))
   149→      const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60))
   150→      const seconds = Math.floor((diff % (1000 * 60)) / 1000)
   151→
   152→      if (days > 0) {
   153→        setTimeLeft(`${days}d ${hours}h ${minutes}m ${seconds}s`)
   154→      } else {
   155→        setTimeLeft(`${hours}h ${minutes}m ${seconds}s`)
   156→      }
   157→    }
   158→
   159→    calculateTime()
   160→    const interval = setInterval(calculateTime, 1000)
   161→    return () => clearInterval(interval)
   162→  }, [matchTime])
   163→
   164→  return <span className="font-medium text-sm text-foreground">{timeLeft}</span>
   165→}

   297→// Generated Team Card
   298→function GeneratedTeamCard({ team, leftTeamName, rightTeamName, teamIndex }: {
   299→  team: GeneratedTeam
   300→  leftTeamName: string
   301→  rightTeamName: string
   302→  teamIndex: number
   303→}) {
   304→  const [expanded, setExpanded] = useState(false)
   305→  const leftPlayers = team.players.filter(p => p.team_name === leftTeamName)
   306→  const rightPlayers = team.players.filter(p => p.team_name === rightTeamName)
   307→  const totalCredits = team.players.reduce((sum, p) => sum + p.credits, 0)
   308→
   309→  return (
   310→    <div className="bg-white rounded-lg border border-gray-200 shadow-sm overflow-hidden mb-2">
   311→      {/* Header */}
   312→      <div
   313→        className="flex items-center justify-between p-2.5 cursor-pointer hover:bg-gray-50"
   314→        onClick={() => setExpanded(!expanded)}
   315→      >
   316→        <div className="flex items-center gap-2">
   317→          <span className="text-xs font-bold text-[#6C63FF]">Team {teamIndex}</span>
   318→          <Badge className="bg-[#00bfa5] text-white text-[9px] border-0 px-1.5">
   319→            C: {team.captain.name}
   320→          </Badge>
   321→          <Badge className="bg-[#00D4AA] text-white text-[9px] border-0 px-1.5">
   322→            VC: {team.viceCaptain.name}
   323→          </Badge>
   324→        </div>
   325→        <div className="flex items-center gap-2">
   326→          <span className="text-[10px] text-gray-500">{totalCredits.toFixed(1)} cr</span>
   327→          <span className="text-[10px] text-gray-500">{leftPlayers.length}-{rightPlayers.length}</span>
   328→          {expanded ? <ChevronUp className="w-3.5 h-3.5 text-gray-400" /> : <ChevronDown className="w-3.5 h-3.5 text-gray-400" />}
   329→        </div>
   330→      </div>
   331→
   332→      {/* Expanded content */}
   333→      {expanded && (
   334→        <div className="border-t border-gray-100">
   335→          {/* Players by team */}
   336→          <div className="grid grid-cols-2 divide-x divide-gray-100">
   337→            <div className="p-2">
   338→              <p className="text-[10px] font-bold text-[#6C63FF] mb-1">{leftTeamName} ({leftPlayers.length})</p>
   339→              {leftPlayers.map(p => (
   340→                <PlayerRow
   341→                  key={p.pl_id}
   342→                  player={p}
   343→                  isCaptain={p.pl_id === team.captain.pl_id}
   344→                  isViceCaptain={p.pl_id === team.viceCaptain.pl_id}
   345→                />
   346→              ))}
   347→            </div>
   348→            <div className="p-2">
   349→              <p className="text-[10px] font-bold text-[#00D4AA] mb-1">{rightTeamName} ({rightPlayers.length})</p>
   350→              {rightPlayers.map(p => (
   351→                <PlayerRow
   352→                  key={p.pl_id}
   353→                  player={p}
   354→                  isCaptain={p.pl_id === team.captain.pl_id}
   355→                  isViceCaptain={p.pl_id === team.viceCaptain.pl_id}
   356→                />
   357→              ))}
   358→            </div>
   359→          </div>
   360→        </div>
   361→      )}
   362→    </div>
   363→  )
   364→}
