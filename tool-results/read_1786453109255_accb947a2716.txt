   265→   265→/**
   266→   266→ * Normalize Dream11 contest data into our standard format.
   267→   267→ * Handles various response structures from the Dream11 API.
   268→   268→ */
   269→   269→function normalizeDream11Contests(data: any, matchId: string): PlatformContest[] {
   270→   270→  const contests: PlatformContest[] = [];
   271→   271→
   272→   272→  if (!data || typeof data !== 'object') return contests;
   273→   273→
   274→   274→  // Try to find contest arrays in the response
   275→   275→  let contestArrays: any[][] = [];
   276→   276→
   277→   277→  // Direct array
   278→   278→  if (Array.isArray(data)) {
   279→   279→    contestArrays.push(data);
   280→   280→  }
   281→   281→
   282→   282→  // Common field names that might contain contest arrays
   283→   283→  const arrayFields = ['contests', 'contestList', 'contest_list', 'data', 'result', 'items', 'categories'];
   284→   284→  for (const field of arrayFields) {
   285→   285→    if (Array.isArray(data[field])) {
   286→   286→      contestArrays.push(data[field]);
   287→   287→    }
   288→   288→  }
   289→   289→
   290→   290→  // Categories with nested contests
   291→   291→  if (Array.isArray(data.categories)) {
   292→   292→    for (const cat of data.categories) {
   293→   293→      if (cat && typeof cat === 'object') {
   294→   294→        for (const key of ['contests', 'contestList', 'list', 'challengeList', 'challenges']) {
   295→   295→          if (Array.isArray(cat[key])) {
   296→   296→            contestArrays.push(cat[key]);
   297→   297→          }
   298→   298→        }
   299→   299→      }
   300→   300→    }
   301→   301→  }
   302→   302→
   303→   303→  // Heuristic: find any array of objects that look like contests
   304→   304→  if (contestArrays.length === 0) {
   305→   305→    for (const [key, val] of Object.entries(data)) {
   306→   306→      if (Array.isArray(val) && val.length > 0 && typeof val[0] === 'object') {
   307→   307→        const sample = val[0];
   308→   308→        if (sample.contestId || sample.contest_id || sample.id || sample.entryFee || sample.prizePool || sample.challengeId) {
   309→   309→          contestArrays.push(val);
   310→   310→        }
   311→   311→      }
   312→   312→    }
   313→   313→  }
   314→   314→
   315→   315→  // Normalize each contest
   316→   316→  for (const arr of contestArrays) {
   317→   317→    for (const item of arr) {
   318→   318→      if (!item || typeof item !== 'object') continue;
   319→   319→      const contest = parsePlatformContest(item, matchId, 'dream11');
   320→   320→      if (contest.id) contests.push(contest);
   321→   321→    }
   322→   322→  }
   323→   323→
   324→   324→  // Deduplicate by ID
   325→   325→  const seen = new Set<string>();
   326→   326→  return contests.filter(c => {
   327→   327→    if (seen.has(c.id)) return false;
   328→   328→    seen.add(c.id);
   329→   329→    return true;
   330→   330→  });
   331→   331→}

   429→   429→

   455→   455→        platform: 'my11circle',

   466→   466→        _debug: debugInfo,

   508→   508→function normalizeMy11CircleContests(data: any, matchId: string): PlatformContest[] {
   509→   509→  const contests: PlatformContest[] = [];
   510→   510→  if (!data || typeof data !== 'object') return contests;
   511→   511→
   512→   512→  let contestArrays: any[][] = [];
   513→   513→
   514→   514→  if (Array.isArray(data)) {
   515→   515→    contestArrays.push(data);
   516→   516→  }
   517→   517→
   518→   518→  const arrayFields = ['contests', 'contestList', 'contest_list', 'data', 'result', 'items', 'challengeList', 'challenges'];
   519→   519→  for (const field of arrayFields) {
   520→   520→    if (Array.isArray(data[field])) {
   521→   521→      contestArrays.push(data[field]);
   522→   522→    }
   523→   523→  }
   524→   524→
   525→   525→  // Heuristic
   526→   526→  if (contestArrays.length === 0) {
   527→   527→    for (const [key, val] of Object.entries(data)) {
   528→   528→      if (Array.isArray(val) && val.length > 0 && typeof val[0] === 'object') {
   529→   529→        const sample = val[0];
   530→   530→        if (sample.contestId || sample.contest_id || sample.id || sample.entryFee || sample.challengeId) {
   531→   531→          contestArrays.push(val);
   532→   532→        }
   533→   533→      }
   534→   534→    }
   535→   535→  }
   536→   536→
   537→   537→  for (const arr of contestArrays) {
   538→   538→    for (const item of arr) {
   539→   539→      if (!item || typeof item !== 'object') continue;
   540→   540→      const contest = parsePlatformContest(item, matchId, 'my11circle');
   541→   541→      if (contest.id) contests.push(contest);
   542→   542→    }
   543→   543→  }
   544→   544→
   545→   545→  const seen = new Set<string>();
   546→   546→  return contests.filter(c => {
   547→   547→    if (seen.has(c.id)) return false;
   548→   548→    seen.add(c.id);
   549→   549→    return true;
   550→   550→  });
   551→   551→}

   555→   555→/**
   556→   556→ * Parse a single contest from platform response into our standard format.
   557→   557→ * Handles various field naming conventions from Dream11 and My11Circle.
   558→   558→ *
   559→   559→ * IMPORTANT: Preserves the REAL platform contest ID — never generates fake IDs.

   561→   561→function parsePlatformContest(raw: Record<string, unknown>, matchId: string, platform: string): PlatformContest {
   562→   562→  const totalSpots = (raw.totalSpots as number) || (raw.total_spots as number) || (raw.size as number) || (raw.totalSpot as number) || (raw.maxTeam as number) || (raw.maximumSpot as number) || 0;
   563→   563→  const filledSpots = (raw.filledSpots as number) || (raw.filled_spots as number) || (raw.joined as number) || (raw.spot_filled as number) || (raw.totalJoined as number) || (raw.spotFilled as number) || 0;
   564→   564→  // REAL platform contest ID — never fake
   565→   565→  const id = String(raw.id || raw.contestId || raw.contest_id || raw.challengeId || raw.challenge_id || raw._id || '');
   566→   566→
   567→   567→  return {
   568→   568→    id,
   569→   569→    name: String(raw.name || raw.contestName || raw.contest_name || raw.challengeName || raw.challenge_name || raw.title || 'Contest'),
   570→   570→    entryFee: (raw.entryFee as number) || (raw.entry_fee as number) || (raw.entry as number) || (raw.joinAmount as number) || (raw.buyIn as number) || (raw.joiningAmount as number) || 0,
   571→   571→    prizePool: (raw.prizePool as number) || (raw.prize_pool as number) || (raw.prize as number) || (raw.winningAmount as number) || (raw.totalWinnings as number) || (raw.totalPrize as number) || 0,
   572→   572→    totalSpots,
   573→   573→    filledSpots,
   574→   574→    remainingSpots: Math.max(0, totalSpots - filledSpots),
   575→   575→    joinable: totalSpots === 0 || filledSpots < totalSpots,
   576→   576→    matchId,
   577→   577→    platform,
   578→   578→    _raw: raw,
   579→   579→  };
   580→   580→}
