     1→'use client';
     2→
     3→import React, { useState, useCallback, useMemo, useEffect } from 'react';
     4→import {
     5→  Trophy, X, CheckCircle2, Loader2, ChevronRight, ChevronLeft,
     6→  AlertCircle, Users, DollarSign, Filter, RefreshCw, KeyRound,
     7→} from 'lucide-react';
     8→import {
     9→  JCMatch, JCContest, JCTeam, JCJoinItem, JCProgress,
    10→  JCContestFetchResult,
    11→  parseContest, formatCurrency, getContestTypeColor,
    12→  getExistingPlatformTeams, getPlatformContests, normalizeContests,
    13→  buildJoinItems, getJoinKey,
    14→  verifySession, initiateSessionRefresh, completeSessionRefresh,
    15→} from '@/lib/join-contest-service';
    16→
    17→// ============ Step Enum ============
    18→type Step = 'matches' | 'teams' | 'contests' | 'joining' | 'result';
    19→
    20→// ============ Contest Error State ============
    21→interface ContestErrorState {
    22→  matchId: string;
    23→  error: string;
    24→  errorType: 'auth' | 'network' | 'parse' | 'invalid_match' | 'api_fail';
    25→}
    26→
    27→// ============ Props ============
    28→interface JoinContestDialogProps {
    29→  open: boolean;
    30→  onClose: () => void;
    31→  matches: JCMatch[];
    32→  fantasyAccounts: Record<string, { authToken: string; mobileNumber: string; my11circleChallenge?: string | null } | null>;
    33→  /** Callback to update the parent's fantasy account state after a successful token refresh */
    34→  onAccountUpdate?: (platform: string, account: { authToken: string; mobileNumber: string; my11circleChallenge?: string | null; my11circleUserId?: string | null; linkedAt: string }) => void;
    35→}
    36→
    37→// ============ Main Dialog ============
    38→export default function JoinContestDialog({
    39→  open, onClose, matches, fantasyAccounts, onAccountUpdate,
    40→}: JoinContestDialogProps) {
    41→  // Step state
    42→  const [step, setStep] = useState<Step>('matches');
    43→  const [platform, setPlatform] = useState<string>('dream11');
    44→
    45→  // Match selection
    46→  const [selectedMatchIds, setSelectedMatchIds] = useState<Set<string | number>>(new Set());
    47→
    48→  // Team selection — EXISTING PLATFORM TEAMS
    49→  const [platformTeams, setPlatformTeams] = useState<JCTeam[]>([]);
    50→  const [selectedTeamIds, setSelectedTeamIds] = useState<Set<string | number>>(new Set());
    51→  const [mixedTeamMode, setMixedTeamMode] = useState(false);
    52→  const [loadingTeams, setLoadingTeams] = useState(false);
    53→  const [teamsError, setTeamsError] = useState<string | null>(null);
    54→  const [teamsTokenExpired, setTeamsTokenExpired] = useState(false);
    55→
    56→  // Contest selection — with proper error states
    57→  const [contestsMap, setContestsMap] = useState<Map<string, JCContest[]>>(new Map());
    58→  const [selectedContestIds, setSelectedContestIds] = useState<Set<string>>(new Set());
    59→  const [loadingContests, setLoadingContests] = useState(false);
    60→  const [contestErrors, setContestErrors] = useState<ContestErrorState[]>([]);
    61→  const [contestTokenExpired, setContestTokenExpired] = useState(false);
    62→
    63→  // Join progress
    64→  const [joinItems, setJoinItems] = useState<JCJoinItem[]>([]);
    65→  const [progress, setProgress] = useState<JCProgress>({ current: 0, total: 0, status: 'idle' });
    66→
    67→  // Result
    68→  const [resultItems, setResultItems] = useState<JCJoinItem[]>([]);
    69→
    70→  // Session refresh state
    71→  const [sessionRefreshState, setSessionRefreshState] = useState<{
    72→    active: boolean;
    73→    step: 'verifying' | 'sending_otp' | 'entering_otp' | 'verifying_otp' | 'success' | 'failed';
    74→    otpState?: string | null;
    75→    challenge?: string | null;
    76→    reasonCode?: string | null;
    77→    message?: string;
    78→  }>({ active: false, step: 'verifying', message: '' });
    79→  const [refreshOtp, setRefreshOtp] = useState('');
    80→
    81→  // Use the LATEST token — after a successful refresh, use the new token
    82→  const effectiveAccount = useMemo(() => {
    83→    return fantasyAccounts[platform];
    84→  }, [fantasyAccounts, platform]);
    85→
    86→  // Available platforms from linked accounts
    87→  const availablePlatforms = useMemo(() => {
    88→    const platforms: string[] = [];
    89→    if (fantasyAccounts.dream11?.authToken) platforms.push('dream11');
    90→    if (fantasyAccounts.my11circle?.authToken) platforms.push('my11circle');
    91→    return platforms;
    92→  }, [fantasyAccounts]);
    93→
    94→  const account = effectiveAccount;
    95→
    96→  // Reset on close
    97→  const handleClose = () => {
    98→    setStep('matches');
    99→    setSelectedMatchIds(new Set());
   100→    setPlatformTeams([]);
   101→    setSelectedTeamIds(new Set());
   102→    setMixedTeamMode(false);
   103→    setLoadingTeams(false);
   104→    setTeamsError(null);
   105→    setTeamsTokenExpired(false);
   106→    setContestsMap(new Map());
   107→    setSelectedContestIds(new Set());
   108→    setLoadingContests(false);
   109→    setContestErrors([]);
   110→    setContestTokenExpired(false);
   111→    setJoinItems([]);
   112→    setProgress({ current: 0, total: 0, status: 'idle' });
   113→    setResultItems([]);
   114→    setSessionRefreshState({ active: false, step: 'verifying', message: '' });
   115→    setRefreshOtp('');
   116→    onClose();
   117→  };
   118→
   119→  // ============ Session Refresh Handlers ============
   120→
   121→  /**
   122→   * Handle session expiry by initiating a token refresh flow.
   123→   * This uses the existing connected account's mobile number to re-authenticate.
   124→   * The wizard state (selected match, teams, contests) is PRESERVED.
   125→   */
   126→  const handleSessionExpired = useCallback(async () => {
   127→    const acc = effectiveAccount;
   128→    if (!acc?.mobileNumber) {
   129→      console.error('[JOIN CONTEST] Cannot refresh session — no mobile number stored for account');
   130→      setSessionRefreshState({
   131→        active: true,
   132→        step: 'failed',
   133→        message: 'No mobile number stored for this account. Please reconnect manually.',
   134→      });
   135→      return;
   136→    }
   137→
   138→    console.log('[JOIN CONTEST] Session expired — initiating refresh for', platform, acc.mobileNumber.slice(0, 4) + '****');
   139→
   140→    // Step 1: Send OTP
   141→    setSessionRefreshState({ active: true, step: 'sending_otp', message: `Sending OTP to ${acc.mobileNumber}...` });
   142→
   143→    const refreshResult = await initiateSessionRefresh(platform, acc.mobileNumber);
   144→
   145→    if (!refreshResult.success) {
   146→      console.error('[JOIN CONTEST] Session refresh OTP send failed:', refreshResult.message);
   147→      setSessionRefreshState({
   148→        active: true,
   149→        step: 'failed',
   150→        message: refreshResult.message || 'Failed to send OTP. Please try again.',
   151→      });
   152→      return;
   153→    }
   154→
   155→    // Step 2: Wait for user to enter OTP
   156→    console.log('[JOIN CONTEST] OTP sent — waiting for user to enter OTP');
   157→    setSessionRefreshState({
   158→      active: true,
   159→      step: 'entering_otp',
   160→      otpState: refreshResult.state,
   161→      challenge: refreshResult.challenge,
   162→      reasonCode: refreshResult.reasonCode,
   163→      message: `OTP sent to ${acc.mobileNumber}. Enter the OTP to refresh your session.`,
   164→    });
   165→  }, [platform, effectiveAccount]);
   166→
   167→  /**
   168→   * Complete the session refresh by verifying the OTP entered by the user.
   169→   * On success, update the token in the parent state and retry the failed operation.
   170→   */
   171→  const handleCompleteRefresh = useCallback(async () => {
   172→    if (!refreshOtp || refreshOtp.length < 4) return;
   173→
   174→    const acc = effectiveAccount;
   175→    if (!acc?.mobileNumber) return;
   176→
   177→    console.log('[JOIN CONTEST] Verifying refresh OTP...');
   178→    setSessionRefreshState(prev => ({ ...prev, step: 'verifying_otp', message: 'Verifying OTP...' }));
   179→
   180→    const result = await completeSessionRefresh(
   181→      platform,
   182→      acc.mobileNumber,
   183→      refreshOtp,
   184→      sessionRefreshState.otpState,
   185→      sessionRefreshState.challenge,
   186→      sessionRefreshState.reasonCode,
   187→    );
   188→
   189→    if (!result.success || !result.token) {
   190→      console.error('[JOIN CONTEST] Session refresh OTP verification failed:', result.message);
   191→      setSessionRefreshState(prev => ({
   192→        ...prev,
   193→        step: 'failed',
   194→        message: result.message || 'OTP verification failed. Please try again.',
   195→      }));
   196→      return;
   197→    }
   198→
   199→    console.log('[JOIN CONTEST] Session refresh SUCCESS — new token obtained, length:', result.token.length);
   200→
   201→    // Update the parent's fantasy account state with the new token
   202→    if (onAccountUpdate) {
   203→      onAccountUpdate(platform, {
   204→        authToken: result.token,
   205→        mobileNumber: acc.mobileNumber,
   206→        my11circleChallenge: result.my11circleChallenge || acc.my11circleChallenge,
   207→        my11circleUserId: result.my11circleUserId || null,
   208→        linkedAt: new Date().toISOString(),
   209→      });
   210→    }
   211→
   212→    // Mark refresh as successful
   213→    setSessionRefreshState({
   214→      active: true,
   215→      step: 'success',
   216→      message: 'Session refreshed successfully!',
   217→    });
   218→
   219→    // Clear token expired flags
   220→    setTeamsTokenExpired(false);
   221→    setContestTokenExpired(false);
   222→    setTeamsError(null);
   223→    setContestErrors([]);
   224→
   225→    // After a brief delay, auto-retry the failed operation
   226→    setTimeout(() => {
   227→      setSessionRefreshState({ active: false, step: 'verifying', message: '' });
   228→      setRefreshOtp('');
   229→
   230→      // Retry based on current step — preserve all selections
   231→      if (step === 'teams') {
   232→        loadPlatformTeams();
   233→      } else if (step === 'contests') {
   234→        loadContests();
   235→      }
   236→    }, 1500);
   237→  }, [refreshOtp, platform, effectiveAccount, sessionRefreshState, onAccountUpdate, step]);
   238→
   239→  // ============ Match Selection ============
   240→  const toggleMatch = (id: string | number) => {
   241→    setSelectedMatchIds(prev => {
   242→      const next = new Set(prev);
   243→      next.has(id) ? next.delete(id) : next.add(id);
   244→      return next;
   245→    });
   246→  };
   247→
   248→  const selectAllMatches = () => {
   249→    setSelectedMatchIds(new Set(matches.map(m => m.id)));
   250→  };
   251→
   252→  // ============ Load Existing Platform Teams ============
   253→  const loadPlatformTeams = useCallback(async (selectedPlatform?: string) => {
   254→    const plat = selectedPlatform || platform;
   255→    const acc = fantasyAccounts[plat];
   256→    if (!acc?.authToken) {
   257→      setTeamsError('No connected account found. Connect your platform account to load existing teams.');
   258→      setPlatformTeams([]);
   259→      return;
   260→    }
   261→
   262→    console.log('[JOIN CONTEST] Connected account detected — Platform:', plat, 'Mobile:', acc.mobileNumber?.slice(0, 4) + '****');
   263→    setLoadingTeams(true);
   264→    setTeamsError(null);
   265→    setTeamsTokenExpired(false);
   266→    // Preserve team selections on retry (only clear on first load)
   267→    if (platformTeams.length === 0) {
   268→      setSelectedTeamIds(new Set());
   269→    }
   270→
   271→    const allTeams: JCTeam[] = [];
   272→    let firstError: string | null = null;
   273→    let anyTokenExpired = false;
   274→
   275→    for (const matchId of selectedMatchIds) {
   276→      const result = await getExistingPlatformTeams(plat, matchId, acc.authToken);
   277→      if (result.teams.length > 0) {
   278→        allTeams.push(...result.teams);
   279→      }
   280→      if (result.tokenExpired) {
   281→        anyTokenExpired = true;
   282→        firstError = result.error || 'Session expired';
   283→      } else if (result.error && !firstError) {
   284→        firstError = result.error;
   285→      }
   286→    }
   287→
   288→    setPlatformTeams(allTeams);
   289→    setTeamsError(allTeams.length === 0 ? firstError : null);
   290→    setTeamsTokenExpired(anyTokenExpired);
   291→    setLoadingTeams(false);
   292→
   293→    // If session expired and we have a connected account, auto-trigger refresh
   294→    if (anyTokenExpired && acc.mobileNumber) {
   295→      console.log('[JOIN CONTEST] Token expired — auto-triggering session refresh');
   296→      handleSessionExpired();
   297→    }
   298→  }, [selectedMatchIds, platform, fantasyAccounts, platformTeams.length, handleSessionExpired]);
   299→
   300→  // ============ Team Selection ============
   301→  const toggleTeam = (id: string | number) => {
   302→    setSelectedTeamIds(prev => {
   303→      const next = new Set(prev);
   304→      next.has(id) ? next.delete(id) : next.add(id);
   305→      return next;
   306→    });
   307→  };
   308→
   309→  const selectAllTeams = () => {
   310→    setSelectedTeamIds(new Set(platformTeams.map(t => t.id)));
   311→  };
   312→
   313→  const deselectAllTeams = () => {
   314→    setSelectedTeamIds(new Set());
   315→  };
   316→
   317→  // ============ Contest Loading — with proper error differentiation ============
   318→  const loadContests = useCallback(async () => {
   319→    if (!account?.authToken) return;
   320→    setLoadingContests(true);
   321→    setContestErrors([]);
   322→    setContestTokenExpired(false);
   323→    // Preserve contest selections on retry (only clear on first load)
   324→    if (contestsMap.size === 0) {
   325→      setSelectedContestIds(new Set());
   326→    }
   327→
   328→    console.log('[JOIN CONTEST] Connected account detected — Platform:', platform, 'Mobile:', account.mobileNumber?.slice(0, 4) + '****');
   329→
   330→    const newMap = new Map<string, JCContest[]>();
   331→    const errors: ContestErrorState[] = [];
   332→    let anyTokenExpired = false;
   333→
   334→    for (const matchId of selectedMatchIds) {
   335→      const match = matches.find(m => m.id === matchId);
   336→
   337→      // Use the match ID as the platform match ID.
   338→      // The TG API returns numeric match IDs (e.g., 113672) that are compatible
   339→      // with Dream11/My11Circle platform APIs.
   340→      // CRITICAL: This must be a numeric ID, NOT a display name like "MO vs SUL".
   341→      const platformMatchId = matchId;
   342→      const numericMatchId = typeof matchId === 'string' ? parseInt(matchId, 10) : matchId;
   343→
   344→      console.log('[JOIN CONTEST] Selected match:', match?.left_team_name, 'vs', match?.right_team_name);
   345→      console.log('[JOIN CONTEST] Platform:', platform);
   346→      console.log('[JOIN CONTEST] Platform Match ID:', platformMatchId, '(numeric:', numericMatchId, ')');
   347→      console.log('[JOIN CONTEST] Account ID:', account.mobileNumber || 'unknown');
   348→      console.log('[JOIN CONTEST] Sport Index:', match?.sport_index ?? 0);
   349→
   350→      // Validate: matchId must be numeric
   351→      if (isNaN(numericMatchId) || numericMatchId <= 0) {
   352→        console.error('[JOIN CONTEST] INVALID matchId:', matchId, '— not a valid numeric ID. This should be a platform match ID like 113672, not a display name.');
   353→        errors.push({
   354→          matchId: String(matchId),
   355→          error: `Invalid match ID "${matchId}". Must be a numeric platform match ID.`,
   356→          errorType: 'invalid_match',
   357→        });
   358→        newMap.set(String(matchId), []);
   359→        continue;
   360→      }
   361→
   362→      const result: JCContestFetchResult = await getPlatformContests(
   363→        platform,
   364→        platformMatchId,
   365→        account.authToken,
   366→        match?.sport_index ?? 0,
   367→        account.my11circleChallenge || undefined,
   368→      );
   369→
   370→      console.log('[JOIN CONTEST] Contest count:', result.contests.length);
   371→      console.log('[JOIN CONTEST] Error type:', result.errorType);
   372→
   373→      if (result.errorType === 'auth') {
   374→        anyTokenExpired = true;
   375→        errors.push({
   376→          matchId: String(matchId),
   377→          error: result.error || 'Session expired',
   378→          errorType: 'auth',
   379→        });
   380→        // Still set empty array for this match
   381→        newMap.set(String(matchId), []);
   382→      } else if (result.errorType && result.errorType !== 'none') {
   383→        errors.push({
   384→          matchId: String(matchId),
   385→          error: result.error || 'Failed to load contests',
   386→          errorType: result.errorType,
   387→        });
   388→        newMap.set(String(matchId), []);
   389→      } else {
   390→        // Success (possibly 0 contests — that's a valid state)
   391→        newMap.set(String(matchId), result.contests);
   392→      }
   393→
   394→      // Validate: contest.matchId should match our matchId
   395→      for (const contest of result.contests) {
   396→        if (contest.matchId && contest.matchId !== platformMatchId) {
   397→          console.warn('[JOIN CONTEST] Contest matchId mismatch:', contest.matchId, '!==', platformMatchId, 'for contest:', contest.id);
   398→        }
   399→      }
   400→
   401→      console.log('[JOIN CONTEST] Normalized contests for match', platformMatchId, ':', result.contests.length);
   402→    }
   403→
   404→    setContestsMap(newMap);
   405→    setContestErrors(errors);
   406→    setContestTokenExpired(anyTokenExpired);
   407→    setLoadingContests(false);
   408→
   409→    // If session expired and we have a connected account, auto-trigger refresh
   410→    if (anyTokenExpired && account.mobileNumber) {
   411→      console.log('[JOIN CONTEST] Token expired — auto-triggering session refresh');
   412→      handleSessionExpired();
   413→    }
   414→  }, [selectedMatchIds, platform, account, matches, contestsMap.size, handleSessionExpired]);
   415→
   416→  const toggleContest = (id: string) => {
   417→    setSelectedContestIds(prev => {
   418→      const next = new Set(prev);
   419→      next.has(id) ? next.delete(id) : next.add(id);
   420→      return next;
   421→    });
   422→  };
   423→
   424→  const selectAllContests = () => {
   425→    const allIds: string[] = [];
   426→    for (const contests of contestsMap.values()) {
   427→      for (const c of contests) {
   428→        if (c.joinAvailable) allIds.push(c.id);
   429→      }
   430→    }
   431→    setSelectedContestIds(new Set(allIds));
   432→  };
   433→
   434→  const deselectAllContests = () => {
   435→    setSelectedContestIds(new Set());
   436→  };
   437→
   438→  // Total contest count across all matches
   439→  const totalContestCount = useMemo(() => {
   440→    let count = 0;
   441→    for (const contests of contestsMap.values()) {
   442→      count += contests.length;
   443→    }
   444→    return count;
   445→  }, [contestsMap]);
   446→
   447→  // Total joinable contest count
   448→  const totalJoinableCount = useMemo(() => {
   449→    let count = 0;
   450→    for (const contests of contestsMap.values()) {
   451→      count += contests.filter(c => c.joinAvailable).length;
   452→    }
   453→    return count;
   454→  }, [contestsMap]);
   455→
   456→  // ============ Join Execution ============
   457→  const executeJoin = useCallback(async () => {
   458→    const selectedMatches = matches.filter(m => selectedMatchIds.has(m.id));
   459→    const selectedContestsForJoin = new Map<string, JCContest[]>();
   460→    for (const [matchId, contests] of contestsMap.entries()) {
   461→      selectedContestsForJoin.set(matchId, contests.filter(c => selectedContestIds.has(c.id)));
   462→    }
   463→
   464→    const items = buildJoinItems(selectedMatches, selectedContestsForJoin, selectedTeamIds, platformTeams, platform);
   465→
   466→    if (items.length === 0) {
   467→      setStep('result');
   468→      return;
   469→    }
   470→
   471→    setJoinItems(items);
   472→    setProgress({ current: 0, total: items.length, status: 'joining' });
   473→    setStep('joining');
   474→
   475→    const updatedItems = [...items];
   476→
   477→    for (let i = 0; i < items.length; i++) {
   478→      const item = items[i];
   479→      setProgress(prev => ({ ...prev, current: i + 1 }));
   480→      setJoinItems(prev => prev.map((it, idx) =>
   481→        idx === i ? { ...it, status: 'processing' } : it
   482→      ));
   483→
   484→      try {
   485→        const match = matches.find(m => String(m.id) === String(item.matchId));
   486→        // Ensure matchId is numeric for the join API
   487→        const numericMatchId = typeof item.matchId === 'string' ? parseInt(item.matchId, 10) : item.matchId;
   488→        const res = await fetch('/api/fantasy/join-contest', {
   489→          method: 'POST',
   490→          headers: { 'Content-Type': 'application/json' },
   491→          body: JSON.stringify({
   492→            fantasyApp: item.platform || platform,
   493→            matchId: numericMatchId,      // Numeric platform match ID
   494→            authToken: account?.authToken, // Platform session token from verify-otp
   495→            teamId: item.teamId,           // REAL platform team ID (from list-of-teams)
   496→            contestId: item.contestId,     // REAL platform contest ID
   497→            sportIndex: match?.sport_index ?? 0,
   498→            challenge: account?.my11circleChallenge || undefined,
   499→          }),
   500→        });
   501→        const data = await res.json();
   502→
   503→        const status = data.status === 'success' ? 'success'
   504→          : data.status === 'already_joined' ? 'already_joined' : 'fail';
   505→
   506→        updatedItems[i] = { ...updatedItems[i], status, message: data.message };
   507→        setJoinItems([...updatedItems]);
   508→      } catch {
   509→        updatedItems[i] = { ...updatedItems[i], status: 'fail', message: 'Network error' };
   510→        setJoinItems([...updatedItems]);
   511→      }
   512→
   513→      // Rate limit between joins
   514→      await new Promise(r => setTimeout(r, 300));
   515→    }
   516→
   517→    setResultItems([...updatedItems]);
   518→    setProgress(prev => ({ ...prev, status: 'done' }));
   519→    setStep('result');
   520→  }, [matches, selectedMatchIds, contestsMap, selectedContestIds, selectedTeamIds, platformTeams, platform, account]);
   521→
   522→  // ============ Result Stats ============
   523→  const resultStats = useMemo(() => {
   524→    const items = resultItems.length > 0 ? resultItems : joinItems;
   525→    return {
   526→      total: items.length,
   527→      success: items.filter(i => i.status === 'success').length,
   528→      alreadyJoined: items.filter(i => i.status === 'already_joined').length,
   529→      fail: items.filter(i => i.status === 'fail').length,
   530→    };
   531→  }, [resultItems, joinItems]);
   532→
   533→  if (!open) return null;
   534→
   535→  // ============ RENDER ============
   536→  return (
   537→    <div className="fixed inset-0 z-50 bg-black/60 flex items-center justify-center p-4" onClick={handleClose}>
   538→      <div className="relative bg-white rounded-2xl shadow-2xl w-full max-w-2xl max-h-[90vh] overflow-hidden flex flex-col" onClick={e => e.stopPropagation()}>
   539→        {/* Header */}
   540→        <div className="flex items-center justify-between px-6 py-4 border-b bg-gradient-to-r from-green-600 to-emerald-600 text-white">
   541→          <div className="flex items-center gap-3">
