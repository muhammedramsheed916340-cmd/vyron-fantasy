     6→// ============ Platform Name Normalization ============

     8→/**
     9→ * Normalize platform names to consistent lowercase identifiers.
    10→ * The TG API may return varied casing in fantasy_id_list[].name
    11→ * (e.g., "Dream11", "dream11", "My11Circle", "my11circle").
    12→ * All internal references use lowercase: "dream11", "my11circle".

    14→export function normalizePlatformName(name: string): string {
    15→  return name.toLowerCase().replace(/[\s_-]/g, '');

    18→/**
    19→ * Resolve a player's platform-specific ID from their fantasy_id_list.
    20→ * Uses case-insensitive matching to handle API casing variations.
    21→ */
    22→export function resolvePlatformPlayerId(
    23→  player: TGPlayer,
    24→  platform: string,
    25→): number | null {
    26→  if (!player.fantasy_id_list || player.fantasy_id_list.length === 0) return null;
    27→  const target = normalizePlatformName(platform);
    28→  const found = player.fantasy_id_list.find(
    29→    f => normalizePlatformName(f.name) === target
    30→  );
    31→  return found ? found.id : null;
    32→}

    62→// ============ Lineup Utilities ============
    63→
    64→/**
    65→ * Determine lineup mode based on player data.
    66→ * AFTER LINEUP: enough players have playing === 1 to form valid teams.
    67→ *   Requires at least 11 confirmed total AND at least 4 from each team
    68→ *   (due to max-7-from-one-team constraint: 11 - 7 = 4).
    69→ *   If only a few players are confirmed (partial lineup), we stay in
    70→ *   'before' mode so team generation doesn't fail with too few eligible players.
    71→ * BEFORE LINEUP: lineup not fully confirmed yet — all players eligible.
    72→ *
    73→ * BUG FIX: Previously used players.some(p => p.playing === 1) which
    74→ * switched to 'after' mode when even ONE player was confirmed, causing
    75→ * getEligiblePlayers to filter out all non-confirmed players and leaving
    76→ * too few to form valid 11-player teams → 0 teams generated.
    77→ */
    78→export function getLineupMode(players: TGPlayer[]): 'before' | 'after' {
    79→  const confirmed = players.filter(p => p.playing === 1);
    80→
    81→  // Need at least 11 confirmed players total to form a valid team
    82→  if (confirmed.length < 11) return 'before';
    83→
    84→  // Need at least 4 confirmed from each team
    85→  // (max 7 from one team → need at least 11-7=4 from the other)
    86→  const teamNames = [...new Set(players.map(p => p.team_name))];
    87→  for (const team of teamNames) {
    88→    if (team && confirmed.filter(p => p.team_name === team).length < 4) {
    89→      return 'before';
    90→    }
    91→  }
    92→
    93→  return 'after';
    94→}
    95→
    96→/**
    97→ * Filter players based on lineup mode.
    98→ * BEFORE LINEUP: all players are eligible (no one confirmed OUT yet).
    99→ * AFTER LINEUP: ONLY players with playing === 1 are eligible.
   100→ */
   101→export function getEligiblePlayers(
   102→  allPlayers: TGPlayer[],
   103→  avoidPlayerIds: Set<number> = new Set(),
   104→): TGPlayer[] {
   105→  const mode = getLineupMode(allPlayers);
   106→
   107→  return allPlayers.filter(p => {
   108→    // Avoid players are always excluded
   109→    if (avoidPlayerIds.has(p.pl_id)) return false;
   110→
   111→    if (mode === 'after') {
   112→      // AFTER LINEUP: ONLY confirmed Playing XI players
   113→      return p.playing === 1;
   114→    }
   115→
   116→    // BEFORE LINEUP: all players eligible
   117→    return true;
   118→  });
   119→}

   121→/**
   122→ * Check if a specific player is eligible based on lineup mode.

   124→export function isPlayerEligible(
   125→  player: TGPlayer,
   126→  allPlayers: TGPlayer[],
   127→  avoidPlayerIds: Set<number> = new Set(),
   128→): { eligible: boolean; reason?: string } {
   129→  if (avoidPlayerIds.has(player.pl_id)) {
   130→    return { eligible: false, reason: 'AVOID' };
   131→  }
   132→
   133→  const mode = getLineupMode(allPlayers);
   134→
   135→  if (mode === 'after' && player.playing !== 1) {
   136→    return { eligible: false, reason: 'OUT / NOT IN PLAYING XI' };
   137→  }
   138→
   139→  return { eligible: true };
   140→}

   142→/**
   143→ * Validate a team for lineup eligibility.
   144→ * Returns list of invalid players with reasons.
   145→ */
   146→export function validateTeamForLineup(
   147→  team: GeneratedTeam,
   148→  allPlayers: TGPlayer[],
   149→  avoidPlayerIds: Set<number> = new Set(),
   150→): { valid: boolean; invalidPlayers: { player: TGPlayer; reason: string }[] } {
   151→  const mode = getLineupMode(allPlayers);
   152→  const invalidPlayers: { player: TGPlayer; reason: string }[] = [];
   153→
   154→  for (const player of team.players) {
   155→    if (avoidPlayerIds.has(player.pl_id)) {
   156→      invalidPlayers.push({ player, reason: 'AVOID' });
   157→      continue;
   158→    }
   159→
   160→    if (mode === 'after' && player.playing !== 1) {
   161→      invalidPlayers.push({ player, reason: 'OUT / NOT IN PLAYING XI' });
   162→    }
   163→  }
   164→
   165→  return { valid: invalidPlayers.length === 0, invalidPlayers };
   166→}

   179→/**
   180→ * Get all valid role combinations for an 11-player cricket team.
   181→ */
   182→export function getAllValidCombinations(): RoleCombination[] {
   183→  const combos: RoleCombination[] = [];
   184→  for (let wk = MIN_WK; wk <= MAX_WK; wk++) {
   185→    for (let bat = MIN_BAT; bat <= MAX_BAT; bat++) {
   186→      for (let ar = MIN_AR; ar <= MAX_AR; ar++) {
   187→        const bowl = CRICKET_TEAM_SIZE - wk - bat - ar;
   188→        if (bowl < MIN_BOWL || bowl > MAX_BOWL) continue;
   189→        combos.push({ wk, bat, ar, bowl });
   190→      }
   191→    }
   192→  }
   193→  return combos;
   194→}

   196→/**
   197→ * Check if a role combination is compatible with a set of fixed players.
   198→ */
   199→export function isCombinationCompatibleWithFixed(
   200→  combo: RoleCombination,
   201→  fixedPlayers: TGPlayer[],
   202→): boolean {
   203→  const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   204→  const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   205→  const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   206→  const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   207→
   208→  // Fixed players must not exceed combination totals
   209→  if (fixedWK > combo.wk) return false;
   210→  if (fixedBat > combo.bat) return false;
   211→  if (fixedAR > combo.ar) return false;
   212→  if (fixedBowl > combo.bowl) return false;
   213→
   214→  // Remaining slots must be fillable
   215→  const remainingPlayers = CRICKET_TEAM_SIZE - fixedPlayers.length;
   216→  const neededWK = combo.wk - fixedWK;
   217→  const neededBat = combo.bat - fixedBat;
   218→  const neededAR = combo.ar - fixedAR;
   219→  const neededBowl = combo.bowl - fixedBowl;
   220→
   221→  return (neededWK + neededBat + neededAR + neededBowl) === remainingPlayers;
   222→}

   224→/**
   225→ * Get combinations compatible with fixed players and available player pool.
   226→ */
   227→export function getCompatibleCombinations(
   228→  fixedPlayers: TGPlayer[],
   229→  eligiblePlayers: TGPlayer[],
   230→): RoleCombination[] {
   231→  const all = getAllValidCombinations();
   232→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   233→  const remaining = eligiblePlayers.filter(p => !fixedIds.has(p.pl_id));
   234→
   235→  const remWK = remaining.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   236→  const remBat = remaining.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   237→  const remAR = remaining.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   238→  const remBowl = remaining.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   239→
   240→  return all.filter(combo => {
   241→    if (!isCombinationCompatibleWithFixed(combo, fixedPlayers)) return false;
   242→
   243→    const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   244→    const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   245→    const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   246→    const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   247→
   248→    if (remWK < combo.wk - fixedWK) return false;
   249→    if (remBat < combo.bat - fixedBat) return false;
   250→    if (remAR < combo.ar - fixedAR) return false;
   251→    if (remBowl < combo.bowl - fixedBowl) return false;
   252→
   253→    return true;
   254→  });
   255→}

   257→/**
   258→ * Auto-select the best combination based on player pool and category.
   259→ * For multiple teams, rotates combinations intelligently.
   260→ */
   261→export function autoSelectCombination(
   262→  eligiblePlayers: TGPlayer[],
   263→  category: string,
   264→  fixedPlayers: TGPlayer[] = [],
   265→  teamIndex: number = 0,
   266→): RoleCombination {
   267→  const compatible = getCompatibleCombinations(fixedPlayers, eligiblePlayers);
   268→
   269→  if (compatible.length === 0) {
   270→    return { wk: 1, bat: 4, ar: 2, bowl: 4 };
   271→  }
   272→
   273→  // Category-preferred patterns
   274→  const preferredPatterns: RoleCombination[] =
   275→    category === 'H2H'
   276→      ? [{ wk: 1, bat: 5, ar: 2, bowl: 3 }, { wk: 1, bat: 4, ar: 2, bowl: 4 }, { wk: 1, bat: 3, ar: 3, bowl: 4 }]
   277→      : category === 'SL'
   278→      ? [{ wk: 1, bat: 4, ar: 2, bowl: 4 }, { wk: 1, bat: 3, ar: 3, bowl: 4 }, { wk: 2, bat: 3, ar: 2, bowl: 4 }]
   279→      : [ // Mega GL
   280→        { wk: 1, bat: 3, ar: 2, bowl: 5 }, { wk: 1, bat: 4, ar: 1, bowl: 5 },
   281→        { wk: 1, bat: 3, ar: 3, bowl: 4 }, { wk: 1, bat: 4, ar: 2, bowl: 4 },
   282→        { wk: 2, bat: 3, ar: 2, bowl: 4 }, { wk: 1, bat: 5, ar: 1, bowl: 4 },
   283→        { wk: 2, bat: 3, ar: 1, bowl: 5 }, { wk: 1, bat: 4, ar: 3, bowl: 3 },
   284→        { wk: 2, bat: 4, ar: 1, bowl: 4 }, { wk: 1, bat: 3, ar: 1, bowl: 6 },
   285→        { wk: 2, bat: 2, ar: 3, bowl: 4 }, { wk: 1, bat: 5, ar: 2, bowl: 3 },
   286→      ];
   287→
   288→  const preferredCompatible = preferredPatterns.filter(p =>
   289→    compatible.some(c => c.wk === p.wk && c.bat === p.bat && c.ar === p.ar && c.bowl === p.bowl)
   290→  );
   291→
   292→  const pool = preferredCompatible.length > 0 ? preferredCompatible : compatible;
   293→  const index = teamIndex % pool.length;
   294→
   295→  return pool[index];
   296→}

   298→/**
   299→ * Validate a manual combination against available players.
   300→ */
   301→export function validateCombination(
   302→  combo: RoleCombination,
   303→  eligiblePlayers: TGPlayer[],
   304→  fixedPlayers: TGPlayer[] = [],
   305→): { valid: boolean; errors: string[] } {
   306→  const errors: string[] = [];
   307→
   308→  if (combo.wk + combo.bat + combo.ar + combo.bowl !== CRICKET_TEAM_SIZE) {
   309→    errors.push(`Total players must be 11 (got ${combo.wk + combo.bat + combo.ar + combo.bowl})`);
   310→  }
   311→
   312→  if (combo.wk < MIN_WK || combo.wk > MAX_WK) errors.push(`WK must be ${MIN_WK}-${MAX_WK}`);
   313→  if (combo.bat < MIN_BAT || combo.bat > MAX_BAT) errors.push(`BAT must be ${MIN_BAT}-${MAX_BAT}`);
   314→  if (combo.ar < MIN_AR || combo.ar > MAX_AR) errors.push(`AR must be ${MIN_AR}-${MAX_AR}`);
   315→  if (combo.bowl < MIN_BOWL || combo.bowl > MAX_BOWL) errors.push(`BOWL must be ${MIN_BOWL}-${MAX_BOWL}`);
   316→
   317→  if (fixedPlayers.length > 0 && !isCombinationCompatibleWithFixed(combo, fixedPlayers)) {
   318→    errors.push('Combination is not compatible with selected fixed players');
   319→  }
   320→
   321→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
   322→  const remaining = eligiblePlayers.filter(p => !fixedIds.has(p.pl_id));
   323→
   324→  const fixedWK = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   325→  const fixedBat = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   326→  const fixedAR = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   327→  const fixedBowl = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   328→
   329→  const needWK = Math.max(0, combo.wk - fixedWK);
   330→  const needBat = Math.max(0, combo.bat - fixedBat);
   331→  const needAR = Math.max(0, combo.ar - fixedAR);
   332→  const needBowl = Math.max(0, combo.bowl - fixedBowl);
   333→
   334→  const availWK = remaining.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
   335→  const availBat = remaining.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
   336→  const availAR = remaining.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;
   337→  const availBowl = remaining.filter(p => p.role === PLAYER_ROLES.BOWLER).length;
   338→
   339→  if (availWK < needWK) errors.push(`Need ${needWK} WK but only ${availWK} available`);
   340→  if (availBat < needBat) errors.push(`Need ${needBat} BAT but only ${availBat} available`);
   341→  if (availAR < needAR) errors.push(`Need ${needAR} AR but only ${availAR} available`);
   342→  if (availBowl < needBowl) errors.push(`Need ${needBowl} BOWL but only ${availBowl} available`);
   343→
   344→  return { valid: errors.length === 0, errors };
   345→}

   425→function decryptField(encrypted: string): string {
   426→  try {
   427→    const bytes = CryptoJS.AES.decrypt(encrypted, ENCRYPTION_KEY);
   428→    const decrypted = bytes.toString(CryptoJS.enc.Utf8);
   429→    if (!decrypted) return encrypted;
   430→    return decrypted;
   431→  } catch {
   432→    return encrypted;
   433→  }
   434→}

   508→export async function fetchMatches(sport: string = 'cricket'): Promise<TGMatch[]> {
   509→  try {
   510→    const response = await fetch(`${TG_API_BASE}/fantasy/matches/${sport}`, {
   511→      next: { revalidate: 60 },
   512→    });
   513→
   514→    if (!response.ok) {
   515→      throw new Error(`Failed to fetch matches: ${response.status}`);
   516→    }
   517→
   518→    const data = await response.json();
   519→
   520→    if (data.status !== 'success' || !Array.isArray(data.data)) {
   521→      return [];
   522→    }
   523→
   524→    const matches: TGMatch[] = [];
   525→    for (const encryptedMatch of data.data) {
   526→      const match = decryptMatchData(encryptedMatch);
   527→      if (match) {
   528→        matches.push(match);
   529→      }
   530→    }
   531→
   532→    return matches;
   533→  } catch (error) {
   534→    console.error('Error fetching matches:', error);
   535→    return [];
   536→  }
   537→}

   655→/**
   656→ * Create a unique signature for a team based on sorted player IDs + C/VC IDs.
   657→ * Used for deduplication — two teams with identical signatures are duplicates.
   658→ */
   659→export function makeTeamSignature(players: TGPlayer[], captain: TGPlayer, viceCaptain: TGPlayer): string {
   660→  const playerIds = players.map(p => p.pl_id).sort((a, b) => a - b).join(',');
   661→  return `${playerIds}|C${captain.pl_id}|VC${viceCaptain.pl_id}`;

   664→/**
   665→ * Deduplicate and validate a list of generated teams.
   666→ * Removes exact duplicate player combinations.
   667→ * Separates valid and invalid teams based on lineup eligibility.
   668→ *
   669→ * Returns: { valid, invalid, duplicateCount, invalidLineupCount }
   670→ */
   671→export function deduplicateAndValidateTeams(
   672→  teams: GeneratedTeam[],
   673→  allPlayers: TGPlayer[],
   674→  avoidPlayerIds: Set<number> = new Set(),
   675→): {
   676→  valid: GeneratedTeam[];
   677→  invalid: GeneratedTeam[];
   678→  duplicateCount: number;
   679→  invalidLineupCount: number;
   680→} {
   681→  const seen = new Set<string>();
   682→  const valid: GeneratedTeam[] = [];
   683→  const invalid: GeneratedTeam[] = [];
   684→  let duplicateCount = 0;
   685→  let invalidLineupCount = 0;
   686→
   687→  for (let i = 0; i < teams.length; i++) {
   688→    const team = teams[i];
   689→    const sig = makeTeamSignature(team.players, team.captain, team.viceCaptain);
   690→
   691→    // Skip duplicates
   692→    if (seen.has(sig)) {
   693→      duplicateCount++;
   694→      continue;
   695→    }
   696→    seen.add(sig);
   697→
   698→    // Check lineup validity
   699→    const validation = validateTeamForLineup(team, allPlayers, avoidPlayerIds);
   700→    if (!validation.valid) {
   701→      invalidLineupCount++;
   702→      invalid.push({ ...team, id: invalid.length + 1 });
   703→    } else {
   704→      valid.push({ ...team, id: valid.length + 1 });
   705→    }
   706→  }
   707→
   708→  return { valid, invalid, duplicateCount, invalidLineupCount };
   709→}

   724→function seededRandom(seed: number): () => number {
   725→  let s = seed;
   726→  return () => {
   727→    s = (s * 16807 + 0) % 2147483647;
   728→    return (s - 1) / 2147483646;
   729→  };
   730→}

   732→function shuffleArray<T>(arr: T[], rng: () => number): T[] {
   733→  const result = [...arr];
   734→  for (let i = result.length - 1; i > 0; i--) {
   735→    const j = Math.floor(rng() * (i + 1));
   736→    [result[i], result[j]] = [result[j], result[i]];
   737→  }
   738→  return result;
   739→}

   741→// ============ PRO PLAYER SELECTION ENGINE ============
   742→// Each team tries to rank in the top 1-100.
   743→// Strategy: Must-have players + smart scoring + limited differential.
   744→
   745→/**
   746→ * Score a player for team selection (higher = better pick).
   747→ * Used for the MAIN generation engine (not just autoSelect).
   748→ * This is the PRO scoring that replaces the broken sort+shuffle approach.
   749→ */
   750→function playerScoreForGen(p: TGPlayer, category: string): number {
   751→  if (category === 'H2H') {
   752→    // H2H: Safest possible — selection % is king
   753→    return (
   754→      p.selected_by * 0.50 +
   755→      p.captain_percentage * 0.25 +
   756→      p.points * 0.15 +
   757→      p.vice_captain_percentage * 0.05 +
   758→      (p.playing === 1 ? 12 : 0) +
   759→      (p.role === PLAYER_ROLES.ALL_ROUNDER ? 3 : 0)
   760→    );
   761→  } else if (category === 'SL') {
   762→    // SL: Balanced — form + selection + captain history
   763→    return (
   764→      p.selected_by * 0.35 +
   765→      p.captain_percentage * 0.25 +
   766→      p.points * 0.25 +
   767→      p.vice_captain_percentage * 0.10 +
   768→      (p.playing === 1 ? 12 : 0) +
   769→      (p.role === PLAYER_ROLES.ALL_ROUNDER ? 3 : 0)
   770→    );
   771→  } else {
   772→    // Mega GL: Still prioritize core quality, differential comes from C/VC rotation
   773→    // and 2-3 "punt" players — NOT from making the whole team weak
   774→    return (
   775→      p.selected_by * 0.30 +
   776→      p.captain_percentage * 0.20 +
   777→      p.points * 0.25 +
   778→      p.vice_captain_percentage * 0.15 +
   779→      (p.playing === 1 ? 12 : 0) +
   780→      (p.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0) +
   781→      (p.role === PLAYER_ROLES.WICKET_KEEPER ? 1 : 0)
   782→    );
   783→  }
   784→}
   785→
   786→/**
   787→ * Smart select: Picks top players by score, with controlled randomness.
   788→ * - Keeps top `keepTop` players fixed (highest scored).
   789→ * - Randomly selects from the remaining pool for variety.
   790→ * - This FIXES the critical bug where shuffleArray after sort destroyed all ordering.
   791→ */
   792→function smartSelect(
   793→  pool: TGPlayer[],
   794→  count: number,
   795→  scoreFn: (p: TGPlayer) => number,
   796→  rng: () => number,
   797→  keepTopRatio: number = 0.7, // Keep top 70% of picks as best-scored
   798→): TGPlayer[] {
   799→  if (pool.length <= count) return [...pool];
   800→
   801→  const sorted = [...pool].sort((a, b) => scoreFn(b) - scoreFn(a));
   802→  const keepTop = Math.max(1, Math.floor(count * keepTopRatio));
   803→  const randomSlots = count - keepTop;
   804→
   805→  // Top players — always include these (they're the best picks)
   806→  const top = sorted.slice(0, keepTop);
   807→  const topIds = new Set(top.map(p => p.pl_id));
   808→
   809→  // Remaining pool (excluding already picked top players)
   810→  const remaining = sorted.filter(p => !topIds.has(p.pl_id));
   811→
   812→  // Random selection from remaining for variety/diversification
   813→  const shuffled = shuffleArray(remaining, rng);
   814→  const randomPicks = shuffled.slice(0, randomSlots);
   815→
   816→  return [...top, ...randomPicks];
   817→}

   819→/**
   820→ * Identify must-have players — players selected by >70% of users.
   821→ * These should be in EVERY team. Missing them = guaranteed bad rank.
   822→ */
   823→function getMustHavePlayers(
   824→  pool: TGPlayer[],
   825→  threshold: number = 70,
   826→): TGPlayer[] {
   827→  return pool.filter(p => p.selected_by >= threshold && p.playing === 1);

   830→/**
   831→ * Score an entire team's quality (higher = better team).
   832→ * Used to reject low-quality teams and retry.
   833→ */
   834→function scoreTeamQuality(team: TGPlayer[], category: string): number {
   835→  let score = 0;
   836→  const scoreFn = (p: TGPlayer) => playerScoreForGen(p, category);
   837→
   838→  for (const p of team) {
   839→    score += scoreFn(p);
   840→    // Bonus for must-have players (selected_by > 70%)
   841→    if (p.selected_by >= 70) score += 8;
   842→    // Bonus for confirmed playing
   843→    if (p.playing === 1) score += 3;
   844→    // Penalty for very low selection (< 10%) unless Mega GL
   845→    if (p.selected_by < 10 && category !== 'MEGA GL') score -= 5;
   846→  }
   847→
   848→  // Bonus for role diversity
   849→  const roles = new Set(team.map(p => p.role));
   850→  score += roles.size * 3;
   851→
   852→  // Bonus for team balance (close to 6-5 or 5-6 split)
   853→  const teamNames = [...new Set(team.map(p => p.team_name))];
   854→  if (teamNames.length === 2) {
   855→    const c1 = team.filter(p => p.team_name === teamNames[0]).length;
   856→    const c2 = team.filter(p => p.team_name === teamNames[1]).length;
   857→    const balance = Math.min(c1, c2); // Closer to 5.5 is better
   858→    score += balance * 2;
   859→  }
   860→
   861→  return score;
   862→}

   864→/**
   865→ * Category-specific C/VC scoring functions.
   866→ * Each category has a genuinely different strategy for selecting Captain and Vice-Captain.
   867→ *
   868→ * Mega GL: Prioritize differential captains (low selection % but high points).
   869→ *          Wider C/VC rotation across teams — different C/VC per team.
   870→ * SL:      Prioritize consistent/form players. Higher weight on selection % and captain %.
   871→ *          More stable C/VC — top 2 players usually.
   872→ * H2H:     Prioritize most-selected, safest players. Highest weight on selection %.
   873→ *          Most conservative C/VC — always the safest picks.
   874→ */
   875→function scoreCaptain(player: TGPlayer, category: string): number {
   876→  if (category === 'H2H') {
   877→    // H2H: Safest picks — highest selection % and captain % weighted most
   878→    return player.selected_by * 0.45 + player.captain_percentage * 0.35 + player.points * 0.2;
   879→  } else if (category === 'SL') {
   880→    // SL: Balanced — form players with good captain record
   881→    return player.captain_percentage * 0.35 + player.selected_by * 0.35 + player.points * 0.3;
   882→  } else {
   883→    // Mega GL: Differential — rotate captains, some differential (lower selection but higher upside)
   884→    return player.points * 0.4 + player.captain_percentage * 0.3 + player.selected_by * 0.2 + player.vice_captain_percentage * 0.1;
   885→  }
   886→}
   887→
   888→function scoreViceCaptain(player: TGPlayer, category: string): number {
   889→  if (category === 'H2H') {
   890→    return player.selected_by * 0.4 + player.vice_captain_percentage * 0.35 + player.points * 0.25;
   891→  } else if (category === 'SL') {
   892→    return player.vice_captain_percentage * 0.35 + player.selected_by * 0.35 + player.points * 0.3;
   893→  } else {
   894→    // Mega GL: Wider VC rotation
   895→    return player.points * 0.35 + player.vice_captain_percentage * 0.3 + player.selected_by * 0.2 + player.captain_percentage * 0.15;
   896→  }
   897→}
   898→
   899→/**
   900→ * Select Captain and Vice-Captain based on category strategy.
   901→ * Mega GL: Rotates C/VC across team members (wider differential).
   902→ * SL: Picks the top 2 most reliable.
   903→ * H2H: Picks the safest 2.
   904→ */
   905→function selectCaptainViceCaptain(
   906→  team: TGPlayer[],
   907→  category: string,
   908→  teamIndex: number,
   909→): { captain: TGPlayer; viceCaptain: TGPlayer } {
   910→  const isMegaGL = category !== 'H2H' && category !== 'SL';
   911→
   912→  if (isMegaGL) {
   913→    // Mega GL: Sort by captain score, then rotate C/VC position based on teamIndex
   914→    // This gives wider C/VC distribution across teams
   915→    const sorted = [...team].sort((a, b) => scoreCaptain(b, category) - scoreCaptain(a, category));
   916→    // Rotate: sometimes pick #1 as C and #2 as VC, sometimes #1 as C and #3 as VC, etc.
   917→    const cIdx = 0; // Always pick top scorer as C
   918→    let vcIdx = 1 + (teamIndex % Math.min(4, sorted.length - 1)); // Rotate VC among top 2-5
   919→    if (vcIdx >= sorted.length) vcIdx = 1;
   920→    return { captain: sorted[cIdx], viceCaptain: sorted[vcIdx] };
   921→  } else {
   922→    // H2H and SL: Pick top 2 by category-specific scoring
   923→    const sorted = [...team].sort((a, b) => scoreCaptain(b, category) - scoreCaptain(a, category));
   924→    return { captain: sorted[0], viceCaptain: sorted.length > 1 ? sorted[1] : sorted[0] };
   925→  }
   926→}

  1177→

  1339→export function generateExtraTeams(input: ExtraTeamGenInput): GenerationResult {
  1340→  const startTime = Date.now();

  1342→    fixedPlayers,
  1343→    captainOptions,
  1344→    viceCaptainOptions,
  1345→    leftPlayers,
  1346→    rightPlayers,
  1347→    category,
  1348→    count,
  1349→    seed = Date.now(),
  1350→    avoidPlayerIds = new Set(),
  1351→    combination = null,
  1352→    combinationMode: inputComboMode = 'auto',
  1353→  } = input;
  1354→
  1355→  const effectiveComboMode = combination ? 'manual' : inputComboMode;
  1356→
  1357→  // Apply lineup-aware filtering to the full player pool
  1358→  const allPlayersRaw = [...leftPlayers, ...rightPlayers];
  1359→  const lineupMode = getLineupMode(allPlayersRaw);

  1362→  const rightTeamName = rightPlayers[0]?.team_name || 'B';

  1365→  if (lineupMode === 'after' && allPlayers.length < 14) {
  1366→    console.warn(`[EXTRA TEAM GEN] Only ${allPlayers.length} eligible after lineup, including probable players as fallback.`);
  1367→    allPlayers = allPlayersRaw.filter(p => !avoidPlayerIds.has(p.pl_id) && p.playing !== -1);
  1368→  }

  1371→  const fixedIds = new Set(fixedPlayers.map(p => p.pl_id));
  1372→  const remainingPool = allPlayers.filter(p => !fixedIds.has(p.pl_id));
  1373→

  1375→  const remWK = remainingPool.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER);
  1376→  const remBat = remainingPool.filter(p => p.role === PLAYER_ROLES.BATSMAN);

  1381→  const fixedWKCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length;
  1382→  const fixedBatCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.BATSMAN).length;
  1383→  const fixedARCount = fixedPlayers.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length;

  1385→

  1387→  const remainingCredits = MAX_CREDITS - fixedCredits;
  1388→  const neededPlayers = CRICKET_TEAM_SIZE - fixedPlayers.length; // 3
  1389→
  1390→  const teams: GeneratedTeam[] = [];
  1391→
  1392→  // Pre-validate: fixed players must have credits <= 100
  1393→  if (fixedCredits > MAX_CREDITS) {
  1394→    return { teams, debug: makeExtraDebug(0, category, effectiveComboMode, count, lineupMode, allPlayers.length, allPlayersRaw.length, seed, 'INVALID_FIXED_CREDITS') };
  1395→  }
  1396→
  1397→  // Pre-validate: fixed players must respect max 7 from one team
  1398→  const fixedLeftCount = fixedPlayers.filter(p => p.team_name === leftTeamName).length;
  1399→  const fixedRightCount = fixedPlayers.filter(p => p.team_name === rightTeamName).length;
  1400→  if (fixedLeftCount > MAX_FROM_ONE_TEAM || fixedRightCount > MAX_FROM_ONE_TEAM) {
  1401→    return { teams, debug: makeExtraDebug(0, category, effectiveComboMode, count, lineupMode, allPlayers.length, allPlayersRaw.length, seed, 'INVALID_TEAM_DIST') };
  1402→  }
  1403→
  1404→  // Generate all valid C/VC combinations from the 5C and 5VC options
  1405→  // C must be in the team, VC must be in the team, C ≠ VC
  1406→  const cVCCombos: { c: TGPlayer; vc: TGPlayer }[] = [];
  1407→  for (const c of captainOptions) {
  1408→    for (const vc of viceCaptainOptions) {
  1409→      if (c.pl_id !== vc.pl_id) {
  1410→        cVCCombos.push({ c, vc });
  1411→      }
  1412→    }
  1413→  }
  1414→  // If no valid combos, return empty
  1415→  if (cVCCombos.length === 0) {
  1416→    return { teams, debug: makeExtraDebug(0, category, effectiveComboMode, count, lineupMode, allPlayers.length, allPlayersRaw.length, seed, 'NO_VALID_CVC') };
  1417→  }
  1418→
  1419→  // Category-specific role distribution preferences for remaining players
  1420→  const isH2H = category === 'H2H';
  1421→  const isSL = category === 'SL';
  1422→
  1423→  // EACH CATEGORY HAS ITS OWN DISTINCT PATTERNS — never reuse Mega GL patterns for H2H/SL
  1424→  const categoryPatterns: number[][] = isH2H
  1425→    ? [
  1426→        [1, 5, 2, 3], [1, 4, 2, 4], [1, 3, 3, 4], [1, 4, 3, 3],
  1427→        [2, 4, 1, 4], [1, 5, 1, 4],
  1428→      ]
  1429→    : isSL
  1430→    ? [
  1431→        [1, 4, 2, 4], [1, 3, 3, 4], [2, 3, 2, 4], [1, 5, 2, 3],
  1432→        [1, 4, 3, 3], [2, 4, 1, 4], [1, 3, 2, 5],
  1433→      ]
  1434→    : [ // Mega GL — most differential/varied patterns
  1435→        [1, 3, 2, 5], [1, 4, 1, 5], [1, 3, 3, 4], [1, 4, 2, 4],
  1436→        [2, 3, 2, 4], [1, 5, 1, 4], [2, 3, 1, 5], [1, 4, 3, 3],
  1437→        [2, 4, 1, 4], [1, 3, 1, 6], [2, 2, 3, 4], [1, 5, 2, 3],
  1438→      ];
  1439→
  1440→  // Build list of valid target distributions: [WK, BAT, AR, BOWL]
  1441→  const validDistributions: number[][] = [];
  1442→  for (let wk = MIN_WK; wk <= MAX_WK; wk++) {
  1443→    for (let bat = MIN_BAT; bat <= MAX_BAT; bat++) {
  1444→      for (let ar = MIN_AR; ar <= MAX_AR; ar++) {
  1445→        const bowl = CRICKET_TEAM_SIZE - wk - bat - ar;
  1446→        if (bowl < MIN_BOWL || bowl > MAX_BOWL) continue;
  1447→        // Check compatibility with fixed players
  1448→        const nWK = Math.max(0, wk - fixedWKCount);
  1449→        const nBat = Math.max(0, bat - fixedBatCount);
  1450→        const nAR = Math.max(0, ar - fixedARCount);
  1451→        const nBowl = Math.max(0, bowl - fixedBowlCount);
  1452→        if (nWK + nBat + nAR + nBowl !== neededPlayers) continue;
  1453→        // Check we have enough remaining players for each role
  1454→        if (remWK.length < nWK || remBat.length < nBat || remAR.length < nAR || remBowl.length < nBowl) continue;
  1455→        // For Mega GL / SL / H2H: prefer category-specific patterns but include all valid ones
  1456→        validDistributions.push([wk, bat, ar, bowl]);
  1457→      }
  1458→    }
  1459→  }
  1460→
  1461→  // Sort: prefer category-specific patterns first, then others
  1462→  if (categoryPatterns.length > 0) {
  1463→    const patternSet = new Set(categoryPatterns.map(p => p.join(',')));
  1464→    validDistributions.sort((a, b) => {
  1465→      const aInCat = patternSet.has(a.join(',')) ? 0 : 1;
  1466→      const bInCat = patternSet.has(b.join(',')) ? 0 : 1;
  1467→      return aInCat - bInCat;
  1468→    });
  1469→  }

  1742→// Player scoring for auto-selection: higher = better pick
  1743→function playerScore(p: TGPlayer): number {
  1744→  return (
  1745→    p.selected_by * 0.35 +       // selection % - most important for GL
  1746→    p.captain_percentage * 0.25 + // captain % - indicates trust
  1747→    p.points * 0.2 +             // points - performance indicator
  1748→    p.vice_captain_percentage * 0.1 + // VC %
  1749→    (p.playing === 1 ? 10 : 0) + // confirmed playing bonus
  1750→    (p.role === PLAYER_ROLES.ALL_ROUNDER ? 3 : 0) + // AR bonus (flexible)
  1751→    (p.role === PLAYER_ROLES.WICKET_KEEPER ? 1 : 0) // WK slight bonus
  1752→  );
  1753→}

  1755→export function autoSelectExtraPlayers(
  1756→  leftPlayers: TGPlayer[],
  1757→  rightPlayers: TGPlayer[],
  1758→  avoidPlayerIds: Set<number> = new Set(),
  1759→): AutoSelectResult {
  1760→  const allPlayersRaw = [...leftPlayers, ...rightPlayers];
  1761→  const leftTeamName = leftPlayers[0]?.team_name || 'A';
  1762→  const rightTeamName = rightPlayers[0]?.team_name || 'B';
  1763→
  1764→  // Use lineup-aware eligibility filter
  1765→  const eligible = getEligiblePlayers(allPlayersRaw, avoidPlayerIds);
  1766→
  1767→  const eligibleLeft = eligible.filter(p => p.team_name === leftTeamName);
  1768→  const eligibleRight = eligible.filter(p => p.team_name === rightTeamName);
  1769→
  1770→  // Sort by score descending
  1771→  const sortByScore = (a: TGPlayer, b: TGPlayer) => playerScore(b) - playerScore(a);
  1772→  const sortedLeft = [...eligibleLeft].sort(sortByScore);
  1773→  const sortedRight = [...eligibleRight].sort(sortByScore);
  1774→
  1775→  // === SELECT 8 FIX PLAYERS ===
  1776→  // Strategy: balanced 4-5 from each team, role-diverse
  1777→  const fixedPlayers: TGPlayer[] = [];
  1778→  const fixedIds = new Set<number>();
  1779→
  1780→  const targetLeft = Math.min(4, sortedLeft.length);
  1781→  const targetRight = Math.min(4, sortedRight.length);
  1782→
  1783→  // Pick diverse players ensuring role coverage
  1784→  const pickDiverse = (pool: TGPlayer[], count: number, existing: TGPlayer[]): TGPlayer[] => {
  1785→    const picked: TGPlayer[] = [];
  1786→    const usedIds = new Set(existing.map(p => p.pl_id));
  1787→    const existingRoles = {
  1788→      wk: existing.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER).length,
  1789→      bat: existing.filter(p => p.role === PLAYER_ROLES.BATSMAN).length,
  1790→      ar: existing.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER).length,
  1791→      bowl: existing.filter(p => p.role === PLAYER_ROLES.BOWLER).length,
  1792→    };
  1793→
  1794→    // First pass: fill role gaps
  1795→    const roleNeeds = [
  1796→      { role: PLAYER_ROLES.WICKET_KEEPER, need: Math.max(0, 1 - existingRoles.wk), pool: pool.filter(p => p.role === PLAYER_ROLES.WICKET_KEEPER && !usedIds.has(p.pl_id)) },
  1797→      { role: PLAYER_ROLES.ALL_ROUNDER, need: Math.max(0, 1 - existingRoles.ar), pool: pool.filter(p => p.role === PLAYER_ROLES.ALL_ROUNDER && !usedIds.has(p.pl_id)) },
  1798→      { role: PLAYER_ROLES.BATSMAN, need: Math.max(0, 2 - existingRoles.bat), pool: pool.filter(p => p.role === PLAYER_ROLES.BATSMAN && !usedIds.has(p.pl_id)) },
  1799→      { role: PLAYER_ROLES.BOWLER, need: Math.max(0, 2 - existingRoles.bowl), pool: pool.filter(p => p.role === PLAYER_ROLES.BOWLER && !usedIds.has(p.pl_id)) },
  1800→    ];
  1801→
  1802→    for (const rn of roleNeeds) {
  1803→      for (let i = 0; i < rn.need && picked.length < count; i++) {
  1804→        const candidate = rn.pool.find(p => !picked.some(pp => pp.pl_id === p.pl_id));
  1805→        if (candidate) picked.push(candidate);
  1806→      }
  1807→    }
  1808→
  1809→    // Second pass: fill remaining with top-scored players
  1810→    const remaining = pool.filter(p => !usedIds.has(p.pl_id) && !picked.some(pp => pp.pl_id === p.pl_id));
  1811→    for (const p of remaining) {
  1812→      if (picked.length >= count) break;
  1813→      picked.push(p);
  1814→    }
  1815→
  1816→    return picked;
  1817→  };
  1818→
  1819→  const leftPicks = pickDiverse(sortedLeft, targetLeft, []);
  1820→  for (const p of leftPicks) { fixedPlayers.push(p); fixedIds.add(p.pl_id); }
  1821→
  1822→  const rightPicks = pickDiverse(sortedRight, targetRight, fixedPlayers);
  1823→  for (const p of rightPicks) { fixedPlayers.push(p); fixedIds.add(p.pl_id); }
  1824→
  1825→  // Fill to 8 if needed
  1826→  if (fixedPlayers.length < 8) {
  1827→    const remaining = eligible.filter(p => !fixedIds.has(p.pl_id)).sort(sortByScore);
  1828→    for (const p of remaining) {
  1829→      if (fixedPlayers.length >= 8) break;
  1830→      fixedPlayers.push(p); fixedIds.add(p.pl_id);
  1831→    }
  1832→  }
  1833→
  1834→  // Trim to 8 if over
  1835→  while (fixedPlayers.length > 8) {
  1836→    let minIdx = 0; let minScore = playerScore(fixedPlayers[0]);
  1837→    for (let i = 1; i < fixedPlayers.length; i++) {
  1838→      const s = playerScore(fixedPlayers[i]);
  1839→      if (s < minScore) { minScore = s; minIdx = i; }
  1840→    }
  1841→    const removed = fixedPlayers.splice(minIdx, 1)[0];
  1842→    fixedIds.delete(removed.pl_id);

  1844→

  1846→  const cCandidates = eligible
  1847→    .filter(p => !avoidPlayerIds.has(p.pl_id))
  1848→    .sort((a, b) => {
  1849→      const scoreA = a.captain_percentage * 0.4 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
  1850→      const scoreB = b.captain_percentage * 0.4 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 5 : 0);
  1851→      return scoreB - scoreA;
  1852→    })

  1854→
  1855→  // === SELECT 5 VICE CAPTAIN OPTIONS ===
  1856→  const vcCandidates: TGPlayer[] = [];
  1857→  const vcPool = eligible
  1858→    .filter(p => !avoidPlayerIds.has(p.pl_id))
  1859→    .sort((a, b) => {
  1860→      const scoreA = a.vice_captain_percentage * 0.35 + a.selected_by * 0.3 + a.points * 0.2 + (a.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
  1861→      const scoreB = b.vice_captain_percentage * 0.35 + b.selected_by * 0.3 + b.points * 0.2 + (b.role === PLAYER_ROLES.ALL_ROUNDER ? 4 : 0);
  1862→      return scoreB - scoreA;
  1863→    });
  1864→
  1865→  for (const p of vcPool) {
  1866→    if (vcCandidates.length >= 5) break;
  1867→    vcCandidates.push(p);
  1868→  }
  1869→
  1870→  return {
  1871→    fixedPlayers: fixedPlayers.slice(0, 8),
  1872→    captainOptions: cCandidates.slice(0, 5),
  1873→    viceCaptainOptions: vcCandidates.slice(0, 5),
  1874→  };
  1875→}
