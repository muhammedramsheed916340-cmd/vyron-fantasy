   132→   132→// Countdown timer component
   133→   133→function CountdownTimer({ matchTime }: { matchTime: string }) {
   134→   134→  const [timeLeft, setTimeLeft] = useState('')
   135→   135→
   136→   136→  useEffect(() => {
   137→   137→    const calculateTime = () => {
   138→   138→      const now = new Date().getTime()
   139→   139→      const matchDate = new Date(matchTime).getTime()
   140→   140→      const diff = matchDate - now
   141→   141→
   142→   142→      if (diff <= 0) {
   143→   143→        setTimeLeft('Started')
   144→   144→        return
   145→   145→      }
   146→   146→
   147→   147→      const days = Math.floor(diff / (1000 * 60 * 60 * 24))
   148→   148→      const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60))
   149→   149→      const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60))
   150→   150→      const seconds = Math.floor((diff % (1000 * 60)) / 1000)
   151→   151→
   152→   152→      if (days > 0) {
   153→   153→        setTimeLeft(`${days}d ${hours}h ${minutes}m ${seconds}s`)
   154→   154→      } else {
   155→   155→        setTimeLeft(`${hours}h ${minutes}m ${seconds}s`)
   156→   156→      }
   157→   157→    }
   158→   158→
   159→   159→    calculateTime()
   160→   160→    const interval = setInterval(calculateTime, 1000)
   161→   161→    return () => clearInterval(interval)
   162→   162→  }, [matchTime])
   163→   163→
   164→   164→  return <span className="font-medium text-sm text-foreground">{timeLeft}</span>
   165→   165→}
